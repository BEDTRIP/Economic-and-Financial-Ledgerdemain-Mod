# Every state with a financial centre: a stock exchange method on while the country's market sells that stock and the
# economy is not cooperative or command, off otherwise (R8б.4: by the building group, the types' list --
# zz_ef_fc_activate_pm, scripted_effects/ld_fc_types.txt).
financial_center_production_methods = {
	every_scope_state = {
		limit = { building_financial_centre_any = yes }
		zz_ef_fc_pm_switch = { K = manufacture }
		zz_ef_fc_pm_switch = { K = agricultural }
		zz_ef_fc_pm_switch = { K = mining }
		zz_ef_fc_pm_switch = { K = railroad }
	}
}

# from common/scripted_effects/01_financial_scripted_effects.txt, lines 20297-20333
# State scope: the method pair pm_$K$_stock_exchange / pm_no_$K$_stock_exchange of the state's financial centres.
zz_ef_fc_pm_switch = {
	if = {
		limit = {
			owner = {
				$K$_stock_market_goods_sell_orders > 0
				NOT = {
					OR = {
						has_law = law_type:law_cooperative_ownership
						has_law = law_type:law_command_economy
					}
				}
			}
			any_scope_building = {
				is_building_group = bg_financial_centre
				has_active_production_method = pm_no_$K$_stock_exchange
			}
		}
		zz_ef_fc_activate_pm = { PM = pm_$K$_stock_exchange }
	}
	if = {
		limit = {
			owner = {
				OR = {
					has_law = law_type:law_cooperative_ownership
					has_law = law_type:law_command_economy
					$K$_stock_market_goods_sell_orders = 0
				}
			}
			any_scope_building = {
				is_building_group = bg_financial_centre
				has_active_production_method = pm_$K$_stock_exchange
			}
		}
		zz_ef_fc_activate_pm = { PM = pm_no_$K$_stock_exchange }
	}
}
