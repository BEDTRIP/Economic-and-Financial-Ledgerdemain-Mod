# 2. Начало set_construction_regulator_level, после set_local_variable regulator_level -- снос регулятора, у которого
#    уровней больше, чем у сектора штата (через 4 недели; следующий блок того же эффекта строил его заново пустым):
    if = {
        limit = {
                state_has_building_type_levels = {
                    target = bt:building_construction_regulator
                    value > local_var:regulator_level
                }
        }
        if = {
            limit = {
                OR = {
                    NOT = {
                        has_variable = week_count
                    }
                    var:week_count >= 4  
                }
            }
            remove_building = "building_construction_regulator"
            set_variable = {
                name = week_count
                value = 0
            }
        }
        else = {
            change_variable = {
                name = week_count
                add = 1
            }
        }
    }
