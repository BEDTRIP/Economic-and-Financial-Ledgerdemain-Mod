# из common/scripted_effects/ld_subject_metal.txt, строки 18–80 (на 6.10.2026), zz_ef_subject_metal_step (разовый пересчёт металла субъектов на 12-й неделе)
# 6.10 (М, Д2.13, night of stage 2): no longer called -- a subject's metal is only what it bought; kept for
# history. В2.5 below stays (metal moves with its state, a pair).
# Country scope, once (var:zz_ef_subject_metal_done).
zz_ef_subject_metal_step = {
	if = {
		limit = {
			NOT = { has_variable = zz_ef_subject_metal_done }
			OR = {
				is_subject = yes
				NOT = { has_modifier = has_central_bank }
			}
			zz_ef_metal_in_gold > 0
		}
		set_variable = { name = zz_ef_subject_metal_done value = yes }
		# the metal 40% of the money supply would need, in gold
		set_variable = { name = zz_ef_metal_target_gold value = zz_ef_metal_target_in_gold }
		if = {
			limit = { is_subject = yes }
			# to the overlord's CB, in its standard
			overlord = {
				random_scope_state = {
					limit = { is_capital = yes }
					if = {
						limit = { NOT = { has_modifier = central_bank_historic_place } }
						add_modifier = { name = central_bank_historic_place }
					}
					if = {
						limit = { owner = { has_law = law_type:law_silver_standard } }
						if = { limit = { NOT = { has_variable = silver_state_1 } } set_variable = { name = silver_state_1 value = 0 } }
						change_variable = {
							name = silver_state_1
							add = {
								value = root.var:zz_ef_metal_target_gold
								multiply = gold_to_silver_rate
							}
						}
					}
					else = {
						if = { limit = { NOT = { has_variable = gold_state_1 } } set_variable = { name = gold_state_1 value = 0 } }
						change_variable = { name = gold_state_1 add = root.var:zz_ef_metal_target_gold }
					}
				}
			}
			every_scope_state = {
				limit = { has_modifier = central_bank_historic_place }
				if = { limit = { has_variable = gold_state_1 } set_variable = { name = gold_state_1 value = 0 } }
				if = { limit = { has_variable = silver_state_1 } set_variable = { name = silver_state_1 value = 0 } }
			}
		}
		else = {
			# independent, no CB: scale its own metal to the target
			set_variable = { name = zz_ef_metal_scale value = var:zz_ef_metal_target_gold }
			change_variable = { name = zz_ef_metal_scale divide = zz_ef_metal_in_gold }
			every_scope_state = {
				limit = { has_modifier = central_bank_historic_place }
				if = { limit = { has_variable = gold_state_1 } change_variable = { name = gold_state_1 multiply = owner.var:zz_ef_metal_scale } }
				if = { limit = { has_variable = silver_state_1 } change_variable = { name = silver_state_1 multiply = owner.var:zz_ef_metal_scale } }
			}
		}
		debug_log = "EFM|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|subject_metal|subj [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_is_subject')|0]|target_gold [THIS.GetCountry.MakeScope.Var('zz_ef_metal_target_gold').GetValue|0]"
	}
}
