# from common/scripted_effects/ld_anchor.txt, the end of the file
# The CB's reserves into the anchor's metal, as on a change of standard (ld_standard_switch.txt): gold -- the silver
# into gold at silver_to_gold_rate, silver -- the gold into silver; bimetallism and no metal -- as they are.
zz_ef_anchor_metal_step = {
	if = {
		limit = { zz_ef_metal_gold = yes }
		every_scope_state = {
			limit = {
				has_modifier = central_bank_historic_place
				has_variable = silver_state_1
				var:silver_state_1 > 0
			}
			if = {
				limit = { NOT = { has_variable = gold_state_1 } }
				set_variable = { name = gold_state_1 value = 0 }
			}
			change_variable = { name = gold_state_1 add = { value = var:silver_state_1 multiply = silver_to_gold_rate } }
			set_variable = { name = silver_state_1 value = 0 }
		}
	}
	else_if = {
		limit = { zz_ef_metal_silver = yes }
		every_scope_state = {
			limit = {
				has_modifier = central_bank_historic_place
				has_variable = gold_state_1
				var:gold_state_1 > 0
			}
			if = {
				limit = { NOT = { has_variable = silver_state_1 } }
				set_variable = { name = silver_state_1 value = 0 }
			}
			change_variable = { name = silver_state_1 add = { value = var:gold_state_1 multiply = gold_to_silver_rate } }
			set_variable = { name = gold_state_1 value = 0 }
		}
	}
}
