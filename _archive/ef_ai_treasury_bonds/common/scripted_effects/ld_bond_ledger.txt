# Holder scope, E&F's treasury slot $N$ taken into the register: the slot free again (as E&F's maturity leaves it).
zz_ef_bl_slot_free = {
	set_variable = { name = ai_buy_bond_$N$_on value = 0 }
	set_variable = { name = ai_interest_purchasing_$N$ value = 0 }
	set_variable = { name = ai_bond_quantity_$N$ value = 0 }
	set_variable = { name = ai_quantity_purchasing_$N$ value = 0 }
	set_variable = { name = ai_selle_bond_maturity_yers_time_$N$ value = 0 }
	set_variable = { name = ai_interest_calculated_$N$ value = 0 }
	set_variable = { name = ai_total_bond_value_$N$ value = 0 }
	set_variable = { name = ai_interest_at_term_calculated_$N$ value = 0 }
	set_variable = { name = ai_interest_per_month_calculated_$N$ value = 0 }
	set_variable = { name = ai_selle_bond_maturity_yers_$N$ value = 0 }
}
# treasury slot 1: E&F's purchase ai_buy_bond_1 (ai_buy_bond_1_on / ai_seller_country_general_1 /
# ai_total_bond_value_1) -> the register, the slot freed for E&F's next purchase
zz_ef_bl_slot_1 = {
	if = {
		limit = {
			has_variable = ai_buy_bond_1_on
			var:ai_buy_bond_1_on = 1
			has_variable = ai_seller_country_general_1
			has_variable = ai_total_bond_value_1
			exists = var:ai_seller_country_general_1
			var:ai_seller_country_general_1 = { is_country_type = recognized }
		}
		var:ai_seller_country_general_1 = { save_scope_as = zz_ef_bl_seller }
		set_variable = { name = zz_ef_bl_x value = var:ai_total_bond_value_1 }
		zz_ef_bl_register = { H = tr FB = bl INTO = treasury }
		zz_ef_bl_slot_free = { N = 1 }
	}
}

# Holder scope: a purchase of var:zz_ef_bl_x from scope:zz_ef_bl_seller. The part is at most the seller's room
# (principal - the parts held abroad); the rest back where it came from ($INTO$: treasury | pool); the part -- a claim
# in our map zz_ef_rq_$H$_b, its price into the seller's pool (its banks sold a part of their claim on the engine's debt).
zz_ef_bl_register = {
	set_variable = { name = zz_ef_bl_y value = var:zz_ef_bl_x }
	change_variable = { name = zz_ef_bl_y subtract = scope:zz_ef_bl_seller.zz_ef_bl_room }
	if = {
		limit = { var:zz_ef_bl_y > 0 }
		change_variable = { name = zz_ef_bl_x subtract = var:zz_ef_bl_y }
		set_variable = { name = zz_ef_bl_z value = var:zz_ef_bl_x }
		set_variable = { name = zz_ef_bl_x value = var:zz_ef_bl_y }
		zz_ef_bl_into_$INTO$ = yes
		change_variable = { name = zz_ef_f_$FB$_refund add = var:zz_ef_bl_y }
		set_variable = { name = zz_ef_bl_x value = var:zz_ef_bl_z }
		remove_variable = zz_ef_bl_z
	}
	if = {
		limit = { var:zz_ef_bl_x > 0 }
		change_variable = { name = zz_ef_f_$FB$_buy add = var:zz_ef_bl_x }
		zz_ef_rq_add = { H = $H$ K = b D = scope:zz_ef_bl_seller V = var:zz_ef_bl_x }
		scope:zz_ef_bl_seller = {
			zz_ef_bl_acc_add = { F = sold V = root.var:zz_ef_bl_x }
			add_investment_pool = root.var:zz_ef_bl_x
			zz_ef_bl_seller_capital = { V = root.var:zz_ef_bl_x }
		}
	}
	remove_variable = zz_ef_bl_x
	remove_variable = zz_ef_bl_y
}
# Holder scope, a private banks' purchase with no seller in E&F's list (Britain 1838.1, 1.95M): no debt to
# hold a part of -- back to our pool (only existing debt is bought).
zz_ef_pb_no_seller = {
	add_investment_pool = var:zz_ef_bl_x
	change_variable = { name = zz_ef_f_pb_refund add = var:zz_ef_bl_x }
	if = { limit = { zz_ef_logs_on = yes } debug_log = "EFP|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|no_seller [THIS.GetCountry.MakeScope.Var('zz_ef_bl_x').GetValue|0]" }
	remove_variable = zz_ef_bl_x
}
# private banks' slot 1: E&F's purchase ai_privat_bank_bond_1 / _renforcement (ai_privat_bank_bond_value_1, the
# seller from its list ai_privat_bank_seller_country_general_1) -> the register; E&F's value back to 0
zz_ef_pb_slot_1 = {
	if = {
		limit = {
			has_variable = ai_privat_bank_bond_value_1
			var:ai_privat_bank_bond_value_1 > 0
		}
		set_variable = { name = zz_ef_bl_x value = var:ai_privat_bank_bond_value_1 }
		if = {
			limit = { has_variable_list = ai_privat_bank_seller_country_general_1 }
			random_in_list = {
				variable = ai_privat_bank_seller_country_general_1
				limit = { is_country_type = recognized }
				save_scope_as = zz_ef_pbsel_1
			}
		}
		if = {
			limit = { exists = scope:zz_ef_pbsel_1 }
			scope:zz_ef_pbsel_1 = { save_scope_as = zz_ef_bl_seller }
			zz_ef_bl_register = { H = bk FB = pb INTO = pool }
		}
		else = {
			zz_ef_pb_no_seller = yes
		}
		set_variable = { name = ai_privat_bank_bond_value_1 value = 0 }
		if = { limit = { has_variable = ai_privat_bank_interest_calculated_1 } set_variable = { name = ai_privat_bank_interest_calculated_1 value = 0 } }
	}
}
