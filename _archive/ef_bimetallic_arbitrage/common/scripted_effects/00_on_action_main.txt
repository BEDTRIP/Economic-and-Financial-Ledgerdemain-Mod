# Вырезано из common/scripted_effects/00_on_action_main.txt, строки 823-963 до выреза: блок биметаллического арбитража в central_bank_ef_on_yearly_pulse_country (после currency_crisis_count_reset_condition … economic_crisis_count_reset_condition).
	#################################################
	#Arbitrage: bimetalism arbi pre 1873
	#################################################
	if = {
		limit = {
			game_date < 1873.1.1
			has_law = law_type:law_bimetallism_standard
		}

		#gold_drain
		if = {
			limit = {
				gold_to_silver_rate > bimetallic_rate_gold_to_silver
				misalignment_rate_gold_drain >= 0.05
			}

			random_in_global_list = {
				variable = global_arbitrage_bank_variable_list_ordered #private bank
				save_scope_as = arbitrage_privat_bank

			}
			every_scope_state = {
				if = {
					limit = {
						has_modifier = central_bank_historic_place
						is_capital = yes
					}
					save_scope_as = central_bank_site_seller
				}
			}

			set_variable = {
				name = gold_state_1_fix
				value = gold_lost
			}
			set_variable = {
				name = silver_state_1_fix
				value = silver_gained
			}

			private_bank_arbitrage_gold_drain = yes

			#Country arbitrage
			scope:central_bank_site_seller = {
				change_variable = {
					name = gold_state_1
					subtract = owner.var:gold_state_1_fix

				}
				change_variable = {
					name = silver_state_1
					add = owner.var:silver_state_1_fix
				}
			}

			#event:
			root = {
				save_scope_as = target_country
			}
			scope:arbitrage_privat_bank.owner = {
				save_scope_as = privat_bank_country
			}
			scope:privat_bank_country = {
				capital= {
					save_scope_as = privat_bank_country_capital_state
				}
				if = {
					limit = {
						root = {
							is_player = yes
						}
					}
					trigger_event = 00_ef_economic_event.95
				}
			}
		}
		#silver_drain
		if = {
			limit = {
				gold_to_silver_rate < bimetallic_rate_gold_to_silver
				misalignment_rate_silver_drain >= 0.05
			}
			random_in_global_list = {
				variable = global_arbitrage_bank_variable_list_ordered #private bank
				save_scope_as = arbitrage_privat_bank
			}
			every_scope_state = {
				if = {
					limit = {
						has_modifier = central_bank_historic_place
						is_capital = yes
					}
					save_scope_as = central_bank_site_seller
				}
			}

			set_variable = {
				name = gold_state_1_fix
				value = gold_gained
			}
			set_variable = {
				name = silver_state_1_fix
				value = silver_lost
			}

			private_bank_arbitrage_silver_drain = yes

			#Country arbitrage
			scope:central_bank_site_seller = {
				change_variable = {
					name = gold_state_1
					add = owner.var:gold_state_1_fix
				}
				change_variable = {
					name = silver_state_1
					subtract = owner.var:silver_state_1_fix
				}
			}

			#event:
			root = {
				save_scope_as = target_country
			}
			scope:arbitrage_privat_bank.owner = {
				save_scope_as = privat_bank_country
			}
			scope:privat_bank_country = {
				capital= {
					save_scope_as = privat_bank_country_capital_state
				}
				if = {
					limit = {
						root = {
							is_player = yes
						}
					}
					trigger_event = 00_ef_economic_event.96
				}
			}
		}
	}
