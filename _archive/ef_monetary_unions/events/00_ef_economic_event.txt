	type = country_event

	title = 00_ef_economic_event.10.t
	desc = 00_ef_economic_event.10.d
	flavor = 00_ef_economic_event.10.f

	dlc = dlc_ef
	duration = 3 #0 pas de duration

	event_image = {
		texture = "gfx/event_pictures/00_ef_economic_event_10.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/latin_monetary_union.dds"

	}

	immediate = { 
	}

		name = 00_ef_economic_event.10.a  
		trigger = {
			not = {
				or = {
					market_owner_is_root_univ_with_buiding = yes
					is_subject_custom_trigger = yes
				}
			}

			}

			type = latin_monetary_union_je_1

			value = 1
		} 
	}
	option = {
		name = 00_ef_economic_event.10.b
		default_option = yes
	} 
}

00_ef_economic_event.23 = { #image ok text en
	type = country_event
	placement = scope:capital_state

	#hidden = yes

	title = 00_ef_economic_event.23.t
	desc = 00_ef_economic_event.23.d
	flavor = 00_ef_economic_event.23.f

	dlc = dlc_ef
	duration = 3 #0 pas de duration

	event_image = {
		texture = "gfx/event_pictures/00_ef_economic_event_23.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/scandinavian_monetary_union.dds"

	trigger = {
	}

	immediate = {  
		create_treaty = {
			name = treaty_name_scandinavian_monetary_union
			first_country = scope:target_country
			second_country = scope:scandinavian_monetary_union_initiator 

			is_draft = no 
			binding_period = { years = 20 } 

			articles_to_create = {
				{ 	#scandinavian_monetary_union_treaty
					article = scandinavian_monetary_union_treaty
				}
			}
		}
		#law and money value target modification
		every_country = {
			limit = {
				var:scandinavian_monetary_union_on_discution = 1 
			}   
			#money_value_target_pre_set = yes
			activate_law = law_type:law_gold_standard
			central_bank_production_methods = yes
			if = { limit = { has_modifier = foreign_exchange_controls } remove_modifier = foreign_exchange_controls }
			add_modifier = {
				name = foreign_exchange_controls
				months = 12
			}
			reset_debt_in_currency_for_monetary_systeme_transition = yes
			set_variable = {
				name = money_value_target_1
				value = global_var:money_value_median
			}
			trade_balance = yes
			add_modifier = {
				name = monetary_systeme_transition
				months = 120
			} 
		}

		#set le money value
		#changer le standart
	} 

	option = {
		name = 00_ef_economic_event.23.a
		default_option = yes

		#ajouter +5% de prestige
	}  
}
