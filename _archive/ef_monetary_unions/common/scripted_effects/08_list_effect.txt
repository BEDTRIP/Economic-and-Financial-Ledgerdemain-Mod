				country_has_primary_culture = cu:swedish

			}

				or = {
					market_owner_is_root_univ_with_buiding = yes
					is_subject_custom_trigger = yes
				}
			} 	
        }
        root = {
            add_to_variable_list = {
                name = scandinavian_leader_list_1
                target = prev
            }
        }
    }
    root  = {
        if = { 
            clear_variable_list = list_before_order
            every_in_list = {
                variable = scandinavian_leader_list_1
                root = {
                    add_to_variable_list = {
                        name = list_before_order
                        target = prev
                    }
                } 
            }

			#local

            ordered_in_list = {
                variable = list_before_order
                order_by = national_capacity_in_gold

                max = 5

                    add_to_variable_list = {
                        name = scandinavian_leader_list_1_ordered_1
                        target = prev
                    } 
                }
            }
			
			#global
            clear_global_variable_list = scandinavian_leader_list_1_ordered_1

                variable = scandinavian_leader_list_1_ordered_1
                order_by = national_capacity_in_gold

                max = 5
                check_range_bounds = no
                root = {
                    add_to_global_variable_list = {
                        name = scandinavian_leader_list_1_ordered_1
                        target = prev
                    } 
                }
            }

            clear_global_variable_list = scandinavian_leader_list_1_country_1

                order_by = national_capacity_in_gold
