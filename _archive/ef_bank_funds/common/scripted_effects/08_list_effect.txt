#dinar 
global_arbitrage_bank_dinar_variable_list = { 
   	every_country = {
		every_company = {
			limit = {
				or = {
					type = company_type:company_Rothschild_Bank_gbr  
					type = company_type:company_Rothschild_Bank_fra  
					type = company_type:company_Rothschild_Bank_ger  
					type = company_type:company_Rothschild_Bank_aus  
					type = company_type:company_Rothschild_Bank_ita  
					type = company_type:company_BanqueDeParisEtDesPaysBas  
					type = company_type:company_BankJPMorgan  
					type = company_type:company_DeutscheBank  
					type = company_type:company_Bankenverein  
					type = company_type:company_BancaCommercialeItaliana  
					type = company_type:company_DeNederlandscheBank  
					type = company_type:company_amsterdamschebank  
					type = company_type:company_RotterdamscheBankvereeniging  
					type = company_type:company_Mitsubishiexchangehousebank   
				}
			}
			root = {
				add_to_variable_list = {
					name = global_arbitrage_bank_dinar_variable_list_1
					target = prev
				}
			}
		} 
	}  
 
	if = {
		limit = { exists = root }
		root  = {
			if = {
				clear_variable_list = list_before_order
				every_in_list = {
					variable = global_arbitrage_bank_dinar_variable_list_1
					root = {
						add_to_variable_list = {
							name = list_before_order
							target = prev
						}
					}
				}
				clear_global_variable_list = global_arbitrage_bank_dinar_variable_list_1_ordered
				if = {
					ordered_in_list = {
						variable = list_before_order
						order_by = stockpiling_dinar_private_bank_gui
						max = 100
						check_range_bounds = no
						root = {
							add_to_global_variable_list = {
								name = global_arbitrage_bank_dinar_variable_list_1_ordered
								target = prev
							}
						}
					}
				}
			}
			clear_variable_list = global_arbitrage_bank_dinar_variable_list_1
		}
	}
}
