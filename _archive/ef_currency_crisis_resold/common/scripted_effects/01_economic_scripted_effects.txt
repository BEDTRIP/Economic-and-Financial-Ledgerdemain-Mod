    #dinar
	sell_dinar_currency_crisis = {
	scope:seller = {
		save_scope_as = seller_root
		every_scope_state = {
			limit = {
				has_modifier = central_bank_historic_place
				is_capital = yes
			}
			save_scope_as = central_bank_site_root
		}
	}
	scope:target_country = {
		save_scope_as = overrated_currency_country_owner
		every_scope_state = {
			limit = {
				has_modifier = central_bank_historic_place
				is_capital = yes
			}
			save_scope_as = central_bank_site_overrated_currency_owner
		}
	}
	if = {
		limit = {
			exists = scope:overrated_currency_country_owner
			exists = scope:central_bank_site_root
			exists = scope:central_bank_site_overrated_currency_owner
			scope:overrated_currency_country_owner = {
				has_law = law_type:law_dinar_currency
				sell_currency_crisis_1 = yes
			}
		}
		scope:seller_root = {
			# all of it, at the parity, as far as the issuer's metal goes
			set_variable = { name = stockpiling_overrated_quantity_var value = stockpiling_dinar_state }
			set_variable = {
				name = zz_ef_crisis_due
				value = {
					value = var:stockpiling_overrated_quantity_var
					multiply = scope:overrated_currency_country_owner.money_value_target_real
					min = 0
				}
			}
			zz_ef_crisis_redeem = yes
			if = {
				limit = {
					has_variable = stockpiling_dinar_c_in_reserve_currency
					var:stockpiling_dinar_c_in_reserve_currency > 0
				}
				set_variable = { name = stockpiling_dinar_c_in_reserve_currency value = 0 }
			}
			scope:central_bank_site_root = {
				if = { limit = { has_variable = stockpiling_dinar_state_1 } set_variable = { name = stockpiling_dinar_state_1 value = 0 } }
				if = { limit = { has_variable = stockpiling_dinar_reserve_currency_state_1 } set_variable = { name = stockpiling_dinar_reserve_currency_state_1 value = 0 } }
			}
		}
		scope:overrated_currency_country_owner = {
			reset_debt_in_national_currency = yes
		}
	}
}

		save_scope_as = seller_root
		every_scope_state = {
			limit = {
				has_modifier = central_bank_historic_place
				is_capital = yes
			}
			save_scope_as = central_bank_site_root
		}
	}
	scope:target_country = {
		save_scope_as = overrated_currency_country_owner
		every_scope_state = {
			limit = {
				has_modifier = central_bank_historic_place
				is_capital = yes
			}
			save_scope_as = central_bank_site_overrated_currency_owner
		}
	}
	if = {
		limit = {
			exists = scope:overrated_currency_country_owner
			exists = scope:central_bank_site_root
			exists = scope:central_bank_site_overrated_currency_owner
			scope:overrated_currency_country_owner = {
				has_law = law_type:law_eco_ariary_currency
				sell_currency_crisis_1 = yes
			}
		}
		scope:seller_root = {
			# all of it, at the parity, as far as the issuer's metal goes
			set_variable = { name = stockpiling_overrated_quantity_var value = stockpiling_eco_ariary_state }
			set_variable = {
				name = zz_ef_crisis_due
				value = {
					value = var:stockpiling_overrated_quantity_var
					multiply = scope:overrated_currency_country_owner.money_value_target_real
					min = 0
				}
			}
			zz_ef_crisis_redeem = yes
			if = {
				limit = {
					has_variable = stockpiling_eco_ariary_c_in_reserve_currency
					var:stockpiling_eco_ariary_c_in_reserve_currency > 0
				}
				set_variable = { name = stockpiling_eco_ariary_c_in_reserve_currency value = 0 }
			}
			scope:central_bank_site_root = {
				if = { limit = { has_variable = stockpiling_eco_ariary_state_1 } set_variable = { name = stockpiling_eco_ariary_state_1 value = 0 } }
				if = { limit = { has_variable = stockpiling_eco_ariary_reserve_currency_state_1 } set_variable = { name = stockpiling_eco_ariary_reserve_currency_state_1 value = 0 } }
			}
		}
		scope:overrated_currency_country_owner = {
			reset_debt_in_national_currency = yes
		}
	}
}
