	si_sort_by_country_indice = {
		scope = country
		saved_scopes = {
			player_country
		}

		is_shown = {
			always = yes
		}

		is_valid = {
			always = yes
		}

		effect = {
			scope:player_country = {
				clear_variable_list = list_before_order
				every_in_list = {
					variable = country_index_variable_list_ordered_1
					scope:player_country = {
						add_to_variable_list = {
							name = list_before_order
							target = prev
						}
					}
				}
				clear_variable_list = country_index_variable_list_ordered_1
				if = {
					limit = {NOT = { has_variable = fi_sort_by_country_rank } }
					ordered_in_list = {
						variable = list_before_order
						order_by = country_indice
						max = 1000
						check_range_bounds = no
						scope:player_country = {
							add_to_variable_list = {
								name = country_index_variable_list_ordered_1
								target = prev
							}
						}
					}
					set_variable = fi_sort_by_country_rank
				}
				else = {
					ordered_in_list = {
						variable = list_before_order
						order_by = { value = 0 subtract = country_indice}
						max = 1000
						check_range_bounds = no
						scope:player_country = {
							add_to_variable_list = {
								name = country_index_variable_list_ordered_1
								target = prev
							}
						}
					}
					remove_variable = fi_sort_by_country_rank
				}
				clear_variable_list = list_before_order
			}
		}
