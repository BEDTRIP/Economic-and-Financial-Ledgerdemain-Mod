				# Cas 1 : le pays n'a NI bimetallisme NI silver standard → passe toujours
				AND = {
					NOT = { has_law = law_type:law_bimetallism_standard }
					NOT = { has_law = law_type:law_silver_standard }
				}

				# Cas 2 : le pays a l'un des deux standards → passe seulement avant 1873
				AND = {
					OR = {
						has_law = law_type:law_bimetallism_standard
						has_law = law_type:law_silver_standard
					}
					game_date < 1873.1.1
				}
			}
		}
		set_variable = rise_base_rate
		privat_bank_buy_currency = yes
