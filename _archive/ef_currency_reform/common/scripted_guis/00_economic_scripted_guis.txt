currency_introduction = {
	is_valid = {
		has_modifier = extreme_weak_currency #çà veux dire çà money_value_in_gold < 0.01
		#on exclu les pays qui sont deja entré dans la bankruptcy :
		has_central_bank_FS_NISO = yes #un  pays  en  fiat  standart  à cause  d'une première  crise ne refera pas de crise en boucle, il sera bloqué en fiat  standart
		not = {
			or = {
				#has_modifier = economic_instability_country # модификатора нет
				has_modifier = currency_crisis_country
				has_modifier = central_bank_bankruptcy_country
				has_modifier = financial_crash_country
				has_modifier = economic_crisis_country
				#has_modifier = declared_bankruptcy
			}
		}
		#var:extreme_weak_currency_solution_count_player >= 3 # à 2ans tout pil donc un peu bof bof mais à voir
		extreme_weak_currency_solution_count_sc >= 3
	}
	effect = {
		root = {
			save_scope_as = target_country
		}
		currency_law_list = yes

		#Reset de la balance une nouvelle fois pour etre sure de tout clean
		set_variable = {
			name = central_bank_bankruptcy
			value = 1
		}
		every_country = {
			if = {
				limit  = {
					has_modifier = has_central_bank
					not = {
						this = root
					}
				}
				save_scope_as = seller
				all_currency_resold = yes
			}
		}
		reset_law_event_currency = yes
		reset_debt_currency_reserve_and_export_value = yes #uniquement pour le joueur
		set_variable = {
			name = central_bank_bankruptcy
			value = 0
		}
		##################################

		activate_law = law_type:law_no_market_liquidity
		introduction_new_currency = yes
		activate_law = law_type:law_silver_standard
		set_variable = {
			name = extreme_weak_currency_solution_count
			value = 0
		}
		change_variable = {
			name = new_currency_count
			add = 1 #0
		}
		if = { limit = { has_modifier = new_currency } remove_modifier = new_currency }
		add_modifier = {
			name = new_currency
			months = 60
		}
		money_value_global_var = yes
 		reset_balance = yes
	}
}
