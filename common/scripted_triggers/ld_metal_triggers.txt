# The metal of the standard (Д.R8а.6). A metal standard and an exchange one -- by the law: gold, silver, bimetallism
# and their exchange standards (gold / silver / bimetallic exchange). The external exchange standard (a subject pegged
# to another's currency) -- the metal of its anchor's law (var:zz_ef_anchor, zz_ef_anchor_update, ld_anchor.txt); an
# anchor on fiat -- no metal. The anchor is never itself on the external standard (zz_ef_anchor_update takes the
# anchor's anchor), so the _law triggers below are read one level deep.
zz_ef_metal_gold_law = {
	OR = {
		has_law = law_type:law_gold_standard
		has_law = law_type:law_gold_exchange_standard
	}
}
zz_ef_metal_silver_law = {
	OR = {
		has_law = law_type:law_silver_standard
		has_law = law_type:law_silver_exchange_standard
	}
}
zz_ef_metal_bimet_law = {
	OR = {
		has_law = law_type:law_bimetallism_standard
		has_law = law_type:law_bimetallism_exchange_standard
	}
}
# the external exchange standard with an anchor
zz_ef_anchored = {
	has_law = law_type:law_external_exchange_standard
	has_variable = zz_ef_anchor
}
zz_ef_metal_gold = {
	OR = {
		zz_ef_metal_gold_law = yes
		AND = { zz_ef_anchored = yes var:zz_ef_anchor = { zz_ef_metal_gold_law = yes } }
	}
}
zz_ef_metal_silver = {
	OR = {
		zz_ef_metal_silver_law = yes
		AND = { zz_ef_anchored = yes var:zz_ef_anchor = { zz_ef_metal_silver_law = yes } }
	}
}
zz_ef_metal_bimet = {
	OR = {
		zz_ef_metal_bimet_law = yes
		AND = { zz_ef_anchored = yes var:zz_ef_anchor = { zz_ef_metal_bimet_law = yes } }
	}
}
# an exchange standard of its own (gold / silver / bimetallic exchange): E&F's one exchange standard, its value by the
# reference currency (money_value_0)
zz_ef_exchange_std = {
	OR = {
		has_law = law_type:law_gold_exchange_standard
		has_law = law_type:law_silver_exchange_standard
		has_law = law_type:law_bimetallism_exchange_standard
	}
}

# a CB on a standard backed by metal (metal, exchange, external on a metal anchor) -- its metal is the cover of the money
# (scripted_effects/ld_metal_accounts.txt; Д.R8а.6, В-R8а.1)
zz_ef_cb_metal_standard = {
	has_modifier = has_central_bank
	OR = {
		zz_ef_metal_gold = yes
		zz_ef_metal_silver = yes
		zz_ef_metal_bimet = yes
	}
}
