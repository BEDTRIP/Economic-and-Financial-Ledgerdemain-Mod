	has_modifier = has_central_bank
	has_modifier = global_monetary_reference
	money_value_in_gold > global_var:money_value_median
}
is_reference_currency_no = {
	has_modifier = has_central_bank
	has_modifier = global_monetary_reference
	money_value_in_gold < global_var:money_value_median
}
is_strong_currency = {
	has_modifier = has_central_bank
	not = {
		has_modifier = global_monetary_reference
		money_value_in_gold < 0.01
	}
	zz_ef_currency_strength > 1.25
	var:central_bank_bankruptcy = 0
}
is_balanced_currency = {
	has_modifier = has_central_bank
	not = {
		has_modifier = global_monetary_reference
		money_value_in_gold < 0.01
		zz_ef_currency_strength > 1.25
	}
	zz_ef_currency_strength <= 1.25
	zz_ef_currency_strength >= 0.75
	var:central_bank_bankruptcy = 0
}
is_weak_currency = {
	has_modifier = has_central_bank
	not = {
		has_modifier = global_monetary_reference
		money_value_in_gold < 0.01
		zz_ef_currency_strength >= 0.75
	}
	zz_ef_currency_strength < 0.75
	money_value_in_gold > 0.01
	var:central_bank_bankruptcy = 0
}
is_extreme_weak_currency = {
	has_modifier = has_central_bank
	not = {
		has_modifier = global_monetary_reference
		is_subject = yes
	}
	OR = {
		money_value_in_gold < 0.01
		var:central_bank_bankruptcy >= 1
	}
}
