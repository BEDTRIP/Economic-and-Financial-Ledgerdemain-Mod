# R0.2 (7.10): the EF* debug logs (debug_log "EFW|…", "EFX|…" …) are written only when this holds -- the game
# rule "Ledgerdemain: debug logs" (game_rules/ld_game_rules.txt, off by default) or the global variable
# zz_ef_logs_on, set from the console by the hidden event zz_ef_logs.1 (events/ld_logs_events.txt; the sandbox
# run script sends "event zz_ef_logs.1").
zz_ef_logs_on = {
	OR = {
		has_game_rule = ld_debug_logs_on
		has_global_variable = zz_ef_logs_on
	}
}

# Country scope: the countries the per-country logs follow -- the player and the big economies (GDP over 20M).
zz_ef_log_country = {
	OR = {
		is_player = yes
		gdp > 20000000
	}
}

# Country scope: the CB's metal moved this week without a known posting (var:zz_ef_f_mt_oth_g / _s, ld_metal_accounts.txt)
# by more than the log's noise: 1 000 gold units or 10 000 silver units either way (EFQ).
zz_ef_metal_rest_logged = {
	OR = {
		var:zz_ef_f_mt_oth_g > 1000
		var:zz_ef_f_mt_oth_g < -1000
		var:zz_ef_f_mt_oth_s > 10000
		var:zz_ef_f_mt_oth_s < -10000
	}
}
