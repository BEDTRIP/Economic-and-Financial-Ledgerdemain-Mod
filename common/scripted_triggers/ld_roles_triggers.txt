# the country's role in the money model -- var:zz_ef_role 1 = A (full), 2 = B (simplified),
# 3 = V (decentralized: outside the money model). Set by the monthly world pass (scripted_effects/ld_roles.txt).
# No role yet (the first days of a game) -- neither A nor B nor V: the callers treat it as before (full step).

zz_ef_role_is_a = {
	has_variable = zz_ef_role
	var:zz_ef_role = 1
}

zz_ef_role_is_b = {
	has_variable = zz_ef_role
	var:zz_ef_role = 2
}

zz_ef_role_is_v = {
	has_variable = zz_ef_role
	var:zz_ef_role = 3
}

# Category A (Д.R8.1): the player, the great and the major powers, a share of the world's GDP or trade at least the
# threshold (the world's sums -- the last yearly pass, ld_roles.txt).
zz_ef_role_significant = {
	OR = {
		is_player = yes
		country_rank >= rank_value:major_power
		zz_ef_role_gdp_share >= zz_ef_role_a_gdp_share
		zz_ef_role_trade_share >= zz_ef_role_a_trade_share
	}
}
