# The exchange's trading (scripted_effects/ld_exchange_trade.txt; понятия «Биржа», «Решение держателя»).

# who buys what (the bonds: the bond register serves the treasury's and the banks' parts; the treasury buys them in
# E&F's slots)
zz_ef_xch_buys_cb_m = { always = yes }
zz_ef_xch_buys_bk_m = { always = yes }
zz_ef_xch_buys_pp_m = { always = yes }
zz_ef_xch_buys_bk_b = { always = yes }
zz_ef_xch_buys_tr_b = { always = no }
# a CB on a metal-backed standard with its cover under the floor of redemption (25 %) sells foreign money that is not in
# its cover for metal (Д.R8б.26)
zz_ef_xch_dump_cb_m = {
	zz_ef_cb_metal_standard = yes
	var:zz_ef_xe_rcb > 0
	zz_ef_cb_cover < zz_ef_value_floor_cover
}
zz_ef_xch_dump_bk_m = { always = no }
zz_ef_xch_dump_pp_m = { always = no }
zz_ef_xch_dump_bk_b = { always = no }
zz_ef_xch_dump_tr_b = { always = no }

# Holder scope, scope:zz_ef_xch_d: the debtor is of our metal group (Д.R8б.15) -- gold, silver, bimetallic (an exchange
# standard -- its metal's group, the external one -- its anchor's), or neither (fiat, none): its money is cheaper to
# leave.
zz_ef_xch_same_metal = {
	OR = {
		AND = {
			zz_ef_metal_gold = yes
			scope:zz_ef_xch_d = { zz_ef_metal_gold = yes }
		}
		AND = {
			zz_ef_metal_silver = yes
			scope:zz_ef_xch_d = { zz_ef_metal_silver = yes }
		}
		AND = {
			zz_ef_metal_bimet = yes
			scope:zz_ef_xch_d = { zz_ef_metal_bimet = yes }
		}
		AND = {
			NOT = { zz_ef_metal_gold = yes }
			NOT = { zz_ef_metal_silver = yes }
			NOT = { zz_ef_metal_bimet = yes }
			scope:zz_ef_xch_d = {
				NOT = { zz_ef_metal_gold = yes }
				NOT = { zz_ef_metal_silver = yes }
				NOT = { zz_ef_metal_bimet = yes }
			}
		}
	}
}
# Player scope: the window of the exchange shows money (not bonds)
zz_ef_xwin_kind_is_money = {
	OR = {
		NOT = { has_variable = zz_ef_xpl_kind }
		var:zz_ef_xpl_kind = 0
	}
}
# Country scope: the country puts orders in this week -- A every week, B every 4th week (its phase, as its GUI bridge,
# ld_money_model.txt)
zz_ef_xch_posts_this_week = {
	OR = {
		NOT = { zz_ef_role_is_b = yes }
		NOT = { has_variable = zz_ef_week_phase }
		var:zz_ef_week_phase = zz_ef_sched_week_mod4
	}
}
# Country scope: its week of the 4 (its phase): the fund's capital re-valued
zz_ef_xch_phase_week = {
	OR = {
		NOT = { has_variable = zz_ef_week_phase }
		var:zz_ef_week_phase = zz_ef_sched_week_mod4
	}
}
