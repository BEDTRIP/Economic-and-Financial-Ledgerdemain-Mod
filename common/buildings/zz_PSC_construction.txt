
REPLACE:building_construction_sector = {
	building_group = bg_construction
	icon = "gfx/interface/icons/building_icons/construction_camp.dds"
	city_type = city
	levels_per_mesh = 5
	
	# while the country bans private construction sectors
	# (ld_pb_css_private_ban_buttons.txt) the AI government does not
	# privatize its own. A script value here is per the engine docs
	# (buildings.md: "scope is country the building is located in"); nobody
	# in the set uses the block form yet -- if debug.log complains about this
	# line, fall back to the plain 0.25.
	ai_nationalization_desire = {
		value = 0.25
		if = {
			limit = { has_variable = zz_pb_ef_css_private_ban }
			value = 10
		}
	}

	unlocking_technologies = {
		urbanization
	}
	production_method_groups = {
		pmg_base_building_construction_sector
		# E&F: currency-as-liquidity layer (selected via laws, PMs defined in E&F)
		pmg_market_liquidity
	}
	
	# Credit: KuromiAK for this construction sector weighting
	# Construction sector should generally be preferred over other equal-priority spending items
	ai_value = {
		value = 2500
		
		# Prefer states with iron
		if = {
			limit = {
				has_potential_resource = building_iron_mine
			}
			add = 500
		}

		# Kickstart: if this is the capital state and it has no construction sector (and none is being built), strongly prefer building one.
		if = {
			limit = {
				is_capital = yes
				NOR = {
					has_building = building_construction_sector
					any_scope_building = {
						is_building_type = building_construction_sector
						is_under_construction = yes
					}
				}
			}
			add = 100000
		}

		# If construction goods are overpriced on the market more than the country's chosen construction spending level,
		# give AI extra incentive to build construction sectors (they produce construction goods in PSC chain).
		add = ai_construction_sector_price_pressure_bonus

		# Encourage first sector
		if = {
			limit = {
				NOT = {
					has_building = building_construction_sector
				}
			}
			add = 1000
			add = {
				value = gdp
				divide = 3000
			}
		}

		# `exists = owner` first: ai_value is evaluated for unowned states too, and
		# `owner` comes back as none there ("Value of wrong type … Got value of type 'none'").
		# The same base_rate_percentage race as the ZZ FIX below: a country without
		# the variable yet gets no AI penalty.
		# at or over the limit the AI does not
		# build sectors at all; it also takes the
		# excess down (on_actions/ld_pb_ai_sector_downsize.txt).
		if = {
			limit = {
				exists = owner
				owner = { has_variable = base_rate_percentage }
				owner.building_ef_private_construction_lvl >= owner.building_urban_center_lvl_by_base_rate
			}
			value = 0
		}
	}

	required_construction = construction_cost_very_low
	ownership_type = self

	# Limit total Construction Sectors by the (E&F) Urban Center-based cap.
	# NOTE: `can_build_private` is evaluated in a non-building scope, so `level` is invalid here.
	# We rely on E&F script_values; in pb+ef we remap `building_ef_private_construction_lvl` to sum `building_construction_sector`.
	#
	# `owner ?= { }` rather than `owner.x <= owner.y`: this is also evaluated for
	# states with no owner, where the plain form logs "Value of wrong type … 'none'".
	# The safe form skips the check there, which is the right answer anyway --
	# nobody is privately funding a building in a state nobody owns.
	#
	# ZZ FIX: building_urban_center_lvl_by_base_rate divides through E&F's
	# var:base_rate_percentage, a COUNTRY variable E&F only backfills on a country's
	# first monthly pulse (new_country_var_ef, called from ef_on_monthly_pulse_recurence).
	# A state whose owner is younger than that first pulse (freshly released/formed/
	# independent) has no such variable yet, so building_urban_center_lvl_by_base_rate
	# itself comes back none and the comparison below would throw "Value of wrong
	# type … 'none'" on every evaluation. has_variable gates on readiness; the un-adjusted PSC
	# cap (building_urban_center_lvl) covers a country that hasn't been backfilled yet.
	can_build_private = {
		owner ?= {
			# the country banned private construction sectors.
			NOT = { has_variable = zz_pb_ef_css_private_ban }
			# a trigger has no if/else here ("Unknown trigger type: if", error.log) -- the same choice with OR / AND
			OR = {
				AND = {
					has_variable = base_rate_percentage
					building_ef_private_construction_lvl <= building_urban_center_lvl_by_base_rate
				}
				AND = {
					NOT = { has_variable = base_rate_percentage }
					building_ef_private_construction_lvl <= building_urban_center_lvl
				}
			}
		}
	}

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_light_industry.dds"
}
