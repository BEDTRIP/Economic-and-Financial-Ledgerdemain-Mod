#############################################
# A new country is set up the moment it exists (zz_ef_country_init, scripted_effects/ld_country_init.txt), not at
# its first monthly pulse: E&F's and our scripts read its variables before that.
#
# On on_country_formed / on_become_independent root is the new country; the four on_country_released_as_* fire with
# root -- the overlord and scope:target -- the new country. zz_ef_country_init works in any of them.
#
# Each vanilla on_action only lists our own on_action, which carries the effect: an on_action holds one effect, and
# the most recent definition wins (debug.log: "There is more than one 'effect' defined") -- an effect of ours put
# straight into the vanilla on_action would replace vanilla's and other mods'.
#############################################

on_country_formed = {
	on_actions = { zz_ef_newcountry_on_country_formed }
}

zz_ef_newcountry_on_country_formed = {
	effect = {
		zz_ef_country_init = yes
		# (Д.R3.7) the historical currency of the new tag -- an event with a choice
		# (events/ld_currency_formed_events.txt)
		if = {
			limit = { has_modifier = has_central_bank }
			zz_ef_cf_find = yes
			if = {
				limit = { has_variable = zz_ef_cf_target }
				trigger_event = { id = ld_currency_formed.1 days = 1 }
			}
		}
	}
}

on_become_independent = {
	on_actions = { zz_ef_newcountry_on_become_independent }
}

zz_ef_newcountry_on_become_independent = {
	effect = {
		zz_ef_country_init = yes
	}
}

on_country_released_as_independent = {
	on_actions = { zz_ef_newcountry_on_country_released_as_independent }
}

zz_ef_newcountry_on_country_released_as_independent = {
	effect = {
		scope:target = { zz_ef_country_init = yes }
	}
}
