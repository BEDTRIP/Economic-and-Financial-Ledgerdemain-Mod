#############################################
# EF.39 / EF.48 (2026-09-30) -- the money model's steps
# (scripted_effects/zz_ef_money_model.txt). Every country.
# Monthly: our own on_action listed under on_monthly_pulse_country (additive);
# it also (re)starts the weekly chain. Weekly: a self-renewing chain, the
# engine has no weekly country pulse.
#############################################

on_monthly_pulse_country = {
	on_actions = {
		zz_ef_money_model_monthly
	}
}

zz_ef_money_model_monthly = {
	# R1а.3 (Д.2): role V (decentralized) is outside the money model
	trigger = {
		NOT = { zz_ef_role_is_v = yes }
	}
	effect = {
		zz_ef_money_model_monthly_step = yes
		zz_ef_money_week_start = yes
	}
}

# Daily until the engine's budget tick, then the chain below.
zz_ef_money_week_probe = {
	effect = {
		zz_ef_money_week_probe_step = yes
	}
}

# The week's day of each country: var:zz_ef_week_slot 0..6 (zz_ef_money_week_slot_assign), so the countries'
# steps and GUI-bridge receivers spread over the 7 days instead of all landing on the budget tick's day.
zz_ef_money_model_weekly = {
	effect = {
		# R1а.3 (Д.2): a country that became V stops its chain (and with it the GUI bridge)
		if = {
			limit = { zz_ef_role_is_v = yes }
			if = { limit = { has_variable = zz_ef_week_alive } remove_variable = zz_ef_week_alive }
			if = { limit = { has_variable = zz_ef_week_slot } remove_variable = zz_ef_week_slot }
		}
		else_if = {
			limit = { NOT = { has_variable = zz_ef_week_slot } }
			zz_ef_money_week_slot_assign = yes
		}
		else = {
			zz_ef_money_model_step = yes
			trigger_event = { on_action = zz_ef_money_model_weekly days = 7 }
		}
	}
}
