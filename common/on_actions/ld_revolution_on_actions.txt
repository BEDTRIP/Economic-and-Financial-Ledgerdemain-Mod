# R3 (the user, 8.10): a won revolution -- the winner takes the same monetary system (План проекта, «Денежная система
# при победе революции»). Step 0: a probe -- what exists when (EFV log).
on_revolution_end = {
	on_actions = { zz_ef_rv_on_revolution_end }
}
zz_ef_rv_on_revolution_end = {
	effect = {
		scope:target ?= { set_variable = { name = zz_ef_rv_from value = root } }
		if = { limit = { zz_ef_logs_on = yes } debug_log = "EFV|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|rev_end|root_rev [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_rv_is_rev_v')|0]|target [SCOPE.sC('target').GetNameNoFormatting]|target_rev [SCOPE.sC('target').MakeScope.ScriptValue('zz_ef_rv_is_rev_v')|0]" }
	}
}

	on_actions = { zz_ef_rv_on_revolution_start }
}

		# the new country is set up at once: its pulses come before E&F's monthly hub (scripted_effects/ld_country_init.txt)

		if = { limit = { zz_ef_logs_on = yes } debug_log = "EFV|[TimeKeeper.GetCurrentDate.GetString]||rev_start|target [SCOPE.sC('target').GetNameNoFormatting]|target_rev [SCOPE.sC('target').MakeScope.ScriptValue('zz_ef_rv_is_rev_v')|0]|target_std [SCOPE.sC('target').MakeScope.ScriptValue('zz_ef_std_code')|0]|target_cb [SCOPE.sC('target').MakeScope.ScriptValue('zz_ef_has_cb')|0]" }
	}
