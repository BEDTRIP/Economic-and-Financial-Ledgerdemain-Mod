#############################################
# EF.24 + EF.25, EF.28 -- monthly, per country:
#   0. the share of literate rich pops and the stock issuance cut (EF.28);
#   1. capitalization averages and the stock index
#      (scripted_effects/ld_capitalization_average.txt), then the
#      12 monthly marks and the crash check (ld_capitalization_crash.txt);
#   2. mass shareholding, which reads the updated averages
#      (script_values/ld_mass_shareholding_values.txt).
# One on_action, so the order is guaranteed.
#
# Additive: our own on_action listed under on_monthly_pulse_country, never a
# bare `effect` in the vanilla one (an on_action holds one effect and the
# last loaded wins -- rules, section 1).
#
# The ratio is stored in a variable first and the modifier reads the
# variable: the multiplier of add_modifier is not evaluated in the scope of
# the effect (rules, section 2).
#############################################

# run by the scheduler (scripted_effects/ld_scheduler.txt, zz_ef_sched_monthly), not on on_monthly_pulse_country.

zz_ef_capitalization_monthly = {
	effect = {
		### stock issuance by the share of literate rich pops
		### (scripted_effects/ld_stock_issue_literacy.txt).
		zz_ef_stock_issue_update = yes
		### listed companies -- the "publicly traded" HQ method by script (ld_listing.txt)
		zz_ef_listing_update = yes
		zz_ef_cap_monthly_update = yes
		zz_ef_listing_log = yes
		zz_ef_cap_monthly_crash_check = yes

		### EF.25
		if = { limit = { has_modifier = zz_ef_mass_shareholding } remove_modifier = zz_ef_mass_shareholding }
		set_variable = {
			name = zz_ef_msh_points
			value = zz_ef_mass_shareholding_points
		}
		if = {
			limit = { var:zz_ef_msh_points > 0 }
			add_modifier = {
				name = zz_ef_mass_shareholding
				multiplier = var:zz_ef_msh_points
			}
		}
	}
}
