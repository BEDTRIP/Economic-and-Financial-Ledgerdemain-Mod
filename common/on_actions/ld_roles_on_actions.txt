# R1а.3 (Д.2, Д.3): the countries' roles, once a month for the whole world (scripted_effects/ld_roles.txt);
# R1а.4: the scheduler's month (its number, the day of the month) and its restart if lost (ld_scheduler.txt).
on_monthly_pulse = {
	on_actions = {
		zz_ef_roles_monthly
	}
}

zz_ef_roles_monthly = {
	effect = {
		if = {
			limit = { has_global_variable = zz_ef_month_n }
			change_global_variable = { name = zz_ef_month_n add = 1 }
		}
		else = {
			set_global_variable = { name = zz_ef_month_n value = 1 }
		}
		set_global_variable = { name = zz_ef_dom value = 0 }
		zz_ef_roles_world_pass = yes
		zz_ef_sched_ensure = yes
	}
}
