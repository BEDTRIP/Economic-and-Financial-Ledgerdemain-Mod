law_bimetallism_standard = {
	group = lawgroup_monetary_system
	institution = institution_economic_central_bank

	icon = "gfx/interface/icons/law_icons/bimetallism_standard.dds"

	unlocking_technologies = {
		metalique_standard
	}
	# also from an exchange standard (Д.R8а.6)
	unlocking_laws = {
		law_silver_standard
		law_gold_standard
		law_gold_exchange_standard
		law_silver_exchange_standard
		law_bimetallism_exchange_standard
	}
	
	can_enact = {
		# a follower of a currency union keeps its head's (Д.R8в.19)
		zz_ef_mu_own_laws = yes
		NOT = { 
			has_law = law_type:law_isolationism
			zz_ef_monetary_crisis_now = yes
		}
	} 

	on_activate = {
		# the legal ratio gold : silver (Д.R8а.11) -- the country's own if it had one, else 15
		if = {
			limit = { NOT = { has_variable = zz_ef_bimet_ratio } }
			set_variable = { name = zz_ef_bimet_ratio value = 15 }
		}
		on_activate_monetary_system_law = yes  

		every_country = { 
			limit = { 
				this = root
				has_modifier = has_central_bank
				Not = {
					or = {
						market_owner_is_root_univ_with_buiding = yes
						is_subject_custom_trigger = yes
					}
				} 
			} 
			capital= {
				save_scope_as = capital_state
			} 
			save_scope_as = target_country

			every_country = {
				limit = {
					is_player = yes  
				}   
				trigger_event = 00_ef_economic_event.26
			}    
			change_global_variable = {
				name = global_silver_production_cumulative_production
				add = {
					value = global_silver_production
					multiply = 0.03
				}
			} 
		}   
	}

	progressiveness = 50
	 
	ai_will_do = {
		always = no
	}
}
