# (Д.R3.2, Д.R4а.3) the central bank group (law_groups/ld_central_bank.txt): the law founds and closes the CB
# (scripted_effects/ld_cb_law.txt) -- the state and the private one with the technology central_banking, none -- always.
# The kinds behave apart (Д.R4а.5): the CB's profit, its loan to the treasury (zz_ef_cb_loan_cap), the rate rule's
# weights (zz_ef_cb_kind_w_value). The AI founds a state CB once it has the technology, a private one -- the player.
law_ld_cb_state = {
	group = lawgroup_ld_central_bank
	icon = "gfx/interface/icons/law_icons/gold_standard.dds"
	progressiveness = 0
	on_activate = {
		zz_ef_cb_found = yes
	}
	unlocking_technologies = { central_banking }
	can_enact = {
		NOT = { has_law = law_type:law_no_monetary_system }
	}
	ai_will_do = {
		has_technology_researched = central_banking
	}
	ai_enact_weight_modifier = {
		value = zz_ef_cb_law_ai_weight
	}
}
law_ld_cb_private = {
	group = lawgroup_ld_central_bank
	icon = "gfx/interface/icons/law_icons/large_monetary_policy.dds"
	progressiveness = 0
	on_activate = {
		zz_ef_cb_found = yes
	}
	unlocking_technologies = { central_banking }
	can_enact = {
		NOT = { has_law = law_type:law_no_monetary_system }
	}
	ai_will_do = { always = no }
}
law_ld_cb_none = {
	group = lawgroup_ld_central_bank
	icon = "gfx/interface/icons/law_icons/no_monetary_system.dds"
	progressiveness = 0
	on_activate = {
		zz_ef_cb_close = yes
	}
	ai_will_do = { always = no }
}
