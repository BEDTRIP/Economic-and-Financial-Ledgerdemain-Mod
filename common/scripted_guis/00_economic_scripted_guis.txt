transfert_loot_state = {
	is_shown = {
		or = {
			and = {
				owner = player
				#var:financial_center_site_var = 1
				or = {
					has_modifier = central_bank_place
					has_modifier = central_bank_historic_place
				}
				is_capital = no
			}
			and = {
				has_modifier = central_bank_historic_place
				owner = {
					is_at_war = yes
					has_war_with = player
				}
				controller = player
			}
		}
		not = {
			has_modifier = looted_state
		}
	}
}

looting_state_button = { #enemy_capital_is_occuped_gui
	is_valid = {
		save_temporary_scope_as = occupied_capital
		owner = {
			is_at_war = yes
			has_war_with = player
		}
		player = {
			is_at_war = yes
			var:looting_1_year = 0
		}
		not  = {
			controller = owner #root
		}
	}
	effect = {
		every_country = {
			if  = {
				limit = {
					is_player = yes
				}
				set_variable  = {
					name = looting_1_year
					value = 1
				}
				every_scope_state = {
					limit = {
						has_modifier = central_bank_historic_place
						is_capital = yes
					}
					save_scope_as = central_bank_site_capital
				}
			}
		}
		save_scope_as = occupied_capital
		enemy_capital_is_occuped = yes
	}
}

modifier_inflation = {
	effect = {
		if = {
			limit = {
				inflation_value > 0.02
			}
			add_modifier = {
				name = inflation_country
				multiplier = inflation_value
			}
		}
		else_if = {
			limit = {
				inflation_value < -0.02
			}
			add_modifier = {
				name = deflation_country
				multiplier = inflation_value
			}
		}
	}
}

devaluation_rate_increase = {
	is_shown = {
		#var:money_value_target <= 10
	}
	effect = {
		if = {
			limit = {
				has_variable = devaluation_rate
				var:devaluation_rate < 200 #500
			}
			change_variable = {
				name = devaluation_rate
				add = 5
			}
			if = {
				limit = {
					has_variable = devaluation_rate
					var:devaluation_rate > 65
					var:devaluation_rate <= 70

				}
				set_variable = {
					name = devaluation_time_1
					value = 5
				}
			}
			if = {
				limit = {
					has_variable = devaluation_rate
					var:devaluation_rate > 110
					var:devaluation_rate <= 115

				}
				set_variable = {
					name = devaluation_time_1
					value = 7.5
				}
			}
			if = {
				limit = {
					has_variable = devaluation_rate
					var:devaluation_rate > 155
					var:devaluation_rate <= 160

				}
				set_variable = {
					name = devaluation_time_1
					value = 10
				}
			}
		}
	}
}

set_money_value_target = {
	is_valid = {
		not = {
			has_modifier = foreign_exchange_controls
		}
	}
	effect = {
		set_variable = {
			name = money_value_target_1_old
			value = var:money_value_target_1
		}
		set_variable = {
			name = money_value_target_1
			value = var:money_value_target_2
		}
		if = { limit = { has_modifier = foreign_exchange_controls } remove_modifier = foreign_exchange_controls }
		add_modifier = {
			name = foreign_exchange_controls
			months = 12
		}
		currency_strength_modifier = yes

		#arbitrage
		if = {
			limit = {
				var:money_value_target_1_old > var:money_value_target_1
			}
			if = { limit = { has_modifier = devaluation_currency_target } remove_modifier = devaluation_currency_target }
			add_modifier = {
				name = devaluation_currency_target
				months = 12
			}
		}
		if = {
			limit = {
				var:money_value_target_1_old < var:money_value_target_1
			}
			if = { limit = { has_modifier = revaluation_currency_target } remove_modifier = revaluation_currency_target }
			add_modifier = {
				name = revaluation_currency_target
				months = 12
			}
		}
	}
}

transfer_gold_to_central_bank_metal_reserves_increase = {
	is_shown = {
	}

	effect = {
		if = {
			limit = {
				has_variable = transfer_gold_to_central_bank_metal_reserves
			}
			change_variable = {
				name = transfer_gold_to_central_bank_metal_reserves
				add = 10000000
			}
		}
	}
}
transfer_gold_to_central_bank_metal_reserves_reduce = {
	is_shown = {
	}
	is_valid = {
		var:transfer_gold_to_central_bank_metal_reserves > 10000000
	}
	effect = {
		if = {
			limit = {
				has_variable = transfer_gold_to_central_bank_metal_reserves
				var:transfer_gold_to_central_bank_metal_reserves > 10000000
			}
			change_variable = {
				name = transfer_gold_to_central_bank_metal_reserves
				subtract = 10000000
			}
		}
	}
}

transfer_gold_to_central_bank_metal_reserves = {
	effect = {
		add_treasury = {
			value = 0
			subtract = var:transfer_gold_to_central_bank_metal_reserves
		}

		if = {
			limit = {
				or = {
					has_law = law_type:law_silver_standard
					has_law = law_type:law_bimetallism_standard
				}
			}
			every_scope_state = {
				if = {
					limit = {
						has_modifier = central_bank_historic_place
						is_capital = yes
					}
					change_variable = {
						name = silver_state_1
						add = root.transfer_gold_to_central_bank_metal_reserves_sc
					}

				}
			}
		}
		if = {
			limit = {
				or = {
					has_law = law_type:law_gold_standard
					has_law = law_type:law_gold_exchange_standard
				}
			}
			every_scope_state = {
				if = {
					limit = {
						has_modifier = central_bank_historic_place
						is_capital = yes
					}
					change_variable = {
						name = gold_state_1
						add = root.transfer_gold_to_central_bank_metal_reserves_sc
					}

				}
			}
		}


	}
}

################################

################################

transfer_gold_to_central_bank_metal_reserves_visible = {
	is_valid = {
		gold_reserves_var > var:transfer_gold_to_central_bank_metal_reserves
	}
}
