# The game in debug mode turns the EF* logs on (gui/ld_logs_hook.gui; the console event zz_ef_logs.1 does the same),
# unless the console event zz_ef_logs.2 switched them off for this game.
zz_ef_logs_debug_sg = {
	scope = country
	effect = {
		if = {
			limit = {
				NOT = { has_global_variable = zz_ef_logs_on }
				NOT = { has_global_variable = zz_ef_logs_off }
			}
			set_global_variable = zz_ef_logs_on
		}
	}
}
