		# Д.R4б.15: a window of 1 year(s) -- the base at the count 1, the change at the count 3 (half-yearly calls),
		# the next window's base the same value; a country's first call is skipped (a new country's value may not be
		# set yet), its change counts once a window is full (var:zz_ef_mv_ok_1)
		if = {
			limit = { NOT = { has_variable = zz_ef_mv_seen_1 } }
			set_variable = zz_ef_mv_seen_1
		}
		else = {
			change_variable = {
				name = money_value_fixe_var_1_1Y
				add = 1
			}

			if = {
				limit = {
					var:money_value_fixe_var_1_1Y = 1
				}
				set_variable = {
					name = money_value_var_half_1_year
					value = money_value_rapported_inflation
				}
			}
			if = {
				limit = {
					var:money_value_fixe_var_1_1Y = 3
				}
				set_variable = {
					name = money_value_var_1_year
					value = money_value_rapported_inflation
				}
				set_variable = {
					name = money_value_dif_1
					value = 0
				}
				change_variable = {
					name = money_value_dif_1
					add = var:money_value_var_1_year
				}
				change_variable = { #test
					name = money_value_dif_1
					subtract = var:money_value_var_half_1_year
				}
				change_variable = {
					name = money_value_dif_1
					divide = var:money_value_var_half_1_year
				}

				# the next window starts here: its base -- this value
				set_variable = { name = money_value_fixe_var_1_1Y value = 1 }
				set_variable = { name = money_value_var_half_1_year value = var:money_value_var_1_year }
				set_variable = zz_ef_mv_ok_1

						}
					}

# currency are zeroed once: no reserve currency from exports and no import debt any more (the monthly

money_value_target_pre_set = {
	# a new country without E&F's variables yet ("Value of wrong type... none" in money_value_*_standard: they read
	# var:law_* of E&F) -- a revolutionary (r1008_094935: 653 errors; new_country_var_ef runs at once only for formed and
	# released countries, ld_new_country_immediate_init.txt), Peru-Bolivia; the target stays as it was
	if = {
		limit = {
			has_variable = zz_ef_country_vars_set
			money_supply > 0
		}
		# Si standard argent
		if = {
			limit = {
				#has_law = law_type:law_silver_standard
				var:law_silver_standard ?= 1
			}
			#bimetalique
			set_variable = {
				name = money_value_target_1_bimetallic
				value = money_value_bimetallism_standard
			}
			#gold
			set_variable = {
				name = money_value_target_1_gold
				value = money_value_gold_standard
			}
		}
		# Si standard bimétallique
		if = {
			limit = {
				#has_law = law_type:law_bimetallism_standard
				var:law_bimetallism_standard ?= 1
			}
			#silver
			set_variable = {
				name = money_value_target_1_silver
				value = money_value_silver_standard
			}
			#gold
			set_variable = {
				name = money_value_target_1_gold
				value = money_value_gold_standard
			}
		}
		# Si standard or
		if = {
			limit = {
				#has_law = law_type:law_gold_standard
				var:law_gold_standard ?= 1
			}
			#silver
			set_variable = {
				name = money_value_target_1_silver
				value = money_value_silver_standard
			}
			#bimetalique
			set_variable = {
				name = money_value_target_1_bimetallic
				value = money_value_bimetallism_standard
			}
		}
	}
}

on_activate_monetary_system_law = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = extreme_weak_currency_solution_count }
				var:extreme_weak_currency_solution_count = 0
			}
		}
		zz_ef_std_switch_before = yes
		money_value_target_pre_set = yes
		central_bank_production_methods = yes
		if = { limit = { has_modifier = foreign_exchange_controls } remove_modifier = foreign_exchange_controls }
		add_modifier = {
			name = foreign_exchange_controls
			months = 12
		}
		# the parity: a metal one before, a metal standard now -- the old one in the new metal, the reserves converted
		# (scripted_effects/ld_standard_switch.txt); else E&F's
		if = {
			limit = { NOT = { zz_ef_std_switch_metal = yes } }
			set_variable = {
				name = money_value_target_1
				value = money_value_target_pre_set
			}
		}
		zz_ef_std_switch_after = yes
		trade_balance = yes
		if = { limit = { has_modifier = monetary_systeme_transition } remove_modifier = monetary_systeme_transition }
		add_modifier = {
			name = monetary_systeme_transition
			months = 12
		}
		set_reset_monetary_system_status = yes
	}
	# the head of a currency union: its followers take the new standard at once (ld_monetary_union.txt, Д.R8в.19)
	zz_ef_mu_head_law_changed = yes
}
