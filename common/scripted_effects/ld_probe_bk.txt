# Probe R4б.1 (Р26): where a subject-bank lives (Д.R4б.1, Д.R4б.2). Runs by itself on every country's monthly pulse
# (common/on_actions/ld_probe_bk_on_actions.txt); goes to _archive/ld_probe_bk/ after the run.
#  (а) variables and maps on a company and in a script container (create_container, parent = the company), read from the
#      country (sum into a country variable) and from the window (gui/ld_probe_bk.gui reads Company.MakeScope.Var and
#      hands it back);
#  (б) the company's levels of the "Bank" building -- script has no trigger for them (company_owned_levels counts all
#      its buildings): the window reads Company.GetBuildingTypeLevels and hands them to script;
#  (в) the price of maps: every company of a country with banks (an upper bound of the bank companies) and two subjects
#      per country with banks do 4 rounds (a month of weeks) of read / remove / add on a 30-key map (the claims
#      register's write, ld_claims.txt); run inside the event ld_probe_bk.1 -- its own line in the script profiler.
# Log: EFZ|date|country|bk_probe|... (company lines -- bk_comp).

# COUNTRY scope, monthly.
ld_probe_bk_month = {
	if = {
		limit = { NOT = { has_global_variable_list = ld_pb_keys } }
		ordered_country = {
			order_by = gdp
			max = 30
			check_range_bounds = no
			add_to_global_variable_list = { name = ld_pb_keys target = this }
		}
	}
	save_temporary_scope_as = ld_pb_cty
	set_variable = { name = ld_pb_comp_lv value = 0 }
	set_variable = { name = ld_pb_comp_n value = 0 }
	set_variable = { name = ld_pb_sum_n value = 0 }
	set_variable = { name = ld_pb_load_n value = 0 }
	set_variable = { name = ld_pb_bank_lv value = ld_probe_bk_levels_v }
	set_variable = { name = ld_pb_gov_lv value = ld_probe_bk_gov_levels_v }
	every_company = {
		limit = { company_owned_levels > 0 }
		if = {
			limit = { NOT = { has_variable = ld_pb_n } }
			set_variable = { name = ld_pb_n value = 0 }
		}
		change_variable = { name = ld_pb_n add = 1 }
		scope:ld_pb_cty = { change_variable = { name = ld_pb_sum_n add = prev.var:ld_pb_n } }
		# the window: this company's bank levels (read next month)
		if = {
			limit = { NOT = { is_target_in_global_variable_list = { name = ld_pb_companies target = this } } }
			add_to_global_variable_list = { name = ld_pb_companies target = this }
		}
		if = {
			limit = {
				has_variable = ld_pb_lv
				var:ld_pb_lv > 0
			}
			scope:ld_pb_cty = {
				change_variable = { name = ld_pb_comp_lv add = prev.var:ld_pb_lv }
				change_variable = { name = ld_pb_comp_n add = 1 }
			}
		}
		# the load: every company of a country with banks (more than the bank companies -- an upper bound)
		if = {
			limit = { scope:ld_pb_cty.var:ld_pb_bank_lv > 0 }
			ld_probe_bk_map_work = yes
			ld_probe_bk_box = yes
			ld_probe_bk_map_count = yes
			scope:ld_pb_cty = { change_variable = { name = ld_pb_load_n add = 1 } }
			if = {
				limit = { zz_ef_logs_on = yes }
				debug_log = "EFZ|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCompany.GetCountry.GetNameNoFormatting]|bk_comp|[THIS.GetCompany.GetNameNoFormatting]|lv [THIS.GetCompany.MakeScope.Var('ld_pb_lv').GetValue|0]|lv_self [THIS.GetCompany.MakeScope.Var('ld_pb_lv2').GetValue|0]|five [THIS.GetCompany.MakeScope.Var('ld_pb_five').GetValue|0]|five_i [THIS.GetCompany.MakeScope.Var('ld_pb_five_i').GetValue|0]|n [THIS.GetCompany.MakeScope.Var('ld_pb_n').GetValue|0]|n_gui [THIS.GetCompany.MakeScope.Var('ld_pb_n_gui').GetValue|0]|map [THIS.GetCompany.MakeScope.Var('ld_pb_ms').GetValue|0]|box_n [THIS.GetCompany.MakeScope.Var('ld_pb_box_n').GetValue|0]|box_map [THIS.GetCompany.MakeScope.Var('ld_pb_box_ms').GetValue|0]"
			}
		}
	}
	if = {
		limit = { var:ld_pb_bank_lv > 0 }
		# «госбанки» and «частные банкиры»: two subjects on the country
		ld_probe_bk_map_work = yes
		ld_probe_bk_map_work = yes
		ld_probe_bk_map_count = yes
		if = {
			limit = { zz_ef_logs_on = yes }
			debug_log = "EFZ|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|bk_probe|bank_lv [THIS.GetCountry.MakeScope.Var('ld_pb_bank_lv').GetValue|0]|gov_lv [THIS.GetCountry.MakeScope.Var('ld_pb_gov_lv').GetValue|0]|comp_lv [THIS.GetCountry.MakeScope.Var('ld_pb_comp_lv').GetValue|0]|comp_n [THIS.GetCountry.MakeScope.Var('ld_pb_comp_n').GetValue|0]|sum_n [THIS.GetCountry.MakeScope.Var('ld_pb_sum_n').GetValue|0]|load_n [THIS.GetCountry.MakeScope.Var('ld_pb_load_n').GetValue|0]|map [THIS.GetCountry.MakeScope.Var('ld_pb_ms').GetValue|0]|gui_list [THIS.GetCountry.MakeScope.ScriptValue('ld_probe_bk_gui_list_v')|0]"
		}
	}
}

# SUBJECT scope (company or country): 4 rounds over the 30 keys -- read, remove, add (a week's claims write each).
ld_probe_bk_map_work = {
	every_in_global_list = {
		variable = ld_pb_keys
		save_temporary_scope_as = ld_pb_k
		prev = {
			ld_probe_bk_map_op = yes
			ld_probe_bk_map_op = yes
			ld_probe_bk_map_op = yes
			ld_probe_bk_map_op = yes
		}
	}
}

# SUBJECT scope, scope:ld_pb_k -- the key.
ld_probe_bk_map_op = {
	set_variable = { name = ld_pb_old value = 0 }
	if = {
		limit = { is_key_in_variable_map = { name = ld_pb_map target = scope:ld_pb_k } }
		set_variable = { name = ld_pb_old value = "variable_map(ld_pb_map|scope:ld_pb_k)" }
		remove_from_variable_map = { name = ld_pb_map key = scope:ld_pb_k }
	}
	set_variable = { name = ld_pb_new value = { value = var:ld_pb_old add = 1 } }
	add_to_variable_map = { name = ld_pb_map key = scope:ld_pb_k value = ld_probe_bk_new_v }
}

# SUBJECT scope: the map's size into var:ld_pb_ms.
ld_probe_bk_map_count = {
	set_variable = { name = ld_pb_ms value = 0 }
	if = {
		limit = { has_variable_map = ld_pb_map }
		every_key_in_variable_map = {
			variable = ld_pb_map
			prev = { change_variable = { name = ld_pb_ms add = 1 } }
		}
	}
}

# COMPANY scope: a script container with the company as parent -- a variable and a 30-key map in it, read back.
ld_probe_bk_box = {
	if = {
		limit = { NOT = { has_variable = ld_pb_box } }
		create_container = {
			parent = this
			tags = { ld_pb }
			save_scope_as = ld_pb_newbox
		}
		set_variable = { name = ld_pb_box value = scope:ld_pb_newbox }
		var:ld_pb_box = {
			set_variable = { name = ld_pb_bn value = 0 }
			every_in_global_list = {
				variable = ld_pb_keys
				save_temporary_scope_as = ld_pb_k
				prev = {
					set_variable = { name = ld_pb_new value = 7 }
					add_to_variable_map = { name = ld_pb_map key = scope:ld_pb_k value = ld_probe_bk_new_v }
				}
			}
		}
	}
	var:ld_pb_box = {
		change_variable = { name = ld_pb_bn add = 1 }
		ld_probe_bk_map_count = yes
	}
	set_variable = { name = ld_pb_box_n value = var:ld_pb_box.var:ld_pb_bn }
	set_variable = { name = ld_pb_box_ms value = var:ld_pb_box.var:ld_pb_ms }
}
