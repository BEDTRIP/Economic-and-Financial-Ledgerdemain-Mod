# E&F's world numbers of the month, once for the whole world (Д.R3б.4), not in every central bank's monthly hub:
# the world's gold and silver production -> silver_to_gold_rate (one price of the month for all), the reference
# country's data, the median's insurance at the start (money_value_median 0 or 1).
# Country scope, root -- the country ranked first (the currency law lists are kept on root): the event ld_world_month.1,
# from on_monthly_pulse (on_actions/ld_roles_on_actions.txt) and from the start setup (ld_start_setup.txt).
zz_ef_world_month_ef = {
	global_gold_silver_production = yes
	if = {
		limit = {
			or = {
				global_var:money_value_median = 0
				global_var:money_value_median = 1
			}
		}
		currency_law_list = yes
		money_value_global_var = yes
		median_currency_value = yes
	}
	every_country = {
		limit = { has_modifier = global_monetary_reference }
		global_monetary_reference_global_var_fixe = yes
	}
	# the world's exchange for the countries with no other (ld_exchange.txt)
	zz_ef_xch_world_set = yes
	# the register of claims adds up (ld_claims.txt, the EFI line) -- a check for the logs only
	if = {
		limit = { zz_ef_logs_on = yes }
		zz_ef_rq_world_check = yes
		zz_ef_fund_log = yes
		zz_ef_xch_month_log = yes
		set_global_variable = { name = zz_ef_fc_world_n value = 0 }
		every_country = {
			limit = { zz_ef_has_exchange = yes }
			change_global_variable = { name = zz_ef_fc_world_n add = 1 }
		}
		set_global_variable = { name = zz_ef_xch_n_1 value = 0 }
		set_global_variable = { name = zz_ef_xch_n_2 value = 0 }
		set_global_variable = { name = zz_ef_xch_n_3 value = 0 }
		set_global_variable = { name = zz_ef_xch_n_4 value = 0 }
		set_global_variable = { name = zz_ef_xch_n_5 value = 0 }
		every_country = {
			if = { limit = { var:zz_ef_xch_how ?= 1 } change_global_variable = { name = zz_ef_xch_n_1 add = 1 } }
			else_if = { limit = { var:zz_ef_xch_how ?= 2 } change_global_variable = { name = zz_ef_xch_n_2 add = 1 } }
			else_if = { limit = { var:zz_ef_xch_how ?= 3 } change_global_variable = { name = zz_ef_xch_n_3 add = 1 } }
			else_if = { limit = { var:zz_ef_xch_how ?= 4 } change_global_variable = { name = zz_ef_xch_n_4 add = 1 } }
			else_if = { limit = { var:zz_ef_xch_how ?= 5 } change_global_variable = { name = zz_ef_xch_n_5 add = 1 } }
		}
		if = { limit = { zz_ef_logs_on = yes } debug_log = "EFH|[TimeKeeper.GetCurrentDate.GetString]|WORLD|exchanges [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_fc_world_n_v')|0]|own [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_xch_n_1_v')|0]|market [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_xch_n_2_v')|0]|bloc [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_xch_n_3_v')|0]|lever [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_xch_n_4_v')|0]|world [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_xch_n_5_v')|0]" }
	}
}

# Any scope: the event on the country ranked first.
zz_ef_world_month_fire = {
	every_country = {
		limit = { global_country_ranking = 1 }
		trigger_event = { id = ld_world_month.1 days = 0 }
	}
}
