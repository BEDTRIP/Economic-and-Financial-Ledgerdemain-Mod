# E&F's currency introduction (introduction_new_currency, 09_introduction_building_lvl.txt -- the technology
# currency_standards and macro_facilities_bc) is wrapped here.
# Д.R8в.7: a country with money and a parity of its own keeps them -- its standard's law and parity are put back as they
# were before the call (Hanover leaving the union stays at its 3.32, not E&F's random 15-30); a country with no parity
# takes one from the money that went on its land -- its market owner's parity x the owner's money against its parity
# (in the new standard's metal).
# E&F's metal "from nothing" (introduction_of_<currency> adds to the capital's gold_state_1 / silver_state_1 the money in
# E&F's count + its stock + building savings x the new parity) is switched off:
#   * before the country's first step (var:zz_ef_metal_started): untouched -- E&F's start, which the first step takes;
#   * later: the capital's metal is put back as it was before the call; a new CB starts with no metal and buys it on
#     the market.

zz_ef_cur_intro_before = {
	# the standard and the parity before (Д.R8в.7)
	set_variable = { name = zz_ef_cur_intro_std value = zz_ef_std_code }
	set_variable = { name = zz_ef_cur_intro_par value = 0 }
	if = {
		limit = { has_variable = money_value_target_1 }
		set_variable = { name = zz_ef_cur_intro_par value = var:money_value_target_1 }
	}
	# history can call it for a country with no capital yet
	if = {
		limit = { exists = capital }
		if = { limit = { has_variable = zz_ef_cur_intro_first } remove_variable = zz_ef_cur_intro_first }
		if = {
			limit = { NOT = { has_variable = country_already_central_bank } }
			set_variable = { name = zz_ef_cur_intro_first value = yes }
		}
		capital = {
			set_variable = { name = zz_ef_cur_intro_gold value = 0 }
			set_variable = { name = zz_ef_cur_intro_silver value = 0 }
			if = {
				limit = { has_variable = gold_state_1 }
				set_variable = { name = zz_ef_cur_intro_gold value = var:gold_state_1 }
			}
			if = {
				limit = { has_variable = silver_state_1 }
				set_variable = { name = zz_ef_cur_intro_silver value = var:silver_state_1 }
			}
		}
	}
}

zz_ef_cur_intro_after = {
	# a country with money and a parity: its standard and parity back (Д.R8в.7)
	if = {
		limit = { var:zz_ef_cur_intro_par > zz_ef_cur_intro_par_min }
		zz_ef_cur_intro_law_back = yes
		set_variable = { name = money_value_target_1 value = var:zz_ef_cur_intro_par }
	}
	# no parity before: the money that went on its land -- its market owner's
	else_if = {
		limit = {
			exists = market
			NOT = { market.owner = this }
			market.owner = {
				has_variable = money_value_target_1
				var:money_value_target_1 > zz_ef_cur_intro_par_min
			}
		}
		market.owner = {
			save_temporary_scope_as = zz_ef_ci_o
			set_variable = { name = zz_ef_ci_vtp value = zz_ef_value_to_parity }
		}
		set_variable = {
			name = money_value_target_1
			value = {
				value = scope:zz_ef_ci_o.var:money_value_target_1
				multiply = scope:zz_ef_ci_o.var:zz_ef_ci_vtp
			}
		}
		scope:zz_ef_ci_o = { remove_variable = zz_ef_ci_vtp }
		if = {
			limit = {
				zz_ef_metal_silver = yes
				NOT = { market.owner = { zz_ef_metal_silver = yes } }
			}
			change_variable = { name = money_value_target_1 multiply = gold_to_silver_rate }
		}
		else_if = {
			limit = {
				NOT = { zz_ef_metal_silver = yes }
				market.owner = { zz_ef_metal_silver = yes }
			}
			change_variable = { name = money_value_target_1 multiply = silver_to_gold_rate }
		}
	}
	if = { limit = { zz_ef_logs_on = yes } debug_log = "EFM|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|cur_intro_parity|std_before [THIS.GetCountry.MakeScope.Var('zz_ef_cur_intro_std').GetValue|0]|par_before [THIS.GetCountry.MakeScope.Var('zz_ef_cur_intro_par').GetValue|4]|std [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_std_code')|0]|parity [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cb_valuation')|4]" }
	remove_variable = zz_ef_cur_intro_std
	remove_variable = zz_ef_cur_intro_par
	# the new currency as data (scripted_effects/ld_currency_var.txt)
	zz_ef_cur_set = yes
	if = {
		limit = { exists = capital }
		# 6.10: from the country's first step on -- E&F's metal at an introduction in the first
		# 13 weeks was left to the week-12 rescale, and the world line of metal read +370M gold and +4.4 bn silver from
		# nothing in one week (17.2.1836); before the first step it is E&F's start, which the first step takes as stock
		if = {
			limit = {
				has_variable = zz_ef_metal_started
			}
			capital = {
				set_variable = { name = zz_ef_cur_intro_g_ef value = 0 }
				if = {
					limit = { has_variable = gold_state_1 }
					set_variable = { name = zz_ef_cur_intro_g_ef value = var:gold_state_1 }
				}
				set_variable = { name = gold_state_1 value = var:zz_ef_cur_intro_gold }
				set_variable = { name = silver_state_1 value = var:zz_ef_cur_intro_silver }
			}
			# a first currency got 40% of its M2 in metal from nothing (the norm) -- removed; its CB
			# buys metal on the market like every CB (its building's input, zz_ef_cb_metal_purchase_step)
			set_variable = { name = zz_ef_cur_intro_norm value = 0 }
			if = { limit = { zz_ef_logs_on = yes } debug_log = "EFM|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|cur_intro|first [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cur_intro_first_v')|0]|gold_ef [THIS.GetCountry.GetCapital.MakeScope.Var('zz_ef_cur_intro_g_ef').GetValue|0]|gold_kept [THIS.GetCountry.GetCapital.MakeScope.Var('zz_ef_cur_intro_gold').GetValue|0]|norm [THIS.GetCountry.MakeScope.Var('zz_ef_cur_intro_norm').GetValue|0]|parity [THIS.GetCountry.MakeScope.Var('money_value_target_1').GetValue|4]" }
			capital = {
				if = { limit = { has_variable = zz_ef_cur_intro_g_ef } remove_variable = zz_ef_cur_intro_g_ef }
			}
		}
		capital = {
			if = { limit = { has_variable = zz_ef_cur_intro_gold } remove_variable = zz_ef_cur_intro_gold }
			if = { limit = { has_variable = zz_ef_cur_intro_silver } remove_variable = zz_ef_cur_intro_silver }
		}
		if = { limit = { has_variable = zz_ef_cur_intro_first } remove_variable = zz_ef_cur_intro_first }
	}
}

# The standard's law of var:zz_ef_cur_intro_std (zz_ef_std_code) back, if the introduction changed it.
zz_ef_cur_intro_law_back = {
	if = {
		limit = { NOT = { zz_ef_std_code = var:zz_ef_cur_intro_std } }
		if = { limit = { var:zz_ef_cur_intro_std = 1 } activate_law = law_type:law_gold_standard }
		else_if = { limit = { var:zz_ef_cur_intro_std = 2 } activate_law = law_type:law_silver_standard }
		else_if = { limit = { var:zz_ef_cur_intro_std = 3 } activate_law = law_type:law_bimetallism_standard }
		else_if = { limit = { var:zz_ef_cur_intro_std = 4 } activate_law = law_type:law_gold_exchange_standard }
		else_if = { limit = { var:zz_ef_cur_intro_std = 5 } activate_law = law_type:law_fiat_standard }
		else_if = { limit = { var:zz_ef_cur_intro_std = 6 } activate_law = law_type:law_external_exchange_standard }
		else_if = { limit = { var:zz_ef_cur_intro_std = 7 } activate_law = law_type:law_silver_exchange_standard }
		else_if = { limit = { var:zz_ef_cur_intro_std = 8 } activate_law = law_type:law_bimetallism_exchange_standard }
	}
}
