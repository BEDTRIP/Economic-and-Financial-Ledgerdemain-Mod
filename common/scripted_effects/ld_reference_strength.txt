#############################################
# the reference currency keeps its value / parity for everyone's strength
# (script_values/ld_reference_currency_values.txt). Monthly, from the money model's monthly step.
#############################################

zz_ef_reference_strength_step = {
	if = {
		limit = { has_modifier = global_monetary_reference }
		set_global_variable = { name = zz_ef_ref_vtp value = zz_ef_value_to_parity }
		# the reference's CB rate for the dependent standards' corridor (zz_ef_cb_rate_floor / _ceiling)
		if = {
			limit = { has_variable = base_rate_percentage }
			set_global_variable = { name = zz_ef_ref_rate value = var:base_rate_percentage }
		}
		# the reference's price index for the real rate (zz_ef_currency_trade_m)
		if = {
			limit = { has_variable = zz_ef_prev_price }
			set_global_variable = { name = zz_ef_ref_cpi value = var:zz_ef_prev_price }
		}
	}
}

# the trade modifier of the currency's strength (static_modifiers/ld_currency_trade.txt).
# Monthly, after the reference's step; countries with a central bank, not the reference.
zz_ef_currency_trade_step = {
	if = { limit = { has_modifier = zz_ef_currency_trade } remove_modifier = zz_ef_currency_trade }
	if = {
		limit = {
			has_modifier = has_central_bank
			NOT = { has_modifier = global_monetary_reference }
			has_global_variable = zz_ef_ref_vtp
			NOT = { zz_ef_currency_trade_m = 0 }
		}
		add_modifier = {
			name = zz_ef_currency_trade
			multiplier = zz_ef_currency_trade_m
		}
	}
}
