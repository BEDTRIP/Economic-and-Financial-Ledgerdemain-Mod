#############################################
# the reference currency keeps its value / parity for everyone's strength
# (script_values/ld_reference_currency_values.txt). Monthly, the world's pass (zz_ef_world_month_ef), whatever the
# reference's standard and role.
#############################################

zz_ef_reference_strength_step = {
	# a fiat currency of an older save without its worth at the transition
	every_country = {
		limit = {
			has_law = law_type:law_fiat_standard
			NOT = { has_variable = zz_ef_fiat_v0 }
			has_variable = zz_ef_std_parity_new
		}
		zz_ef_fiat_v0_set = yes
	}
	every_country = {
		limit = { has_modifier = global_monetary_reference }
		set_global_variable = { name = zz_ef_ref_vtp value = zz_ef_value_to_parity }
		# the reference's price index for the real rate (zz_ef_currency_trade_m) and the fiat currencies
		if = {
			limit = { has_variable = zz_ef_prev_price }
			set_global_variable = { name = zz_ef_ref_cpi value = var:zz_ef_prev_price }
		}
	}
}

# the trade modifier of the currency's strength (static_modifiers/ld_currency_trade.txt) -- on the country's states, so a
# member of another's market trades by its own money too (Д.R8в.1). Monthly, after the reference's step; every country
# with money of its own (its monetary authority -- the CB or the banks), not the reference.
zz_ef_currency_trade_step = {
	if = { limit = { has_modifier = zz_ef_currency_trade } remove_modifier = zz_ef_currency_trade }
	if = {
		limit = {
			zz_ef_has_currency_value = yes
			NOT = { has_modifier = global_monetary_reference }
			has_global_variable = zz_ef_ref_vtp
			NOT = { zz_ef_currency_trade_m = 0 }
		}
		add_modifier = {
			name = zz_ef_currency_trade
			multiplier = zz_ef_currency_trade_m
		}
	}
}
