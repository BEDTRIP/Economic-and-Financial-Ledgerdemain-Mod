# ---- the start in balance (R1б.1: Д.1, Д.2, В4) -----------------------------
# Country scope, the first step of a new game, once (the user, 7.10: "in balance from the first day"):
#   * the parity stays E&F's historical one (Д.1: the pound 7.32, the franc 0.75 -- the rate between currencies and the
#     labels in national units); the engine's money is one unit for every country and the metal is counted in it at a
#     fixed rate (zz_ef_metal_per_money: 1 gold unit per unit of money, a silver standard -- silver at the market rate);
#   * the CB's metal (a metal standard with a CB) = 40% of M2 -- once, at the first step where the engine has filled
#     the buildings' cash (zz_ef_cb_metal_start_step, var:zz_ef_cb_start_due); E&F's metal from history is not taken
#     (its scale is ~10 times our money); until then the CB does not trade metal (Д.2);
#   * В4 (the user, 8.10): E&F's metal of a country with a CB stays in its CB states, also when the country has no metal
#     standard yet (a subject joins its overlord's through the currency zone after its first step -- Hanover, Lower
#     Canada, Finland got 44-132 GDP of it as the pops' metal, r1008_062130); the CB's start scales it to 40% of M2 (the
#     currency at its E&F parity), gold and silver by one factor (E&F's ratio). A country without a CB: its CB states'
#     metal goes to the pops;
#   * E&F's metal outside a CB state (history writes it before the country has a CB -- Bavaria) -- the pops' too (Д.R1.1);
#   * the pops' metal: 5% of GDP with a CB, 20% without (В4: a traditional economy saves in metal), or E&F's metal of a
#     CB-less country if more; once;
#   * the banks' metal: the reserve norm x the start deposits -- at the first receiver (zz_ef_metal_start_banks).
# Everything here is the start's stock, not mining: logged (EFM|...|metal_start) and summed into
# global_var:zz_ef_wm_start_gold / _silver for the world line.
zz_ef_metal_start_step = {
	zz_ef_metal_start_cb_before = yes
	set_variable = { name = zz_ef_ms_ef_g value = 0 }
	set_variable = { name = zz_ef_ms_ef_s value = 0 }
	# the CB's 40% comes once the money exists and the country is a CB on a metal standard (zz_ef_cb_metal_start_step) --
	# a subject joins its overlord's standard through the currency zone after its first step, E&F writes its metal
	# again then (r1007_204439: the Dutch East Indies 1.8 bn silver, Finland, New South Wales, Hanover)
	if = {
		limit = { has_modifier = has_central_bank }
		set_variable = { name = zz_ef_cb_start_due value = yes }
	}
	if = {
		limit = { NOT = { has_modifier = has_central_bank } }
		# В4: E&F's metal of a country without a CB -- the pops' (with a CB it waits for the CB's start, above)
		set_variable = { name = zz_ef_ms_ef_g value = gold_state_native_for_stockpile }
		set_variable = { name = zz_ef_ms_ef_s value = silver_state_native_for_stockpile }
		every_scope_state = {
			limit = { has_modifier = central_bank_historic_place }
			if = { limit = { has_variable = gold_state_1 } set_variable = { name = gold_state_1 value = 0 } }
			if = { limit = { has_variable = silver_state_1 } set_variable = { name = silver_state_1 value = 0 } }
		}
	}
	# Д.R1.1 (8.10): E&F's metal outside a CB state -- history writes it into var:central_bank_location before the
	# country has a CB (r1008_001240: Bavaria 37.1M gold in a state with no central_bank_historic_place; the first step
	# did not see it, the CB made later counted it as its metal from nothing) -- the pops' too (В4), and 0. Only a
	# country with no metal standard: on a metal standard the CB made in the first month gets that state and the CB's
	# start scales it to 40% of M2 (zz_ef_cb_metal_start_step; r1008_005426: taken for all, the pops got 111 bn silver)
	every_scope_state = {
		limit = {
			NOT = { has_modifier = central_bank_historic_place }
			owner = {
				NOR = {
					has_law = law_type:law_gold_standard
					has_law = law_type:law_silver_standard
					has_law = law_type:law_bimetallism_standard
				}
			}
		}
		if = {
			limit = { has_variable = gold_state_1 }
			owner = { change_variable = { name = zz_ef_ms_ef_g add = PREV.var:gold_state_1 } }
			set_variable = { name = gold_state_1 value = 0 }
		}
		if = {
			limit = { has_variable = silver_state_1 }
			owner = { change_variable = { name = zz_ef_ms_ef_s add = PREV.var:silver_state_1 } }
			set_variable = { name = silver_state_1 value = 0 }
		}
	}
	zz_ef_metal_start_cb_after = yes
	# the pops' metal: the norm, or E&F's metal (CB-less countries) if more -- in money (gold units), split by the standard
	set_variable = { name = zz_ef_ms_pop value = zz_ef_popm_start_money }
	set_variable = {
		name = zz_ef_ms_ef
		value = {
			value = var:zz_ef_ms_ef_s
			multiply = silver_to_gold_rate
			add = var:zz_ef_ms_ef_g
		}
	}
	if = {
		limit = { var:zz_ef_ms_ef > var:zz_ef_ms_pop }
		set_variable = { name = zz_ef_ms_pop value = var:zz_ef_ms_ef }
	}
	set_variable = { name = zz_ef_popm_gold value = var:zz_ef_ms_pop }
	change_variable = { name = zz_ef_popm_gold multiply = { value = 1 subtract = zz_ef_metal_silver_share } }
	set_variable = { name = zz_ef_popm_silver value = var:zz_ef_ms_pop }
	change_variable = { name = zz_ef_popm_silver multiply = zz_ef_metal_silver_share }
	change_variable = { name = zz_ef_popm_silver multiply = gold_to_silver_rate }
	zz_ef_wm_add = { KEY = start_gold VALUE = var:zz_ef_popm_gold }
	zz_ef_wm_add = { KEY = start_silver VALUE = var:zz_ef_popm_silver }
	set_variable = { name = zz_ef_ms_what value = 1 }
	zz_ef_metal_start_log = yes
	remove_variable = zz_ef_ms_pop
	remove_variable = zz_ef_ms_ef
	remove_variable = zz_ef_ms_ef_g
	remove_variable = zz_ef_ms_ef_s
}

# Д.2: the CB's metal = 40% of M2, once, at the first step after the start deposits where M2 is over 10% of GDP (the
# engine fills the buildings' cash in the first weeks). E&F's metal from history is scaled to it (with the foreign currency in the reserves, so
# the cover reads 40%); a CB with no metal gets it in a CB state. The change is the start's stock (world line).
zz_ef_cb_metal_start_step = {
	if = {
		limit = {
			has_variable = zz_ef_cb_start_due
			zz_ef_cb_metal_standard = yes
			# after the first receiver: the start deposits are in M2 (r1007_200927: at the first step the cover was set to
			# 40% before them, and read 0.27-0.30 once they came in)
			has_variable = zz_ef_pop_savings
			zz_ef_agg_m2_to_gdp > 0.1
			# a CB state to hold the metal (E&F writes the CB's metal in the first month by its own formula -- r1007_200927:
			# Denmark, Bolivia, Brazil, Qing... 0 at the first step, 22M gold + 344M silver a week later); the start scales
			# what E&F wrote and counts it as the start's stock (below)
			any_scope_state = { has_modifier = central_bank_historic_place }
		}
		remove_variable = zz_ef_cb_start_due
		zz_ef_metal_start_cb_before = yes
		if = {
			limit = {
				zz_ef_cb_metal_standard = yes
				zz_ef_cb_cover > 0
				central_bank_reserves > 1
			}
			set_variable = { name = zz_ef_metal_rescale value = zz_ef_cover_normal }
			change_variable = { name = zz_ef_metal_rescale divide = zz_ef_cb_cover }
			every_scope_state = {
				limit = { has_modifier = central_bank_historic_place }
				if = {
					limit = { has_variable = gold_state_1 }
					change_variable = { name = gold_state_1 multiply = owner.var:zz_ef_metal_rescale }
				}
				if = {
					limit = { has_variable = silver_state_1 }
					change_variable = { name = silver_state_1 multiply = owner.var:zz_ef_metal_rescale }
				}
			}
			if = {
				limit = { has_variable = zz_ef_fx_metal }
				change_variable = { name = zz_ef_fx_metal multiply = var:zz_ef_metal_rescale }
			}
			remove_variable = zz_ef_metal_rescale
		}
		else_if = {
			limit = {
				zz_ef_cb_metal_standard = yes
				any_scope_state = { has_modifier = central_bank_historic_place }
			}
			set_variable = { name = zz_ef_metal_rescale value = zz_ef_agg_m2 }
			change_variable = { name = zz_ef_metal_rescale multiply = zz_ef_cover_normal }
			change_variable = { name = zz_ef_metal_rescale multiply = zz_ef_metal_per_money }
			random_scope_state = {
				limit = { has_modifier = central_bank_historic_place }
				if = {
					limit = { owner = { has_law = law_type:law_silver_standard } }
					if = { limit = { NOT = { has_variable = silver_state_1 } } set_variable = { name = silver_state_1 value = 0 } }
					change_variable = { name = silver_state_1 add = owner.var:zz_ef_metal_rescale }
				}
				else = {
					if = { limit = { NOT = { has_variable = gold_state_1 } } set_variable = { name = gold_state_1 value = 0 } }
					change_variable = { name = gold_state_1 add = owner.var:zz_ef_metal_rescale }
				}
			}
			remove_variable = zz_ef_metal_rescale
		}
		zz_ef_metal_start_cb_after = yes
		# E&F's metal written since the last snapshot (its CB made in the first month, r1007_201854: Denmark 21.5M gold +
		# 333M silver a week after the first step) is the start's stock too, not «прочее» of the CB's metal
		if = {
			limit = { has_variable = zz_ef_mt_prev_g }
			set_variable = { name = zz_ef_ms_g0 value = { value = var:zz_ef_ms_g0 subtract = var:zz_ef_mt_prev_g } }
			set_variable = { name = zz_ef_ms_s0 value = { value = var:zz_ef_ms_s0 subtract = var:zz_ef_mt_prev_s } }
			change_variable = { name = zz_ef_mt_start_adj_g add = var:zz_ef_ms_g0 }
			change_variable = { name = zz_ef_mt_start_adj_s add = var:zz_ef_ms_s0 }
			zz_ef_wm_add = { KEY = start_gold VALUE = var:zz_ef_ms_g0 }
			zz_ef_wm_add = { KEY = start_silver VALUE = var:zz_ef_ms_s0 }
		}
		set_variable = { name = zz_ef_ms_what value = 3 }
		zz_ef_metal_start_log = yes
	}
}

# The CB's metal before / after a start operation (the first step, the CB's 40%): the change is the start's stock
# (world line), not a purchase.
zz_ef_metal_start_cb_before = {
	set_variable = { name = zz_ef_ms_g0 value = gold_state_native_for_stockpile }
	set_variable = { name = zz_ef_ms_s0 value = silver_state_native_for_stockpile }
}
zz_ef_metal_start_cb_after = {
	set_variable = { name = zz_ef_ms_g1 value = gold_state_native_for_stockpile }
	change_variable = { name = zz_ef_ms_g1 subtract = var:zz_ef_ms_g0 }
	set_variable = { name = zz_ef_ms_s1 value = silver_state_native_for_stockpile }
	change_variable = { name = zz_ef_ms_s1 subtract = var:zz_ef_ms_s0 }
	# В3: the recount is known to the reconciliation of the CB's metal
	if = { limit = { NOT = { has_variable = zz_ef_mt_start_adj_g } } set_variable = { name = zz_ef_mt_start_adj_g value = 0 } }
	if = { limit = { NOT = { has_variable = zz_ef_mt_start_adj_s } } set_variable = { name = zz_ef_mt_start_adj_s value = 0 } }
	change_variable = { name = zz_ef_mt_start_adj_g add = var:zz_ef_ms_g1 }
	change_variable = { name = zz_ef_mt_start_adj_s add = var:zz_ef_ms_s1 }
	# the first step: E&F's metal before it is nobody's -- the whole stock after is the start's
	if = {
		limit = { NOT = { has_variable = zz_ef_metal_started } }
		set_variable = { name = zz_ef_metal_started value = yes }
		set_variable = { name = zz_ef_ms_g1 value = gold_state_native_for_stockpile }
		set_variable = { name = zz_ef_ms_s1 value = silver_state_native_for_stockpile }
	}
	zz_ef_wm_add = { KEY = start_gold VALUE = var:zz_ef_ms_g1 }
	zz_ef_wm_add = { KEY = start_silver VALUE = var:zz_ef_ms_s1 }
}

# The first receiver, after the start deposits (zz_ef_pop_savings_step): the banks' metal = the reserve norm x the
# deposits, in the standard's metal.
zz_ef_metal_start_banks = {
	set_variable = { name = zz_ef_ms_bank value = zz_ef_pop_deposits }
	change_variable = { name = zz_ef_ms_bank multiply = zz_ef_reserve_norm }
	change_variable = { name = zz_ef_ms_bank multiply = zz_ef_gold_per_money_par }
	set_variable = { name = zz_ef_bankm_gold value = var:zz_ef_ms_bank }
	change_variable = { name = zz_ef_bankm_gold multiply = { value = 1 subtract = zz_ef_metal_silver_share } }
	set_variable = { name = zz_ef_bankm_silver value = var:zz_ef_ms_bank }
	change_variable = { name = zz_ef_bankm_silver multiply = zz_ef_metal_silver_share }
	change_variable = { name = zz_ef_bankm_silver multiply = gold_to_silver_rate }
	remove_variable = zz_ef_ms_bank
	zz_ef_wm_add = { KEY = start_gold VALUE = var:zz_ef_bankm_gold }
	zz_ef_wm_add = { KEY = start_silver VALUE = var:zz_ef_bankm_silver }
	set_variable = { name = zz_ef_ms_what value = 2 }
	zz_ef_metal_start_log = yes
}

# what: 1 the first step (pops; E&F's metal of CB-less countries), 2 the banks at the first receiver, 3 the CB's 40%
zz_ef_metal_start_log = {
	if = { limit = { zz_ef_logs_on = yes } debug_log = "EFM|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|metal_start|what [THIS.GetCountry.MakeScope.Var('zz_ef_ms_what').GetValue|0]|cb_gold [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cbm_gold')|0]|cb_silver [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cbm_silver')|0]|pop_gold [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_popm_gold')|0]|pop_silver [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_popm_silver')|0]|bank_gold [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_bankm_gold')|0]|bank_silver [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_bankm_silver')|0]|parity [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cb_valuation')|4]|cover [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cb_cover')|3]|m2 [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_agg_m2')|0]|gdp [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_raw_gdp')|0]" }
}

# ---- step 8: the world line of metal ------------------------------------------
# global_var:zz_ef_wm_<KEY> += VALUE -- cumulative since the start (never reset): start_* the start's stock,
# buy_* bought on the market, sell_* sold to it; written by zz_ef_world_acc with the accounts' sums (EFG).
zz_ef_wm_add = {
	# the log's line (EFV) only -- with the logs on (В4)
	if = {
		limit = { zz_ef_logs_on = yes }
		if = {
			limit = { NOT = { has_global_variable = zz_ef_wm_$KEY$ } }
			set_global_variable = { name = zz_ef_wm_$KEY$ value = 0 }
		}
		change_global_variable = { name = zz_ef_wm_$KEY$ add = $VALUE$ }
	}
}

# ---- steps 4, 5, 7: the week's purchases and sales ---------------------------------
# Country scope, the weekly step (zz_ef_money_model_step, before the CB card's ledger). What the buildings bought /
# sold under last week's modifiers goes to the accounts (goods x 100 gold / x 775 silver units); then the modifiers
# for the coming week are set. The CB's purchase was monthly (Д2.7а, zz_ef_cb_metal_purchase_step) -- weekly now.
# var:zz_ef_f_mt_<what>_<metal>: units this week (cb_in, cb_out, bank_in, pop_in, buyback); var:zz_ef_cb_stock_acc
# keeps the CB's change for its card (UI.11).
zz_ef_metal_week_step = {
	# В3 (the user, 7.10): the CB's metal against its known moves since last week's step -- what is left (oth) moved
	# without the market or our pairs (crisis redemption, a standard switch...). Known: the clearing at the last receiver
	# (var:zz_ef_f_hume, the standard's metal), the start's recount (var:zz_ef_mt_start_adj_*), the player's forex buttons
	# (var:zz_ef_mt_fx_*, R2); this week's own purchases come after the snapshot. EFQ line when it is not ~0.
	zz_ef_metal_reconcile = yes
	# what was bought / sold this week
	# the CB's and the banks' input (goods) once each: the pops' part below is the states' consumption less these two
	# (= zz_ef_pop_<metal>_goods, with the same floor at 0)
	set_variable = { name = zz_ef_t_cb_in_g value = zz_ef_cb_in_gold_goods }
	set_variable = { name = zz_ef_t_cb_in_s value = zz_ef_cb_in_silver_goods }
	set_variable = { name = zz_ef_t_bank_in_g value = zz_ef_bank_in_gold_goods }
	set_variable = { name = zz_ef_t_bank_in_s value = zz_ef_bank_in_silver_goods }
	# R1а.4: a step after a gap -- the buildings' input of the weeks the step covers
	change_variable = { name = zz_ef_t_cb_in_g multiply = zz_ef_step_weeks_v }
	change_variable = { name = zz_ef_t_cb_in_s multiply = zz_ef_step_weeks_v }
	change_variable = { name = zz_ef_t_bank_in_g multiply = zz_ef_step_weeks_v }
	change_variable = { name = zz_ef_t_bank_in_s multiply = zz_ef_step_weeks_v }
	set_variable = { name = zz_ef_f_mt_cb_in_g value = var:zz_ef_t_cb_in_g }
	change_variable = { name = zz_ef_f_mt_cb_in_g multiply = zz_ef_gold_units_per_good }
	set_variable = { name = zz_ef_f_mt_cb_in_s value = var:zz_ef_t_cb_in_s }
	change_variable = { name = zz_ef_f_mt_cb_in_s multiply = zz_ef_silver_units_per_good }
	set_variable = { name = zz_ef_f_mt_cb_out_g value = zz_ef_cb_out_gold_goods }
	change_variable = { name = zz_ef_f_mt_cb_out_g multiply = zz_ef_gold_units_per_good }
	change_variable = { name = zz_ef_f_mt_cb_out_g multiply = zz_ef_step_weeks_v }
	set_variable = { name = zz_ef_f_mt_cb_out_s value = zz_ef_cb_out_silver_goods }
	change_variable = { name = zz_ef_f_mt_cb_out_s multiply = zz_ef_silver_units_per_good }
	change_variable = { name = zz_ef_f_mt_cb_out_s multiply = zz_ef_step_weeks_v }
	set_variable = { name = zz_ef_f_mt_bank_in_g value = var:zz_ef_t_bank_in_g }
	change_variable = { name = zz_ef_f_mt_bank_in_g multiply = zz_ef_gold_units_per_good }
	set_variable = { name = zz_ef_f_mt_bank_in_s value = var:zz_ef_t_bank_in_s }
	change_variable = { name = zz_ef_f_mt_bank_in_s multiply = zz_ef_silver_units_per_good }
	set_variable = {
		name = zz_ef_f_mt_pop_in_g
		value = {
			value = zz_ef_state_gold_cons
			multiply = zz_ef_step_weeks_v
			subtract = var:zz_ef_t_cb_in_g
			subtract = var:zz_ef_t_bank_in_g
			min = 0
		}
	}
	change_variable = { name = zz_ef_f_mt_pop_in_g multiply = zz_ef_gold_units_per_good }
	set_variable = {
		name = zz_ef_f_mt_pop_in_s
		value = {
			value = zz_ef_state_silver_cons
			multiply = zz_ef_step_weeks_v
			subtract = var:zz_ef_t_cb_in_s
			subtract = var:zz_ef_t_bank_in_s
			min = 0
		}
	}
	change_variable = { name = zz_ef_f_mt_pop_in_s multiply = zz_ef_silver_units_per_good }
	# the CB: bought - sold, into (out of) its CB state; a sale never more than it holds
	if = {
		limit = { var:zz_ef_f_mt_cb_out_g > gold_state_native_for_stockpile }
		set_variable = { name = zz_ef_f_mt_cb_out_g value = gold_state_native_for_stockpile }
	}
	if = {
		limit = { var:zz_ef_f_mt_cb_out_s > silver_state_native_for_stockpile }
		set_variable = { name = zz_ef_f_mt_cb_out_s value = silver_state_native_for_stockpile }
	}
	set_variable = { name = zz_ef_cb_stock_before value = central_bank_reserves }
	if = {
		limit = {
			any_scope_state = { has_modifier = central_bank_historic_place }
		}
		random_scope_state = {
			limit = { has_modifier = central_bank_historic_place }
			if = { limit = { NOT = { has_variable = gold_state_1 } } set_variable = { name = gold_state_1 value = 0 } }
			if = { limit = { NOT = { has_variable = silver_state_1 } } set_variable = { name = silver_state_1 value = 0 } }
			change_variable = { name = gold_state_1 add = owner.var:zz_ef_f_mt_cb_in_g }
			change_variable = { name = gold_state_1 subtract = owner.var:zz_ef_f_mt_cb_out_g }
			change_variable = { name = silver_state_1 add = owner.var:zz_ef_f_mt_cb_in_s }
			change_variable = { name = silver_state_1 subtract = owner.var:zz_ef_f_mt_cb_out_s }
		}
	}
	else = {
		# no CB state: what a building_bank bought has nowhere to go -- the pops' (logged as cb_in anyway)
		change_variable = { name = zz_ef_f_mt_pop_in_g add = var:zz_ef_f_mt_cb_in_g }
		change_variable = { name = zz_ef_f_mt_pop_in_s add = var:zz_ef_f_mt_cb_in_s }
		set_variable = { name = zz_ef_f_mt_cb_in_g value = 0 }
		set_variable = { name = zz_ef_f_mt_cb_in_s value = 0 }
		set_variable = { name = zz_ef_f_mt_cb_out_g value = 0 }
		set_variable = { name = zz_ef_f_mt_cb_out_s value = 0 }
	}
	# the banks and the pops
	if = { limit = { NOT = { has_variable = zz_ef_bankm_gold } } set_variable = { name = zz_ef_bankm_gold value = 0 } }
	if = { limit = { NOT = { has_variable = zz_ef_bankm_silver } } set_variable = { name = zz_ef_bankm_silver value = 0 } }
	if = { limit = { NOT = { has_variable = zz_ef_popm_gold } } set_variable = { name = zz_ef_popm_gold value = 0 } }

	change_variable = { name = zz_ef_popm_silver add = var:zz_ef_f_mt_pop_in_s }
	# the world line: bought on / sold to the market
	zz_ef_wm_add = { KEY = buy_gold VALUE = var:zz_ef_f_mt_cb_in_g }
	zz_ef_wm_add = { KEY = buy_gold VALUE = var:zz_ef_f_mt_bank_in_g }
	zz_ef_wm_add = { KEY = buy_gold VALUE = var:zz_ef_f_mt_pop_in_g }
	zz_ef_wm_add = { KEY = buy_silver VALUE = var:zz_ef_f_mt_cb_in_s }
	zz_ef_wm_add = { KEY = buy_silver VALUE = var:zz_ef_f_mt_bank_in_s }
	zz_ef_wm_add = { KEY = buy_silver VALUE = var:zz_ef_f_mt_pop_in_s }
	zz_ef_wm_add = { KEY = sell_gold VALUE = var:zz_ef_f_mt_cb_out_g }
	zz_ef_wm_add = { KEY = sell_silver VALUE = var:zz_ef_f_mt_cb_out_s }

	# step 7: the CB buys back from the pops under 40% cover (a pair: metal pops -> CB, new money -> their savings)
	set_variable = { name = zz_ef_f_mt_buyback value = 0 }
	set_variable = { name = zz_ef_f_mt_buyback_money value = 0 }
	if = {
		limit = {
			has_modifier = has_central_bank
			has_variable = money_value_target_1
			var:money_value_target_1 > 0
			OR = {
				has_law = law_type:law_gold_standard
				has_law = law_type:law_silver_standard
				has_law = law_type:law_bimetallism_standard
			}
			has_variable = zz_ef_pop_savings
			any_scope_state = { has_modifier = central_bank_historic_place }
			zz_ef_cb_buyback_units > 0
		}
		set_variable = { name = zz_ef_f_mt_buyback value = zz_ef_cb_buyback_units }
		set_variable = { name = zz_ef_f_mt_buyback_money value = var:zz_ef_f_mt_buyback }
		change_variable = { name = zz_ef_f_mt_buyback_money divide = zz_ef_cb_valuation }
		change_variable = { name = zz_ef_f_mt_buyback_money multiply = zz_ef_cb_buyback_premium }
		random_scope_state = {
			limit = { has_modifier = central_bank_historic_place }
			if = {
				limit = { owner = { has_law = law_type:law_silver_standard } }
				change_variable = { name = silver_state_1 add = owner.var:zz_ef_f_mt_buyback }
			}
			else = {
				change_variable = { name = gold_state_1 add = owner.var:zz_ef_f_mt_buyback }
			}
		}
		if = {
			limit = { has_law = law_type:law_silver_standard }
			change_variable = { name = zz_ef_popm_silver subtract = var:zz_ef_f_mt_buyback }
		}
		else = {
			change_variable = { name = zz_ef_popm_gold subtract = var:zz_ef_f_mt_buyback }
		}
		change_variable = { name = zz_ef_pop_savings add = var:zz_ef_f_mt_buyback_money }
		# R1б (В3, 8.10): paid in new banknotes -- the CB's debt to the pops
		change_variable = { name = zz_ef_notes add = var:zz_ef_f_mt_buyback_money }
	}
	# the CB card (UI.11): its reserves' change by purchases, sales and the buy-back
	if = {
		limit = { NOT = { has_variable = zz_ef_cb_stock_acc } }
		set_variable = { name = zz_ef_cb_stock_acc value = 0 }
	}
	change_variable = { name = zz_ef_cb_stock_acc add = central_bank_reserves }
	change_variable = { name = zz_ef_cb_stock_acc subtract = var:zz_ef_cb_stock_before }
	remove_variable = zz_ef_cb_stock_before

	# В3: the snapshot after this week's own moves
	set_variable = { name = zz_ef_mt_prev_g value = gold_state_native_for_stockpile }
	set_variable = { name = zz_ef_mt_prev_s value = silver_state_native_for_stockpile }
	# the modifiers for the coming week
	set_variable = { name = zz_ef_mt_bank_levels value = zz_ef_bank_levels }
	set_variable = { name = zz_ef_mt_bank_g value = zz_ef_bank_buy_gold_goods }
	set_variable = { name = zz_ef_mt_bank_s value = zz_ef_bank_buy_silver_goods }
	set_variable = { name = zz_ef_mt_bank_mult_g value = zz_ef_bank_mult_gold }
	set_variable = { name = zz_ef_mt_bank_mult_s value = zz_ef_bank_mult_silver }
	# no bank levels -- no building of the type (a level-0 building is gone), so nothing to remove or add
	if = {
		limit = { var:zz_ef_mt_bank_levels > 0 }
		every_scope_building = {
			limit = { is_building_type = building_zz_ef_bank }
			if = { limit = { has_modifier = zz_ef_bank_gold_buy } remove_modifier = zz_ef_bank_gold_buy }
			if = { limit = { has_modifier = zz_ef_bank_silver_buy } remove_modifier = zz_ef_bank_silver_buy }
			add_modifier = {
				name = zz_ef_bank_gold_buy
				multiplier = root.var:zz_ef_mt_bank_mult_g
			}
			add_modifier = {
				name = zz_ef_bank_silver_buy
				multiplier = root.var:zz_ef_mt_bank_mult_s
			}
		}
	}
	set_variable = { name = zz_ef_mt_cb_mult value = 0 }
	set_variable = { name = zz_ef_mt_cb_sell value = 0 }
	if = {
		limit = {
			has_modifier = has_central_bank
			has_variable = money_value_target_1
			var:money_value_target_1 > 0
			OR = {
				has_law = law_type:law_gold_standard
				has_law = law_type:law_silver_standard
				has_law = law_type:law_bimetallism_standard
			}
		}
		set_variable = { name = zz_ef_mt_cb_mult value = zz_ef_cb_buy_mult }
		set_variable = { name = zz_ef_mt_cb_sell value = zz_ef_cb_sell_units }
		if = {
			limit = { has_law = law_type:law_silver_standard }
			change_variable = { name = zz_ef_mt_cb_sell divide = zz_ef_silver_units_per_good }
		}
		else = {
			change_variable = { name = zz_ef_mt_cb_sell divide = zz_ef_gold_units_per_good }
		}
	}
	every_scope_building = {
		limit = { is_building_type = building_bank }
		if = { limit = { has_modifier = zz_ef_cb_metal_buy } remove_modifier = zz_ef_cb_metal_buy }
		if = { limit = { has_modifier = zz_ef_cb_gold_sell } remove_modifier = zz_ef_cb_gold_sell }
		if = { limit = { has_modifier = zz_ef_cb_silver_sell } remove_modifier = zz_ef_cb_silver_sell }
		if = {
			limit = { NOT = { root.var:zz_ef_mt_cb_mult = 0 } }
			add_modifier = {
				name = zz_ef_cb_metal_buy
				multiplier = root.var:zz_ef_mt_cb_mult
			}
		}
		if = {
			limit = { root.var:zz_ef_mt_cb_sell > 0 }
			if = {
				limit = { root = { has_law = law_type:law_silver_standard } }
				add_modifier = {
					name = zz_ef_cb_silver_sell
					multiplier = root.var:zz_ef_mt_cb_sell
				}
			}
			else = {
				add_modifier = {
					name = zz_ef_cb_gold_sell
					multiplier = root.var:zz_ef_mt_cb_sell
				}
			}
		}
	}
	if = {
		limit = {
			OR = {
				is_player = yes
				gdp > 20000000
			}
		}
		if = { limit = { zz_ef_logs_on = yes } debug_log = "EFT|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|cbg # В3 (7.10): the CB's metal now minus last week's snapshot minus the known moves (see zz_ef_metal_week_step).
# var:zz_ef_f_mt_oth_g / _s: this week's rest; logged per country (EFQ) over 1 000 gold / 10 000 silver units, and summed
# into global_var:zz_ef_wm_oth_gold / _silver for the world line.
zz_ef_metal_reconcile = {
	set_variable = { name = zz_ef_f_mt_oth_g value = 0 }
	set_variable = { name = zz_ef_f_mt_oth_s value = 0 }
	if = {
		limit = { has_variable = zz_ef_mt_prev_g }
		set_variable = { name = zz_ef_f_mt_oth_g value = gold_state_native_for_stockpile }
		change_variable = { name = zz_ef_f_mt_oth_g subtract = var:zz_ef_mt_prev_g }
		set_variable = { name = zz_ef_f_mt_oth_s value = silver_state_native_for_stockpile }
		change_variable = { name = zz_ef_f_mt_oth_s subtract = var:zz_ef_mt_prev_s }
		# the clearing at the last receiver, in the standard's metal (gold for gold and bimetallism)
		if = {
			limit = { has_variable = zz_ef_f_hume }
			if = {
				limit = { has_law = law_type:law_silver_standard }
				change_variable = { name = zz_ef_f_mt_oth_s subtract = var:zz_ef_f_hume }
			}
			else = {
				change_variable = { name = zz_ef_f_mt_oth_g subtract = var:zz_ef_f_hume }
			}
		}
		if = {
			limit = { has_variable = zz_ef_mt_start_adj_g }
			change_variable = { name = zz_ef_f_mt_oth_g subtract = var:zz_ef_mt_start_adj_g }
			change_variable = { name = zz_ef_f_mt_oth_s subtract = var:zz_ef_mt_start_adj_s }
		}
		# R2 (8.10): the player's forex buttons -- the metal both CBs moved (scripted_effects/ld_fx_buttons.txt)
		if = {
			limit = { has_variable = zz_ef_mt_fx_g }
			change_variable = { name = zz_ef_f_mt_oth_g subtract = var:zz_ef_mt_fx_g }
			change_variable = { name = zz_ef_f_mt_oth_s subtract = var:zz_ef_mt_fx_s }
		}
		zz_ef_wm_add = { KEY = oth_gold VALUE = var:zz_ef_f_mt_oth_g }
		zz_ef_wm_add = { KEY = oth_silver VALUE = var:zz_ef_f_mt_oth_s }
		if = {
			limit = {
				OR = {
					var:zz_ef_f_mt_oth_g > 1000
					var:zz_ef_f_mt_oth_g < -1000
					var:zz_ef_f_mt_oth_s > 10000
					var:zz_ef_f_mt_oth_s < -10000
				}
			}
			if = { limit = { zz_ef_logs_on = yes } debug_log = "EFQ|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|oth_g [THIS.GetCountry.MakeScope.Var('zz_ef_f_mt_oth_g').GetValue|0]|oth_s [THIS.GetCountry.MakeScope.Var('zz_ef_f_mt_oth_s').GetValue|0]|cbg [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cbm_gold')|0]|cbs [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cbm_silver')|0]|hume [THIS.GetCountry.MakeScope.Var('zz_ef_f_hume').GetValue|0]|std [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_std_code')|0]|cb [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_has_cb')|0]|subj [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_is_subject')|0]" }
		}
	}
	set_variable = { name = zz_ef_mt_start_adj_g value = 0 }
	set_variable = { name = zz_ef_mt_start_adj_s value = 0 }
	if = { limit = { has_variable = zz_ef_mt_fx_g } remove_variable = zz_ef_mt_fx_g }
	if = { limit = { has_variable = zz_ef_mt_fx_s } remove_variable = zz_ef_mt_fx_s }
}
