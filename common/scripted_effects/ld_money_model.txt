#############################################
# the money model, in the engine's money, on
# the budget's own rhythm: a WEEKLY step (bank credit from the central bank
# at the key rate, its interest to the treasury, the ledger of each account's
# change, M0/M1/M2 history) and a monthly one (E&F's currency good, which
# E&F moves monthly; EF.31's modifier). Numbers and the why:
# script_values/ld_money_model_values.txt. Started from
# on_actions/ld_money_model_on_actions.txt.
#
# The engine has no weekly pulse for countries (script_docs on_actions.log:
# monthly and slower only), so the week is a chain: the weekly on_action
# re-triggers itself 7 days later (trigger_event = { on_action days = 7 },
# as LLWA and PSC do). The chain starts right after the engine's weekly
# budget tick: a daily probe (at most 8 days) waits until the treasury
# (gold_reserves - principal) changes. Then every 7-day window holds exactly
# one tick, and the budget read at the step is the tick in the window.
#
# E&F's investement_pool_borrowing (yearly, market owners) and its replacement (the
# private banks' interest on foreign bonds into the pool) -- _archive/ld_privbank_interest/ (R1б, В7).
#############################################

# the issuer paid the interest on its bonds held by other treasuries as an expense line -- replaced
# by the bond ledger, which pays the holder out of the issuer's pool.

# ---- the weekly step --------------------------------------------------------

zz_ef_money_model_step = {
	# Business cash: the sum over the buildings, once a week (run 15: the
	# estimate from the credit limit -- credit - base - 1.7 x GDP -- swung by
	# millions a week, M2 by +-30%).
	set_variable = { name = zz_ef_bcash value = zz_ef_building_cash_sum }
	# the Banks' cash -- the banks' own money, out of M1 / M2
	set_variable = { name = zz_ef_bkcash value = zz_ef_bank_cash_sum }
	# the trade, one measure, once a step
	zz_ef_trade_step = yes
	# the CB's metal rescaled this step, in money -- 0 since R1б.1 (no rescale; the CB card's line)
	set_variable = { name = zz_ef_f_cb_rescale value = 0 }
	# foreign currency in the reserves, before the rescale
	zz_ef_fx_metal_update = yes
	# each foreign currency's weekly change -- read only by the «+X a week» of the CB currencies table (ld_clearing.txt):
	# the vitrine, players only (Д.R3б.3)
	if = {
		limit = { is_player = yes }
		zz_ef_cbfx_week_step = yes
	}
	# CB states taken from a country send their metal back to it
	zz_ef_cb_state_owner_step = yes
	# the issuer pays the week's interest to the treasuries holding its bonds
	# bonds as parts of the engine's debt (generated, scripted_effects/ld_bond_ledger.txt); replaces
	# Е.5's issuer-pays expense line (zz_ef_bond_interest_step)
	zz_ef_bond_ledger_step = yes
	if = {
		limit = { NOT = { has_variable = zz_ef_weeks_run } }
		set_variable = { name = zz_ef_weeks_run value = 0 }
	}
	if = {
		limit = { var:zz_ef_weeks_run < 100 }
		change_variable = { name = zz_ef_weeks_run add = zz_ef_step_weeks_v }
	}
	# the start in balance, once (scripted_effects/ld_metal_accounts.txt) -- the parity stays
	# E&F's historical one (the rate between currencies, the labels), the money is the engine's (one unit for every
	# country); the pops' and CB-less countries' metal now, the CB's 40% of M2 once the money exists. No CB holds a
	# foreign currency at the start (ld_nr_deposits.txt). zz_ef_model_started -- the country's first step is done.
	if = {
		limit = { NOT = { has_variable = zz_ef_model_started } }
		set_variable = zz_ef_model_started
		zz_ef_metal_start_step = yes
	}
	zz_ef_cb_metal_start_step = yes
	# the start's window: a CB not on a metal-backed standard half a year in (a CB with no monetary standard) takes no
	# start metal later -- a standard adopted years on buys its metal
	if = {
		limit = {
			has_variable = zz_ef_cb_start_due
			NOT = { zz_ef_cb_metal_standard = yes }
			var:zz_ef_weeks_run >= 26
		}
		remove_variable = zz_ef_cb_start_due
	}

	# The budget's week, as the engine counted it at its last tick.
	set_variable = { name = zz_ef_f_budget value = zz_ef_budget_week }
	set_variable = { name = zz_ef_f_contrib value = investment_pool_gross_income }
	set_variable = { name = zz_ef_f_transfer value = zz_ef_pool_transfer_week }
	# a step after a gap -- the flows measured for one week x the weeks the step covers
	zz_ef_sw = { F = contrib }
	zz_ef_sw = { F = transfer }

	# Bank credit from the central bank: the debt goes to GDP x a share by the
	# rate, repaid from the pool when
	# over it (at most the debt); interest at the key rate goes
	# pool -> CB -> treasury (the CB's profit).
	if = {
		limit = { NOT = { has_variable = zz_ef_bank_cb_debt } }
		set_variable = { name = zz_ef_bank_cb_debt value = 0 }
	}
	set_variable = { name = zz_ef_f_cb_interest value = zz_ef_cb_interest }
	zz_ef_sw = { F = cb_interest }
	set_variable = { name = zz_ef_f_cb_borrow value = zz_ef_cb_borrow }
	set_variable = { name = zz_ef_f_cb_repay value = zz_ef_cb_repay }
	add_investment_pool = var:zz_ef_f_cb_borrow
	change_variable = { name = zz_ef_bank_cb_debt add = var:zz_ef_f_cb_borrow }
	add_investment_pool = {
		value = var:zz_ef_f_cb_repay
		multiply = -1
	}
	change_variable = { name = zz_ef_bank_cb_debt subtract = var:zz_ef_f_cb_repay }
	# the interest -- the banks pay it (their capital) to the CB, the CB's profit to the treasury
	zz_ef_post_eng_paid = { FROM = pool TO = treasury PAYER = zz_ef_bank_capital V = var:zz_ef_f_cb_interest }

	# consols -- the CB's bonds sold are the treasury's perpetual debt to pops: interest treasury -> pops'
	# savings, buy-back out of the surplus over the engine's limit first (scripted_effects/ld_consols.txt)
	zz_ef_consol_step = yes

	# R2 closing: the treasury's surplus over the engine's limit no longer goes into the pool by itself
	# (Д4.5 removed): the private sector builds on its own and the banks' money; the state adds to
	# the pool by its own decision (E&F's transfert_currency_to_investement_pool_button -- a posting in R4). The flow stays 0.
	set_variable = { name = zz_ef_f_tr_pool value = 0 }
	# EF.48 item 5.
	zz_ef_business_credit_step = yes
	# the pops' deposits = their contributions to the pool by the engine
	zz_ef_pop_deposits_step = yes

	# The CB coined half of the mined gold / silver -- removed: metal is taken from
	# the market only; the mines sell all of it, and whoever needs metal buys it (CB, banks, pops). The mint's log
	# keys (EFR mint / mint_tr) stay 0.
	set_variable = { name = zz_ef_f_mint value = 0 }
	set_variable = { name = zz_ef_f_mint_tr value = 0 }
	# Stage 2, night 6.10 (steps 4, 5, 7): the metal accounts -- the week's purchases of the CB, the banks and the pops,
	# the CB's sale and buy-back, the buildings' modifiers for the coming week (scripted_effects/ld_metal_accounts.txt)
	zz_ef_metal_week_step = yes

	# History rings: a week ago
	# (w1..4), 13 blocks of 4 weeks (q1..13, a year), 20 blocks of 13 weeks
	# (r1..20, 5 years); plus the money in circulation and GDP for the inflation
	# (q only). Filled from last week's value; each macro skips a quantity not
	# yet recorded.
	if = {
		limit = { NOT = { has_variable = zz_ef_ring_q } }
		set_variable = { name = zz_ef_ring_q value = 0 }
		set_variable = { name = zz_ef_ring_r value = 0 }
	}
	zz_ef_ring_w_push = { M = agg0 }
	zz_ef_ring_w_push = { M = agg1 }
	zz_ef_ring_w_push = { M = agg2 }
	zz_ef_ring_w_push = { M = agg3 }
	change_variable = { name = zz_ef_ring_q add = zz_ef_step_weeks_v }
	if = {
		limit = { var:zz_ef_ring_q >= 4 }
		set_variable = { name = zz_ef_ring_q value = 0 }
		zz_ef_ring_q_push = { M = agg0 }
		zz_ef_ring_q_push = { M = agg1 }
		zz_ef_ring_q_push = { M = agg2 }
		zz_ef_ring_q_push = { M = agg3 }
		zz_ef_ring_q_push = { M = circ }
		zz_ef_ring_q_push = { M = gdp }
		# prices settle in the first two months of a game (run 17: the USA's index
		# 144 -> 99 by spring 1836, read as -37% a year): the year ring of prices
		# starts with week 9
		if = {
			limit = { var:zz_ef_weeks_run >= 9 }
			zz_ef_ring_q_push = { M = price }
		}
	}
	change_variable = { name = zz_ef_ring_r add = zz_ef_step_weeks_v }
	if = {
		limit = { var:zz_ef_ring_r >= 13 }
		set_variable = { name = zz_ef_ring_r value = 0 }
		zz_ef_ring_r_push = { M = agg0 }
		zz_ef_ring_r_push = { M = agg1 }
		zz_ef_ring_r_push = { M = agg2 }
		zz_ef_ring_r_push = { M = agg3 }
	}

	# what moves the CB's account (its metal at parity), for the
	# CB card -- each a week, in money; the rest is E&F's own metal operations.
	# Revaluation: the reserves of last step at the new parity minus at the old
	# one = cbm_prev x (parity_prev / parity_now - 1) -- devaluation laws and
	# re-anchors.
	# Hume: the metal zz_ef_cb_hume_step added last week (after last week's
	# ledger, so it is in this week's change), at the parity.
	set_variable = { name = zz_ef_f_cb_reval value = 0 }
	set_variable = { name = zz_ef_f_cb_hume_m value = 0 }
	set_variable = { name = zz_ef_f_cb_stock value = 0 }
	if = {
		limit = {
			has_variable = money_value_target_1
			var:money_value_target_1 > 0
		}
		if = {
			limit = {
				has_variable = zz_ef_prev_cbm
				has_variable = zz_ef_prev_cbval
			}
			set_variable = { name = zz_ef_f_cb_reval value = var:zz_ef_prev_cbval }
			change_variable = { name = zz_ef_f_cb_reval divide = zz_ef_cb_valuation }
			change_variable = { name = zz_ef_f_cb_reval subtract = 1 }
			change_variable = { name = zz_ef_f_cb_reval multiply = var:zz_ef_prev_cbm }
		}
		if = {
			limit = {
				has_variable = zz_ef_f_hume
				zz_ef_cb_metal_money > 0
			}
			set_variable = { name = zz_ef_f_cb_hume_m value = var:zz_ef_f_hume }
			change_variable = { name = zz_ef_f_cb_hume_m divide = zz_ef_cb_valuation }
		}
		# E&F's CB building stored metal since the last step (its monthly stockpiling), at the parity -- a line of the CB card
		if = {
			limit = { has_variable = zz_ef_cb_stock_acc }
			set_variable = { name = zz_ef_f_cb_stock value = var:zz_ef_cb_stock_acc }
			change_variable = { name = zz_ef_f_cb_stock divide = zz_ef_cb_valuation }
		}
		set_variable = { name = zz_ef_prev_cbval value = zz_ef_cb_valuation }
	}
	else_if = {
		limit = { has_variable = zz_ef_prev_cbval }
		remove_variable = zz_ef_prev_cbval
	}
	if = {
		limit = { has_variable = zz_ef_cb_stock_acc }
		remove_variable = zz_ef_cb_stock_acc
	}
	# Week-on-week change of each account (0 in the first week).
	zz_ef_money_ledger_delta = { ACC = pool VALUE = investment_pool }
	zz_ef_money_ledger_delta = { ACC = buildings VALUE = zz_ef_building_cash }
	zz_ef_money_ledger_delta = { ACC = tc VALUE = zz_ef_tc_cash }
	zz_ef_money_ledger_delta = { ACC = treasury VALUE = zz_ef_treasury }
	zz_ef_money_ledger_delta = { ACC = bonds VALUE = zz_ef_bank_bonds }
	zz_ef_money_ledger_delta = { ACC = tbonds VALUE = zz_ef_treasury_bonds }
	zz_ef_money_ledger_delta = { ACC = m2 VALUE = zz_ef_m2 }
	zz_ef_money_ledger_delta = { ACC = cbm VALUE = zz_ef_cb_money }
	# abroad = the net foreign position (bonds + the CB's foreign currency - our currency abroad)
	zz_ef_money_ledger_delta = { ACC = abroad VALUE = zz_ef_abroad_net }
	zz_ef_money_ledger_delta = { ACC = agg0 VALUE = zz_ef_agg_m0 }
	zz_ef_money_ledger_delta = { ACC = agg1 VALUE = zz_ef_agg_m1 }
	zz_ef_money_ledger_delta = { ACC = agg2 VALUE = zz_ef_agg_m2 }
	zz_ef_money_ledger_delta = { ACC = agg3 VALUE = zz_ef_agg_m3 }
	zz_ef_money_ledger_delta = { ACC = circ VALUE = zz_ef_circulation }
	zz_ef_money_ledger_delta = { ACC = gdp VALUE = gdp }
	zz_ef_money_ledger_delta = { ACC = price VALUE = zz_ef_price_index }
	# building levels the government owns (privatization)
	zz_ef_money_ledger_delta = { ACC = govlv VALUE = zz_ef_gov_levels }
	# 4.10 (the money tooltip): pops' savings and deposits, for the cash-at-hand and deposits lines
	zz_ef_money_ledger_delta = { ACC = savings VALUE = zz_ef_pop_savings }
	zz_ef_money_ledger_delta = { ACC = deposits VALUE = zz_ef_pop_deposits }
	# the reconciliation by stocks -- «прочее» (scripted_effects/ld_ledger.txt)
	zz_ef_reconcile = yes

	# Sandbox log (tools/run_vic3_sandbox.ps1 reads debug.log): one line a
	# week for the player and the big economies.
	if = {
		limit = {
			zz_ef_log_country = yes
		}
		if = { limit = { zz_ef_logs_on = yes } debug_log = "EFW|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|m0 [THIS.GetCountry.MakeScope.Var('zz_ef_prev_agg0').GetValue|0]|m1 [THIS.GetCountry.MakeScope.Var('zz_ef_prev_agg1').GetValue|0]|m2 [THIS.GetCountry.MakeScope.Var('zz_ef_prev_agg2').GetValue|0]|m3 [THIS.GetCountry.MakeScope.Var('zz_ef_prev_agg3').GetValue|0]|circ [THIS.GetCountry.MakeScope.Var('zz_ef_prev_circ').GetValue|0]|cbm [THIS.GetCountry.MakeScope.Var('zz_ef_prev_cbm').GetValue|0]|abroad [THIS.GetCountry.MakeScope.Var('zz_ef_prev_abroad').GetValue|0]|bonds_abr [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_foreign_assets')|0]|eng [THIS.GetCountry.MakeScope.Var('zz_ef_prev_m2').GetValue|0]|d_treasury [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_d_treasury')|0]|d_buildings [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_d_buildings')|0]|d_pool [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_d_pool')|0]|d_m2 [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_d_m2')|0]|budget [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_f_budget')|0]|contrib [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_f_contrib')|0]|transfer [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_f_transfer')|0]|cb_borrow [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_f_cb_borrow')|0]|cb_repay [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_f_cb_repay')|0]|cb_debt [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_bank_cb_debt')|0]|gdp [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_raw_gdp')|0]|infl [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_inflation')|3]|g_m3 [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_circ_growth_year')|3]|g_gdp [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_gdp_growth_year')|3]|mv0 	}

	# The GUI bridge (gui/ld_money_hook.gui): the budget's lines outside
	# the accounts (minting, additional income and expenses, diplomacy,
	# treaties, power bloc, supply network fees, tolls, piracy) are GUI-only;
	# the hook reads them for every country in the list and hands them back
	# through scripted_guis/ld_money_hook.txt, which computes the leak.
	# Last week's values are dropped, so a week without the GUI reads 0.
	# The deposit operations the receiver made after the previous step fell
	# into the window just closed: moved to var:zz_ef_w_dep_* for the leak.
	zz_ef_money_window_roll = { F = dep_int }

	zz_ef_money_window_roll = { F = cc_repay }
	zz_ef_money_window_roll = { F = cc_int }
	zz_ef_money_window_roll = { F = cc_net }
	zz_ef_money_window_roll = { F = hume_money }
	# other CBs' deposits put into / drawn out of the pool
	zz_ef_money_window_roll = { F = nr_dep }
	# 4.10 (the cash-at-hand card): what went into pops' savings at the last receiver -- the window the
	# ledger's change of M0 / savings / deposits covers
	zz_ef_money_window_roll = { F = inflow }
	# the clearing's currency at the last receiver, for the abroad card
	zz_ef_money_window_roll = { F = clr_cur_out }
	zz_ef_money_window_roll = { F = clr_fx_in }

	# B steps weekly, its bridge stays every 4th week (its phase); the step takes the latest numbers

		limit = {
			OR = {
				NOT = { zz_ef_role_is_b = yes }
				NOT = { has_variable = zz_ef_week_phase }

		}
	}
	# the bridge's part of the step, on its latest numbers
	zz_ef_bridge_apply = yes
}

# the GUI bridge is transport only. Called by the bridge (gui/ld_money_hook.gui) with
# the GUI-only numbers: scope:ext = the week's budget lines outside the accounts (settled, GetTrendValue), scope:abr =
# their part with other countries, scope:g_* = the engine's values the GUI alone sees. Once per queued step (the
# pending var is consumed; on_start may fire twice). It only writes them into the country's variables
# (var:zz_ef_hk_*, var:zz_ef_g_*); every calculation is the scheduler's step (zz_ef_bridge_apply), which takes the
# latest numbers received -- a week without the bridge (minimized window, multiplayer) reuses the last ones.
zz_ef_money_hook_receive = {
	# Diagnostics for the log: every call, also the ones that find nothing pending.
	if = {
		limit = { zz_ef_logs_on = yes }
		if = {
			limit = { NOT = { has_global_variable = zz_ef_hook_calls } }
			set_global_variable = { name = zz_ef_hook_calls value = 0 }
		}
		change_global_variable = { name = zz_ef_hook_calls add = 1 }
	}
	if = {
		limit = { has_variable = zz_ef_hook_pending }
		remove_variable = zz_ef_hook_pending
		remove_list_global_variable = { name = zz_ef_hook_countries target = this }
		# the bridge's numbers default to 0 here; the widget's AddScope overrides them (the engine's check sees no setter of a
		# scope the GUI passes -- "Event target 'g_…' is used but is never set")
		if = { limit = { NOT = { exists = scope:g_aint } } save_scope_value_as = { name = g_aint value = 0 } }
		if = { limit = { NOT = { exists = scope:g_tfees } } save_scope_value_as = { name = g_tfees value = 0 } }
		if = { limit = { NOT = { exists = scope:g_imp } } save_scope_value_as = { name = g_imp value = 0 } }
		if = { limit = { NOT = { exists = scope:g_exp } } save_scope_value_as = { name = g_exp value = 0 } }
		if = { limit = { NOT = { exists = scope:g_surplus } } save_scope_value_as = { name = g_surplus value = 0 } }
		if = { limit = { NOT = { exists = scope:g_fown } } save_scope_value_as = { name = g_fown value = 0 } }
		if = { limit = { NOT = { exists = scope:g_aown } } save_scope_value_as = { name = g_aown value = 0 } }
		if = { limit = { NOT = { exists = scope:g_poolchg } } save_scope_value_as = { name = g_poolchg value = 0 } }
		if = { limit = { NOT = { exists = scope:g_pcons } } save_scope_value_as = { name = g_pcons value = 0 } }
		if = { limit = { NOT = { exists = scope:g_grate } } save_scope_value_as = { name = g_grate value = 0 } }
		if = { limit = { NOT = { exists = scope:g_maxcred } } save_scope_value_as = { name = g_maxcred value = 0 } }
		if = { limit = { NOT = { exists = scope:g_wgdp } } save_scope_value_as = { name = g_wgdp value = 0 } }
		if = { limit = { NOT = { exists = scope:g_wrate } } save_scope_value_as = { name = g_wrate value = 0 } }
		if = { limit = { NOT = { exists = scope:g_ipay } } save_scope_value_as = { name = g_ipay value = 0 } }
		if = { limit = { NOT = { exists = scope:g_sdebt } } save_scope_value_as = { name = g_sdebt value = 0 } }
		if = { limit = { NOT = { exists = scope:g_sdebtf } } save_scope_value_as = { name = g_sdebtf value = 0 } }
		set_variable = { name = zz_ef_hk_ext value = scope:ext }

# Pops' savings: var:zz_ef_pop_savings (S) = the banknotes they hold (var:zz_ef_notes, M0, the CB's debt -- R1б, В3)
# + their deposits var:zz_ef_pop_deposits (D). R1а.6 (Д.6, the user 7.10: "the pool is money in the bank, a contribution
# to the pool is putting money into a deposit"): the deposits are the pops' contributions to the pool by the engine
# (zz_ef_pop_deposits_step, at the step) -- no money of ours is added to the pool for them (the probe 7.10: our
# add_investment_pool is not in investment_pool_gross_income). The banknotes move only by postings: the consols, the
# CB's buy-back of the pops' metal. The leak (the reconciliation's rest of the week) is «прочее» (ld_ledger.txt), not
# savings. Interest on deposits: the banks' capital -> D. The start (once): the deposits out of the pool as it is,
# the banknotes 0, the banks' metal by their reserve norm.
zz_ef_pop_savings_step = {
	if = {
		limit = { NOT = { has_variable = zz_ef_pop_savings } }
		# the pops' start savings (zz_ef_pop_start_savings) are deposits only: R1б -- the banknotes in
		# circulation (M0, var:zz_ef_notes) start at 0
		# the deposits are the pool as it is (no money into it) -- the norm, at most 90% of the pool
		# (Хвост R1: the rest is the banks' capital); out of the
		# banks' capital the first reconciliation set (the pool less the book)
		set_variable = { name = zz_ef_pop_deposits value = 0 }
		set_variable = { name = zz_ef_t_dep0 value = zz_ef_pop_start_deposits }
		zz_ef_post = { FROM = zz_ef_bank_capital TO = zz_ef_pop_deposits V = var:zz_ef_t_dep0 }
		remove_variable = zz_ef_t_dep0
		set_variable = { name = zz_ef_pop_savings value = var:zz_ef_pop_deposits }
		set_variable = { name = zz_ef_notes value = 0 }
		# the banks' start metal by the reserve norm (scripted_effects/ld_metal_accounts.txt)
		zz_ef_metal_start_banks = yes
	}
	# the leak no longer goes into the savings (it is «прочее»)
	# three guesses of the hotfix removed (_archive/ld_pop_savings_guesses/) -- the pool's
	# unexplained loss credited to the pops' savings as "levels sold", the treasury's event spending as savings, 10% a
	# week of the savings over a norm written off "into wealth"; with the deposits = the engine's contributions there is
	# nothing to drain, and an unexplained change is «прочее».
	# cash <-> deposits by the cash norm -- _archive/ld_cash_norm/ (behaviour -- R7)

	set_variable = { name = zz_ef_f_dep_int value = zz_ef_dep_interest }
	zz_ef_sw = { F = dep_int }
	# credited to the deposits, paid by the banks' capital (no cash leaves the pool)
	zz_ef_post = { FROM = zz_ef_bank_capital TO = zz_ef_pop_deposits V = var:zz_ef_f_dep_int }
	change_variable = { name = zz_ef_pop_savings add = var:zz_ef_f_dep_int }
}

# at the step after the business credit: the pops' contributions to the pool (the engine's gross income
# of the pool, x the weeks of the step) are their deposits -- less the part that services the business credit (it
# repays the banks' loans, zz_ef_f_bc_paid). The pool's private construction (its transfer to the treasury) less the
# banks' new business credit is paid out of the deposits for building levels: deposits -> the private sector
#. The planned economy has no pool -- no deposits.
zz_ef_pop_deposits_step = {
	set_variable = { name = zz_ef_f_dep_contrib value = 0 }
	set_variable = { name = zz_ef_f_dep_build value = 0 }
	if = {
		limit = {
			has_variable = zz_ef_pop_savings
			NOT = { has_law = law_type:law_command_economy }
		}
		set_variable = {
			name = zz_ef_f_dep_contrib
			value = {
				value = zz_ef_v_f_contrib
				subtract = zz_ef_v_f_bc_paid
				min = 0
			}
		}
		change_variable = { name = zz_ef_pop_deposits add = var:zz_ef_f_dep_contrib }
		change_variable = { name = zz_ef_pop_savings add = var:zz_ef_f_dep_contrib }
		set_variable = {
			name = zz_ef_f_dep_build
			value = {
				value = zz_ef_v_f_transfer
				subtract = zz_ef_v_f_bc_new
				min = 0
				max = {
					value = var:zz_ef_pop_deposits
					min = 0
				}
			}
		}
		zz_ef_post = { FROM = zz_ef_pop_deposits TO = zz_ef_pop_claims_private V = var:zz_ef_f_dep_build }
		change_variable = { name = zz_ef_pop_savings subtract = var:zz_ef_f_dep_build }
	}
	# what entered the pops' savings this step (the cash-at-hand card: zz_ef_v_w_inflow)
	set_variable = { name = zz_ef_f_inflow value = var:zz_ef_f_dep_contrib }
	change_variable = { name = zz_ef_f_inflow subtract = var:zz_ef_f_dep_build }
}

zz_ef_world_acc = {
	if = {
		limit = { NOT = { has_global_variable = zz_ef_wld_window } }
		zz_ef_world_window_open = yes
	}
	change_global_variable = { name = zz_ef_wld_waown add = zz_ef_v_g_aown }

	change_global_variable = { name = zz_ef_wld_wimp add = zz_ef_v_f_imp }
	change_global_variable = { name = zz_ef_wld_wfee add = zz_ef_v_f_tfees }
	if = {
		limit = {
			zz_ef_clr_gpm_own > 0
			gdp > 0
		}
		change_global_variable = { name = zz_ef_wld_gpm_wsum add = { value = zz_ef_clr_gpm_own multiply = gdp } }
		change_global_variable = { name = zz_ef_wld_gpm_w add = gdp }
	}
	# the log's world sums (EFG) -- only with the logs on (В4)
	if = {
		limit = { zz_ef_logs_on = yes }
		change_global_variable = { name = zz_ef_wld_n add = 1 }
		change_global_variable = { name = zz_ef_wld_waint add = zz_ef_v_f_aint }
		# the same in the engine's money, every country (its payments abroad are in it), and the
		# signed sums of the flow's parts -- a part whose world sum is not ~0 has a payment without a counterpart
		if = {
			limit = { var:zz_ef_f_ext_net > 0 }
			change_global_variable = { name = zz_ef_wld_in_m add = var:zz_ef_f_ext_net }
		}
		else = {
			change_global_variable = { name = zz_ef_wld_out_m subtract = var:zz_ef_f_ext_net }
		}
		change_global_variable = { name = zz_ef_wld_trade_m add = zz_ef_v_f_trade }
		change_global_variable = { name = zz_ef_wld_abr_m add = { value = zz_ef_v_f_abr subtract = zz_ef_tr_abr_overlap } }
		change_global_variable = { name = zz_ef_wld_div_m add = zz_ef_v_f_div }

			name = zz_ef_wld_bond_m
			add = {
				value = var:zz_ef_f_ext_net
				subtract = zz_ef_v_f_trade
				subtract = zz_ef_v_f_abr
				add = zz_ef_tr_abr_overlap
				subtract = zz_ef_v_f_div
			}
		}
		# the members' flows (gold) their heads settled this week
		change_global_variable = { name = zz_ef_wld_sub add = zz_ef_v_f_clr_sub }
		# the world's average (set by the window block above) is final for this receiver: one pass of the price chain
		set_variable = { name = zz_ef_t_gpm value = zz_ef_clr_gold_per_money }
		if = {
			limit = { var:zz_ef_t_gpm > 0 }
			set_variable = { name = zz_ef_wld_g value = var:zz_ef_f_ext_net }
			change_variable = { name = zz_ef_wld_g multiply = var:zz_ef_t_gpm }
			if = {
				limit = { var:zz_ef_wld_g > 0 }
				change_global_variable = { name = zz_ef_wld_in add = var:zz_ef_wld_g }
				if = {
					limit = { is_subject = yes }
					change_global_variable = { name = zz_ef_wld_subj_in add = var:zz_ef_wld_g }
				}
			}
			else = {
				change_global_variable = { name = zz_ef_wld_out subtract = var:zz_ef_wld_g }
				if = {
					limit = { is_subject = yes }
					change_global_variable = { name = zz_ef_wld_subj_out subtract = var:zz_ef_wld_g }
				}
			}
			if = {
				limit = { NOT = { has_modifier = has_central_bank } }
				if = {
					limit = { var:zz_ef_wld_g < 0 }
					change_variable = { name = zz_ef_wld_g multiply = -1 }
				}
				change_global_variable = { name = zz_ef_wld_nocl add = var:zz_ef_wld_g }
			}
			remove_variable = zz_ef_wld_g
		}
		else = {
			change_global_variable = { name = zz_ef_wld_n0 add = 1 }
		}
		if = {
			limit = {
				has_variable = zz_ef_f_hume
				NOT = { var:zz_ef_f_hume = 0 }
			}
			set_variable = { name = zz_ef_wld_g value = var:zz_ef_f_hume }
			if = {
				limit = { zz_ef_metal_silver = yes }
				change_variable = { name = zz_ef_wld_g multiply = silver_to_gold_rate }
			}
			change_global_variable = { name = zz_ef_wld_hume add = var:zz_ef_wld_g }

				limit = { var:zz_ef_wld_g < 0 }
				change_variable = { name = zz_ef_wld_g multiply = -1 }
			}
			change_global_variable = { name = zz_ef_wld_hume_abs add = var:zz_ef_wld_g }
			remove_variable = zz_ef_wld_g
		}
	}
}

# consumer credit, in the receiver after the savings (its pool
# operations fall into the next week's window, like the deposits). Values:
# script_values/ld_money_model_values.txt.
zz_ef_consumer_credit_step = {
	if = {
		limit = {
			has_modifier = has_central_bank
			has_variable = zz_ef_dependents
			var:zz_ef_dependents > 0
		}
		if = {
			limit = { NOT = { has_variable = zz_ef_cc_debt } }
			set_variable = { name = zz_ef_cc_debt value = 0 }
		}
		set_variable = { name = zz_ef_f_cc_issue value = zz_ef_cc_issue }
		set_variable = { name = zz_ef_f_cc_repay value = zz_ef_cc_repay }
		set_variable = { name = zz_ef_f_cc_int value = zz_ef_cc_interest }
		zz_ef_sw = { F = cc_issue }
		zz_ef_sw = { F = cc_repay }
		zz_ef_sw = { F = cc_int }
		# the interest due is the banks' income (paid now, or added to the debt below) -- their capital
		change_variable = { name = zz_ef_bank_capital add = var:zz_ef_f_cc_int }
		set_variable = { name = zz_ef_f_cc_net value = var:zz_ef_f_cc_issue }
		change_variable = { name = zz_ef_f_cc_net subtract = var:zz_ef_f_cc_repay }
		change_variable = { name = zz_ef_f_cc_net subtract = var:zz_ef_f_cc_int }
		# pops cannot pay more than the floor (values file): cut the repayment,
		# then the interest, which is added to the debt
		set_variable = { name = zz_ef_cc_short value = zz_ef_cc_floor_week }
		change_variable = { name = zz_ef_cc_short subtract = var:zz_ef_f_cc_net }
		if = {
			limit = { var:zz_ef_cc_short > 0 }
			if = {
				limit = { var:zz_ef_cc_short > var:zz_ef_f_cc_repay }
				change_variable = { name = zz_ef_cc_short subtract = var:zz_ef_f_cc_repay }
				set_variable = { name = zz_ef_f_cc_repay value = 0 }
				change_variable = { name = zz_ef_f_cc_int subtract = var:zz_ef_cc_short }
				change_variable = { name = zz_ef_cc_debt add = var:zz_ef_cc_short }
			}
			else = {
				change_variable = { name = zz_ef_f_cc_repay subtract = var:zz_ef_cc_short }
			}
			set_variable = { name = zz_ef_f_cc_net value = zz_ef_cc_floor_week }
		}
		remove_variable = zz_ef_cc_short
		change_variable = { name = zz_ef_cc_debt add = var:zz_ef_f_cc_issue }
		change_variable = { name = zz_ef_cc_debt subtract = var:zz_ef_f_cc_repay }
		# pool side
		add_investment_pool = {
			value = var:zz_ef_f_cc_net
			multiply = -1
		}
	}
}

# The dependents' surcharge for the coming week: the consumer credit's net
# flow (zz_ef_cc_wage_add). Every country with dependents: without a CB it is
# 0 and clears the surcharge of the savings spending tried in run 14.
zz_ef_pop_extra_income_step = {
	if = {

			var:zz_ef_dependents > 0
		}
		if = { limit = { has_modifier = zz_ef_consumer_credit } remove_modifier = zz_ef_consumer_credit }
		add_modifier = {
			name = zz_ef_consumer_credit
			multiplier = zz_ef_cc_wage_add
		}
	}
}

}

# A seller redeems a currency in crisis for the
# issuer's metal (sell_<currency>_currency_crisis, 01_economic_scripted_effects.txt). SELLER scope; var:zz_ef_crisis_due =
# the holdings at the issuer's parity, in the issuer's metal. The issuer pays
# from its CB capital's stock, at most what it holds; the seller gets it in
# its own standard's metal at the market rate (silver_to_gold_rate). Was:
# the issuer's metal set to 0, the sellers' stocks to 0, nobody paid.
zz_ef_crisis_redeem = {
	set_variable = { name = zz_ef_crisis_paid value = 0 }
	set_variable = { name = zz_ef_crisis_got value = 0 }
	if = {
		limit = {
			scope:overrated_currency_country_owner = { zz_ef_metal_silver = yes }
			scope:central_bank_site_overrated_currency_owner = { has_variable = silver_state_1 }
		}
		set_variable = {
			name = zz_ef_crisis_paid
			value = {
				value = var:zz_ef_crisis_due
				max = scope:central_bank_site_overrated_currency_owner.var:silver_state_1
				min = 0
			}
		}
		scope:central_bank_site_overrated_currency_owner = {
			change_variable = { name = silver_state_1 subtract = scope:seller_root.var:zz_ef_crisis_paid }
		}
		set_variable = { name = zz_ef_crisis_got value = var:zz_ef_crisis_paid }
		if = {
			limit = { NOT = { zz_ef_metal_silver = yes } }
			change_variable = { name = zz_ef_crisis_got multiply = silver_to_gold_rate }
		}
	}
	else_if = {
		limit = {
			scope:overrated_currency_country_owner = { NOT = { has_law = law_type:law_fiat_standard } }
			scope:central_bank_site_overrated_currency_owner = { has_variable = gold_state_1 }
		}
		set_variable = {
			name = zz_ef_crisis_paid
			value = {
				value = var:zz_ef_crisis_due
				max = scope:central_bank_site_overrated_currency_owner.var:gold_state_1
				min = 0
			}
		}
		scope:central_bank_site_overrated_currency_owner = {
			change_variable = { name = gold_state_1 subtract = scope:seller_root.var:zz_ef_crisis_paid }
		}
		set_variable = { name = zz_ef_crisis_got value = var:zz_ef_crisis_paid }
		if = {
			limit = { zz_ef_metal_silver = yes }
			change_variable = { name = zz_ef_crisis_got multiply = gold_to_silver_rate }
		}
	}
	scope:central_bank_site_root = {
		if = {
			limit = { owner = { zz_ef_metal_silver = yes } }
			if = {
				limit = { NOT = { has_variable = silver_state_1 } }
				set_variable = { name = silver_state_1 value = 0 }
			}
			change_variable = { name = silver_state_1 add = scope:seller_root.var:zz_ef_crisis_got }
		}
		else = {
			if = {
				limit = { NOT = { has_variable = gold_state_1 } }
				set_variable = { name = gold_state_1 value = 0 }
			}
			change_variable = { name = gold_state_1 add = scope:seller_root.var:zz_ef_crisis_got }
		}
	}
	if = {
		limit = { var:zz_ef_crisis_due > 0 }
		if = { limit = { zz_ef_logs_on = yes } debug_log = "EFC|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|due [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_crisis_due')|0]|paid [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_crisis_paid')|0]" }
		scope:overrated_currency_country_owner = {
			if = { limit = { zz_ef_logs_on = yes } debug_log = "EFC|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|issuer|crisis [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_crisis_flag')|0]|std [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_std_code')|0]|mv [THIS.GetCountry.MakeScope.ScriptValue('money_value')|4]|mv0 [THIS.GetCountry.MakeScope.ScriptValue('money_value_0')|4]|vtp [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_value_to_parity')|3]|cover [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cb_cover')|3]" }
		}
	}
}

# business credit with repayment (the user, 1.10 evening:
# "repayment in any case, the point is a realistic money turnover"). Until
# now only values existed and the debt was never kept (always 0). Private
# construction paid with the banks' borrowed money (the CB's credit and pops'
# deposits, by their share of the pool) becomes the businesses' debt; due a
# week: 1/260 of it (5 years) + interest (key + 3 pp). It is paid from the
# buildings' contributions to the pool -- real money, business cash -> pool,
# out of M1. When the contributions fall short, zz_ef_debt_service raises the
# owners' contribution share (profits go to the bank instead of dividends),
# a step a week up to +0.3; when they cover the due by 20% it steps down.
# What is still unpaid is added to the debt.
zz_ef_business_credit_step = {
	if = {
		limit = { NOT = { has_variable = zz_ef_bc_debt } }
		set_variable = { name = zz_ef_bc_debt value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = zz_ef_bc_svc } }
		set_variable = { name = zz_ef_bc_svc value = 0 }
	}
	set_variable = { name = zz_ef_f_bc_new value = zz_ef_bc_new }
	set_variable = { name = zz_ef_f_bc_int value = zz_ef_bc_interest }
	set_variable = { name = zz_ef_f_bc_due value = zz_ef_bc_due }
	set_variable = { name = zz_ef_f_bc_paid value = zz_ef_bc_paid }
	# a step after a gap -- the flows measured for one week x the weeks the step covers
	if = {
		limit = { var:zz_ef_step_weeks > 1 }
		zz_ef_sw = { F = bc_int }
		zz_ef_sw = { F = bc_due }
		set_variable = { name = zz_ef_f_bc_paid value = { value = var:zz_ef_f_bc_due max = { value = zz_ef_v_f_contrib min = 0 } } }
	}
	change_variable = { name = zz_ef_bc_debt add = var:zz_ef_f_bc_new }
	change_variable = { name = zz_ef_bc_debt add = var:zz_ef_f_bc_int }
	# the interest accrues -- the banks' claim grows, so does their capital (ld_ledger.txt, the banks' book)
	change_variable = { name = zz_ef_bank_capital add = var:zz_ef_f_bc_int }
	change_variable = { name = zz_ef_bc_debt subtract = var:zz_ef_f_bc_paid }
	if = {
		limit = { var:zz_ef_bc_debt < 0 }
		set_variable = { name = zz_ef_bc_debt value = 0 }
	}
	# the controller of the owners' contribution share
	if = {
		limit = {
			var:zz_ef_f_bc_due > zz_ef_v_f_contrib
			var:zz_ef_bc_svc < zz_ef_bc_svc_max
		}
		change_variable = { name = zz_ef_bc_svc add = 1 }
	}
	else_if = {
		limit = {
			var:zz_ef_bc_svc > 0
			zz_ef_bc_contrib_slack > 0
		}
		change_variable = { name = zz_ef_bc_svc subtract = 1 }
	}
	if = { limit = { has_modifier = zz_ef_debt_service } remove_modifier = zz_ef_debt_service }
	if = {
		limit = { var:zz_ef_bc_svc > 0 }
		add_modifier = {
			name = zz_ef_debt_service
			multiplier = var:zz_ef_bc_svc
		}
	}
}

# Night 2, item 5: the government's rate to key + premium (static_
# modifiers/ld_gov_rate.txt, values file). Ours cancels the other
# multipliers (total 1) and tops the adds up to the target.
zz_ef_gov_rate_step = {
	set_variable = { name = zz_ef_gr_target value = zz_ef_gov_rate_target }
	set_variable = { name = zz_ef_gr_mult_new value = zz_ef_gov_rate_mult_other }
	change_variable = { name = zz_ef_gr_mult_new multiply = -1 }
	set_variable = { name = zz_ef_gr_add_new value = var:zz_ef_gr_target }
	# the engine's rate stayed 11-18% under the target for France and Russia
	# (0.0727 at grt 0.0873) while modifier:country_loan_interest_rate_mult read 0 -- something the script does not see
	# (E&F's banks' prosperity modifiers, -0.05 each?) scales the whole rate. Calibrated by the fact: the target over
	# the engine's own factor (zz_ef_gr_k, weekly: its yearly rate / the adds).
	change_variable = { name = zz_ef_gr_add_new divide = zz_ef_gr_k_v }
	change_variable = { name = zz_ef_gr_add_new subtract = zz_ef_gov_rate_add_other }
	if = { limit = { has_modifier = zz_ef_gov_rate_mult } remove_modifier = zz_ef_gov_rate_mult }
	if = { limit = { has_modifier = zz_ef_gov_rate_add } remove_modifier = zz_ef_gov_rate_add }
	set_variable = { name = zz_ef_gr_mult value = var:zz_ef_gr_mult_new }
	set_variable = { name = zz_ef_gr_add value = var:zz_ef_gr_add_new }
	set_variable = { name = zz_ef_gr_mult_units value = var:zz_ef_gr_mult }
	change_variable = { name = zz_ef_gr_mult_units multiply = 100 }
	set_variable = { name = zz_ef_gr_add_units value = var:zz_ef_gr_add }
	change_variable = { name = zz_ef_gr_add_units multiply = 1000 }
	add_modifier = {
		name = zz_ef_gov_rate_mult
		multiplier = var:zz_ef_gr_mult_units
	}
	add_modifier = {
		name = zz_ef_gov_rate_add
		multiplier = var:zz_ef_gr_add_units
	}
	remove_variable = zz_ef_gr_mult_new
	remove_variable = zz_ef_gr_add_new
}
