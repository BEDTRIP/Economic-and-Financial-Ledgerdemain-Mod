# Any scope, the world's week (zz_ef_world_week_close): the pops', the banks' and the monetary authority's metal are
# the country's accounts (var:zz_ef_popm_* / zz_ef_bankm_* / zz_ef_am_*) -- a country that is gone (annexed, a revolt put down, a union) took them with it,
# and the world's line of metal lost them.
# No on_action marks a country gone, but a gone country stays an object with its variables (exists = yes,
# is_country_alive = no). Every living country is in global list zz_ef_mh_list and keeps its capital's state region
# (var:zz_ef_mh_region -- a region stays, a state may not: annexation merges split states); a gone one's accounts go,
# as they are, to the owner of a state of that region (zeroed with the gone one). In the world's pass, not the
# countries' steps: a country of role V does not step, as keeper or as heir.
zz_ef_metal_heir_world = {
	every_in_global_list = {
		variable = zz_ef_mh_list
		limit = { is_country_alive = no }
		save_scope_as = zz_ef_mh_dead
		# the gone country and, next line, its heir (the sandbox script sits down as the heir when the player's country is
		# gone, Д.R4б.34)
		if = { limit = { zz_ef_logs_on = yes } debug_log = "EFH|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|gone|tag [THIS.GetCountry.GetTagName]" }
		# the heir -- not a decentralized nation: the world's line does not see its accounts; none -- it waits (counted
		# with the world as a gone one's)
		if = {
			limit = {
				has_variable = zz_ef_mh_region
				var:zz_ef_mh_region = {
					any_scope_state = { owner = { NOT = { is_country_type = decentralized } } }
				}
			}
			var:zz_ef_mh_region = {
				random_scope_state = {
					limit = { owner = { NOT = { is_country_type = decentralized } } }
					owner = {
						if = { limit = { zz_ef_logs_on = yes } debug_log = "EFH|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|heir|tag [THIS.GetCountry.GetTagName]" }
						# the engine hands the gone country's debt over too (a revolt put down): the heir's next two reconciliations
						# take a rise of its debt as inherited, not «прочее» (zz_ef_reconcile)
						set_variable = { name = zz_ef_heir_debt_steps value = 2 }
						# and its treasury (a rising put down, a won revolution merged back): its last treasury net of debt, the
						# heir's next two reconciliations take a rise of the treasury up to it as inherited (zz_ef_heir_tr)
						if = {
							limit = {
								scope:zz_ef_mh_dead = {
									has_variable = zz_ef_tr_prev
									var:zz_ef_tr_prev > 0
									# taken at once by the heir whose treasury rose before this pass (zz_ef_reconcile)
									NOT = { has_variable = zz_ef_mh_tr_taken }
								}
							}
							if = { limit = { NOT = { has_variable = zz_ef_heir_tr_due } } set_variable = { name = zz_ef_heir_tr_due value = 0 } }
							change_variable = { name = zz_ef_heir_tr_due add = scope:zz_ef_mh_dead.var:zz_ef_tr_prev }
						}
						zz_ef_metal_heir_take = { V = popm_gold }
						zz_ef_metal_heir_take = { V = popm_silver }
						zz_ef_metal_heir_take = { V = bankm_gold }
						zz_ef_metal_heir_take = { V = bankm_silver }
						# the authority's metal is a known move of the heir's reconciliation (zz_ef_metal_reconcile)
						if = {
							limit = { scope:zz_ef_mh_dead = { has_variable = zz_ef_am_gold } }
							zz_ef_mt_known_add = { M = g V = scope:zz_ef_mh_dead.var:zz_ef_am_gold }
						}
						if = {
							limit = { scope:zz_ef_mh_dead = { has_variable = zz_ef_am_silver } }
							zz_ef_mt_known_add = { M = s V = scope:zz_ef_mh_dead.var:zz_ef_am_silver }
						}
						zz_ef_metal_heir_take = { V = am_gold }
						zz_ef_metal_heir_take = { V = am_silver }
						# the treasury's hoard (Д.R4а.13)
						zz_ef_metal_heir_take = { V = trm_gold }
						zz_ef_metal_heir_take = { V = trm_silver }
						# its claims and the claims on its money (Д.R4а.8, ld_claims.txt)
						zz_ef_rq_heir_take = yes
						if = { limit = { zz_ef_logs_on = yes } debug_log = "EFM|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|metal_heir|pop_g [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_popm_gold')|0]|pop_s [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_popm_silver')|0]|bank_g [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_bankm_gold')|0]|bank_s [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_bankm_silver')|0]" }
					}
				}
			}
			remove_list_global_variable = { name = zz_ef_mh_list target = scope:zz_ef_mh_dead }
		}
	}
	every_country = {
		limit = { exists = capital }
		zz_ef_mh_keep = yes
	}
}
# Country scope: in the list of the countries with metal accounts, its capital's region kept (zz_ef_metal_heir_world);
# also at its metal start (ld_metal_accounts.txt) -- a country gone before the world's next pass is in the list.
zz_ef_mh_keep = {
	if = {
		limit = { exists = capital }
		set_variable = { name = zz_ef_mh_region value = capital.state_region }
	}
	if = {
		limit = { NOT = { is_target_in_global_variable_list = { name = zz_ef_mh_list target = this } } }
		add_to_global_variable_list = { name = zz_ef_mh_list target = this }
	}
}
# Country scope (the heir), scope:zz_ef_mh_dead -- the gone country: var:zz_ef_$V$ += its account, the account -> 0.
zz_ef_metal_heir_take = {
	if = {
		limit = { scope:zz_ef_mh_dead = { has_variable = zz_ef_$V$ } }
		if = {
			limit = { NOT = { has_variable = zz_ef_$V$ } }
			set_variable = { name = zz_ef_$V$ value = 0 }
		}
		change_variable = { name = zz_ef_$V$ add = scope:zz_ef_mh_dead.var:zz_ef_$V$ }
		scope:zz_ef_mh_dead = { set_variable = { name = zz_ef_$V$ value = 0 } }
	}
}
