# The construction regulator's purchase by the market (Д.R8в.24). Values -- script_values/ld_construction_market_values.txt;
# docs/construction.md.

# Scope: state with the regulator. Weekly, before the state's spending turns into points (set_construction_point_demand):
# the shares of the four goods move toward the market's split (supply x (cheapest / price)^2), the regulator's input of
# each good is its share of 10 units a point (the method takes 10 of each, goods_input_<good>_mult = share - 1). The
# points the state's spending buys -- PSC's price model at the mix's base price (state_construction_allocation).
zz_ef_reg_market_step = {
	save_temporary_scope_value_as = { name = zz_ef_reg_pmin value = zz_ef_reg_pmin }
	set_local_variable = {
		name = zz_ef_reg_w_sum
		value = {
			value = zz_ef_reg_w_wood
			add = zz_ef_reg_w_iron
			add = zz_ef_reg_w_steel
			add = zz_ef_reg_w_arc
		}
	}
	if = {
		limit = { local_var:zz_ef_reg_w_sum > 0 }
		zz_ef_reg_share_move = { G = wood }
		zz_ef_reg_share_move = { G = iron }
		zz_ef_reg_share_move = { G = steel }
		zz_ef_reg_share_move = { G = arc }
	}
	else = {
		# nothing on offer: wood, as the regulator's default method
		set_variable = { name = zz_ef_reg_s_wood value = 1 }
		set_variable = { name = zz_ef_reg_s_iron value = 0 }
		set_variable = { name = zz_ef_reg_s_steel value = 0 }
		set_variable = { name = zz_ef_reg_s_arc value = 0 }
	}
	# the multipliers are local variables of the state: the regulator's block reads them (as PSC's
	# construction_throughput_mult); a script value or a variable of the state read from inside the building is 'none'
	zz_ef_reg_mult_local = { G = wood }
	zz_ef_reg_mult_local = { G = iron }
	zz_ef_reg_mult_local = { G = steel }
	zz_ef_reg_mult_local = { G = arc }
	b:building_construction_regulator = {
		zz_ef_reg_input_set = { G = wood }
		zz_ef_reg_input_set = { G = iron }
		zz_ef_reg_input_set = { G = steel }
		zz_ef_reg_input_set = { G = arc }
	}
}

# Scope: state. $G$'s share one step toward its market share (no variable yet -- at once).
zz_ef_reg_share_move = {
	set_local_variable = {
		name = zz_ef_reg_t
		value = {
			value = zz_ef_reg_w_$G$
			divide = local_var:zz_ef_reg_w_sum
		}
	}
	if = {
		limit = { has_variable = zz_ef_reg_s_$G$ }
		set_variable = {
			name = zz_ef_reg_s_$G$
			value = {
				value = local_var:zz_ef_reg_t
				subtract = var:zz_ef_reg_s_$G$
				multiply = zz_ef_reg_step
				add = var:zz_ef_reg_s_$G$
			}
		}
	}
	else = {
		set_variable = { name = zz_ef_reg_s_$G$ value = local_var:zz_ef_reg_t }
	}
}

# Scope: state. The multiplier of $G$'s input: its share - 1.
zz_ef_reg_mult_local = {
	set_local_variable = {
		name = zz_ef_reg_m_$G$
		value = {
			value = var:zz_ef_reg_s_$G$
			subtract = 1
		}
	}
}

# Scope: the regulator. Its input of $G$ = the share x the method's 10 units a point.
zz_ef_reg_input_set = {
	if = {
		limit = { has_modifier = zz_ef_reg_in_$G$ }
		remove_modifier = zz_ef_reg_in_$G$
	}
	add_modifier = {
		name = zz_ef_reg_in_$G$
		multiplier = local_var:zz_ef_reg_m_$G$
	}
}

# Scope: country, at the end of PSC's week (set_construction_point_demand). Log EFC: the construction spending, what the
# states got of it, the points it bought, the rest not given out, the construction goods made, the government part.
zz_ef_reg_log = {
	if = {
		limit = { zz_ef_logs_on = yes }
		debug_log = "EFC|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|total [THIS.GetCountry.MakeScope.Var('total_construction_spending').GetValue|0]|alloc [THIS.GetCountry.MakeScope.Var('zz_ef_c_alloc').GetValue|0]|pts [THIS.GetCountry.MakeScope.Var('zz_ef_c_pts').GetValue|1]|rest [THIS.GetCountry.MakeScope.Var('psc_remaining_construction_spending').GetValue|0]|prod [THIS.GetCountry.MakeScope.Var('national_production').GetValue|0]|gov [THIS.GetCountry.MakeScope.Var('limited_government_construction_spending').GetValue|0]|wtot [THIS.GetCountry.MakeScope.Var('weighted_total').GetValue|3]"
	}
	remove_variable = zz_ef_c_alloc
	remove_variable = zz_ef_c_pts
}
