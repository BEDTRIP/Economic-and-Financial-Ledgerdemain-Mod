# The clearing of payments with abroad (R8в, шаг 2; понятия «Клиринг», «Зачёт по группам металла»; Д.R8в.2, Д.R8в.4,
# Д.R8в.11). No house and no head: each country with money of its own pays its receivers itself.
#   the week (the country's step, zz_ef_clr_step): the flow abroad of the week (var:zz_ef_f_ext_net, money) in gold at
#     the country's money against its parity -- its position (var:zz_ef_clr_pos, + to receive / - to pay) and its group
#     of metal (var:zz_ef_clr_gc); a country with no money of its own -- the world line (global_var:zz_ef_clr_wl, gold);
#   the world's close of the week (zz_ef_clr_settle, from zz_ef_world_week_close): the positions netted inside each
#     group of metal (gold, silver, bimetallism by its legal ratio, fiat) -- every payer of the group pays every
#     receiver of the group a share proportional to their positions; the groups' rests across the groups, as far as
#     the world's rests match, less the payer's exchange fee; what the world does not match is not settled (logged);
#   a payment (zz_ef_clr_pair) -- a set (Д.R8в.11): n of it in the payer's money (the receiver's claim on the payer,
#     zz_ef_rq_cb_m / zz_ef_rq_bk_m; n by the strength of the payer's money -- 1 - zz_ef_clr_metal_share), the rest by the
#     receiver's standard: a metal standard -- metal out of the payer's reserves; then the receiver's money out of the
#     payer's reserves; the rest -- the week's debt: the receiver's claim on the payer, and the payer bids for the
#     receiver's money on its exchange at its next step (zz_ef_clr_post_needs, ld_exchange_trade.txt);
#   the monetary authority: the CB (its metal in its capital's CB state, its claims -- cb); with no CB -- the banks
#     (var:zz_ef_bankm_gold / _silver, claims -- bk);
#   the results reach the country's next step (accumulators var:zz_ef_clr_a_<n>, zz_ef_clr_take_acc / _take_hume).
# The receiver sells the claims it does not want on the exchange (R8б). docs/clearing-fx.md.

# ---- the country's week ----------------------------------------------------------------------------------------------

# COUNTRY scope, from zz_ef_cb_hume_step: the week's flow into the position; the results of the last settlement.
zz_ef_clr_step = {
	zz_ef_clr_take_acc = yes
	set_variable = { name = zz_ef_f_clr_net value = var:zz_ef_f_ext_net }
	if = {
		limit = { zz_ef_clr_gpm_own > 0 }
		set_variable = { name = zz_ef_clr_gpm value = zz_ef_clr_gpm_own }
		set_variable = { name = zz_ef_clr_n value = zz_ef_clr_own_share }
		if = {
			limit = { NOT = { has_variable = zz_ef_clr_pos } }
			set_variable = { name = zz_ef_clr_pos value = 0 }
		}
		change_variable = {
			name = zz_ef_clr_pos
			add = {
				value = var:zz_ef_f_clr_net
				multiply = var:zz_ef_clr_gpm
			}
		}
		zz_ef_clr_group_set = yes
		# the parts the world's pools scale (trade, the fees, the dividends from abroad), gold -- settled with the
		# week's own ratios
		set_variable = { name = zz_ef_clr_x value = { value = var:zz_ef_f_exp_my add = zz_ef_v_f_tfees multiply = var:zz_ef_clr_gpm } }
		zz_ef_clr_add = { N = ce V = var:zz_ef_clr_x }
		set_variable = { name = zz_ef_clr_x value = { value = var:zz_ef_f_imp_my multiply = var:zz_ef_clr_gpm } }
		zz_ef_clr_add = { N = ci V = var:zz_ef_clr_x }
		set_variable = { name = zz_ef_clr_x value = { value = zz_ef_v_g_aown multiply = var:zz_ef_clr_gpm } }
		zz_ef_clr_add = { N = ca V = var:zz_ef_clr_x }
		remove_variable = zz_ef_clr_x
		if = {
			limit = { NOT = { is_target_in_global_variable_list = { name = zz_ef_clr_list target = this } } }
			add_to_global_variable_list = { name = zz_ef_clr_list target = this }
		}
	}
	else = {
		# no money of its own (Д.R8а.2): an explicit world account, at the world's average gold a unit
		if = {
			limit = { NOT = { has_global_variable = zz_ef_clr_wl } }
			set_global_variable = { name = zz_ef_clr_wl value = 0 }
		}
		change_global_variable = {
			name = zz_ef_clr_wl
			add = {
				value = var:zz_ef_f_clr_net
				multiply = zz_ef_clr_gold_per_money
			}
		}
	}
}

# The group of metal (Д.R8в.2): 1 gold, 2 silver, 4 fiat (no metal), 100 + 10 x the legal ratio -- bimetallism (one
# ratio, one group). An exchange standard -- its metal's group; the external one -- its anchor's.
zz_ef_clr_group_set = {
	set_variable = { name = zz_ef_clr_gc value = 4 }
	if = {
		limit = { zz_ef_metal_gold = yes }
		set_variable = { name = zz_ef_clr_gc value = 1 }
	}
	else_if = {
		limit = { zz_ef_metal_silver = yes }
		set_variable = { name = zz_ef_clr_gc value = 2 }
	}
	else_if = {
		limit = { zz_ef_metal_bimet = yes }
		set_variable = {
			name = zz_ef_clr_gc
			value = {
				value = bimetallic_rate_gold_to_silver
				multiply = 10
				add = 0.5
				floor = yes
				add = 100
			}
		}
	}
}

# COUNTRY scope: what the settlements since our last step did (var:zz_ef_clr_a_<n>) -- the week's results var:zz_ef_f_clr_<n>.
# The CB's metal (hume) is taken before the metal's week (zz_ef_clr_take_hume), the reconciliation reads it.
zz_ef_clr_take_acc = {
	zz_ef_clr_take = { N = cur_out }
	zz_ef_clr_take = { N = fx_in }
	zz_ef_clr_take = { N = own_back }
	zz_ef_clr_take = { N = fx_out }
	zz_ef_clr_take = { N = metal_g }
	zz_ef_clr_take = { N = paid }
	zz_ef_clr_take = { N = got }
	zz_ef_clr_take = { N = debt }
	zz_ef_clr_take = { N = unm }
	zz_ef_clr_take = { N = fee }
}
zz_ef_clr_take_hume = {
	set_variable = { name = zz_ef_f_hume value = 0 }
	if = {
		limit = { has_variable = zz_ef_clr_a_hume }
		set_variable = { name = zz_ef_f_hume value = var:zz_ef_clr_a_hume }
		remove_variable = zz_ef_clr_a_hume
	}
}
zz_ef_clr_take = {
	set_variable = { name = zz_ef_f_clr_$N$ value = 0 }
	if = {
		limit = { has_variable = zz_ef_clr_a_$N$ }
		set_variable = { name = zz_ef_f_clr_$N$ value = var:zz_ef_clr_a_$N$ }
		remove_variable = zz_ef_clr_a_$N$
	}
}
# COUNTRY scope: var:zz_ef_clr_$N$ += $V$
zz_ef_clr_add = {
	if = {
		limit = { NOT = { has_variable = zz_ef_clr_$N$ } }
		set_variable = { name = zz_ef_clr_$N$ value = 0 }
	}
	change_variable = { name = zz_ef_clr_$N$ add = $V$ }
}
# COUNTRY scope: var:zz_ef_clr_a_$N$ += $V$
zz_ef_clr_acc = {
	if = {
		limit = { NOT = { has_variable = zz_ef_clr_a_$N$ } }
		set_variable = { name = zz_ef_clr_a_$N$ value = 0 }
	}
	change_variable = { name = zz_ef_clr_a_$N$ add = $V$ }
}

# ---- the world's settlement ------------------------------------------------------------------------------------------

# Any scope, the end of the scheduler's week (zz_ef_world_week_close): every position of the week settled.
zz_ef_clr_settle = {
	set_global_variable = { name = zz_ef_clr_w_grp value = 0 }
	set_global_variable = { name = zz_ef_clr_w_x value = 0 }
	set_global_variable = { name = zz_ef_clr_w_unp value = 0 }
	set_global_variable = { name = zz_ef_clr_w_unr value = 0 }
	set_global_variable = { name = zz_ef_clr_w_fee value = 0 }
	set_global_variable = { name = zz_ef_clr_w_debt value = 0 }
	set_global_variable = { name = zz_ef_clr_w_metal value = 0 }
	set_global_variable = { name = zz_ef_clr_w_n value = 0 }
	set_global_variable = { name = zz_ef_clr_w_ng value = 0 }
	set_global_variable = { name = zz_ef_clr_rp value = 0 }
	set_global_variable = { name = zz_ef_clr_rm value = 0 }
	if = {
		limit = { NOT = { has_global_variable = zz_ef_clr_wl } }
		set_global_variable = { name = zz_ef_clr_wl value = 0 }
	}
	clear_global_variable_list = zz_ef_clr_reps
	clear_global_variable_list = zz_ef_clr_rr
	# 0. the week's own ratios of the world's pools (zz_ef_world_window_open has just set them from this week's sums) in
	# place of those the countries' steps used (Ч4)
	set_global_variable = { name = zz_ef_clr_w_corr value = 0 }
	every_in_global_list = {
		variable = zz_ef_clr_list
		limit = { has_variable = zz_ef_clr_pos }
		set_variable = {
			name = zz_ef_clr_x
			value = {
				value = zz_ef_wtr_kr_v
				subtract = global_var:zz_ef_wtr_kr_used
				multiply = var:zz_ef_clr_ce
			}
		}
		change_variable = {
			name = zz_ef_clr_x
			subtract = {
				value = zz_ef_wtr_kp_v
				subtract = global_var:zz_ef_wtr_kp_used
				multiply = var:zz_ef_clr_ci
			}
		}
		change_variable = {
			name = zz_ef_clr_x
			add = {
				value = zz_ef_wdv_r_v
				subtract = global_var:zz_ef_wdv_r_used
				multiply = var:zz_ef_clr_ca
			}
		}
		change_variable = { name = zz_ef_clr_pos add = var:zz_ef_clr_x }
		change_global_variable = { name = zz_ef_clr_w_corr add = var:zz_ef_clr_x }
	}
	# 1. the groups: the first country of a group is its record (var:zz_ef_clr_gp / _gm -- the group's receipts and
	# payments, list zz_ef_clr_grl -- its receivers)
	every_in_global_list = {
		variable = zz_ef_clr_list
		limit = {
			has_variable = zz_ef_clr_pos
			NOT = { var:zz_ef_clr_pos = 0 }
		}
		save_temporary_scope_as = zz_ef_clr_c
		change_global_variable = { name = zz_ef_clr_w_n add = 1 }
		if = {
			limit = { has_variable = zz_ef_clr_rep }
			remove_variable = zz_ef_clr_rep
		}
		random_in_global_list = {
			variable = zz_ef_clr_reps
			limit = { var:zz_ef_clr_gc = scope:zz_ef_clr_c.var:zz_ef_clr_gc }
			scope:zz_ef_clr_c = { set_variable = { name = zz_ef_clr_rep value = prev } }
		}
		if = {
			limit = { NOT = { has_variable = zz_ef_clr_rep } }
			add_to_global_variable_list = { name = zz_ef_clr_reps target = this }
			change_global_variable = { name = zz_ef_clr_w_ng add = 1 }
			set_variable = { name = zz_ef_clr_rep value = this }
			set_variable = { name = zz_ef_clr_gp value = 0 }
			set_variable = { name = zz_ef_clr_gm value = 0 }
			clear_variable_list = zz_ef_clr_grl
		}
		var:zz_ef_clr_rep = {
			if = {
				limit = { scope:zz_ef_clr_c.var:zz_ef_clr_pos > 0 }
				change_variable = { name = zz_ef_clr_gp add = scope:zz_ef_clr_c.var:zz_ef_clr_pos }
				add_to_variable_list = { name = zz_ef_clr_grl target = scope:zz_ef_clr_c }
			}
			else = {
				change_variable = { name = zz_ef_clr_gm subtract = scope:zz_ef_clr_c.var:zz_ef_clr_pos }
			}
		}
	}
	# 2. inside a group: min(receipts, payments) is netted -- the payers' share kp, the receivers' kr; the rests to the
	# world
	every_in_global_list = {
		variable = zz_ef_clr_reps
		set_variable = {
			name = zz_ef_clr_gn
			value = {
				value = var:zz_ef_clr_gp
				max = var:zz_ef_clr_gm
			}
		}
		set_variable = { name = zz_ef_clr_kp value = 0 }
		set_variable = { name = zz_ef_clr_kr value = 0 }
		if = {
			limit = { var:zz_ef_clr_gm > 0 }
			set_variable = { name = zz_ef_clr_kp value = { value = var:zz_ef_clr_gn divide = var:zz_ef_clr_gm } }
		}
		if = {
			limit = { var:zz_ef_clr_gp > 0 }
			set_variable = { name = zz_ef_clr_kr value = { value = var:zz_ef_clr_gn divide = var:zz_ef_clr_gp } }
		}
		change_global_variable = { name = zz_ef_clr_rp add = { value = var:zz_ef_clr_gp subtract = var:zz_ef_clr_gn } }
		change_global_variable = { name = zz_ef_clr_rm add = { value = var:zz_ef_clr_gm subtract = var:zz_ef_clr_gn } }
	}
	# the world's rests: as far as they match -- the payers' share global_var:zz_ef_clr_xp, the receivers' _xr
	set_global_variable = { name = zz_ef_clr_xm value = { value = global_var:zz_ef_clr_rp max = global_var:zz_ef_clr_rm } }
	set_global_variable = { name = zz_ef_clr_xp value = 0 }
	set_global_variable = { name = zz_ef_clr_xr value = 0 }
	if = {
		limit = { global_var:zz_ef_clr_rm > 0 }
		set_global_variable = { name = zz_ef_clr_xp value = { value = global_var:zz_ef_clr_xm divide = global_var:zz_ef_clr_rm } }
	}
	if = {
		limit = { global_var:zz_ef_clr_rp > 0 }
		set_global_variable = { name = zz_ef_clr_xr value = { value = global_var:zz_ef_clr_xm divide = global_var:zz_ef_clr_rp } }
	}
	# the receivers' rests (var:zz_ef_clr_res, list zz_ef_clr_rr); what the world does not match -- not received
	every_in_global_list = {
		variable = zz_ef_clr_list
		limit = {
			has_variable = zz_ef_clr_rep
			var:zz_ef_clr_pos > 0
		}
		set_variable = {
			name = zz_ef_clr_res
			value = {
				value = 1
				subtract = var:zz_ef_clr_rep.var:zz_ef_clr_kr
				multiply = var:zz_ef_clr_pos
			}
		}
		if = {
			limit = { var:zz_ef_clr_res > 0 }
			add_to_global_variable_list = { name = zz_ef_clr_rr target = this }
			set_variable = {
				name = zz_ef_clr_x
				value = {
					value = 1
					subtract = global_var:zz_ef_clr_xr
					multiply = var:zz_ef_clr_res
				}
			}
			zz_ef_clr_acc = { N = unm V = var:zz_ef_clr_x }
			change_global_variable = { name = zz_ef_clr_w_unr add = var:zz_ef_clr_x }
		}
	}
	# 3. the payers: inside the group, then the rest across the groups
	every_in_global_list = {
		variable = zz_ef_clr_list
		limit = {
			has_variable = zz_ef_clr_rep
			var:zz_ef_clr_pos < 0
		}
		save_temporary_scope_as = zz_ef_clr_p
		var:zz_ef_clr_rep = { save_temporary_scope_as = zz_ef_clr_g }
		set_variable = { name = zz_ef_clr_d value = { value = var:zz_ef_clr_pos multiply = -1 } }
		if = {
			limit = { scope:zz_ef_clr_g.var:zz_ef_clr_gn > 0 }
			scope:zz_ef_clr_g = {
				every_in_list = {
					variable = zz_ef_clr_grl
					save_temporary_scope_as = zz_ef_clr_r
					scope:zz_ef_clr_p = {
						set_variable = {
							name = zz_ef_clr_a
							value = {
								value = var:zz_ef_clr_d
								multiply = scope:zz_ef_clr_g.var:zz_ef_clr_kp
								multiply = scope:zz_ef_clr_r.var:zz_ef_clr_pos
								divide = scope:zz_ef_clr_g.var:zz_ef_clr_gp
							}
						}
						change_global_variable = { name = zz_ef_clr_w_grp add = var:zz_ef_clr_a }
						zz_ef_clr_pair = yes
					}
				}
			}
		}
		set_variable = {
			name = zz_ef_clr_d
			value = {
				value = 1
				subtract = scope:zz_ef_clr_g.var:zz_ef_clr_kp
				multiply = var:zz_ef_clr_d
			}
		}
		if = {
			limit = { var:zz_ef_clr_d > 0 }
			# not matched by the world -- not paid
			set_variable = {
				name = zz_ef_clr_x
				value = {
					value = 1
					subtract = global_var:zz_ef_clr_xp
					multiply = var:zz_ef_clr_d
					multiply = -1
				}
			}
			zz_ef_clr_acc = { N = unm V = var:zz_ef_clr_x }
			change_global_variable = { name = zz_ef_clr_w_unp subtract = var:zz_ef_clr_x }
			if = {
				limit = { global_var:zz_ef_clr_xp > 0 }
				set_variable = { name = zz_ef_clr_fx value = zz_ef_xch_fee_v }
				every_in_global_list = {
					variable = zz_ef_clr_rr
					save_temporary_scope_as = zz_ef_clr_r
					scope:zz_ef_clr_p = {
						set_variable = {
							name = zz_ef_clr_a
							value = {
								value = var:zz_ef_clr_d
								multiply = global_var:zz_ef_clr_xp
								multiply = scope:zz_ef_clr_r.var:zz_ef_clr_res
								divide = global_var:zz_ef_clr_rp
							}
						}
						# across the groups -- through the exchange: its fee is not paid on
						set_variable = { name = zz_ef_clr_x value = { value = var:zz_ef_clr_a multiply = var:zz_ef_clr_fx } }
						zz_ef_clr_acc = { N = fee V = var:zz_ef_clr_x }
						change_global_variable = { name = zz_ef_clr_w_fee add = var:zz_ef_clr_x }
						change_variable = { name = zz_ef_clr_a subtract = var:zz_ef_clr_x }
						change_global_variable = { name = zz_ef_clr_w_x add = var:zz_ef_clr_a }
						zz_ef_clr_pair = yes
					}
				}
				remove_variable = zz_ef_clr_fx
			}
		}
	}
	if = {
		limit = { zz_ef_logs_on = yes }
		debug_log = "EFN|[TimeKeeper.GetCurrentDate.GetString]|WORLD|n [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_w_n_v')|0]|groups [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_w_ng_v')|0]|grp [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_w_grp_v')|0]|x [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_w_x_v')|0]|fee [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_w_fee_v')|0]|unpaid [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_w_unp_v')|0]|unrec [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_w_unr_v')|0]|metal [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_w_metal_v')|0]|debt [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_w_debt_v')|0]|wl [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_wl_v')|0]|rp [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_rp_v')|0]|rm [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_rm_v')|0]|corr [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_clr_w_corr_v')|0]|mid [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_wld_mid_v')|0]"
	}
	# 4. the week is closed
	every_in_global_list = {
		variable = zz_ef_clr_list
		if = { limit = { has_variable = zz_ef_clr_pos } remove_variable = zz_ef_clr_pos }
		if = { limit = { has_variable = zz_ef_clr_rep } remove_variable = zz_ef_clr_rep }
		if = { limit = { has_variable = zz_ef_clr_d } remove_variable = zz_ef_clr_d }
		if = { limit = { has_variable = zz_ef_clr_res } remove_variable = zz_ef_clr_res }
		if = { limit = { has_variable = zz_ef_clr_a } remove_variable = zz_ef_clr_a }
		if = { limit = { has_variable = zz_ef_clr_x } remove_variable = zz_ef_clr_x }
		if = { limit = { has_variable = zz_ef_clr_ce } remove_variable = zz_ef_clr_ce }
		if = { limit = { has_variable = zz_ef_clr_ci } remove_variable = zz_ef_clr_ci }
		if = { limit = { has_variable = zz_ef_clr_ca } remove_variable = zz_ef_clr_ca }
	}
	clear_global_variable_list = zz_ef_clr_list
	clear_global_variable_list = zz_ef_clr_rr
	set_global_variable = { name = zz_ef_clr_wl value = 0 }
}

# ---- a payment -------------------------------------------------------------------------------------------------------

# PAYER scope (scope:zz_ef_clr_p), scope:zz_ef_clr_r -- the receiver; var:zz_ef_clr_a -- the payment, gold. The set
# (Д.R8в.11): n in the payer's money; the rest -- metal to a metal standard, the receiver's money out of the payer's
# reserves, the week's debt.
zz_ef_clr_pair = {
	zz_ef_clr_acc = { N = paid V = var:zz_ef_clr_a }
	scope:zz_ef_clr_r = { zz_ef_clr_acc = { N = got V = scope:zz_ef_clr_p.var:zz_ef_clr_a } }
	set_variable = { name = zz_ef_clr_x value = { value = var:zz_ef_clr_a multiply = var:zz_ef_clr_n } }
	set_variable = { name = zz_ef_clr_rest value = { value = var:zz_ef_clr_a subtract = var:zz_ef_clr_x } }
	if = {
		limit = { var:zz_ef_clr_x > 0 }
		zz_ef_clr_give_claim = yes
	}
	if = {
		limit = {
			var:zz_ef_clr_rest > 0
			scope:zz_ef_clr_r = { zz_ef_clr_wants_metal = yes }
		}
		zz_ef_clr_give_metal = yes
	}
	if = {
		limit = { var:zz_ef_clr_rest > 0 }
		zz_ef_clr_give_back = yes
	}
	if = {
		limit = { var:zz_ef_clr_rest > 0 }
		set_variable = { name = zz_ef_clr_x value = var:zz_ef_clr_rest }
		zz_ef_clr_give_claim = yes
		zz_ef_clr_acc = { N = debt V = var:zz_ef_clr_rest }
		change_global_variable = { name = zz_ef_clr_w_debt add = var:zz_ef_clr_rest }
		zz_ef_clr_need_add = yes
	}
	remove_variable = zz_ef_clr_rest
}

# The receiver's monetary authority holds claims as cb (a CB with its capital's CB state) or bk (the banks).
zz_ef_clr_give_claim = {
	set_variable = { name = zz_ef_clr_u value = { value = var:zz_ef_clr_x divide = var:zz_ef_clr_gpm } }
	scope:zz_ef_clr_r = {
		set_variable = { name = zz_ef_clr_u value = scope:zz_ef_clr_p.var:zz_ef_clr_u }
		if = {
			limit = { zz_ef_clr_cb_ok = yes }
			zz_ef_rq_add = { H = cb K = m D = scope:zz_ef_clr_p V = var:zz_ef_clr_u }
		}
		else = {
			zz_ef_rq_add = { H = bk K = m D = scope:zz_ef_clr_p V = var:zz_ef_clr_u }
		}
		zz_ef_clr_acc = { N = fx_in V = scope:zz_ef_clr_p.var:zz_ef_clr_x }
		remove_variable = zz_ef_clr_u
	}
	zz_ef_clr_acc = { N = cur_out V = var:zz_ef_clr_u }
	remove_variable = zz_ef_clr_u
}

# Metal out of the payer's reserves (gold value var:zz_ef_clr_mg, at most what it has) into the receiver's, each in its
# own metal at the world's price (silver_to_gold_rate).
zz_ef_clr_give_metal = {
	set_variable = {
		name = zz_ef_clr_mg
		value = {
			value = zz_ef_clr_metal_av_g
			max = var:zz_ef_clr_rest
			min = 0
		}
	}
	if = {
		limit = { var:zz_ef_clr_mg > 0 }
		zz_ef_clr_metal_move = { SIGN = -1 }
		scope:zz_ef_clr_r = {
			set_variable = { name = zz_ef_clr_mg value = scope:zz_ef_clr_p.var:zz_ef_clr_mg }
			zz_ef_clr_metal_move = { SIGN = 1 }
			zz_ef_clr_acc = { N = metal_g V = var:zz_ef_clr_mg }
			remove_variable = zz_ef_clr_mg
		}
		change_global_variable = { name = zz_ef_clr_w_metal add = var:zz_ef_clr_mg }
		set_variable = { name = zz_ef_clr_x value = { value = var:zz_ef_clr_mg multiply = -1 } }
		zz_ef_clr_acc = { N = metal_g V = var:zz_ef_clr_x }
		change_variable = { name = zz_ef_clr_rest subtract = var:zz_ef_clr_mg }
	}
	remove_variable = zz_ef_clr_mg
}
# COUNTRY scope: var:zz_ef_clr_mg gold of metal in ($SIGN$ = 1) / out (-1). The CB -- its standard's metal in its
# capital's CB state (silver for silver, else gold); the banks -- gold first out, their standard's metal in.
zz_ef_clr_metal_move = {
	if = {
		limit = { zz_ef_clr_cb_ok = yes }
		set_variable = { name = zz_ef_clr_mn value = { value = var:zz_ef_clr_mg multiply = $SIGN$ } }
		if = {
			limit = { zz_ef_metal_silver = yes }
			change_variable = { name = zz_ef_clr_mn multiply = gold_to_silver_rate }
		}
		every_scope_state = {
			limit = {
				has_modifier = central_bank_historic_place
				is_capital = yes
			}
			if = {
				limit = { owner = { zz_ef_metal_silver = yes } }
				if = {
					limit = { NOT = { has_variable = silver_state_1 } }
					set_variable = { name = silver_state_1 value = 0 }
				}
				change_variable = { name = silver_state_1 add = owner.var:zz_ef_clr_mn }
			}
			else = {
				if = {
					limit = { NOT = { has_variable = gold_state_1 } }
					set_variable = { name = gold_state_1 value = 0 }
				}
				change_variable = { name = gold_state_1 add = owner.var:zz_ef_clr_mn }
			}
		}
		zz_ef_clr_acc = { N = hume V = var:zz_ef_clr_mn }
		remove_variable = zz_ef_clr_mn
	}
	else = {
		if = {
			limit = { NOT = { has_variable = zz_ef_bankm_gold } }
			set_variable = { name = zz_ef_bankm_gold value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = zz_ef_bankm_silver } }
			set_variable = { name = zz_ef_bankm_silver value = 0 }
		}
		set_variable = { name = zz_ef_clr_mn value = $SIGN$ }
		if = {
			limit = { var:zz_ef_clr_mn > 0 }
			if = {
				limit = { zz_ef_metal_silver = yes }
				change_variable = { name = zz_ef_bankm_silver add = { value = var:zz_ef_clr_mg multiply = gold_to_silver_rate } }
			}
			else = {
				change_variable = { name = zz_ef_bankm_gold add = var:zz_ef_clr_mg }
			}
			remove_variable = zz_ef_clr_mn
		}
		else = {
			set_variable = { name = zz_ef_clr_mn value = { value = var:zz_ef_clr_mg max = var:zz_ef_bankm_gold } }
			change_variable = { name = zz_ef_bankm_gold subtract = var:zz_ef_clr_mn }
			change_variable = {
				name = zz_ef_bankm_silver
				subtract = {
					value = var:zz_ef_clr_mg
					subtract = var:zz_ef_clr_mn
					multiply = gold_to_silver_rate
				}
			}
			remove_variable = zz_ef_clr_mn
		}
	}
}

# The receiver's money out of the payer's reserves: its claim comes home to the receiver and is gone.
zz_ef_clr_give_back = {
	if = {
		limit = { zz_ef_clr_cb_ok = yes }
		zz_ef_rq_get = { MAP = zz_ef_rq_cb_m D = scope:zz_ef_clr_r }
	}
	else = {
		zz_ef_rq_get = { MAP = zz_ef_rq_bk_m D = scope:zz_ef_clr_r }
	}
	set_variable = {
		name = zz_ef_clr_u
		value = {
			value = var:zz_ef_clr_rest
			divide = scope:zz_ef_clr_r.var:zz_ef_clr_gpm
			max = var:zz_ef_rq_h
			min = 0
		}
	}
	if = {
		limit = { var:zz_ef_clr_u > 0 }
		set_variable = { name = zz_ef_rq_neg value = { value = var:zz_ef_clr_u multiply = -1 } }
		if = {
			limit = { zz_ef_clr_cb_ok = yes }
			zz_ef_rq_add = { H = cb K = m D = scope:zz_ef_clr_r V = var:zz_ef_rq_neg }
		}
		else = {
			zz_ef_rq_add = { H = bk K = m D = scope:zz_ef_clr_r V = var:zz_ef_rq_neg }
		}
		set_variable = { name = zz_ef_clr_x value = { value = var:zz_ef_clr_u multiply = scope:zz_ef_clr_r.var:zz_ef_clr_gpm } }
		zz_ef_clr_acc = { N = fx_out V = var:zz_ef_clr_x }
		scope:zz_ef_clr_r = { zz_ef_clr_acc = { N = own_back V = scope:zz_ef_clr_p.var:zz_ef_clr_u } }
		change_variable = { name = zz_ef_clr_rest subtract = var:zz_ef_clr_x }
	}
	remove_variable = zz_ef_clr_u
}

# The week's debt in the receiver's money (map zz_ef_clr_need on the payer, units of the receiver's money): the payer
# bids for it on its exchange at its next step (zz_ef_clr_post_needs).
zz_ef_clr_need_add = {
	set_variable = {
		name = zz_ef_clr_u
		value = {
			value = var:zz_ef_clr_rest
			divide = scope:zz_ef_clr_r.var:zz_ef_clr_gpm
		}
	}
	set_variable = { name = zz_ef_clr_h value = 0 }
	if = {
		limit = { is_key_in_variable_map = { name = zz_ef_clr_need target = scope:zz_ef_clr_r } }
		set_variable = { name = zz_ef_clr_h value = "variable_map(zz_ef_clr_need|scope:zz_ef_clr_r)" }
		remove_from_variable_map = { name = zz_ef_clr_need key = scope:zz_ef_clr_r }
	}
	change_variable = { name = zz_ef_clr_h add = var:zz_ef_clr_u }
	add_to_variable_map = { name = zz_ef_clr_need key = scope:zz_ef_clr_r value = zz_ef_clr_h_v }
	remove_variable = zz_ef_clr_u
	remove_variable = zz_ef_clr_h
}

# COUNTRY scope, its exchange step (zz_ef_xch_step, after its own orders): last settlement's debt in each receiver's
# money (map zz_ef_clr_need) -- a bid for it at the top step on the exchange of its route, for the next payments; a
# receiver its authority already bids for this step is left alone. The map is emptied.
zz_ef_clr_post_needs = {
	if = {
		limit = { has_variable_map = zz_ef_clr_need }
		if = {
			limit = {
				has_variable = zz_ef_xch
				has_global_variable = zz_ef_sched_week
				var:zz_ef_xch = { zz_ef_has_exchange = yes }
			}
			var:zz_ef_xch = { save_temporary_scope_as = zz_ef_xch_e }
			set_variable = { name = zz_ef_xo_e value = var:zz_ef_xch }
			set_variable = { name = zz_ef_xo_week value = global_var:zz_ef_sched_week }
			save_temporary_scope_as = zz_ef_clr_p
			every_key_in_variable_map = {
				variable = zz_ef_clr_need
				save_temporary_scope_as = zz_ef_xch_d
				scope:zz_ef_clr_p = {
					if = {
						limit = {
							NOT = { zz_ef_xch_banned = yes }
							NOT = { scope:zz_ef_xch_d = { is_country_type = decentralized } }
						}
						set_variable = { name = zz_ef_xo_a value = "variable_map(zz_ef_clr_need|scope:zz_ef_xch_d)" }
						set_variable = { name = zz_ef_xo_s value = zz_ef_xq_top }
						if = {
							limit = {
								zz_ef_clr_cb_ok = yes
								NOT = { is_key_in_variable_map = { name = zz_ef_xa_cb_m target = scope:zz_ef_xch_d } }
							}
							zz_ef_xch_order_put = { T = cb K = m SIDE = b }
						}
						else_if = {
							limit = {
								NOT = { zz_ef_clr_cb_ok = yes }
								NOT = { is_key_in_variable_map = { name = zz_ef_xa_bk_m target = scope:zz_ef_xch_d } }
							}
							zz_ef_xch_order_put = { T = bk K = m SIDE = b }
						}
					}
				}
			}
		}
		clear_variable_map = zz_ef_clr_need
	}
}

# PLAYER scope, weekly: the week's change of each foreign money the CB holds (map zz_ef_cbfx_d by issuer; last week's
# holdings -- map zz_ef_cbfx_p), for the table of the CB's reserves.
zz_ef_cbfx_week_step = {
	if = { limit = { has_variable_map = zz_ef_cbfx_d } clear_variable_map = zz_ef_cbfx_d }
	if = {
		limit = { has_variable_map = zz_ef_rq_cb_m }
		zz_ef_rq_keys_take = { MAP = zz_ef_rq_cb_m }
		save_temporary_scope_as = zz_ef_cbfx_h
		every_in_list = {
			variable = zz_ef_rq_keys
			save_temporary_scope_as = zz_ef_cbfx_k
			scope:zz_ef_cbfx_h = {
				zz_ef_rq_get = { MAP = zz_ef_cbfx_p D = scope:zz_ef_cbfx_k }
				set_variable = { name = zz_ef_cbfx_t value = var:zz_ef_rq_h }
				zz_ef_rq_get = { MAP = zz_ef_rq_cb_m D = scope:zz_ef_cbfx_k }
				change_variable = { name = zz_ef_cbfx_t subtract = var:zz_ef_rq_h }
				change_variable = { name = zz_ef_cbfx_t multiply = -1 }
				add_to_variable_map = { name = zz_ef_cbfx_d key = scope:zz_ef_cbfx_k value = zz_ef_cbfx_t_v }
			}
		}
	}
	if = { limit = { has_variable_map = zz_ef_cbfx_p } clear_variable_map = zz_ef_cbfx_p }
	if = {
		limit = { has_variable_map = zz_ef_rq_cb_m }
		every_in_list = {
			variable = zz_ef_rq_keys
			save_temporary_scope_as = zz_ef_cbfx_k
			scope:zz_ef_cbfx_h = {
				zz_ef_rq_get = { MAP = zz_ef_rq_cb_m D = scope:zz_ef_cbfx_k }
				add_to_variable_map = { name = zz_ef_cbfx_p key = scope:zz_ef_cbfx_k value = zz_ef_rq_h_v }
			}
		}
		clear_variable_list = zz_ef_rq_keys
	}
	remove_variable = zz_ef_cbfx_t
}
