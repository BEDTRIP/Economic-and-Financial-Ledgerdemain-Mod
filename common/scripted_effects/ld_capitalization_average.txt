#############################################
# monthly update of the capitalization averages. Numbers and the why:
# script_values/ld_capitalization_snapshot.txt (averages),
# scripted_effects/ld_capitalization_crash.txt (crash).
#
# COUNTRY scope, once a month, from on_actions/ld_capitalization_on_actions.txt.
#
# Per country, 5 averages (+ 12 monthly marks in ld_capitalization_crash.txt,
# countries with a stock exchange only):
#   zz_ef_cap_avg_ms/_as/_mn/_rr  each stock's value, moving average, weight 1/12
#   zz_ef_cap_half                all four together, weight 1/6 (crash check)
# avg = avg + weight x (snapshot - avg). A new average starts at the first
# snapshot, not at 0 -- no warm-up, and no false crash on old saves.
#
# The snapshot is computed once into a local variable (a pass over the country's companies).
#
# The exchanges are compared by their turnover, not by points (Д.R4б.11,
# zz_ef_xch_month_roll in ld_exchange_trade.txt).
#############################################

zz_ef_cap_monthly_update = {
	# the capitalization is the listed companies' shares (script_values/ld_listing_values.txt, Д.R6.12); E&F's menu shows
	# it over its four stock goods -- equal parts (the goods are no longer produced: their output no longer splits it).
	# Nothing listed -- 0 for all four.
	set_local_variable = { name = zz_s_new value = zz_ef_cap_listed_snapshot }
	set_local_variable = { name = zz_s_ms value = { value = local_var:zz_s_new multiply = 0.25 } }
	set_local_variable = { name = zz_s_as value = { value = local_var:zz_s_new multiply = 0.25 } }
	set_local_variable = { name = zz_s_mn value = { value = local_var:zz_s_new multiply = 0.25 } }
	set_local_variable = { name = zz_s_rr value = { value = local_var:zz_s_new multiply = 0.25 } }
	set_local_variable = { name = zz_s_total value = local_var:zz_s_new }

	# always -- with nothing listed the snapshot is 0, and a country whose
	# averages had not started showed the old formula (weekly output x 520) through the quantities' fallback
	# (Britain 1836: 2.55 GDP with nothing listed).
	if = {
		limit = { always = yes }
		if = {
			limit = { has_variable = zz_ef_cap_avg_ms }
			set_variable = {
				name = zz_ef_cap_avg_ms
				value = {
					value = var:zz_ef_cap_avg_ms
					add = {
						value = local_var:zz_s_ms
						subtract = var:zz_ef_cap_avg_ms
						multiply = zz_ef_cap_weight_year
					}
				}
			}
		}
		else = {
			set_variable = { name = zz_ef_cap_avg_ms value = local_var:zz_s_ms }
		}
		if = {
			limit = { has_variable = zz_ef_cap_avg_as }
			set_variable = {
				name = zz_ef_cap_avg_as
				value = {
					value = var:zz_ef_cap_avg_as
					add = {
						value = local_var:zz_s_as
						subtract = var:zz_ef_cap_avg_as
						multiply = zz_ef_cap_weight_year
					}
				}
			}
		}
		else = {
			set_variable = { name = zz_ef_cap_avg_as value = local_var:zz_s_as }
		}
		if = {
			limit = { has_variable = zz_ef_cap_avg_mn }
			set_variable = {
				name = zz_ef_cap_avg_mn
				value = {
					value = var:zz_ef_cap_avg_mn
					add = {
						value = local_var:zz_s_mn
						subtract = var:zz_ef_cap_avg_mn
						multiply = zz_ef_cap_weight_year
					}
				}
			}
		}
		else = {
			set_variable = { name = zz_ef_cap_avg_mn value = local_var:zz_s_mn }
		}
		if = {
			limit = { has_variable = zz_ef_cap_avg_rr }
			set_variable = {
				name = zz_ef_cap_avg_rr
				value = {
					value = var:zz_ef_cap_avg_rr
					add = {
						value = local_var:zz_s_rr
						subtract = var:zz_ef_cap_avg_rr
						multiply = zz_ef_cap_weight_year
					}
				}
			}
		}
		else = {
			set_variable = { name = zz_ef_cap_avg_rr value = local_var:zz_s_rr }
		}
		if = {
			limit = { has_variable = zz_ef_cap_half }
			set_variable = {
				name = zz_ef_cap_half
				value = {
					value = var:zz_ef_cap_half
					add = {
						value = local_var:zz_s_total
						subtract = var:zz_ef_cap_half
						multiply = zz_ef_cap_weight_half
					}
				}
			}
		}
		else = {
			set_variable = { name = zz_ef_cap_half value = local_var:zz_s_total }
		}
	}
}
