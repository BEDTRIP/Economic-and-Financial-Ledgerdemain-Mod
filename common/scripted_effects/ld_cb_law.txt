# R3.2 (Д.R3.2, 8.10): the central bank law follows the CB until R4 (laws/ld_central_bank_laws.txt). Country scope;
# history (history/global/ld_central_bank_law.txt) and E&F's monthly hub (on_actions/00_ef_on_action.txt).
# A private CB (law_ld_cb_private) is kept as it is -- nothing sets it before R4.
zz_ef_cb_law_sync = {
	if = {
		limit = {
			has_modifier = has_central_bank
			NOT = { has_law = law_type:law_ld_cb_state }
			NOT = { has_law = law_type:law_ld_cb_private }
		}
		activate_law = law_type:law_ld_cb_state
	}
	else_if = {
		limit = {
			NOT = { has_modifier = has_central_bank }
			NOT = { has_law = law_type:law_ld_cb_none }
		}
		activate_law = law_type:law_ld_cb_none
	}
}
