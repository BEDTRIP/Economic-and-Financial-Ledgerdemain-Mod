# The engine's money burnt inside the private sector goes back to the pool (Д.R4б.33, Д.R4б.31: the 2nd kind -- the
# engine's behaviour the script cannot fix, measured and made good). The line «движок: оплачено сверх продаж рынка»
# (var:zz_ef_f_leak, the week's rest by money conservation, ld_money_model_values.txt) -- a buyer pays for its whole
# order, the sellers are paid for what sold -- comes back as deposits of those who lost it: the pops' share
# (zz_ef_leak_back_pop_share) into their deposits, the rest into the companies' deposits (var:zz_ef_comp_dep; the
# buildings' cash is out of the script's reach). The rest, not a list of losses: a loss a later stage posts itself
# (the pops' unspent income -- R7) leaves the rest by itself, nothing returns twice. A week the engine made money
# (the line > 0) takes it out of the same deposits, at most what they and the pool hold. The pool and the book move
# together -- «прочее» does not; the pool's card shows it a week later (var:zz_ef_w_leak_back).

# COUNTRY scope, the step's bridge part, after the line is set (zz_ef_bridge_apply).
zz_ef_leak_back_step = {
	set_variable = { name = zz_ef_f_leak_back value = 0 }
	set_variable = { name = zz_ef_f_leak_back_pop value = 0 }
	if = {
		limit = {
			has_variable = zz_ef_model_started
			has_variable = zz_ef_f_leak
		}
		if = { limit = { NOT = { has_variable = zz_ef_comp_dep } } set_variable = { name = zz_ef_comp_dep value = 0 } }
		if = { limit = { NOT = { has_variable = zz_ef_pop_deposits } } set_variable = { name = zz_ef_pop_deposits value = 0 } }
		set_variable = { name = zz_ef_t_lb_p value = { value = var:zz_ef_f_leak multiply = { value = 0 subtract = zz_ef_leak_back_pop_share } } }
		set_variable = { name = zz_ef_t_lb_c value = { value = var:zz_ef_f_leak multiply = { value = zz_ef_leak_back_pop_share subtract = 1 } } }
		# taken out: at most the deposits, and the two together at most the pool
		if = {
			limit = { var:zz_ef_t_lb_p < 0 }
			set_variable = { name = zz_ef_t_lb_p value = { value = var:zz_ef_t_lb_p min = { value = var:zz_ef_pop_deposits min = 0 multiply = -1 } } }
		}
		if = {
			limit = { var:zz_ef_t_lb_c < 0 }
			set_variable = { name = zz_ef_t_lb_c value = { value = var:zz_ef_t_lb_c min = { value = var:zz_ef_comp_dep min = 0 multiply = -1 } } }
		}
		set_variable = { name = zz_ef_f_leak_back value = { value = var:zz_ef_t_lb_p add = var:zz_ef_t_lb_c } }
		set_variable = { name = zz_ef_t_lb_room value = { value = investment_pool add = var:zz_ef_f_leak_back } }
		if = {
			limit = { var:zz_ef_t_lb_room < 0 }
			set_variable = { name = zz_ef_f_leak_back value = 0 }
			set_variable = { name = zz_ef_t_lb_p value = 0 }
			set_variable = { name = zz_ef_t_lb_c value = 0 }
		}
		change_variable = { name = zz_ef_pop_deposits add = var:zz_ef_t_lb_p }
		# the savings are banknotes + deposits (ld_money_model.txt, zz_ef_pop_savings_step)
		if = { limit = { has_variable = zz_ef_pop_savings } change_variable = { name = zz_ef_pop_savings add = var:zz_ef_t_lb_p } }
		set_variable = { name = zz_ef_f_leak_back_pop value = var:zz_ef_t_lb_p }
		change_variable = { name = zz_ef_comp_dep add = var:zz_ef_t_lb_c }
		zz_ef_eng_add_pool = { V = var:zz_ef_f_leak_back }
		if = {
			limit = { zz_ef_logs_on = yes }
			if = { limit = { NOT = { has_variable = zz_ef_leak_back_cum } } set_variable = { name = zz_ef_leak_back_cum value = 0 } }
			change_variable = { name = zz_ef_leak_back_cum add = var:zz_ef_f_leak_back }
		}
		remove_variable = zz_ef_t_lb_p
		remove_variable = zz_ef_t_lb_c
		remove_variable = zz_ef_t_lb_room
	}
}
