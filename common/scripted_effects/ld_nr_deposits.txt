# Our money held abroad = deposits of non-residents in our banks (понятие «Вклады нерезидентов»; docs/код/banks.md). The
# truth is the register of claims (scripted_effects/ld_claims.txt): the holders' maps on us add up to var:zz_ef_rq_out_m;
# deposits appear only by a posting (Д2.30, Хвост R1).

# COUNTRY scope, weekly after the clearing (zz_ef_bridge_apply, after zz_ef_cb_hume_step).
#   * as a holder: the interest on the foreign money our CB, banks, pops and fund hold -- each issuer's weekly deposit
#     rate, credited as more deposit (zz_ef_rq_interest_step); the issuer pays it out of its banks' capital at its own
#     step;
#   * as an issuer: the holders' interest credited since our last step out of the banks' capital (var:zz_ef_f_nr_int);
#     our weekly rate for the holders (var:zz_ef_rq_rate_w);
#   * every country: the clearing's posting -- our money paid out (var:zz_ef_f_clr_cur_out) less the one redeemed
#     (var:zz_ef_f_clr_own_back), and the exchange's deals in our money (var:zz_ef_xch_nr_acc): money the engine already
#     took for the imports, into / out of the pool as deposits (var:zz_ef_f_nr_dep, at most the pool on the way out);
#   * the holders spend our money on our goods (Д2.33): an export advantage by our money held abroad.
zz_ef_nr_dep_step = {
	set_variable = { name = zz_ef_f_nr_dep value = 0 }
	set_variable = { name = zz_ef_f_nr_int value = 0 }
	if = {
		limit = { zz_ef_am_holder = yes }
		zz_ef_rq_interest_step = { H = cb }
	}
	zz_ef_rq_interest_step = { H = bk }
	zz_ef_rq_interest_step = { H = pp }
	zz_ef_rq_interest_step = { H = fd }
	if = {
		limit = {
			zz_ef_nr_issuer = yes
			has_variable = zz_ef_model_started
		}
		set_variable = {
			name = zz_ef_rq_rate_w
			value = {
				value = zz_ef_deposit_rate
				divide = 52
				min = 0
			}
		}
		if = {
			limit = { has_variable = zz_ef_rq_int_due }
			set_variable = { name = zz_ef_f_nr_int value = var:zz_ef_rq_int_due }
			remove_variable = zz_ef_rq_int_due
			if = { limit = { NOT = { has_variable = zz_ef_bank_capital } } set_variable = { name = zz_ef_bank_capital value = 0 } }
			change_variable = { name = zz_ef_bank_capital subtract = var:zz_ef_f_nr_int }
		}
	}
	# the clearing's posting -- for every country whose money the clearing paid out (an issuer or not: a subject on the
	# external exchange standard, a country without a CB -- their money held abroad is in the book all the same)
	if = {
		limit = { has_variable = zz_ef_model_started }
		if = { limit = { has_variable = zz_ef_f_clr_cur_out } change_variable = { name = zz_ef_f_nr_dep add = var:zz_ef_f_clr_cur_out } }
		if = { limit = { has_variable = zz_ef_f_clr_own_back } change_variable = { name = zz_ef_f_nr_dep subtract = var:zz_ef_f_clr_own_back } }
		# the exchange's deals in our money since our last step (zz_ef_xch_nr_after_m)
		# (var:zz_ef_f_nr_xch -- the same for the abroad card: our money held abroad changed by the exchange)
		set_variable = { name = zz_ef_f_nr_xch value = 0 }
		if = {
			limit = { has_variable = zz_ef_xch_nr_acc }
			change_variable = { name = zz_ef_f_nr_dep add = var:zz_ef_xch_nr_acc }
			set_variable = { name = zz_ef_f_nr_xch value = var:zz_ef_xch_nr_acc }
			remove_variable = zz_ef_xch_nr_acc
		}
		# the holders draw out more than the pool holds (the banks lent it out): the register has already taken the claims
		# off -- the shortfall is paid by the CB as credit to the banks, straight to the holders (the lender of last
		# resort: the banks' debt to the CB grows, the pool does not move); without a CB the banks keep it (their
		# capital). The book then moves as the pool does («прочее» untouched).
		set_variable = { name = zz_ef_f_nr_short value = 0 }
		if = {
			limit = { var:zz_ef_f_nr_dep < 0 }
			set_variable = {
				name = zz_ef_f_nr_short
				value = {
					value = var:zz_ef_f_nr_dep
					multiply = -1
					subtract = {
						value = investment_pool
						min = 0
					}
					min = 0
				}
			}
			change_variable = { name = zz_ef_f_nr_dep add = var:zz_ef_f_nr_short }
		}
		if = {
			limit = { var:zz_ef_f_nr_short > 0 }
			if = {
				limit = { has_modifier = has_central_bank }
				if = { limit = { NOT = { has_variable = zz_ef_bank_cb_debt } } set_variable = { name = zz_ef_bank_cb_debt value = 0 } }
				change_variable = { name = zz_ef_bank_cb_debt add = var:zz_ef_f_nr_short }
			}
			else = {
				if = { limit = { NOT = { has_variable = zz_ef_bank_capital } } set_variable = { name = zz_ef_bank_capital value = 0 } }
				change_variable = { name = zz_ef_bank_capital add = var:zz_ef_f_nr_short }
			}
			if = { limit = { zz_ef_logs_on = yes } debug_log = "EFK|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|nr_short|short [THIS.GetCountry.MakeScope.Var('zz_ef_f_nr_short').GetValue|0]|paid [THIS.GetCountry.MakeScope.Var('zz_ef_f_nr_dep').GetValue|0]|cb [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_has_cb_v')|0]" }
		}
		add_investment_pool = var:zz_ef_f_nr_dep
	}
	if = {
		limit = { has_modifier = zz_ef_fx_holders_demand }
		remove_modifier = zz_ef_fx_holders_demand
	}
	if = {
		limit = {
			zz_ef_nr_issuer = yes
			zz_ef_fx_holders_demand_m > 0
		}
		add_modifier = {
			name = zz_ef_fx_holders_demand
			multiplier = zz_ef_fx_holders_demand_m
		}
	}
}
