### E&F central bank: how it is built -- hand-maintained.
###
### the central bank is the state's (buildings/ef_15_bank.txt).
###
### E&F's spawners (introduction_new_currency and the others in 09_introduction_building_lvl.txt) call this with
### the size E&F wants, and the bank is built or grown to that size, state-owned, with an empty cash register
###. E&F calls its spawners on a pulse with a size growing with gdp_view;
### a bank that is already big enough is left alone.
###
### State scope. $BANK_BLDG_TYPE$ is passed by every caller and must be used: an unused macro argument is a
### fatal compile error that takes the whole effect down (debug.log, jomini_script_argument.cpp:182).

zz_ef_cm_create_owned_bank = {
	if = {
		limit = {
			exists = this
			NOT = {
				any_scope_building = {
					is_building_type = $BANK_BLDG_TYPE$
					level >= $CB_SIZE$
				}
			}
		}
		create_building = {
			building = $BANK_BLDG_TYPE$
			level    = $CB_SIZE$
			reserves = 0
		}
	}
}
