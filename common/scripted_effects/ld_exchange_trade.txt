# The exchange's trading (R8б, шаг 7; понятия «Биржа», «Решение держателя», «Инвестиционный фонд»; Д.R8б.17,
# Д.R8б.19–22, Д.R8б.26). A book per asset (kind of claim x debtor) on each exchange: every holder offers what it holds
# at its price and bids for what it wants at its price, in its weekly step, on the exchange of its route
# (var:zz_ef_xch); once a week (the world's close) each book is cleared at one price -- the step with the most matched;
# the rest goes to the exchange's investment fund at its quotes, as far as its room and stock go; the deals are settled
# at each holder's next step out of its own account (Д.R8б.21). The exchange's country is the other side of every deal:
# a holder abroad pays it / is paid by it through the clearing (the flow abroad, zz_ef_ext_net_week).
#   prices: shares of the fair price (on the debtor: money -- 1, bonds -- zz_ef_xq_fair_b), in steps of zz_ef_xq_w;
#   a holder's gain of holding (Д.R8б.20): the yield (money -- the issuer's deposit rate, bonds -- its government's rate)
#     less its own key rate (a CB's claim in its cover -- against metal, 0; Д.R8б.26), the debtor's risk premium and the
#     cost of leaving (zz_ef_xch_exit_v); it asks the fair price x (1 + the gain, not under 0) and, if the gain is over
#     the costs of buying (the fee and the trade policy's surcharge), bids the fair price x (1 + the gain - the costs)
#     instead of selling; a position under 1 % of its portfolio and a CB short of cover selling what is not in it --
#     the lowest step;
#   who: the CB, the banks and the pops buy and sell money; the banks buy and sell bonds, the treasury sells them; the
#     fund trades money (bonds of the pops, the CB and the fund -- the bond register, R5);
#   the book's maps and the clearing -- scripted_effects/ld_exchange_book.txt (generated); the numbers --
#   script_values/ld_exchange_trade_values.txt.

# ---- the maps --------------------------------------------------------------------------------------------------------

# Scope -- the map's owner, key scope:zz_ef_rq_key: var:zz_ef_xm_v added to MAP's value / set as it.
zz_ef_xmap_add = {
	$MAP$_read = yes
	if = {
		limit = { is_key_in_variable_map = { name = $MAP$ target = scope:zz_ef_rq_key } }
		remove_from_variable_map = { name = $MAP$ key = scope:zz_ef_rq_key }
	}
	change_variable = { name = zz_ef_rq_h add = var:zz_ef_xm_v }
	add_to_variable_map = { name = $MAP$ key = scope:zz_ef_rq_key value = zz_ef_rq_h_v }
}
zz_ef_xmap_set = {
	if = {
		limit = { is_key_in_variable_map = { name = $MAP$ target = scope:zz_ef_rq_key } }
		remove_from_variable_map = { name = $MAP$ key = scope:zz_ef_rq_key }
	}
	add_to_variable_map = { name = $MAP$ key = scope:zz_ef_rq_key value = zz_ef_xm_v_v }
}

# ---- a holder's step -------------------------------------------------------------------------------------------------

# Country scope, weekly (zz_ef_money_model_step, after the fund): what others' deals moved here since our last step (as
# the exchange's country), last week's orders settled, this week's put into the book of our exchange (a B country --
# every 4th week, its phase).
zz_ef_xch_step = {
	set_variable = { name = zz_ef_f_xch_net value = 0 }
	set_variable = { name = zz_ef_f_xch_pool value = 0 }
	set_variable = { name = zz_ef_f_xch_pp value = 0 }
	set_variable = { name = zz_ef_f_xch_tr value = 0 }
	set_variable = { name = zz_ef_f_xch_fee value = 0 }
	if = {
		limit = { has_variable = zz_ef_xch_ext_acc }
		set_variable = { name = zz_ef_f_xch_net value = var:zz_ef_xch_ext_acc }
		remove_variable = zz_ef_xch_ext_acc
	}
	if = {
		limit = { has_variable = zz_ef_xch_pool_acc }
		set_variable = { name = zz_ef_f_xch_pool value = var:zz_ef_xch_pool_acc }
		remove_variable = zz_ef_xch_pool_acc
	}
	zz_ef_xch_settle = yes
	zz_ef_xch_fee_pay = yes
	zz_ef_xch_bimet_arb = yes
	# the orders: A every week, B every 4th (its phase, as its bridge)
	if = {
		limit = {
			zz_ef_xch_posts_this_week = yes
			has_variable = zz_ef_xch
			has_variable = zz_ef_pop_savings
			has_global_variable = zz_ef_sched_week
			var:zz_ef_xch = { zz_ef_has_exchange = yes }
		}
		var:zz_ef_xch = { save_temporary_scope_as = zz_ef_xch_e }
		set_variable = { name = zz_ef_xo_e value = var:zz_ef_xch }
		set_variable = { name = zz_ef_xo_week value = global_var:zz_ef_sched_week }
		zz_ef_xch_post_player = yes
		zz_ef_xch_post_sells = { T = cb K = m }
		zz_ef_xch_post_sells = { T = bk K = m }
		zz_ef_xch_post_sells = { T = pp K = m }
		zz_ef_xch_post_sells = { T = tr K = b }
		zz_ef_xch_post_sells = { T = bk K = b }
		zz_ef_xch_post_buys = yes
	}
}

# Holder scope, scope:zz_ef_xch_d, scope:zz_ef_xch_e: the gain of holding a claim of kind K on the debtor --
# var:zz_ef_xe_base (the yield less the risk premium and the cost of leaving), var:zz_ef_xe_r / var:zz_ef_xe_rcb (the
# key rate to beat for others / for the CB: 0 for a claim in its cover), var:zz_ef_xe_fee (the costs of buying).
zz_ef_xch_eval = {
	zz_ef_xch_eval_y_$K$ = yes
	change_variable = { name = zz_ef_xe_base subtract = scope:zz_ef_xch_d.zz_ef_gov_rate_premium }
	change_variable = { name = zz_ef_xe_base subtract = zz_ef_xch_exit_v }
	set_variable = { name = zz_ef_xe_fee value = zz_ef_xch_fee_buy_v }
	set_variable = { name = zz_ef_xe_r value = zz_ef_money_rate }
	set_variable = { name = zz_ef_xe_rcb value = zz_ef_money_rate }
	zz_ef_xch_eval_cov_$K$ = yes
}
zz_ef_xch_eval_y_m = { set_variable = { name = zz_ef_xe_base value = scope:zz_ef_xch_d.zz_ef_deposit_rate } }
zz_ef_xch_eval_y_b = { set_variable = { name = zz_ef_xe_base value = scope:zz_ef_xch_d.zz_ef_gov_rate_target } }
zz_ef_xch_eval_cov_m = {
	if = {
		limit = {
			zz_ef_exchange_std = yes
			scope:zz_ef_xch_d = { zz_ef_rq_conv_v > 0 }
		}
		set_variable = { name = zz_ef_xe_rcb value = 0 }
	}
}

# Holder scope, scope:zz_ef_xch_e: the sell orders of the holder T for its claims of kind K.
zz_ef_xch_post_sells = {
	if = {
		limit = { has_variable_map = zz_ef_rq_$T$_$K$ }
		set_variable = { name = zz_ef_xo_pf value = zz_ef_xch_portfolio_$T$ }
		save_temporary_scope_as = zz_ef_xo_h
		every_key_in_variable_map = {
			variable = zz_ef_rq_$T$_$K$
			save_temporary_scope_as = zz_ef_xch_d
			scope:zz_ef_xo_h = { zz_ef_xch_sell_one = { T = $T$ K = $K$ } }
		}
	}
}
zz_ef_xch_sell_one = {
	if = {
		limit = {
			NOT = { scope:zz_ef_xch_d = this }
			scope:zz_ef_xch_d = { NOT = { is_country_type = decentralized } }
			NOT = { zz_ef_xch_banned = yes }
			NOT = { is_key_in_variable_map = { name = zz_ef_xa_$T$_$K$ target = scope:zz_ef_xch_d } }
		}
		scope:zz_ef_xch_d = { save_temporary_scope_as = zz_ef_rq_key }
		zz_ef_rq_$T$_$K$_read = yes
		set_variable = { name = zz_ef_xo_a value = var:zz_ef_rq_h }
		if = {
			limit = { var:zz_ef_xo_a > 0 }
			set_variable = { name = zz_ef_xo_s value = 0 }
			set_variable = { name = zz_ef_xo_g value = { value = var:zz_ef_xo_a multiply = scope:zz_ef_xch_d.zz_ef_xq_gold_$K$ } }
			# a small position -- the lowest step, it goes first (Д.R8б.22); else every holding is offered at the holder's
			# price (Д.R8б.32), unless it would buy more of it
			if = {
				limit = { var:zz_ef_xo_g < zz_ef_xo_small_lim_v }
				set_variable = { name = zz_ef_xo_s value = 1 }
			}
			else = {
				zz_ef_xch_eval = { K = $K$ }
				zz_ef_xch_g_$T$ = yes
				if = {
					limit = {
						zz_ef_xch_buys_$T$_$K$ = yes
						var:zz_ef_xe_g > var:zz_ef_xe_fee
					}
					# it would pay over the fair price for more: it keeps what it has
				}
				else_if = {
					limit = { zz_ef_xch_dump_$T$_$K$ = yes }
					set_variable = { name = zz_ef_xo_s value = 1 }
				}
				else = {
					set_variable = { name = zz_ef_xo_s value = zz_ef_xch_ask_step_v }
				}
			}
			if = {
				limit = {
					var:zz_ef_xo_s >= 1
					var:zz_ef_xo_s <= zz_ef_xq_top
				}
				zz_ef_xch_order_put = { T = $T$ K = $K$ SIDE = s }
			}
		}
	}
}
# Holder scope, scope:zz_ef_xch_d, scope:zz_ef_xch_e, var:zz_ef_xo_a (the amount), var:zz_ef_xo_s (the step): the
# order of the holder T for kind K (SIDE = s -- sell, b -- buy) -- into our orders and the exchange's book.
zz_ef_xch_order_put = {
	scope:zz_ef_xch_d = { save_temporary_scope_as = zz_ef_rq_key }
	set_variable = { name = zz_ef_xm_v value = var:zz_ef_xo_a }
	zz_ef_xmap_set = { MAP = zz_ef_xa_$T$_$K$ }
	set_variable = { name = zz_ef_xm_v value = var:zz_ef_xo_s }
	zz_ef_xch_code_$SIDE$ = yes
	zz_ef_xmap_set = { MAP = zz_ef_xp_$T$_$K$ }
	save_temporary_scope_as = zz_ef_xo_h
	scope:zz_ef_xch_e = {
		set_variable = { name = zz_ef_xm_v value = scope:zz_ef_xo_h.var:zz_ef_xo_a }
		set_variable = { name = zz_ef_xm_s value = scope:zz_ef_xo_h.var:zz_ef_xo_s }
		zz_ef_xch_book_add_$K$_$SIDE$ = yes
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = zz_ef_xk_$K$ target = scope:zz_ef_xch_d } } }
			add_to_variable_list = { name = zz_ef_xk_$K$ target = scope:zz_ef_xch_d }
			if = { limit = { NOT = { has_variable = zz_ef_xk_n_$K$ } } set_variable = { name = zz_ef_xk_n_$K$ value = 0 } }
			change_variable = { name = zz_ef_xk_n_$K$ add = 1 }
		}
		set_variable = zz_ef_xk_on
	}
	if = {
		limit = { zz_ef_logs_on = yes }
		zz_ef_xw_add = { N = orders V = 1 }
	}
}

# Holder scope, scope:zz_ef_xch_e: the player's orders from the window (common/scripted_guis/ld_exchange_window.txt),
# before the automatic ones (they leave its debtors alone this step): a buy -- money by the CB, bonds by the treasury, at
# the highest step for what is left of the sum; a sell -- all the holding at the lowest step. A request stays until it
# is filled (a buy -- its sum paid, zz_ef_xpl_paid; a sell -- nothing left to sell) or cancelled (Д.R8б.35).
zz_ef_xch_post_player = {
	save_temporary_scope_as = zz_ef_xo_h
	if = {
		limit = {
			has_variable_map = zz_ef_xpl_buy_m
			has_modifier = has_central_bank
		}
		every_key_in_variable_map = {
			variable = zz_ef_xpl_buy_m
			save_temporary_scope_as = zz_ef_xch_d
			scope:zz_ef_xo_h = { zz_ef_xch_player_buy = { T = cb K = m } }
		}
	}
	if = {
		limit = { has_variable_map = zz_ef_xpl_buy_b }
		every_key_in_variable_map = {
			variable = zz_ef_xpl_buy_b
			save_temporary_scope_as = zz_ef_xch_d
			scope:zz_ef_xo_h = { zz_ef_xch_player_buy = { T = tr K = b } }
		}
	}
	if = {
		limit = { has_variable_map = zz_ef_xpl_sell_m }
		every_key_in_variable_map = {
			variable = zz_ef_xpl_sell_m
			save_temporary_scope_as = zz_ef_xch_d
			scope:zz_ef_xo_h = { zz_ef_xch_player_sell = { T = cb K = m } }
		}
	}
	if = {
		limit = { has_variable_map = zz_ef_xpl_sell_b }
		every_key_in_variable_map = {
			variable = zz_ef_xpl_sell_b
			save_temporary_scope_as = zz_ef_xch_d
			scope:zz_ef_xo_h = { zz_ef_xch_player_sell = { T = tr K = b } }
		}
	}
}
zz_ef_xch_player_buy = {
	if = {
		limit = {
			NOT = { scope:zz_ef_xch_d = this }
			NOT = { zz_ef_xch_banned = yes }
		}
		scope:zz_ef_xch_d = { save_temporary_scope_as = zz_ef_rq_key }
		zz_ef_xpl_buy_$K$_read = yes
		if = {
			limit = { var:zz_ef_rq_h < zz_ef_xpl_left_min }
			remove_from_variable_map = { name = zz_ef_xpl_buy_$K$ key = scope:zz_ef_xch_d }
			set_variable = { name = zz_ef_rq_h value = 0 }
		}
		set_variable = { name = zz_ef_xo_s value = zz_ef_xq_top }
		set_variable = {
			name = zz_ef_xo_a
			value = {
				value = var:zz_ef_rq_h
				multiply = zz_ef_xch_gpm_v
				divide = {
					value = scope:zz_ef_xch_d.zz_ef_xq_gold_$K$
					multiply = zz_ef_xq_ratio_v
					min = 0.001
				}
			}
		}
		if = {
			limit = { var:zz_ef_xo_a > 0 }
			zz_ef_xch_order_put = { T = $T$ K = $K$ SIDE = b }
			zz_ef_xch_player_log = { SIDE = buy K = $K$ }
		}
	}
}
zz_ef_xch_player_sell = {
	if = {
		limit = {
			NOT = { scope:zz_ef_xch_d = this }
			NOT = { zz_ef_xch_banned = yes }
		}
		scope:zz_ef_xch_d = { save_temporary_scope_as = zz_ef_rq_key }
		zz_ef_rq_$T$_$K$_read = yes
		set_variable = { name = zz_ef_xo_a value = var:zz_ef_rq_h }
		set_variable = { name = zz_ef_xo_s value = 1 }
		if = {
			limit = { var:zz_ef_xo_a > 0 }
			zz_ef_xch_order_put = { T = $T$ K = $K$ SIDE = s }
			zz_ef_xch_player_log = { SIDE = sell K = $K$ }
		}
		else = {
			remove_from_variable_map = { name = zz_ef_xpl_sell_$K$ key = scope:zz_ef_xch_d }
		}
	}
}
zz_ef_xch_player_log = {
	if = {
		limit = { zz_ef_logs_on = yes }
		save_temporary_scope_as = zz_ef_xo_h
		scope:zz_ef_xch_d = {
			set_variable = { name = zz_ef_xl_m value = scope:zz_ef_xo_h.var:zz_ef_xo_a }
			# a parameter inside the log's text comes out broken: the side and the kind as numbers
			set_variable = { name = zz_ef_xl_sd value = zz_ef_xl_code_$SIDE$ }
			set_variable = { name = zz_ef_xl_k value = zz_ef_xl_code_$K$ }

			remove_variable = zz_ef_xl_sd
			remove_variable = zz_ef_xl_k
		}
	}
}
# Player scope, scope:zz_ef_xch_d, var:zz_ef_xd_x -- what a deal of its buy request cost: the request's sum less it

# Player scope (the window's «update»): two tables of the kind. «Ours» (global list zz_ef_xwin_own) -- what we hold;
# «the exchange» (zz_ef_xwin_book) -- the assets cleared on our exchange last week and our buyers' short list (our
# market's owner, our bloc's leader, the reference, the deepest markets). On each row country: the fair price in gold a
# unit, last week's price on our exchange (0 -- no deal), the fund's bid / ask in gold, our holding and its worth in gold.
zz_ef_xwin_fill = {
	save_temporary_scope_as = zz_ef_xw_root
	if = { limit = { NOT = { has_variable = zz_ef_xpl_kind } } set_variable = { name = zz_ef_xpl_kind value = 0 } }
	if = { limit = { NOT = { has_variable = zz_ef_xpl_sum } } set_variable = { name = zz_ef_xpl_sum value = zz_ef_xwin_sum_start } }
	clear_global_variable_list = zz_ef_xwin_own
	clear_global_variable_list = zz_ef_xwin_book
	clear_global_variable_list = zz_ef_xwin_ord
	# our requests of the kind (Д.R8б.35): what is left of a buy's sum, a sell -- all we hold
	if = {
		limit = { zz_ef_xwin_kind_is_money = yes }
		zz_ef_xwin_ord_add = { K = m }
	}
	else = {
		zz_ef_xwin_ord_add = { K = b }
	}
	if = {
		limit = {
			zz_ef_xwin_kind_is_money = yes
			has_variable_map = zz_ef_rq_cb_m
		}
		every_key_in_variable_map = {
			variable = zz_ef_rq_cb_m
			add_to_global_variable_list = { name = zz_ef_xwin_own target = this }
		}
	}
	else_if = {
		limit = {
			NOT = { zz_ef_xwin_kind_is_money = yes }
			has_variable_map = zz_ef_rq_tr_b
		}
		every_key_in_variable_map = {
			variable = zz_ef_rq_tr_b
			add_to_global_variable_list = { name = zz_ef_xwin_own target = this }
		}
	}
	if = {
		limit = { has_variable = zz_ef_xch }
		var:zz_ef_xch = { save_temporary_scope_as = zz_ef_xch_e }
		if = {
			limit = {
				zz_ef_xwin_kind_is_money = yes
				scope:zz_ef_xch_e = { has_variable_map = zz_ef_xr_m_c }
			}
			scope:zz_ef_xch_e = {
				every_key_in_variable_map = {
					variable = zz_ef_xr_m_c
					save_temporary_scope_as = zz_ef_xw_c
					scope:zz_ef_xw_root = { zz_ef_xwin_row_add = yes }
				}
			}
		}
		else_if = {
			limit = {
				NOT = { zz_ef_xwin_kind_is_money = yes }
				scope:zz_ef_xch_e = { has_variable_map = zz_ef_xr_b_c }
			}
			scope:zz_ef_xch_e = {
				every_key_in_variable_map = {
					variable = zz_ef_xr_b_c
					save_temporary_scope_as = zz_ef_xw_c
					scope:zz_ef_xw_root = { zz_ef_xwin_row_add = yes }
				}
			}

		}

	}
	# shown in currencies (Д.R8б.33): the holding in the debtor's currency, its worth in ours; money's prices -- ours a
	# unit of the debtor's, bonds' -- % of the principal
	set_variable = { name = zz_ef_xwin_holdc value = { value = var:zz_ef_xwin_hold divide = zz_ef_cur_unit_v } }

# Holder scope, scope:zz_ef_xch_e: the buy orders -- for each buyer and kind the best of a short list of debtors (our
# market's owner, our bloc's leader, the reference currency's country, the three deepest markets) by the gain over the
# costs of buying, if over 0; the amount -- the week's budget at the bid.
zz_ef_xch_post_buys = {
	clear_variable_list = zz_ef_xo_cand

		limit = {
			exists = market
			NOT = { market.owner = this }
		}
		add_to_variable_list = { name = zz_ef_xo_cand target = market.owner }
	}
	if = {
		limit = {
			is_in_power_bloc = yes
			NOT = { power_bloc.power_bloc_leader = this }
		}
		add_to_variable_list = { name = zz_ef_xo_cand target = power_bloc.power_bloc_leader }
	}
	if = {
		limit = {
			has_global_variable = zz_ef_xref
			NOT = { global_var:zz_ef_xref = this }
			NOT = { is_target_in_variable_list = { name = zz_ef_xo_cand target = global_var:zz_ef_xref } }
		}
		add_to_variable_list = { name = zz_ef_xo_cand target = global_var:zz_ef_xref }
	}
	save_temporary_scope_as = zz_ef_xo_h
	every_in_global_list = {
		variable = zz_ef_xdeep
		limit = {
			NOT = { this = scope:zz_ef_xo_h }
			NOT = { scope:zz_ef_xo_h = { is_target_in_variable_list = { name = zz_ef_xo_cand target = prev } } }
		}
		save_temporary_scope_as = zz_ef_xo_k
		scope:zz_ef_xo_h = { add_to_variable_list = { name = zz_ef_xo_cand target = scope:zz_ef_xo_k } }
	}
	set_variable = { name = zz_ef_xo_bv_cb_m value = 0 }
	set_variable = { name = zz_ef_xo_bv_bk_m value = 0 }
	set_variable = { name = zz_ef_xo_bv_pp_m value = 0 }
	set_variable = { name = zz_ef_xo_bv_bk_b value = 0 }
	# the lowest rate to beat: 0 for a CB of an exchange standard (a claim in its cover), else our key rate; a candidate
	# whose yield is not over it is not weighed
	set_variable = { name = zz_ef_xo_rmin value = zz_ef_money_rate }
	if = {
		limit = {
			has_modifier = has_central_bank
			zz_ef_exchange_std = yes
			zz_ef_xch_budget_cb > 0
		}
		set_variable = { name = zz_ef_xo_rmin value = 0 }
	}
	every_in_list = {
		variable = zz_ef_xo_cand
		save_temporary_scope_as = zz_ef_xch_d

					NOT = { zz_ef_xch_banned = yes }

					zz_ef_xch_best = { T = cb K = m }
					zz_ef_xch_best = { T = bk K = m }
					zz_ef_xch_best = { T = pp K = m }
				}
				if = {
					limit = {
						scope:zz_ef_xch_d = { zz_ef_bl_principal > 0 }
						scope:zz_ef_xch_d.zz_ef_gov_rate_target > zz_ef_money_rate
					}
					zz_ef_xch_eval = { K = b }
					zz_ef_xch_best = { T = bk K = b }

	}

	clear_variable_list = zz_ef_xo_cand
	zz_ef_xch_buy_put = { T = cb K = m }
	zz_ef_xch_buy_put = { T = bk K = m }
	zz_ef_xch_buy_put = { T = pp K = m }
	zz_ef_xch_buy_put = { T = bk K = b }
}

# Holder scope, scope:zz_ef_xch_d: an order filled -- by the auction (all of it below the price for a sell, above it for
# a buy, the marginal step's share at it) at the clearing price, its rest by the fund at its quote if the order reaches it.
zz_ef_xch_settle_one = {
	scope:zz_ef_xch_d = { save_temporary_scope_as = zz_ef_rq_key }
	zz_ef_xa_$T$_$K$_read = yes
	set_variable = { name = zz_ef_xo_a value = var:zz_ef_rq_h }
	zz_ef_xp_$T$_$K$_read = yes
	set_variable = { name = zz_ef_xo_p value = var:zz_ef_rq_h }
	scope:zz_ef_xch_e = { zz_ef_xr_$K$_c_read = yes }
	set_variable = { name = zz_ef_xo_c value = scope:zz_ef_xch_e.var:zz_ef_rq_h }
	scope:zz_ef_xch_e = { zz_ef_xr_$K$_ps_read = yes }
	set_variable = { name = zz_ef_xo_ps value = scope:zz_ef_xch_e.var:zz_ef_rq_h }
	scope:zz_ef_xch_e = { zz_ef_xr_$K$_pb_read = yes }
	set_variable = { name = zz_ef_xo_pb value = scope:zz_ef_xch_e.var:zz_ef_rq_h }
	scope:zz_ef_xch_e = { zz_ef_xr_$K$_fs_read = yes }
	set_variable = { name = zz_ef_xo_fs value = scope:zz_ef_xch_e.var:zz_ef_rq_h }
	scope:zz_ef_xch_e = { zz_ef_xr_$K$_fb_read = yes }
	set_variable = { name = zz_ef_xo_fb value = scope:zz_ef_xch_e.var:zz_ef_rq_h }
	set_variable = { name = zz_ef_xo_a1 value = 0 }
	set_variable = { name = zz_ef_xo_a2 value = 0 }
	if = {
		limit = { var:zz_ef_xo_p < 10 }
		if = {
			limit = { var:zz_ef_xo_c > 0 }
			if = {
				limit = { var:zz_ef_xo_p < var:zz_ef_xo_c }
				set_variable = { name = zz_ef_xo_a1 value = var:zz_ef_xo_a }
			}
			else_if = {
				limit = { var:zz_ef_xo_p = var:zz_ef_xo_c }
				set_variable = { name = zz_ef_xo_a1 value = { value = var:zz_ef_xo_a multiply = var:zz_ef_xo_ps } }
			}
		}
		if = {
			limit = { var:zz_ef_xo_p <= scope:zz_ef_xch_e.var:zz_ef_xr_ib }
			set_variable = { name = zz_ef_xo_a2 value = { value = var:zz_ef_xo_a subtract = var:zz_ef_xo_a1 multiply = var:zz_ef_xo_fs } }
		}
		if = {
			limit = { var:zz_ef_xo_a1 > 0 }
			set_variable = { name = zz_ef_xd_a value = { value = var:zz_ef_xo_a1 multiply = -1 } }
			set_variable = { name = zz_ef_xd_s value = var:zz_ef_xo_c }
			set_variable = { name = zz_ef_xd_f value = 0 }
			zz_ef_xch_deal = { T = $T$ K = $K$ }
		}
		if = {
			limit = { var:zz_ef_xo_a2 > 0 }
			set_variable = { name = zz_ef_xd_a value = { value = var:zz_ef_xo_a2 multiply = -1 } }
			set_variable = { name = zz_ef_xd_s value = scope:zz_ef_xch_e.var:zz_ef_xr_ib }
			set_variable = { name = zz_ef_xd_f value = 1 }
			zz_ef_xch_deal = { T = $T$ K = $K$ }
		}
	}
	else = {
		change_variable = { name = zz_ef_xo_p subtract = 10 }
		if = {
			limit = { var:zz_ef_xo_c > 0 }
			if = {
				limit = { var:zz_ef_xo_p > var:zz_ef_xo_c }
				set_variable = { name = zz_ef_xo_a1 value = var:zz_ef_xo_a }
			}
			else_if = {
				limit = { var:zz_ef_xo_p = var:zz_ef_xo_c }
				set_variable = { name = zz_ef_xo_a1 value = { value = var:zz_ef_xo_a multiply = var:zz_ef_xo_pb } }
			}
		}
		if = {
			limit = { var:zz_ef_xo_p >= scope:zz_ef_xch_e.var:zz_ef_xr_ia }
			set_variable = { name = zz_ef_xo_a2 value = { value = var:zz_ef_xo_a subtract = var:zz_ef_xo_a1 multiply = var:zz_ef_xo_fb } }
		}
		if = {
			limit = { var:zz_ef_xo_a1 > 0 }
			set_variable = { name = zz_ef_xd_a value = var:zz_ef_xo_a1 }
			set_variable = { name = zz_ef_xd_s value = var:zz_ef_xo_c }
			set_variable = { name = zz_ef_xd_f value = 0 }
			zz_ef_xch_deal = { T = $T$ K = $K$ }
		}
		if = {
			limit = { var:zz_ef_xo_a2 > 0 }
			set_variable = { name = zz_ef_xd_a value = var:zz_ef_xo_a2 }
			set_variable = { name = zz_ef_xd_s value = scope:zz_ef_xch_e.var:zz_ef_xr_ia }
			set_variable = { name = zz_ef_xd_f value = 1 }
			zz_ef_xch_deal = { T = $T$ K = $K$ }
		}
	}
}

# Holder scope, scope:zz_ef_xch_d, scope:zz_ef_xch_e; var:zz_ef_xd_a -- the claims bought (+) / sold (-), var:zz_ef_xd_s --
# the price step, var:zz_ef_xd_f -- 1 if the other side is the exchange's fund. The claims change hands, the price in
# gold (var:zz_ef_xd_g) is paid in each side's money out of its account (zz_ef_xch_pay_<T>_<K>; the fund -- its deposit
# and the pool of its country); a side abroad of the exchange's country -- a flow abroad of both.
zz_ef_xch_deal = {
	save_temporary_scope_as = zz_ef_xo_h
	# the fund sells: out of its stock first (it may hold less)
	if = {
		limit = {
			var:zz_ef_xd_f = 1
			var:zz_ef_xd_a > 0
		}
		scope:zz_ef_xch_e = {
			set_variable = { name = zz_ef_rq_neg value = { value = scope:zz_ef_xo_h.var:zz_ef_xd_a multiply = -1 } }
			zz_ef_rq_add = { H = fd K = $K$ D = scope:zz_ef_xch_d V = var:zz_ef_rq_neg }
		}
		set_variable = { name = zz_ef_xd_a value = { value = scope:zz_ef_xch_e.var:zz_ef_rq_dlt multiply = -1 } }
	}
	zz_ef_rq_add = { H = $T$ K = $K$ D = scope:zz_ef_xch_d V = var:zz_ef_xd_a }
	set_variable = { name = zz_ef_xd_a value = var:zz_ef_rq_dlt }
	if = {
		limit = { NOT = { var:zz_ef_xd_a = 0 } }
		set_variable = {
			name = zz_ef_xd_g
			value = {
				value = var:zz_ef_xd_a
				multiply = zz_ef_xd_ratio_v
				multiply = scope:zz_ef_xch_d.zz_ef_xq_gold_$K$
			}
		}
		# the buyer pays the exchange's fee, its trade policy's surcharge and a non-resident's fee on top (Д.R8.5, Д.R8б.18,
		# Д.R8б.35): to the owners of
		# the exchange's financial centre (var:zz_ef_xch_fee_acc of its country, zz_ef_xch_fee_pay)
		set_variable = { name = zz_ef_xd_fee value = 0 }
		if = {
			limit = { var:zz_ef_xd_a > 0 }

			}
		}
		# what we pay (+) / get (-) in our money
		set_variable = { name = zz_ef_xd_x value = { value = var:zz_ef_xd_g add = var:zz_ef_xd_fee divide = zz_ef_xch_gpm_v } }
		zz_ef_xch_pay_$T$_$K$ = yes
		if = {
			limit = {
				is_player = yes
				var:zz_ef_xd_a > 0
			}
			zz_ef_xpl_paid = { K = $K$ }
		}
		if = {
			limit = { NOT = { scope:zz_ef_xch_e = this } }
			change_variable = { name = zz_ef_f_xch_net subtract = var:zz_ef_xd_x }
			scope:zz_ef_xch_e = {
				set_variable = { name = zz_ef_xd_xe value = { value = scope:zz_ef_xo_h.var:zz_ef_xd_g add = scope:zz_ef_xo_h.var:zz_ef_xd_fee divide = zz_ef_xch_gpm_v } }
				if = { limit = { NOT = { has_variable = zz_ef_xch_ext_acc } } set_variable = { name = zz_ef_xch_ext_acc value = 0 } }
				change_variable = { name = zz_ef_xch_ext_acc add = var:zz_ef_xd_xe }
			}
		}
		if = {
			limit = { var:zz_ef_xd_f = 1 }
			scope:zz_ef_xch_e = {
				set_variable = { name = zz_ef_xd_xe value = { value = scope:zz_ef_xo_h.var:zz_ef_xd_g divide = zz_ef_xch_gpm_v } }
				# the fund buys: our claims into its stock
				if = {
					limit = { scope:zz_ef_xo_h.var:zz_ef_xd_a < 0 }

					zz_ef_rq_add = { H = fd K = $K$ D = scope:zz_ef_xch_d V = var:zz_ef_rq_neg }
				}
				# its money: what we pay into its deposit and its country's pool, what we get out of them

				if = { limit = { NOT = { has_variable = zz_ef_xch_pool_acc } } set_variable = { name = zz_ef_xch_pool_acc value = 0 } }
				change_variable = { name = zz_ef_xch_pool_acc add = var:zz_ef_xd_xe }
			}
		}
		if = {
			limit = { zz_ef_logs_on = yes }
			zz_ef_xch_deal_count = yes
		}
	}
}

# ---- the metal: the arbitrage of bimetallism (R8б, шаг 9; понятия «Биметаллизм», «Цена серебра к золоту»; Д.R8.10) --

# Country scope (a CB of its own bimetallic standard), weekly from zz_ef_xch_step: the legal ratio of its standard
# (bimetallic_rate_gold_to_silver) against the world's price of gold in silver (gold_to_silver_rate). The metal the law
# overvalues comes into the CB, the other goes out to the world's market (Gresham's law) -- the investment fund of our
# exchange does it and earns the gap less the exchange's fee on both legs (to the exchange's owners). The week's share of
# the CB's undervalued metal: (the gap - the fees) x zz_ef_xch_arb_speed, at most zz_ef_xch_arb_cap and the fund's room.
# The CB's metal moved -- var:zz_ef_f_mt_arb_g / _s (the metal's reconciliation; the world's line: sold / bought).
zz_ef_xch_bimet_arb = {
	set_variable = { name = zz_ef_f_mt_arb_g value = 0 }
	set_variable = { name = zz_ef_f_mt_arb_s value = 0 }
	if = {
		limit = {
			has_modifier = has_central_bank
			zz_ef_metal_bimet_law = yes
			has_variable = zz_ef_xch
			var:zz_ef_xch = {
				zz_ef_has_exchange = yes
				zz_ef_xch_fund_room_v > 0
			}
			zz_ef_xch_arb_net_v > 0
		}
		var:zz_ef_xch = { save_temporary_scope_as = zz_ef_xch_e }
		set_variable = { name = zz_ef_t_arb_share value = { value = zz_ef_xch_arb_net_v multiply = zz_ef_xch_arb_speed max = zz_ef_xch_arb_cap } }
		set_variable = { name = zz_ef_t_arb_room value = { value = scope:zz_ef_xch_e.zz_ef_xch_fund_room_v multiply = scope:zz_ef_xch_e.zz_ef_xch_gpm_v } }
		# the gold value of the metal that leaves: gold if silver is overvalued, silver (at the world's price) if gold is
		if = {
			limit = { zz_ef_xch_arb_gap_v > 0 }
			set_variable = { name = zz_ef_t_arb_out value = { value = gold_state_native_for_stockpile multiply = var:zz_ef_t_arb_share max = var:zz_ef_t_arb_room min = 0 } }
			set_variable = { name = zz_ef_f_mt_arb_g value = { value = var:zz_ef_t_arb_out multiply = -1 } }
			set_variable = { name = zz_ef_f_mt_arb_s value = { value = var:zz_ef_t_arb_out multiply = bimetallic_rate_gold_to_silver } }
			zz_ef_wm_add = { KEY = sell_gold VALUE = var:zz_ef_t_arb_out }
			zz_ef_wm_add = { KEY = buy_silver VALUE = var:zz_ef_f_mt_arb_s }
		}
		else = {
			set_variable = { name = zz_ef_t_arb_out value = { value = silver_state_native_for_stockpile multiply = var:zz_ef_t_arb_share multiply = silver_to_gold_rate max = var:zz_ef_t_arb_room min = 0 } }
			set_variable = { name = zz_ef_f_mt_arb_s value = { value = var:zz_ef_t_arb_out multiply = gold_to_silver_rate multiply = -1 } }
			set_variable = { name = zz_ef_f_mt_arb_g value = { value = var:zz_ef_f_mt_arb_s multiply = -1 divide = bimetallic_rate_gold_to_silver } }
			set_variable = { name = zz_ef_t_arb_s value = { value = var:zz_ef_f_mt_arb_s multiply = -1 } }
			zz_ef_wm_add = { KEY = buy_gold VALUE = var:zz_ef_f_mt_arb_g }
			zz_ef_wm_add = { KEY = sell_silver VALUE = var:zz_ef_t_arb_s }
			remove_variable = zz_ef_t_arb_s
		}
		if = {
			limit = { var:zz_ef_t_arb_out > 0 }
			every_scope_state = {
				limit = {
					has_modifier = central_bank_historic_place
					is_capital = yes
				}
				if = { limit = { NOT = { has_variable = gold_state_1 } } set_variable = { name = gold_state_1 value = 0 } }
				if = { limit = { NOT = { has_variable = silver_state_1 } } set_variable = { name = silver_state_1 value = 0 } }
				change_variable = { name = gold_state_1 add = owner.var:zz_ef_f_mt_arb_g }
				change_variable = { name = silver_state_1 add = owner.var:zz_ef_f_mt_arb_s }
			}
			# the fund's profit -- the gap, less the fees (to the exchange's owners), in its country's money
			set_variable = { name = zz_ef_t_arb_fee value = { value = var:zz_ef_t_arb_out multiply = 2 multiply = scope:zz_ef_xch_e.zz_ef_xch_fee_v } }
			set_variable = { name = zz_ef_t_arb_profit value = { value = var:zz_ef_t_arb_out multiply = zz_ef_xch_arb_net_v } }
			save_temporary_scope_as = zz_ef_xo_h
			scope:zz_ef_xch_e = {
				set_variable = { name = zz_ef_xd_xe value = { value = scope:zz_ef_xo_h.var:zz_ef_t_arb_profit divide = zz_ef_xch_gpm_v } }
				zz_ef_fd_dep_add = { V = var:zz_ef_xd_xe }
				zz_ef_eng_add_pool = { V = var:zz_ef_xd_xe }
				if = { limit = { NOT = { has_variable = zz_ef_xch_pool_acc } } set_variable = { name = zz_ef_xch_pool_acc value = 0 } }
				change_variable = { name = zz_ef_xch_pool_acc add = var:zz_ef_xd_xe }
				if = { limit = { NOT = { has_variable = zz_ef_xch_fee_acc } } set_variable = { name = zz_ef_xch_fee_acc value = 0 } }
				change_variable = { name = zz_ef_xch_fee_acc add = { value = scope:zz_ef_xo_h.var:zz_ef_t_arb_fee divide = zz_ef_xch_gpm_v } }
			}
			if = {
				limit = { zz_ef_logs_on = yes }
				debug_log = "EFD|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|arb|legal [THIS.GetCountry.MakeScope.ScriptValue('bimetallic_rate_gold_to_silver')|2]|world [THIS.GetCountry.MakeScope.ScriptValue('gold_to_silver_rate')|2]|gold [THIS.GetCountry.MakeScope.Var('zz_ef_f_mt_arb_g').GetValue|0]|silver [THIS.GetCountry.MakeScope.Var('zz_ef_f_mt_arb_s').GetValue|0]|profit_g [THIS.GetCountry.MakeScope.Var('zz_ef_t_arb_profit').GetValue|0]|fee_g [THIS.GetCountry.MakeScope.Var('zz_ef_t_arb_fee').GetValue|0]"
				zz_ef_xw_add = { N = arb V = var:zz_ef_t_arb_out }
			}
			remove_variable = zz_ef_t_arb_fee
			remove_variable = zz_ef_t_arb_profit
		}
		remove_variable = zz_ef_t_arb_share
		remove_variable = zz_ef_t_arb_room
		remove_variable = zz_ef_t_arb_out
	}
}

		limit = { has_modifier = global_monetary_reference }

	if = { limit = { has_global_variable = zz_ef_sched_week } set_global_variable = { name = zz_ef_xr_week value = global_var:zz_ef_sched_week } }

	every_country = {
		limit = { is_player = yes }
		zz_ef_xwin_fill = yes
	}
}

# Any scope, with the logs on: the month's world sum N up by V (global_var:zz_ef_xw_<N>).
zz_ef_xw_add = {
	if = { limit = { NOT = { has_global_variable = zz_ef_xw_$N$ } } set_global_variable = { name = zz_ef_xw_$N$ value = 0 } }
	change_global_variable = { name = zz_ef_xw_$N$ add = $V$ }
}
# Holder scope, a deal settled: its value in gold into the month's sums -- the auction counted on the buyer's side,
# the fund's buys and sells on the holder's
zz_ef_xch_deal_count = {
	set_variable = { name = zz_ef_xd_ga value = var:zz_ef_xd_g }
	if = { limit = { var:zz_ef_xd_ga < 0 } change_variable = { name = zz_ef_xd_ga multiply = -1 } }
	if = {
		limit = { var:zz_ef_xd_f = 1 }
		if = {
			limit = { var:zz_ef_xd_a < 0 }
			zz_ef_xw_add = { N = fund_buy V = var:zz_ef_xd_ga }
		}
		else = {
			zz_ef_xw_add = { N = fund_sell V = var:zz_ef_xd_ga }
		}
	}
	else_if = {
		limit = { var:zz_ef_xd_a > 0 }
		zz_ef_xw_add = { N = auction V = var:zz_ef_xd_ga }
	}
	zz_ef_xw_add = { N = deals V = 1 }
	remove_variable = zz_ef_xd_ga
}

# month's orders, assets cleared, deals and their value in gold (the auction, the funds' buys and sells); the pyramid of
# reserves (R11): the exchange-standard CBs' cover (metal + foreign money in it) over their metal; their mutual holdings
