# Subjects-banks of a country (Д.R4б.1; понятие «Банки», docs/banks.md): by the ownership of the "Bank" building's
# levels -- every company with "Bank" levels is its own subject, the government's levels are «госбанки», the rest
# (aristocrats, capitalists, foreign small owners) is «частные банкиры». A subject is a script container (1.13,
# create_container; probe Р26): a company's -- with the company as parent (it goes when the company goes), the two
# aggregates' -- with the country as parent. Its variables (Д.R4б.2): zz_ef_bk_kind (flag:company | gov | pri),
# zz_ef_bk_lv -- its levels, zz_ef_bk_sh -- its share of the country's levels, zz_ef_bk_cty -- the country,
# zz_ef_bk_co -- the company.
#   on the country: var:zz_ef_bk_gov_box, var:zz_ef_bk_pri_box -- the aggregates, list zz_ef_bk_subj -- every subject
#   (rebuilt each step), zz_ef_bk_t / _g / _cl / _ck -- all levels, the government's, the companies', their scale
#   on a company: var:zz_ef_bk_box -- its subject, var:zz_ef_bk_lv -- its "Bank" levels as the window reads them
#   (gui/ld_bank_hook.gui; script has no trigger for a company's levels of one building type)
#   on a company: list zz_ef_bk_books -- its local books abroad, var:zz_ef_bk_lvh -- its levels at home
# A company present abroad (PRESENCE of tools/regen_ef_bank_seed.py) keeps a local book in the host (Д.R9а.2): a subject
# of the host's banks (its accounts in the host's money, var:zz_ef_bk_local), parent -- the company; its levels -- the
# host's "Bank" levels the company's country owns (zz_ef_bk_presence_step). One capital and one bankruptcy -- over all
# its books (zz_ef_bk_fail_step); its other levels abroad count at home.

# COUNTRY scope, its weekly step (zz_ef_money_model_step, after the reconciliation).
zz_ef_bk_subj_step = {
	save_temporary_scope_as = zz_ef_bk_c
	zz_ef_bk_aggregates_ensure = yes
	set_variable = { name = zz_ef_bk_t value = zz_ef_bk_levels_v }
	set_variable = { name = zz_ef_bk_g value = zz_ef_bk_gov_levels_v }
	set_variable = { name = zz_ef_bk_cl value = 0 }
	clear_variable_list = zz_ef_bk_subj
	add_to_variable_list = { name = zz_ef_bk_subj target = var:zz_ef_bk_gov_box }
	add_to_variable_list = { name = zz_ef_bk_subj target = var:zz_ef_bk_pri_box }
	every_company = {
		limit = { company_owned_levels > 0 }
		# the window measures its "Bank" levels (read next step)
		if = {
			limit = { NOT = { is_target_in_global_variable_list = { name = zz_ef_bk_hook target = this } } }
			add_to_global_variable_list = { name = zz_ef_bk_hook target = this }
		}
		zz_ef_bk_home_levels = yes
		if = {
			limit = { zz_ef_bk_company_banks = yes }
			if = {
				limit = { NOT = { has_variable = zz_ef_bk_box } }
				zz_ef_bk_box_create = { KIND = company PARENT = this }
				set_variable = { name = zz_ef_bk_box value = scope:zz_ef_bk_new }
				var:zz_ef_bk_box = { set_variable = { name = zz_ef_bk_co value = prev } }
			}
			# its country now: a company passes with an annexed or split country, its subject with it (the books follow by
			# the weekly allocation; the exceptional events -- R11)
			var:zz_ef_bk_box = {
				set_variable = { name = zz_ef_bk_lv value = prev.var:zz_ef_bk_lvh }
				set_variable = { name = zz_ef_bk_cty value = scope:zz_ef_bk_c }
			}
			scope:zz_ef_bk_c = {
				add_to_variable_list = { name = zz_ef_bk_subj target = prev.var:zz_ef_bk_box }
				change_variable = { name = zz_ef_bk_cl add = prev.var:zz_ef_bk_lvh }
			}
		}
	}
	# gone: a company subject without "Bank" levels -- its accounts and claims to «частные банкиры», the subject closed
	every_company = {
		limit = {
			has_variable = zz_ef_bk_box
			NOT = { zz_ef_bk_company_banks = yes }
		}
		var:zz_ef_bk_box = {
			zz_ef_bk_accounts_move = { TO = scope:zz_ef_bk_c.var:zz_ef_bk_pri_box }
			destroy_container = yes
		}
		remove_variable = zz_ef_bk_box
	}
	# the local books of the companies of other countries present here (Д.R9а.2)
	zz_ef_bk_presence_step = yes
	# the companies' levels within the private ones (the window lags a step)
	set_variable = { name = zz_ef_bk_free value = { value = var:zz_ef_bk_t subtract = var:zz_ef_bk_g min = 0 } }
	set_variable = { name = zz_ef_bk_ck value = 1 }
	if = {
		limit = { var:zz_ef_bk_cl > var:zz_ef_bk_free }
		set_variable = { name = zz_ef_bk_ck value = { value = var:zz_ef_bk_free divide = var:zz_ef_bk_cl } }
	}
	var:zz_ef_bk_gov_box = { set_variable = { name = zz_ef_bk_lv value = scope:zz_ef_bk_c.var:zz_ef_bk_g } }
	set_variable = { name = zz_ef_bk_p value = zz_ef_bk_pri_levels_v }
	var:zz_ef_bk_pri_box = { set_variable = { name = zz_ef_bk_lv value = scope:zz_ef_bk_c.var:zz_ef_bk_p } }
	set_variable = { name = zz_ef_bk_shsum value = 0 }
	every_in_list = {
		variable = zz_ef_bk_subj
		set_variable = { name = zz_ef_bk_sh value = zz_ef_bk_share_v }
		scope:zz_ef_bk_c = { change_variable = { name = zz_ef_bk_shsum add = prev.var:zz_ef_bk_sh } }
	}
	# the "Bank" buildings' profit of the week by the subjects' shares (Д.R4б.13): the engine pays it to the buildings'
	# owners as dividends -- a line of the subject's accounts, its capital does not move (no money a second time)
	set_variable = { name = zz_ef_bk_bprof value = 0 }
	every_scope_state = {
		every_scope_building = {
			limit = { is_building_type = building_zz_ef_bank }
			scope:zz_ef_bk_c = { change_variable = { name = zz_ef_bk_bprof add = prev.weekly_profit } }
		}
	}
	every_in_list = {
		variable = zz_ef_bk_subj
		set_variable = { name = zz_ef_bk_inc value = 0 }
		if = {
			limit = { scope:zz_ef_bk_c.var:zz_ef_bk_shsum > 0 }
			set_variable = { name = zz_ef_bk_inc value = { value = scope:zz_ef_bk_c.var:zz_ef_bk_bprof multiply = var:zz_ef_bk_sh divide = scope:zz_ef_bk_c.var:zz_ef_bk_shsum } }
		}
	}
	if = {
		limit = { zz_ef_logs_on = yes }
		debug_log = "EFJ|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|bank_subj|n [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_bk_subj_n_v')|0]|levels [THIS.GetCountry.MakeScope.Var('zz_ef_bk_t').GetValue|0]|gov [THIS.GetCountry.MakeScope.Var('zz_ef_bk_g').GetValue|0]|comp [THIS.GetCountry.MakeScope.Var('zz_ef_bk_cl').GetValue|0]|comp_k [THIS.GetCountry.MakeScope.Var('zz_ef_bk_ck').GetValue|2]|pri [THIS.GetCountry.MakeScope.Var('zz_ef_bk_p').GetValue|0]|share_sum [THIS.GetCountry.MakeScope.Var('zz_ef_bk_shsum').GetValue|3]|bank_profit [THIS.GetCountry.MakeScope.Var('zz_ef_bk_bprof').GetValue|0]"
		if = {
			limit = { zz_ef_log_country = yes }
			every_company = {
				limit = { has_variable = zz_ef_bk_box }
				set_variable = { name = zz_ef_bk_sh value = var:zz_ef_bk_box.var:zz_ef_bk_sh }
				debug_log = "EFJ|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCompany.GetCountry.GetNameNoFormatting]|bank_subj_co|[THIS.GetCompany.GetNameNoFormatting]|levels [THIS.GetCompany.MakeScope.Var('zz_ef_bk_lv').GetValue|0]|share [THIS.GetCompany.MakeScope.Var('zz_ef_bk_sh').GetValue|3]"
			}
		}
	}
}

# COMPANY scope (zz_ef_bk_subj_step): var:zz_ef_bk_lvh -- its "Bank" levels at home: the window's, less its local
# books'; a local book no host updated for two weeks (the host is gone) is closed -- it is in no country's sum already.
zz_ef_bk_home_levels = {
	save_temporary_scope_as = zz_ef_bk_hc
	set_variable = { name = zz_ef_bk_lvh value = 0 }
	if = {
		limit = { has_variable = zz_ef_bk_lv }
		set_variable = { name = zz_ef_bk_lvh value = var:zz_ef_bk_lv }
	}
	if = {
		limit = { has_variable_list = zz_ef_bk_books }
		every_in_list = {
			variable = zz_ef_bk_books
			limit = { NOT = { has_variable = zz_ef_bk_wk } }
			scope:zz_ef_bk_hc = { add_to_variable_list = { name = zz_ef_bk_stale target = prev } }
		}
		if = {
			limit = { has_variable_list = zz_ef_bk_stale }
			every_in_list = {
				variable = zz_ef_bk_stale
				scope:zz_ef_bk_hc = { remove_list_variable = { name = zz_ef_bk_books target = prev } }
				destroy_container = yes
			}
			clear_variable_list = zz_ef_bk_stale
		}
		every_in_list = {
			variable = zz_ef_bk_books
			scope:zz_ef_bk_hc = { change_variable = { name = zz_ef_bk_lvh subtract = prev.var:zz_ef_bk_lv } }
		}
		if = {
			limit = { var:zz_ef_bk_lvh < 0 }
			set_variable = { name = zz_ef_bk_lvh value = 0 }
		}
	}
}

# COUNTRY scope, the host; scope:zz_ef_bk_c -- it, scope:zz_ef_bk_x -- a company of country $HOME$ present here
# (zz_ef_bk_presence_step): its local book (Д.R9а.2) -- levels = the host's "Bank" levels $HOME$ owns
# (zz_ef_bk_owned_levels), on the host's list of subjects and in its companies' levels; none or bankrupt -- the book
# closes, its accounts and claims to «частные банкиры» of the host.
zz_ef_bk_local_book = {
	set_variable = { name = zz_ef_bk_xl value = 0 }
	every_scope_state = {
		every_scope_building = {
			limit = {
				is_building_type = building_zz_ef_bank
				levels_owned_by_country = { target = $HOME$ value >= 1 }
			}
			zz_ef_bk_owned_levels = { HOME = $HOME$ }
		}
	}
	scope:zz_ef_bk_x = {
		if = {
			limit = {
				has_variable_list = zz_ef_bk_books
				any_in_list = { variable = zz_ef_bk_books var:zz_ef_bk_cty ?= scope:zz_ef_bk_c }
			}
			random_in_list = {
				variable = zz_ef_bk_books
				limit = { var:zz_ef_bk_cty ?= scope:zz_ef_bk_c }
				scope:zz_ef_bk_c = { set_variable = { name = zz_ef_bk_lb value = prev } }
			}
		}
	}
	if = {
		limit = {
			var:zz_ef_bk_xl > 0
			scope:zz_ef_bk_x = { NOT = { has_variable = zz_ef_bk_failed } }
		}
		if = {
			limit = { NOT = { has_variable = zz_ef_bk_lb } }
			zz_ef_bk_box_create = { KIND = company PARENT = scope:zz_ef_bk_x }
			scope:zz_ef_bk_new = {
				set_variable = { name = zz_ef_bk_co value = scope:zz_ef_bk_x }
				set_variable = zz_ef_bk_local
			}
			scope:zz_ef_bk_x = { add_to_variable_list = { name = zz_ef_bk_books target = scope:zz_ef_bk_new } }
			set_variable = { name = zz_ef_bk_lb value = scope:zz_ef_bk_new }
		}
		var:zz_ef_bk_lb = {
			set_variable = { name = zz_ef_bk_lv value = scope:zz_ef_bk_c.var:zz_ef_bk_xl }
			set_variable = { name = zz_ef_bk_wk value = 1 days = zz_ef_bk_book_seen_days }
		}
		add_to_variable_list = { name = zz_ef_bk_subj target = var:zz_ef_bk_lb }
		change_variable = { name = zz_ef_bk_cl add = var:zz_ef_bk_xl }
	}
	else_if = {
		limit = { has_variable = zz_ef_bk_lb }
		var:zz_ef_bk_lb = { zz_ef_bk_accounts_move = { TO = scope:zz_ef_bk_c.var:zz_ef_bk_pri_box } }
		scope:zz_ef_bk_x = { remove_list_variable = { name = zz_ef_bk_books target = scope:zz_ef_bk_c.var:zz_ef_bk_lb } }
		var:zz_ef_bk_lb = { destroy_container = yes }
	}
	if = { limit = { has_variable = zz_ef_bk_lb } remove_variable = zz_ef_bk_lb }
	remove_variable = zz_ef_bk_xl
}

# COUNTRY scope: the two aggregates' subjects, once.
zz_ef_bk_aggregates_ensure = {
	if = {
		limit = { NOT = { has_variable = zz_ef_bk_gov_box } }
		zz_ef_bk_box_create = { KIND = gov PARENT = this }
		set_variable = { name = zz_ef_bk_gov_box value = scope:zz_ef_bk_new }
	}
	if = {
		limit = { NOT = { has_variable = zz_ef_bk_pri_box } }
		zz_ef_bk_box_create = { KIND = pri PARENT = this }
		set_variable = { name = zz_ef_bk_pri_box value = scope:zz_ef_bk_new }
	}
}

# Any scope, scope:zz_ef_bk_c -- the country: a new subject of kind $KIND$ with parent $PARENT$, saved as
# scope:zz_ef_bk_new; its accounts start at zero.
zz_ef_bk_box_create = {
	create_container = {
		parent = $PARENT$
		tags = { zz_ef_bk }
		save_scope_as = zz_ef_bk_new
	}
	scope:zz_ef_bk_new = {
		set_variable = { name = zz_ef_bk_kind value = flag:$KIND$ }
		set_variable = { name = zz_ef_bk_cty value = scope:zz_ef_bk_c }
		set_variable = { name = zz_ef_bk_lv value = 0 }
		set_variable = { name = zz_ef_bk_sh value = 0 }
	}
}

# The bank book by subject (Д.R4б.2): every subject keeps its own accounts -- zz_ef_bk_dep (pops' deposits), _cpd (the
# companies', Д.R4б.33), _fdd (the investment fund's), _nrd (foreign CBs'), _gvd (the treasury's), _cbd (debt to the CB), _cap (capital), _cc / _bc
# (consumer / business loans), _bnd (parts of foreign bonds), _gvl (the loan to the government, Д.R5.2), _inv (fund shares and foreign money bought), _prv (the
# state's levels bought, Д.R4б.23), _mg / _ms (metal) -- and its maps of claims zz_ef_rq_bk_m / _b / _f (the country's maps' keys and kinds, ld_claims.txt). The
# country's accounts stay the books' sum, written where they were; at the country's step each subject account follows
# the country's: what moved since the last step (the residual R = country - sum of subjects) is shared -- new deposits
# and loans by the subjects' capacity (share of levels) x the appeal of their rate, debt to the CB by the liquidity gap,
# metal, bonds, fund shares and claims by the share of levels, a fall by the subjects' own balances; capital (signed)
# always by earning assets. What rounding leaves goes to «частные банкиры», so the sum equals the country's account
# every step (log EFJ|bank_sum). A subject that is gone with its company drops out of the sum -- its balances come back
# as residual.
#   rate (принято ночью): «госбанки» -- the country's deposit rate; a private subject -- plus a premium from its capital
#   ratio against the country's (zz_ef_bk_prem_k x the gap, within zz_ef_bk_prem_max); the appeal of the rate
#   1 + zz_ef_bk_appeal_k x premium. Its loan rate (Д.R9.1) -- the credit rate zz_ef_cc_rate + the same premium; new
#   business credit by its free limit (Д.R9.2) x the appeal of the loan rate 1 - zz_ef_bk_appeal_k x premium.

# COUNTRY scope, its weekly step after zz_ef_bk_subj_step.
zz_ef_bk_book_step = {
	save_temporary_scope_as = zz_ef_bk_c
	set_variable = { name = zz_ef_bk_capr value = zz_ef_bk_cty_capr_v }
	# the leverage at the key rate (Д.R9.2)
	set_variable = { name = zz_ef_bk_lev value = zz_ef_bk_lev_v }
	every_in_list = {
		variable = zz_ef_bk_subj
		set_variable = { name = zz_ef_bk_prem value = zz_ef_bk_prem_v }
		set_variable = { name = zz_ef_bk_rate value = { value = scope:zz_ef_bk_c.zz_ef_deposit_rate add = var:zz_ef_bk_prem } }
		# its loan rate (Д.R9.1): the credit rate + the same premium
		set_variable = { name = zz_ef_bk_lrate value = { value = scope:zz_ef_bk_c.zz_ef_cc_rate add = var:zz_ef_bk_prem } }
	}
	zz_ef_bk_alloc = { A = dep V = zz_ef_ledger_deposits_v W = zz_ef_bk_w_dep_v BYW = no }
	zz_ef_bk_alloc = { A = fdd V = zz_ef_fd_dep_v W = zz_ef_bk_w_dep_v BYW = no }
	zz_ef_bk_alloc = { A = cpd V = zz_ef_comp_dep_v W = zz_ef_bk_w_dep_v BYW = no }
	zz_ef_bk_alloc = { A = nrd V = zz_ef_nr_dep_v W = zz_ef_bk_w_dep_v BYW = no }
	zz_ef_bk_alloc = { A = gvd V = zz_ef_bk_c_gvd_v W = zz_ef_bk_w_dep_v BYW = no }
	zz_ef_bk_alloc = { A = cc V = zz_ef_bk_c_cc_v W = zz_ef_bk_w_dep_v BYW = no }
	zz_ef_bk_alloc = { A = bc V = zz_ef_bk_c_bc_v W = zz_ef_bk_w_bc_v BYW = no }
	zz_ef_bk_alloc = { A = bnd V = zz_ef_bank_bonds W = zz_ef_bk_w_lv_v BYW = no }
	zz_ef_bk_alloc = { A = gvl V = zz_ef_bank_gov_loan_v W = zz_ef_bk_w_lv_v BYW = no }
	zz_ef_bk_alloc = { A = inv V = zz_ef_bk_c_inv_v W = zz_ef_bk_w_lv_v BYW = no }
	zz_ef_bk_alloc = { A = prv V = zz_ef_bk_c_prv_v W = zz_ef_bk_w_lv_v BYW = no }
	zz_ef_bk_alloc = { A = mg V = zz_ef_bankm_gold W = zz_ef_bk_w_lv_v BYW = no }
	zz_ef_bk_alloc = { A = ms V = zz_ef_bankm_silver W = zz_ef_bk_w_lv_v BYW = no }
	zz_ef_bk_alloc = { A = cbd V = zz_ef_bk_c_cbd_v W = zz_ef_bk_w_cbd_v BYW = no }
	zz_ef_bk_alloc = { A = cap V = zz_ef_bk_c_cap_v W = zz_ef_bk_w_cap_v BYW = yes }
	zz_ef_bk_map_alloc = { K = m }
	zz_ef_bk_map_alloc = { K = b }
	zz_ef_bk_map_alloc = { K = f }
	# the banks' free credit limit (Д.R9.2): capital x leverage less loans, summed -- the private construction's lever
	# (zz_ef_psc_credit_f, ld_psc_private_allocation.txt); the business credit's growth above it (the engine spent the
	# pool already) is a credit all the same -- var:zz_ef_bk_over, the log
	set_variable = { name = zz_ef_bk_free_cr value = 0 }
	every_in_list = {
		variable = zz_ef_bk_subj
		set_variable = { name = zz_ef_bk_free value = zz_ef_bk_free_v }
		scope:zz_ef_bk_c = { change_variable = { name = zz_ef_bk_free_cr add = prev.var:zz_ef_bk_free } }
	}
	set_variable = { name = zz_ef_bk_over value = { value = var:zz_ef_bk_r_bc subtract = var:zz_ef_bk_free_cr min = 0 } }
	# the business credit's rate of the country (Д.R9.1): the subjects' loan rates weighted by their business loans --
	# its interest (zz_ef_bc_rate_v, ld_money_model_values.txt) is the sum over the banks
	set_variable = { name = zz_ef_bk_t_bc value = 0 }
	set_variable = { name = zz_ef_bk_t_bci value = 0 }
	every_in_list = {
		variable = zz_ef_bk_subj
		if = {
			limit = { var:zz_ef_bk_bc > 0 }
			scope:zz_ef_bk_c = {
				change_variable = { name = zz_ef_bk_t_bc add = prev.var:zz_ef_bk_bc }
				change_variable = { name = zz_ef_bk_t_bci add = { value = prev.var:zz_ef_bk_bc multiply = prev.var:zz_ef_bk_lrate } }
			}
		}
	}
	if = {
		limit = { var:zz_ef_bk_t_bc > 0 }
		set_variable = { name = zz_ef_bk_bc_rate value = { value = var:zz_ef_bk_t_bci divide = var:zz_ef_bk_t_bc } }
	}
	else_if = {
		limit = { has_variable = zz_ef_bk_bc_rate }
		remove_variable = zz_ef_bk_bc_rate
	}
	remove_variable = zz_ef_bk_t_bc
	remove_variable = zz_ef_bk_t_bci
	if = {
		limit = { zz_ef_logs_on = yes }
		zz_ef_bk_sum_check = yes
		debug_log = "EFJ|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|bank_sum|diff [THIS.GetCountry.MakeScope.Var('zz_ef_bk_diff').GetValue|2]|map_diff [THIS.GetCountry.MakeScope.Var('zz_ef_bk_mdiff').GetValue|2]|n [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_bk_subj_n_v')|0]|r_cap [THIS.GetCountry.MakeScope.Var('zz_ef_bk_r_cap').GetValue|0]|r_dep [THIS.GetCountry.MakeScope.Var('zz_ef_bk_r_dep').GetValue|0]|r_cc [THIS.GetCountry.MakeScope.Var('zz_ef_bk_r_cc').GetValue|0]|r_bc [THIS.GetCountry.MakeScope.Var('zz_ef_bk_r_bc').GetValue|0]|r_cbd [THIS.GetCountry.MakeScope.Var('zz_ef_bk_r_cbd').GetValue|0]|r_inv [THIS.GetCountry.MakeScope.Var('zz_ef_bk_r_inv').GetValue|0]|r_mg [THIS.GetCountry.MakeScope.Var('zz_ef_bk_r_mg').GetValue|0]|capital [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_bk_c_cap_v')|0]|cap_ratio [THIS.GetCountry.MakeScope.Var('zz_ef_bk_capr').GetValue|3]|lev [THIS.GetCountry.MakeScope.Var('zz_ef_bk_lev').GetValue|1]|free [THIS.GetCountry.MakeScope.Var('zz_ef_bk_free_cr').GetValue|0]|pfree [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_pool_free')|0]|over [THIS.GetCountry.MakeScope.Var('zz_ef_bk_over').GetValue|0]|bc_rate [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_bc_rate_v')|4]"
		if = {
			limit = { zz_ef_log_country = yes }
			every_in_list = {
				variable = zz_ef_bk_subj
				zz_ef_bk_subj_log = yes
			}
		}
	}
}

# COUNTRY scope, scope:zz_ef_bk_c: subject account zz_ef_bk_$A$ follows the country's value $V$ (a script value); the
# residual by weight $W$ (a subject script value) when it grows, or always ($BYW$ = yes, a signed account), else by the
# subjects' balances; the rest -- «частные банкиры». var:zz_ef_bk_r_$A$ -- the residual shared (the log).
zz_ef_bk_alloc = {
	set_variable = { name = zz_ef_bk_v value = $V$ }
	set_variable = { name = zz_ef_bk_r value = var:zz_ef_bk_v }
	set_variable = { name = zz_ef_bk_sw value = 0 }
	set_variable = { name = zz_ef_bk_sb value = 0 }
	every_in_list = {
		variable = zz_ef_bk_subj
		if = {
			limit = { NOT = { has_variable = zz_ef_bk_$A$ } }
			set_variable = { name = zz_ef_bk_$A$ value = 0 }
		}
		set_variable = { name = zz_ef_bk_w value = $W$ }
		scope:zz_ef_bk_c = {
			change_variable = { name = zz_ef_bk_r subtract = prev.var:zz_ef_bk_$A$ }
			change_variable = { name = zz_ef_bk_sw add = prev.var:zz_ef_bk_w }
			change_variable = { name = zz_ef_bk_sb add = prev.var:zz_ef_bk_$A$ }
		}
	}
	set_variable = { name = zz_ef_bk_r_$A$ value = var:zz_ef_bk_r }
	if = {
		limit = {
			var:zz_ef_bk_sw > 0
			OR = {
				var:zz_ef_bk_r > 0
				always = $BYW$
				var:zz_ef_bk_sb <= 0
			}
		}
		every_in_list = {
			variable = zz_ef_bk_subj
			change_variable = {
				name = zz_ef_bk_$A$
				add = {
					value = scope:zz_ef_bk_c.var:zz_ef_bk_r
					multiply = var:zz_ef_bk_w
					divide = scope:zz_ef_bk_c.var:zz_ef_bk_sw
				}
			}
		}
	}
	else_if = {
		limit = { var:zz_ef_bk_sb > 0 }
		every_in_list = {
			variable = zz_ef_bk_subj
			change_variable = {
				name = zz_ef_bk_$A$
				add = {
					value = scope:zz_ef_bk_c.var:zz_ef_bk_r
					multiply = var:zz_ef_bk_$A$
					divide = scope:zz_ef_bk_c.var:zz_ef_bk_sb
				}
			}
		}
	}
	set_variable = { name = zz_ef_bk_r value = var:zz_ef_bk_v }
	every_in_list = {
		variable = zz_ef_bk_subj
		scope:zz_ef_bk_c = { change_variable = { name = zz_ef_bk_r subtract = prev.var:zz_ef_bk_$A$ } }
	}
	var:zz_ef_bk_pri_box = { change_variable = { name = zz_ef_bk_$A$ add = scope:zz_ef_bk_c.var:zz_ef_bk_r } }
}

# COUNTRY scope, scope:zz_ef_bk_c: the subjects' maps zz_ef_rq_bk_$K$ follow the country's key by key -- a debtor gone
# from the country's map goes from theirs; a key's residual by the share of levels when it grows, by their holdings when
# it falls; the rest -- «частные банкиры».
zz_ef_bk_map_alloc = {
	every_in_list = {
		variable = zz_ef_bk_subj
		if = {
			limit = { has_variable_map = zz_ef_rq_bk_$K$ }
			save_temporary_scope_as = zz_ef_bk_s
			zz_ef_rq_keys_take = { MAP = zz_ef_rq_bk_$K$ }
			every_in_list = {
				variable = zz_ef_rq_keys
				save_temporary_scope_as = zz_ef_bk_d
				if = {
					limit = {
						NOT = {
							scope:zz_ef_bk_c = { is_key_in_variable_map = { name = zz_ef_rq_bk_$K$ target = scope:zz_ef_bk_d } }
						}
					}
					scope:zz_ef_bk_s = { remove_from_variable_map = { name = zz_ef_rq_bk_$K$ key = scope:zz_ef_bk_d } }
				}
			}
		}
	}
	if = {
		limit = { has_variable_map = zz_ef_rq_bk_$K$ }
		zz_ef_rq_keys_take = { MAP = zz_ef_rq_bk_$K$ }
		every_in_list = {
			variable = zz_ef_rq_keys
			save_temporary_scope_as = zz_ef_bk_d
			scope:zz_ef_bk_c = {
				zz_ef_rq_get = { MAP = zz_ef_rq_bk_$K$ D = scope:zz_ef_bk_d }
				set_variable = { name = zz_ef_bk_kv value = var:zz_ef_rq_h }
				set_variable = { name = zz_ef_bk_kr value = var:zz_ef_rq_h }
				set_variable = { name = zz_ef_bk_ks value = 0 }
				every_in_list = {
					variable = zz_ef_bk_subj
					zz_ef_rq_get = { MAP = zz_ef_rq_bk_$K$ D = scope:zz_ef_bk_d }
					set_variable = { name = zz_ef_bk_kh value = var:zz_ef_rq_h }
					scope:zz_ef_bk_c = {
						change_variable = { name = zz_ef_bk_kr subtract = prev.var:zz_ef_bk_kh }
						change_variable = { name = zz_ef_bk_ks add = prev.var:zz_ef_bk_kh }
					}
				}
				every_in_list = {
					variable = zz_ef_bk_subj
					set_variable = { name = zz_ef_bk_mv value = var:zz_ef_bk_kh }
					if = {
						limit = {
							OR = {
								scope:zz_ef_bk_c.var:zz_ef_bk_kr > 0
								scope:zz_ef_bk_c.var:zz_ef_bk_ks <= 0
							}
						}
						change_variable = { name = zz_ef_bk_mv add = { value = scope:zz_ef_bk_c.var:zz_ef_bk_kr multiply = var:zz_ef_bk_sh } }
					}
					else = {
						change_variable = {
							name = zz_ef_bk_mv
							add = {
								value = scope:zz_ef_bk_c.var:zz_ef_bk_kr
								multiply = var:zz_ef_bk_kh
								divide = scope:zz_ef_bk_c.var:zz_ef_bk_ks
							}
						}
					}
					zz_ef_bk_map_put = { K = $K$ }
				}
				# the rest -- «частные банкиры»
				set_variable = { name = zz_ef_bk_kr value = var:zz_ef_bk_kv }
				every_in_list = {
					variable = zz_ef_bk_subj
					zz_ef_rq_get = { MAP = zz_ef_rq_bk_$K$ D = scope:zz_ef_bk_d }
					scope:zz_ef_bk_c = { change_variable = { name = zz_ef_bk_kr subtract = prev.var:zz_ef_rq_h } }
				}
				var:zz_ef_bk_pri_box = {
					zz_ef_rq_get = { MAP = zz_ef_rq_bk_$K$ D = scope:zz_ef_bk_d }
					set_variable = { name = zz_ef_bk_mv value = { value = var:zz_ef_rq_h add = scope:zz_ef_bk_c.var:zz_ef_bk_kr } }
					zz_ef_bk_map_put = { K = $K$ }
				}
			}
		}
	}
}

# Bankruptcy of a bank (Д.R4б.6; понятие «Банкротство»): once a month a company whose capital is < 0 with no credit from
# the CB closes -- over all its books, home and local abroad (Д.R9а.2: one capital, in gold, zz_ef_rq_gpm_v of each
# book's country; a local book's loss is covered by the rest while the sum holds). Each book's depositors (pops and the
# investment fund) lose its hole pro rata, up to their deposits (posting deposits -> capital in the book's country);
# what a book held goes back as the week's residual to the other banks of its country by their shares
# (zz_ef_bk_book_step); the company is out of the subjects for zz_ef_bk_fail_days (its levels count with «частные
# банкиры»), then new subjects with empty accounts. Принято ночью, проверить: the aggregates («госбанки», «частные
# банкиры») do not close; foreign CBs' and the treasury's deposits do not lose (their registers live elsewhere).
# Logs EFK|bank_fail (a book), EFK|bank_group (a company with books abroad). COUNTRY scope, its monthly pulse
# (zz_ef_bank_monthly).
zz_ef_bk_fail_step = {
	save_temporary_scope_as = zz_ef_bk_c
	every_company = {
		limit = {
			OR = {
				has_variable = zz_ef_bk_box
				has_variable_list = zz_ef_bk_books
			}
		}
		save_temporary_scope_as = zz_ef_bk_gc
		set_variable = { name = zz_ef_bk_gcap value = 0 }
		set_variable = { name = zz_ef_bk_gcbd value = 0 }
		if = {
			limit = { has_variable = zz_ef_bk_box }
			var:zz_ef_bk_box = { zz_ef_bk_group_add = yes }
		}
		if = {
			limit = { has_variable_list = zz_ef_bk_books }
			every_in_list = {
				variable = zz_ef_bk_books
				zz_ef_bk_group_add = yes
			}
			if = {
				limit = { zz_ef_logs_on = yes }
				debug_log = "EFK|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCompany.GetCountry.GetNameNoFormatting]|bank_group|[THIS.GetCompany.GetNameNoFormatting]|cap_gold [THIS.GetCompany.MakeScope.Var('zz_ef_bk_gcap').GetValue|0]|cbd_gold [THIS.GetCompany.MakeScope.Var('zz_ef_bk_gcbd').GetValue|0]|books [THIS.GetCompany.MakeScope.ScriptValue('zz_ef_bk_books_n_v')|0]"
			}
		}
		if = {
			limit = {
				var:zz_ef_bk_gcap < 0
				var:zz_ef_bk_gcbd <= 0
			}
			if = {
				limit = { has_variable = zz_ef_bk_box }
				var:zz_ef_bk_box = {
					zz_ef_bk_box_fail = yes
					destroy_container = yes
				}
				remove_variable = zz_ef_bk_box
			}
			if = {
				limit = { has_variable_list = zz_ef_bk_books }
				every_in_list = {
					variable = zz_ef_bk_books
					zz_ef_bk_box_fail = yes
					destroy_container = yes
				}
				clear_variable_list = zz_ef_bk_books
			}
			set_variable = { name = zz_ef_bk_failed value = 1 days = zz_ef_bk_fail_days }
		}
		remove_variable = zz_ef_bk_gcap
		remove_variable = zz_ef_bk_gcbd
	}
}

# SUBJECT scope (a book of a failing company), scope:zz_ef_bk_gc -- the company: its depositors lose its hole (capital
# below 0) pro rata, up to their deposits -- the posting in its country; log EFK|bank_fail.
zz_ef_bk_box_fail = {
	set_variable = { name = zz_ef_bk_hole value = 0 }
	if = { limit = { has_variable = zz_ef_bk_cap } set_variable = { name = zz_ef_bk_hole value = { value = 0 subtract = var:zz_ef_bk_cap min = 0 } } }
	set_variable = { name = zz_ef_bk_liab value = 0 }
	if = { limit = { has_variable = zz_ef_bk_dep } change_variable = { name = zz_ef_bk_liab add = var:zz_ef_bk_dep } }
	if = { limit = { has_variable = zz_ef_bk_fdd } change_variable = { name = zz_ef_bk_liab add = var:zz_ef_bk_fdd } }
	set_variable = { name = zz_ef_bk_loss value = { value = var:zz_ef_bk_hole max = var:zz_ef_bk_liab } }
	set_variable = { name = zz_ef_bk_xd value = 0 }
	set_variable = { name = zz_ef_bk_xf value = 0 }
	if = {
		limit = { var:zz_ef_bk_liab > 0 }
		if = { limit = { has_variable = zz_ef_bk_dep } set_variable = { name = zz_ef_bk_xd value = { value = var:zz_ef_bk_loss multiply = var:zz_ef_bk_dep divide = var:zz_ef_bk_liab } } }
		if = { limit = { has_variable = zz_ef_bk_fdd } set_variable = { name = zz_ef_bk_xf value = { value = var:zz_ef_bk_loss multiply = var:zz_ef_bk_fdd divide = var:zz_ef_bk_liab } } }
	}
	var:zz_ef_bk_cty = {
		set_variable = { name = zz_ef_bk_fxd value = prev.var:zz_ef_bk_xd }
		set_variable = { name = zz_ef_bk_fxf value = prev.var:zz_ef_bk_xf }
		set_variable = { name = zz_ef_bk_fhole value = prev.var:zz_ef_bk_hole }
		set_variable = { name = zz_ef_bk_fdep value = 0 }
		if = { limit = { prev = { has_variable = zz_ef_bk_dep } } set_variable = { name = zz_ef_bk_fdep value = prev.var:zz_ef_bk_dep } }
		if = {
			limit = { has_variable = zz_ef_pop_deposits }
			zz_ef_post = { FROM = zz_ef_pop_deposits TO = zz_ef_bank_capital V = var:zz_ef_bk_fxd }
			if = { limit = { has_variable = zz_ef_pop_savings } change_variable = { name = zz_ef_pop_savings subtract = var:zz_ef_bk_fxd } }
		}
		if = {
			limit = { has_variable = zz_ef_fd_dep }
			zz_ef_post = { FROM = zz_ef_fd_dep TO = zz_ef_bank_capital V = var:zz_ef_bk_fxf }
		}
	}
	if = {
		limit = { zz_ef_logs_on = yes }
		debug_log = "EFK|[TimeKeeper.GetCurrentDate.GetString]|[THIS.Var('zz_ef_bk_cty').GetCountry.GetNameNoFormatting]|bank_fail|[THIS.Var('zz_ef_bk_co').GetCompany.GetNameNoFormatting]|hole [THIS.Var('zz_ef_bk_cty').GetCountry.MakeScope.Var('zz_ef_bk_fhole').GetValue|0]|dep [THIS.Var('zz_ef_bk_cty').GetCountry.MakeScope.Var('zz_ef_bk_fdep').GetValue|0]|pop_loss [THIS.Var('zz_ef_bk_cty').GetCountry.MakeScope.Var('zz_ef_bk_fxd').GetValue|0]|fund_loss [THIS.Var('zz_ef_bk_cty').GetCountry.MakeScope.Var('zz_ef_bk_fxf').GetValue|0]|levels [THIS.Var('zz_ef_bk_lv').GetValue|0]"
	}
}

# The banks' capital above the norm -- to their owners (Д.R9.3; понятие «Банки»): once a month a subject's capital above
# zz_ef_bk_cap_norm of its earning assets (zz_ef_bk_div_due_v) goes -- a company subject's to its company's account
# (zz_ef_co_acc, the companies' deposits zz_ef_comp_dep), «частные банкиры»' to the pops' deposits and savings (the
# deposits' card line of dividends, zz_ef_f_bk_div_pop), «госбанки»' to the treasury out of the pool's free money (Д.R9.9,
# zz_ef_pool_free: the pool under construction and credit stays; zz_ef_f_bk_div_tr, the pool's and the treasury's
# cards), what is not paid stays capital. The book holds: capital -> deposits leaves the pool as it is, capital and the
# pool -> the treasury moves both; the subject's capital falls with the country's (the week's residual stays 0). Log
# EFJ|bank_div. COUNTRY scope, its monthly pulse (zz_ef_bank_monthly), after the bankruptcies.
zz_ef_bk_div_step = {
	if = {
		limit = {
			has_variable = zz_ef_bk_gov_box
			has_variable_list = zz_ef_bk_subj
			has_variable = zz_ef_bank_capital
		}
		save_temporary_scope_as = zz_ef_bk_c
		set_variable = { name = zz_ef_bk_dv_co value = 0 }
		set_variable = { name = zz_ef_bk_dv_pop value = 0 }
		set_variable = { name = zz_ef_bk_dv_tr value = 0 }
		set_variable = { name = zz_ef_bk_dv_room value = zz_ef_pool_free }
		# the subjects' books are the last week's: their dues are cut so the country's capital keeps the subjects' norm
		# (var:zz_ef_bk_dv_k = (capital - the norms) / the dues, 0..1)
		set_variable = { name = zz_ef_bk_dv_due value = 0 }
		set_variable = { name = zz_ef_bk_dv_k value = var:zz_ef_bank_capital }
		every_in_list = {
			variable = zz_ef_bk_subj
			scope:zz_ef_bk_c = {
				change_variable = { name = zz_ef_bk_dv_due add = prev.zz_ef_bk_div_due_v }
				change_variable = { name = zz_ef_bk_dv_k subtract = prev.zz_ef_bk_cap_hold_v }
			}
		}
		if = {
			limit = { var:zz_ef_bk_dv_due > 0 }
			set_variable = { name = zz_ef_bk_dv_k value = { value = var:zz_ef_bk_dv_k divide = var:zz_ef_bk_dv_due min = 0 max = 1 } }
		}
		else = {
			set_variable = { name = zz_ef_bk_dv_k value = 0 }
		}
		every_in_list = {
			variable = zz_ef_bk_subj
			set_variable = { name = zz_ef_bk_dv value = { value = zz_ef_bk_div_due_v multiply = scope:zz_ef_bk_c.var:zz_ef_bk_dv_k } }
			if = {
				limit = { var:zz_ef_bk_dv > 0 }
				if = {
					limit = {
						var:zz_ef_bk_kind ?= flag:company
						exists = var:zz_ef_bk_co
					}
					var:zz_ef_bk_co = { zz_ef_co_acc_add = { V = prev.var:zz_ef_bk_dv } }
					scope:zz_ef_bk_c = { change_variable = { name = zz_ef_bk_dv_co add = prev.var:zz_ef_bk_dv } }
				}
				else_if = {
					limit = { var:zz_ef_bk_kind ?= flag:gov }
					set_variable = { name = zz_ef_bk_dv value = { value = var:zz_ef_bk_dv max = scope:zz_ef_bk_c.var:zz_ef_bk_dv_room } }
					scope:zz_ef_bk_c = {
						change_variable = { name = zz_ef_bk_dv_tr add = prev.var:zz_ef_bk_dv }
						change_variable = { name = zz_ef_bk_dv_room subtract = prev.var:zz_ef_bk_dv }
					}
				}
				else = {
					scope:zz_ef_bk_c = { change_variable = { name = zz_ef_bk_dv_pop add = prev.var:zz_ef_bk_dv } }
				}
				change_variable = { name = zz_ef_bk_cap subtract = var:zz_ef_bk_dv }
			}
			remove_variable = zz_ef_bk_dv
		}
		# the country's accounts: capital -> the companies' deposits, the pops' deposits; capital and the pool -> treasury
		if = { limit = { NOT = { has_variable = zz_ef_comp_dep } } set_variable = { name = zz_ef_comp_dep value = 0 } }
		zz_ef_post = { FROM = zz_ef_bank_capital TO = zz_ef_comp_dep V = var:zz_ef_bk_dv_co }
		if = { limit = { NOT = { has_variable = zz_ef_pop_deposits } } set_variable = { name = zz_ef_pop_deposits value = 0 } }
		if = { limit = { NOT = { has_variable = zz_ef_pop_savings } } set_variable = { name = zz_ef_pop_savings value = 0 } }
		zz_ef_post = { FROM = zz_ef_bank_capital TO = zz_ef_pop_deposits V = var:zz_ef_bk_dv_pop }
		change_variable = { name = zz_ef_pop_savings add = var:zz_ef_bk_dv_pop }
		if = {
			limit = { var:zz_ef_bk_dv_tr > 0 }
			zz_ef_post_eng_paid = { FROM = pool TO = treasury PAYER = zz_ef_bank_capital V = var:zz_ef_bk_dv_tr }
		}
		# the cards' flows (summed till the week's window roll, ld_money_model.txt)
		if = { limit = { NOT = { has_variable = zz_ef_f_bk_div_pop } } set_variable = { name = zz_ef_f_bk_div_pop value = 0 } }
		if = { limit = { NOT = { has_variable = zz_ef_f_bk_div_tr } } set_variable = { name = zz_ef_f_bk_div_tr value = 0 } }
		change_variable = { name = zz_ef_f_bk_div_pop add = var:zz_ef_bk_dv_pop }
		change_variable = { name = zz_ef_f_bk_div_tr add = var:zz_ef_bk_dv_tr }
		if = {
			limit = { zz_ef_logs_on = yes }
			debug_log = "EFJ|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|bank_div|co [THIS.GetCountry.MakeScope.Var('zz_ef_bk_dv_co').GetValue|0]|pop [THIS.GetCountry.MakeScope.Var('zz_ef_bk_dv_pop').GetValue|0]|tr [THIS.GetCountry.MakeScope.Var('zz_ef_bk_dv_tr').GetValue|0]|pfree [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_pool_free')|0]|capital [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_bk_c_cap_v')|0]|cap_ratio [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_bk_cty_capr_v')|3]"
		}
		remove_variable = zz_ef_bk_dv_room
		remove_variable = zz_ef_bk_dv_due
		remove_variable = zz_ef_bk_dv_k
	}
}
