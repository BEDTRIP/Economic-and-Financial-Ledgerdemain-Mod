#############################################
# the ledger -- the registry of accounts, the posting, the reconciliation
# by stocks. The list of accounts and their variables: docs/код/money-model.md, «Реестр счетов».
#
# Posting: one effect with a pair of accounts, the same amount out of one and into the other. Our accounts are
# country variables (their names are the accounts' names, zz_ef_*); the engine's accounts are the treasury and the
# pool (ENG = treasury | pool), moved by add_treasury / add_investment_pool. $V$ -- one token (var:x, a script value).
#
# Reconciliation by stocks: the engine's money on the date (treasury + buildings' cash + pool)
# against the sum of our accounts on it. The treasury and the buildings' cash are the state's and the businesses'
# own accounts, read from the engine; the pool is the banks' cash, and our accounts say whose it is
# (zz_ef_bank_book_v): deposits of pops and of other CBs, the CB's credit to the banks, the treasury's money in the
# banks, the banks' capital, less their loans (consumer, business). The difference -- the line «прочее»
# (var:zz_ef_other, its change var:zz_ef_f_other) -- is nobody's income: it is logged (EFJ) and summed over the world
# (the world pass). A step missed (a lost chain, a role change) is made good by the next one: the stocks are read
# on the date. The banks' capital starts as the pool less the other accounts (the first reconciliation, «прочее» 0).
#############################################

# Country scope: the accounts that start at zero on the first day (схема 3.1), once.
zz_ef_registry_init = {
	if = {
		limit = { NOT = { has_variable = zz_ef_registry } }
		set_variable = { name = zz_ef_registry value = 1 }
		# the population's claims (stocks, bonds), R6 / R7
		if = { limit = { NOT = { has_variable = zz_ef_pop_claims_stocks } } set_variable = { name = zz_ef_pop_claims_stocks value = 0 } }
		if = { limit = { NOT = { has_variable = zz_ef_pop_claims_bonds } } set_variable = { name = zz_ef_pop_claims_bonds value = 0 } }
		# the net position to the clearing house («долг загранице»), R8
		if = { limit = { NOT = { has_variable = zz_ef_clr_position } } set_variable = { name = zz_ef_clr_position value = 0 } }
		# the treasury's money in the banks (its surplus over the engine's limit, zz_ef_f_tr_pool)
		if = { limit = { NOT = { has_variable = zz_ef_bank_gov_dep } } set_variable = { name = zz_ef_bank_gov_dep value = 0 } }
		# the loans the reconciliation reads
		if = { limit = { NOT = { has_variable = zz_ef_bank_cb_debt } } set_variable = { name = zz_ef_bank_cb_debt value = 0 } }
		if = { limit = { NOT = { has_variable = zz_ef_cc_debt } } set_variable = { name = zz_ef_cc_debt value = 0 } }
		if = { limit = { NOT = { has_variable = zz_ef_bc_debt } } set_variable = { name = zz_ef_bc_debt value = 0 } }
		# the banks' capital: 0 until the first reconciliation sets it
		if = { limit = { NOT = { has_variable = zz_ef_bank_capital } } set_variable = { name = zz_ef_bank_capital value = 0 } }
	}
}

# Our account -> our account.
zz_ef_post = {
	change_variable = { name = $FROM$ subtract = $V$ }
	change_variable = { name = $TO$ add = $V$ }
}
# An engine account (treasury | pool) -> our account.
zz_ef_post_from_eng = {
	zz_ef_eng_sub_$ENG$ = { V = $V$ }
	change_variable = { name = $TO$ add = $V$ }
}
# Our account -> an engine account.
zz_ef_post_to_eng = {
	change_variable = { name = $FROM$ subtract = $V$ }
	zz_ef_eng_add_$ENG$ = { V = $V$ }
}
# Engine account -> engine account (the treasury <-> the pool), with our account of the claim (CLAIM += V).
zz_ef_post_eng = {
	zz_ef_eng_sub_$FROM$ = { V = $V$ }
	zz_ef_eng_add_$TO$ = { V = $V$ }
	change_variable = { name = $CLAIM$ add = $V$ }
}
# Engine account -> engine account, paid by our account PAYER (PAYER -= V: an interest, a fee).
zz_ef_post_eng_paid = {
	zz_ef_eng_sub_$FROM$ = { V = $V$ }
	zz_ef_eng_add_$TO$ = { V = $V$ }
	change_variable = { name = $PAYER$ subtract = $V$ }
}
zz_ef_eng_add_treasury = {
	add_treasury = $V$
	zz_ef_tr_mark = { V = $V$ }
}
zz_ef_eng_sub_treasury = {
	add_treasury = { value = $V$ multiply = -1 }
	zz_ef_tr_unmark = { V = $V$ }
}
# every add_treasury of ours is marked (var:zz_ef_tr_posted, zeroed by the reconciliation), so the
# treasury's «прочее» is its change less the engine's budget less our postings -- what E&F and the rest wrote.
zz_ef_tr_mark = {
	if = { limit = { NOT = { has_variable = zz_ef_tr_posted } } set_variable = { name = zz_ef_tr_posted value = 0 } }
	change_variable = { name = zz_ef_tr_posted add = $V$ }
}
zz_ef_tr_unmark = {
	if = { limit = { NOT = { has_variable = zz_ef_tr_posted } } set_variable = { name = zz_ef_tr_posted value = 0 } }
	change_variable = { name = zz_ef_tr_posted subtract = $V$ }
}
zz_ef_eng_add_pool = { add_investment_pool = $V$ }
zz_ef_eng_sub_pool = { add_investment_pool = { value = $V$ multiply = -1 } }

# Country scope, at the end of the step: «прочее» = the pool's part (the pool - the banks' book, our accounts of it)
# + the treasury's part (R1б, В5: summed each step -- the change of the treasury net of debt since the last step, less
# the engine's budget of the week(s), less our postings var:zz_ef_tr_posted).
zz_ef_reconcile = {
	zz_ef_registry_init = yes
	if = {
		limit = { NOT = { has_variable = zz_ef_other } }
		change_variable = {
			name = zz_ef_bank_capital
			add = {
				value = investment_pool
				subtract = zz_ef_bank_book_v
			}
		}
		set_variable = { name = zz_ef_other value = 0 }
	}
	set_variable = { name = zz_ef_other_prev value = var:zz_ef_other }
	# the change of the levels the country's companies own since the last reconciliation (Д.R4б.22: a pool's step is a
	# purchase of levels only if they grew)
	set_variable = { name = zz_ef_t_complv value = zz_ef_comp_levels_v }
	set_variable = { name = zz_ef_d_complv value = 0 }
	if = {
		limit = { has_variable = zz_ef_complv_prev }
		set_variable = { name = zz_ef_d_complv value = var:zz_ef_t_complv }
		change_variable = { name = zz_ef_d_complv subtract = var:zz_ef_complv_prev }
	}
	set_variable = { name = zz_ef_complv_prev value = var:zz_ef_t_complv }
	remove_variable = zz_ef_t_complv
	# the same for the private owners' levels: the pool pays a new private level as it is built (most multiples the
	# companies' levels do not explain come with a private level more)
	set_variable = { name = zz_ef_t_privlv value = zz_ef_priv_levels }
	set_variable = { name = zz_ef_d_privlv value = 0 }
	if = {
		limit = { has_variable = zz_ef_privlv_prev }
		set_variable = { name = zz_ef_d_privlv value = var:zz_ef_t_privlv }
		change_variable = { name = zz_ef_d_privlv subtract = var:zz_ef_privlv_prev }
	}
	set_variable = { name = zz_ef_privlv_prev value = var:zz_ef_t_privlv }
	remove_variable = zz_ef_t_privlv
	# В.R4б.24 probe (the logs only): the change of our GDP owned abroad and of the foreign-owned GDP here since the last
	# reconciliation (the bridge's numbers) -- whether a pool's multiple with no levels changed at home bought abroad
	if = {
		limit = { zz_ef_logs_on = yes }
		set_variable = { name = zz_ef_d_aown value = 0 }
		set_variable = { name = zz_ef_d_fown value = 0 }
		if = {
			limit = { has_variable = zz_ef_aown_prev }
			set_variable = { name = zz_ef_d_aown value = { value = zz_ef_v_g_aown subtract = var:zz_ef_aown_prev } }
			set_variable = { name = zz_ef_d_fown value = { value = zz_ef_v_g_fown subtract = var:zz_ef_fown_prev } }
		}
		set_variable = { name = zz_ef_aown_prev value = zz_ef_v_g_aown }
		set_variable = { name = zz_ef_fown_prev value = zz_ef_v_g_fown }
	}
	# the engine's purchases of building levels (the price depends on the building); the script has no hook. A step
	# whose levels of the country's companies or private owners grew: the pool part's fall is their price, at most the
	# levels grown x zz_ef_lvl_price_max (Д.R4б.22, Д.R4б.25) -- this line, not «прочее» (zz_ef_lvl_buy_pool, summed).
	# The pool's multiple of zz_ef_lvl_unit (zz_ef_t_pmult) finds the state's levels it paid for. The treasury's multiples
	# -- zz_ef_lvl_buy_tr: positive -- the state sold levels to the banks (privatization, Д.R4б.23: posted pool ->
	# treasury, the banks' asset zz_ef_bank_priv), negative -- the state bought them (nationalization).
	if = { limit = { NOT = { has_variable = zz_ef_lvl_buy_pool } } set_variable = { name = zz_ef_lvl_buy_pool value = 0 } }
	if = { limit = { NOT = { has_variable = zz_ef_lvl_buy_tr } } set_variable = { name = zz_ef_lvl_buy_tr value = 0 } }
	set_variable = { name = zz_ef_f_lvl_buy_pool value = 0 }
	set_variable = { name = zz_ef_f_lvl_buy_tr value = 0 }
	set_variable = { name = zz_ef_f_lvl_priv value = 0 }
	set_variable = { name = zz_ef_t_pmult value = 0 }
	if = { limit = { NOT = { has_variable = zz_ef_bank_priv } } set_variable = { name = zz_ef_bank_priv value = 0 } }
	if = {
		limit = { has_variable = zz_ef_other_pool }
		set_variable = {
			name = zz_ef_t_lvl
			value = {
				value = investment_pool
				subtract = zz_ef_bank_book_v
				subtract = var:zz_ef_lvl_buy_pool
				subtract = var:zz_ef_other_pool
			}
		}
		if = {
			limit = {
				var:zz_ef_t_lvl < 0
				OR = {
					var:zz_ef_d_complv > 0
					var:zz_ef_d_privlv > zz_ef_lvl_privlv_min
				}
			}
			set_variable = { name = zz_ef_f_lvl_buy_pool value = zz_ef_lvl_buy_grown_v }
			change_variable = { name = zz_ef_lvl_buy_pool add = var:zz_ef_f_lvl_buy_pool }
		}
		# the pool's multiple, a purchase or not -- a privatization at home or abroad may be it
		if = {
			limit = { zz_ef_lvl_is_buy_v = 1 }
			set_variable = { name = zz_ef_t_pmult value = zz_ef_lvl_mult_v }
		}
		# a positive multiple: the pool got the price of levels its owners sold (the state's nationalization pays them,
		# the treasury's line «выкуп уровней»; a sale abroad) -- the same line with the opposite sign, not «прочее»
		else_if = {
			limit = { zz_ef_lvl_is_nat_v = 1 }
			set_variable = { name = zz_ef_f_lvl_buy_pool value = zz_ef_lvl_mult_v }
			change_variable = { name = zz_ef_lvl_buy_pool add = var:zz_ef_f_lvl_buy_pool }
		}
		remove_variable = zz_ef_t_lvl
	}
	set_variable = {
		name = zz_ef_other_pool
		value = {
			value = investment_pool
			subtract = zz_ef_bank_book_v
			subtract = var:zz_ef_lvl_buy_pool
		}
	}
	# the treasury's part: from the second reconciliation on (the first one takes the snapshot)
	if = { limit = { NOT = { has_variable = zz_ef_other_tr } } set_variable = { name = zz_ef_other_tr value = 0 } }
	set_variable = { name = zz_ef_f_other_tr value = 0 }
	set_variable = { name = zz_ef_t_trnet value = zz_ef_treasury_net_v }
	if = {
		limit = { has_variable = zz_ef_tr_prev }
		set_variable = { name = zz_ef_f_other_tr value = var:zz_ef_t_trnet }
		change_variable = { name = zz_ef_f_other_tr subtract = var:zz_ef_tr_prev }
		change_variable = { name = zz_ef_f_other_tr subtract = zz_ef_budget_steps_v }
		change_variable = { name = zz_ef_f_other_tr subtract = zz_ef_tr_posted_v }
		# a state seller of building levels (see above)
		set_variable = { name = zz_ef_t_lvl value = { value = var:zz_ef_f_other_tr multiply = -1 } }
		if = {
			limit = { zz_ef_lvl_is_buy_v = 1 }
			set_variable = { name = zz_ef_f_lvl_buy_tr value = { value = zz_ef_lvl_mult_v multiply = -1 } }
			change_variable = { name = zz_ef_lvl_buy_tr add = var:zz_ef_f_lvl_buy_tr }
			change_variable = { name = zz_ef_f_other_tr subtract = var:zz_ef_f_lvl_buy_tr }
			# the buyer is our banks (the engine privatizes to the country's investors, Д.R4б.23): the pool paid, the
			# banks hold the levels -- their asset zz_ef_bank_priv, the book falls with the pool (the capital and the
			# deposits do not move). The pool's fall of the step, if it was taken as a company's purchase, is this one.
			if = {
				limit = { zz_ef_lvl_priv_own_v = 1 }
				set_variable = { name = zz_ef_f_lvl_priv value = var:zz_ef_f_lvl_buy_tr }
				change_variable = { name = zz_ef_bank_priv add = var:zz_ef_f_lvl_priv }
				if = {
					limit = { var:zz_ef_f_lvl_buy_pool < 0 }
					change_variable = { name = zz_ef_lvl_buy_pool subtract = var:zz_ef_f_lvl_buy_pool }
					set_variable = { name = zz_ef_f_lvl_buy_pool value = 0 }
				}
				set_variable = {
					name = zz_ef_other_pool
					value = {
						value = investment_pool
						subtract = zz_ef_bank_book_v
						subtract = var:zz_ef_lvl_buy_pool
					}
				}
			}
			# our pool did not pay (a small pool cannot): the buyer is abroad -- another country's pool of the same week
			# (zz_ef_priv_abroad_seller)
			else = {
				zz_ef_priv_abroad_seller = yes
			}
		}
		# the state as the buyer (the engine's nationalization: the treasury pays a multiple of the unit)
		else_if = {
			limit = { zz_ef_lvl_is_nat_v = 1 }
			set_variable = { name = zz_ef_f_lvl_buy_tr value = { value = zz_ef_lvl_mult_v multiply = -1 } }
			change_variable = { name = zz_ef_lvl_buy_tr add = var:zz_ef_f_lvl_buy_tr }
			change_variable = { name = zz_ef_f_other_tr subtract = var:zz_ef_f_lvl_buy_tr }
		}
		# the same not a whole multiple (the week's other moves blur it): the treasury fell by more than a level's unit in a
		# step whose state's levels grew while the companies' or private owners' fell -- the state bought them, the whole
		# fall; the treasury rose in a step whose state's levels fell while theirs grew -- it sold them, the whole rise
		# (принято ночью)
		else_if = {
			limit = {
				var:zz_ef_f_other_tr > zz_ef_lvl_unit
				zz_ef_v_d_govlv <= -1
				OR = {
					var:zz_ef_d_complv > 0
					var:zz_ef_d_privlv > 0
				}
			}
			set_variable = { name = zz_ef_f_lvl_buy_tr value = var:zz_ef_f_other_tr }
			change_variable = { name = zz_ef_lvl_buy_tr add = var:zz_ef_f_lvl_buy_tr }
			change_variable = { name = zz_ef_f_other_tr subtract = var:zz_ef_f_lvl_buy_tr }
		}
		else_if = {
			limit = {
				var:zz_ef_f_other_tr < zz_ef_lvl_unit_neg
				zz_ef_v_d_govlv >= 1
				OR = {
					var:zz_ef_d_complv < 0
					var:zz_ef_d_privlv < 0
				}
			}
			set_variable = { name = zz_ef_f_lvl_buy_tr value = var:zz_ef_f_other_tr }
			change_variable = { name = zz_ef_lvl_buy_tr add = var:zz_ef_f_lvl_buy_tr }
			change_variable = { name = zz_ef_f_other_tr subtract = var:zz_ef_f_lvl_buy_tr }
		}
		remove_variable = zz_ef_t_lvl
		# the engine writes the debt off -- a declared bankruptcy, or a default cutting the debt down to the credit: the
		# treasury net of debt rises by the debt gone, nobody's money moves -- its own line (zz_ef_bk_woff), not
		# «прочее»: the principal's fall since the last step, at most the treasury part's rise
		set_variable = { name = zz_ef_f_bk_woff value = 0 }
		if = {
			limit = {
				OR = {
					has_modifier = declared_bankruptcy
					in_default = yes
				}
				has_variable = zz_ef_rec_principal
				var:zz_ef_rec_principal > principal
				var:zz_ef_f_other_tr > 0
			}
			set_variable = {
				name = zz_ef_f_bk_woff
				value = {
					value = var:zz_ef_rec_principal
					subtract = principal
					max = var:zz_ef_f_other_tr
				}
			}
			if = { limit = { NOT = { has_variable = zz_ef_bk_woff } } set_variable = { name = zz_ef_bk_woff value = 0 } }
			change_variable = { name = zz_ef_bk_woff add = var:zz_ef_f_bk_woff }
			change_variable = { name = zz_ef_f_other_tr subtract = var:zz_ef_f_bk_woff }
		}
		# the debt of a gone country the engine handed over to its heir (zz_ef_metal_heir_world marks the heir for two
		# steps): the principal's rise, at most the treasury part's fall -- its own line (zz_ef_heir_debt), not «прочее»
		set_variable = { name = zz_ef_f_heir_debt value = 0 }
		# a big rise of the treasury before the world's pass found the gone country (it runs on day 6 of the week): a gone
		# country whose capital's region we now hold is ours to inherit -- its treasury is due here at once, and the world's
		# pass does not hand it to another owner of the region (zz_ef_metal_heir_world; Д.R4б.31, 3rd kind)
		if = {
			limit = {
				var:zz_ef_f_other_tr > zz_ef_other_tr_jump_min
				has_global_variable_list = zz_ef_mh_list
			}
			save_temporary_scope_as = zz_ef_ht_c
			every_in_global_list = {
				variable = zz_ef_mh_list
				limit = {
					is_country_alive = no
					NOT = { has_variable = zz_ef_mh_tr_taken }
					has_variable = zz_ef_tr_prev
					var:zz_ef_tr_prev > 0
					has_variable = zz_ef_mh_region
					var:zz_ef_mh_region = { any_scope_state = { owner = scope:zz_ef_ht_c } }
				}
				set_variable = { name = zz_ef_mh_tr_taken value = yes }
				scope:zz_ef_ht_c = {
					if = { limit = { NOT = { has_variable = zz_ef_heir_tr_due } } set_variable = { name = zz_ef_heir_tr_due value = 0 } }
					change_variable = { name = zz_ef_heir_tr_due add = prev.var:zz_ef_tr_prev }
					if = { limit = { NOT = { has_variable = zz_ef_heir_debt_steps } } set_variable = { name = zz_ef_heir_debt_steps value = 1 } }
				}
			}
		}
		if = {
			limit = { has_variable = zz_ef_heir_debt_steps }
			if = {
				limit = {
					has_variable = zz_ef_rec_principal
					principal > var:zz_ef_rec_principal
					var:zz_ef_f_other_tr < 0
				}
				set_variable = {
					name = zz_ef_f_heir_debt
					value = {
						value = var:zz_ef_f_other_tr
						multiply = -1
						max = { value = principal subtract = var:zz_ef_rec_principal }
					}
				}
				if = { limit = { NOT = { has_variable = zz_ef_heir_debt } } set_variable = { name = zz_ef_heir_debt value = 0 } }
				change_variable = { name = zz_ef_heir_debt add = var:zz_ef_f_heir_debt }
				change_variable = { name = zz_ef_f_other_tr add = var:zz_ef_f_heir_debt }
			}
			# the gone country's treasury (zz_ef_metal_heir_world): the treasury part's rise, at most what is due -- its own
			# line (zz_ef_heir_tr)
			set_variable = { name = zz_ef_f_heir_tr value = 0 }
			if = {
				limit = {
					has_variable = zz_ef_heir_tr_due
					var:zz_ef_heir_tr_due > 0
					var:zz_ef_f_other_tr > 0
				}
				set_variable = {
					name = zz_ef_f_heir_tr
					value = {
						value = var:zz_ef_f_other_tr
						max = var:zz_ef_heir_tr_due
					}
				}
				if = { limit = { NOT = { has_variable = zz_ef_heir_tr } } set_variable = { name = zz_ef_heir_tr value = 0 } }
				change_variable = { name = zz_ef_heir_tr add = var:zz_ef_f_heir_tr }
				change_variable = { name = zz_ef_heir_tr_due subtract = var:zz_ef_f_heir_tr }
				change_variable = { name = zz_ef_f_other_tr subtract = var:zz_ef_f_heir_tr }
			}
			change_variable = { name = zz_ef_heir_debt_steps subtract = 1 }
			if = { limit = { var:zz_ef_heir_debt_steps < 1 } remove_variable = zz_ef_heir_debt_steps }
			if = {
				limit = {
					NOT = { has_variable = zz_ef_heir_debt_steps }
					has_variable = zz_ef_heir_tr_due
				}
				remove_variable = zz_ef_heir_tr_due
			}
		}
		# the treasury handed to a civil war's new country (the engine gives the rebels a part of it): the new countries
		# whose origin is this one, not counted yet -- their treasury net, at most the treasury part's fall -- its own line
		# (zz_ef_cw_tr), not «прочее»
		set_variable = { name = zz_ef_f_cw_tr value = 0 }
		if = {
			limit = { var:zz_ef_f_other_tr < zz_ef_cw_tr_min_neg }
			save_temporary_scope_as = zz_ef_cw_o
			set_variable = { name = zz_ef_t_cw value = 0 }
			every_country = {
				limit = {
					exists = civil_war_origin_country
					civil_war_origin_country = scope:zz_ef_cw_o
					NOT = { has_variable = zz_ef_cw_counted }
				}
				set_variable = { name = zz_ef_cw_counted value = 1 }
				scope:zz_ef_cw_o = { change_variable = { name = zz_ef_t_cw add = prev.zz_ef_treasury_net_v } }
			}
			if = {
				limit = { var:zz_ef_t_cw > 0 }
				set_variable = {
					name = zz_ef_f_cw_tr
					value = {

# days a privatization abroad waits for its other side in the global lists below
@zz_ef_priv_wait_days = 14

# The state's building levels privatized to another country's investors (Д.R4б.23): the seller's treasury got the sum,
# its own pool did not pay; the buyer's pool paid a multiple of the unit the same days with its companies' levels
# unchanged. The countries reconcile on their own days, so whichever comes second finds the other in a global list
# (zz_ef_priv_buyers / zz_ef_priv_sellers, the sum on the country, for zz_ef_priv_wait_days) by the same sum; the
# buyer's banks then hold the levels -- its asset zz_ef_bank_priv (its «прочее» is right at once, or at its next
# reconciliation when the seller came second). Not matched at the buyer's step -- the line «выкуп уровней» of the
# buyer until a seller comes.
# Country scope, the seller (var:zz_ef_f_lvl_buy_tr -- the sum).
zz_ef_priv_abroad_seller = {
	save_temporary_scope_as = zz_ef_pv_s
	set_variable = { name = zz_ef_t_pv_done value = 0 }
	random_in_global_list = {
		variable = zz_ef_priv_buyers
		limit = {
			has_variable = zz_ef_priv_buy_v
			var:zz_ef_priv_buy_v = scope:zz_ef_pv_s.var:zz_ef_f_lvl_buy_tr
		}
		# the buyer's step paid it to its pops as their levels' price (zz_ef_lvl_buy_post) and holds the levels -- kept
		remove_variable = zz_ef_priv_buy_v
		remove_list_global_variable = { name = zz_ef_priv_buyers target = this }
		scope:zz_ef_pv_s = { set_variable = { name = zz_ef_t_pv_done value = 1 } }
	}
	if = {
		limit = { var:zz_ef_t_pv_done = 0 }
		set_variable = { name = zz_ef_priv_sell_v value = var:zz_ef_f_lvl_buy_tr days = @zz_ef_priv_wait_days }
		if = {
			limit = { NOT = { is_target_in_global_variable_list = { name = zz_ef_priv_sellers target = this } } }
			add_to_global_variable_list = { name = zz_ef_priv_sellers target = this days = @zz_ef_priv_wait_days }
		}
	}
	remove_variable = zz_ef_t_pv_done
}
# Country scope, the buyer (var:zz_ef_t_pmult -- the pool's multiple, negative).
zz_ef_priv_abroad_buyer = {
	save_temporary_scope_as = zz_ef_pv_b
	set_variable = { name = zz_ef_t_pbuy value = { value = var:zz_ef_t_pmult multiply = -1 } }
	set_variable = { name = zz_ef_t_pv_done value = 0 }
	random_in_global_list = {
		variable = zz_ef_priv_sellers
		limit = {
			NOT = { this = scope:zz_ef_pv_b }
			has_variable = zz_ef_priv_sell_v
			var:zz_ef_priv_sell_v = scope:zz_ef_pv_b.var:zz_ef_t_pbuy
		}
		remove_variable = zz_ef_priv_sell_v
		remove_list_global_variable = { name = zz_ef_priv_sellers target = this }
		scope:zz_ef_pv_b = {
			zz_ef_priv_abroad_post = { V = var:zz_ef_t_pbuy }
			set_variable = { name = zz_ef_t_pv_done value = 1 }
		}
	}
	if = {
		limit = { var:zz_ef_t_pv_done = 0 }
		set_variable = { name = zz_ef_priv_buy_v value = var:zz_ef_t_pbuy days = @zz_ef_priv_wait_days }
		if = {
			limit = { NOT = { is_target_in_global_variable_list = { name = zz_ef_priv_buyers target = this } } }
			add_to_global_variable_list = { name = zz_ef_priv_buyers target = this days = @zz_ef_priv_wait_days }
		}
		# no seller abroad yet: the engine bought levels from private owners (pops, a change of owner; Д.R4б.31, 2nd
		# kind -- R6 posts the price to the seller, Д.R6.1) -- the line «выкуп уровней», not «прочее»; a seller found
		# later takes it back (zz_ef_priv_abroad_seller)
		change_variable = { name = zz_ef_lvl_buy_pool add = var:zz_ef_t_pmult }
		change_variable = { name = zz_ef_f_lvl_buy_pool add = var:zz_ef_t_pmult }
		set_variable = {
			name = zz_ef_other_pool
			value = {
				value = investment_pool
				subtract = zz_ef_bank_book_v
				subtract = var:zz_ef_lvl_buy_pool
			}
		}
	}
	remove_variable = zz_ef_t_pbuy
	remove_variable = zz_ef_t_pv_done
}
# Country scope, the buyer: its banks paid $V$ for levels abroad -- their asset; the pool's part follows the book
zz_ef_priv_abroad_post = {
	if = { limit = { NOT = { has_variable = zz_ef_bank_priv } } set_variable = { name = zz_ef_bank_priv value = 0 } }
	change_variable = { name = zz_ef_bank_priv add = $V$ }
	if = { limit = { NOT = { has_variable = zz_ef_f_lvl_priv } } set_variable = { name = zz_ef_f_lvl_priv value = 0 } }
	change_variable = { name = zz_ef_f_lvl_priv add = $V$ }
	if = {
		limit = { has_variable = zz_ef_other_pool }
		set_variable = {
			name = zz_ef_other_pool
			value = {
				value = investment_pool
				subtract = zz_ef_bank_book_v
				subtract = var:zz_ef_lvl_buy_pool
			}
		}
	}
}

# Country scope, the end of the reconciliation (Д.R6.1): the pool's price of levels bought from private owners this step
# (var:zz_ef_f_lvl_buy_pool < 0: the companies' or private owners' levels grew, or a multiple with no seller abroad) is
