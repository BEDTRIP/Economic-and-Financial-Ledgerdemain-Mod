# A currency of its own (Д.R8в.7; понятие «Валюта страны»): the technology currency_standards and the central bank's
# building call introduction_new_currency (09_introduction_building_lvl.txt), which calls zz_ef_cur_intro.
# A country with money and a parity of its own keeps them (Hanover leaving a union stays at its 3.32). A country on E&F's
# placeholder (no money system -> fiat at 0.01, law_no_monetary_system_with_tech_currency_standards) takes a metallic
# standard -- silver before 1873, gold after (E&F's dates) -- and a parity: from the money that went on its land -- its
# market owner's parity x the owner's money against its parity (in the new standard's metal); no such money --
# zz_ef_cur_intro_par_silver / _gold. An isolationist market owner stays on fiat (E&F: its market closed to exchange).
# Its name -- the country's own (zz_ef_cur_set, Д.R8а.3). No metal comes with it: the new central bank buys it on the
# market like every central bank (zz_ef_cb_metal_purchase_step).
zz_ef_cur_intro = {
	set_variable = { name = zz_ef_cur_intro_std value = zz_ef_std_code }
	set_variable = { name = zz_ef_cur_intro_par value = 0 }
	if = {
		limit = { has_variable = money_value_target_1 }
		set_variable = { name = zz_ef_cur_intro_par value = var:money_value_target_1 }
	}
	if = {
		limit = {
			zz_ef_cur_placeholder = yes
			NOT = {
				AND = {
					has_law = law_type:law_isolationism
					market_owner_is_root = yes
				}
			}
		}
		if = {
			limit = { game_date < 1873.1.1 }
			activate_law = law_type:law_silver_standard
			set_variable = { name = money_value_target_1 value = zz_ef_cur_intro_par_silver }
		}
		else = {
			activate_law = law_type:law_gold_standard
			set_variable = { name = money_value_target_1 value = zz_ef_cur_intro_par_gold }
		}
		zz_ef_cur_intro_par_from_market = yes
		# the players' message: a new central bank and a new currency (events/00_ef_economic_events.txt)
		if = {
			limit = { NOT = { has_variable = country_already_central_bank } }
			set_variable = country_already_central_bank
			if = {
				limit = { exists = capital }
				capital = { save_scope_as = capital_state }
			}
			save_scope_as = target_country
			every_country = {
				limit = { is_player = yes }
				trigger_event = 00_ef_economic_event.1
			}
		}
	}
	if = { limit = { zz_ef_logs_on = yes } debug_log = "EFM|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|cur_intro_parity|std_before [THIS.GetCountry.MakeScope.Var('zz_ef_cur_intro_std').GetValue|0]|par_before [THIS.GetCountry.MakeScope.Var('zz_ef_cur_intro_par').GetValue|4]|std [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_std_code')|0]|parity [THIS.GetCountry.MakeScope.Var('money_value_target_1').GetValue|4]" }
	remove_variable = zz_ef_cur_intro_std
	remove_variable = zz_ef_cur_intro_par
	# the new currency as data (scripted_effects/ld_currency_var.txt)
	zz_ef_cur_set = yes
}

# The parity from the money that went on the country's land: its market owner's x the owner's money against its parity,
# in the country's metal.
zz_ef_cur_intro_par_from_market = {
	if = {
		limit = {
			exists = market
			NOT = { market.owner = this }
			market.owner = {
				has_variable = money_value_target_1
				var:money_value_target_1 > zz_ef_cur_intro_par_min
			}
		}
		market.owner = {
			save_temporary_scope_as = zz_ef_ci_o
			set_variable = { name = zz_ef_ci_vtp value = zz_ef_value_to_parity }
		}
		set_variable = {
			name = money_value_target_1
			value = {
				value = scope:zz_ef_ci_o.var:money_value_target_1
				multiply = scope:zz_ef_ci_o.var:zz_ef_ci_vtp
			}
		}
		scope:zz_ef_ci_o = { remove_variable = zz_ef_ci_vtp }
		if = {
			limit = {
				zz_ef_metal_silver = yes
				NOT = { market.owner = { zz_ef_metal_silver = yes } }
			}
			change_variable = { name = money_value_target_1 multiply = gold_to_silver_rate }
		}
		else_if = {
			limit = {
				NOT = { zz_ef_metal_silver = yes }
				market.owner = { zz_ef_metal_silver = yes }
			}
			change_variable = { name = money_value_target_1 multiply = silver_to_gold_rate }
		}
	}
}
