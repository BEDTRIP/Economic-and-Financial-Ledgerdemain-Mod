# R8в.1 (Д.R8в.1, Д.R8в.8): the countries of one market. The engine's market is one -- its prices and goods; the money
# and the accounts are each country's. Once a month (zz_ef_msh_world_step, the world's monthly pass ld_world_month.txt)
# every country of a market of two or more gets its share of the market's production and consumption
# (var:zz_ef_msh_p / var:zz_ef_msh_c); the owner -- what the others leave (1 - their sum), so the shares add up to 1.
# The week's trade of a member (zz_ef_trade_net_week, ld_money_model_values.txt) is its share of the market's sales
# abroad and of its internal turnover less its share of the purchases: p x (X + T) - c x (M + T); the members' trade
# adds up to the market's. Between the monthly reads -- the last shares. A country alone in its market has no shares
# and takes the market's trade whole.

# World scope (the monthly pass): the shares of every country in a market it does not own, and its owner's rest.
# One pass over the countries (no market has a list of its members).
zz_ef_msh_world_step = {
	every_country = {
		limit = { has_variable = zz_ef_msh_p }
		remove_variable = zz_ef_msh_p
		remove_variable = zz_ef_msh_c
	}
	every_country = {
		limit = {
			exists = market
			NOT = { market.owner = this }
		}
		save_scope_as = zz_ef_msh_who
		market = { save_scope_as = zz_ef_msh_mk }
		zz_ef_msh_read = { S = production V = zz_ef_msh_p }
		zz_ef_msh_read = { S = consumption V = zz_ef_msh_c }
		market.owner = {
			if = {
				limit = { NOT = { has_variable = zz_ef_msh_p } }
				set_variable = { name = zz_ef_msh_p value = 1 }
				set_variable = { name = zz_ef_msh_c value = 1 }
			}
			change_variable = { name = zz_ef_msh_p subtract = scope:zz_ef_msh_who.var:zz_ef_msh_p }
			change_variable = { name = zz_ef_msh_c subtract = scope:zz_ef_msh_who.var:zz_ef_msh_c }
		}
	}
	# an owner's rest is not below 0 (the members' shares are read within a step of the search)
	every_country = {
		limit = {
			has_variable = zz_ef_msh_p
			market.owner = this
		}
		if = { limit = { var:zz_ef_msh_p < 0 } set_variable = { name = zz_ef_msh_p value = 0 } }
		if = { limit = { var:zz_ef_msh_c < 0 } set_variable = { name = zz_ef_msh_c value = 0 } }
	}
}
