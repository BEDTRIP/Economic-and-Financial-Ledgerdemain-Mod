# The companies' shares (R6.6; Д.R6.3, Д.R6.8, Д.R6.10, Д.R6.11, Д.R6.14; понятия «Компания», «Биржа», «Требование»).
# A share is a claim of kind s on its company (the debtor -- a company scope): the register's maps on the holder country
# zz_ef_rq_pp_s (the pops), zz_ef_rq_bk_s (the banks), zz_ef_rq_co_s (the shares a company has not sold yet -- on its
# country), key -- the company, value -- the shares; the company's var:zz_ef_rq_out_s -- all its shares
# (= var:zz_ef_co_shares), its holder countries -- the lists zz_ef_co_h_pp / zz_ef_co_h_bk (zz_ef_rq_hook_s,
# ld_claims.txt). One kind of share (Д.R6.3); a gone company's shares are lost with it (its key drops out of the maps'
# passes -- the probe R6.1; the holders' totals follow the passes, zz_ef_co_holders_month).
#   listing: the headquarters of the listed companies (ld_listing.txt) mark them, zz_ef_co_go_public / _private;
#   IPO (Д.R6.10): at its first listing the company issues zz_ef_co_shares = its capitalization at the fair price; the
#     share zz_ef_co_ipo_share of them it offers on the exchange (zz_ef_rq_co_s; the buyers' money -- its account), the
#     rest -- to the pops of its country (zz_ef_rq_pp_s), free;
#   the price (Д.R6.11): the capitalization at the fair price = the year's net profit by the account (zz_ef_co_np_y: the
#     income less the outflow, a moving average) / (the government's rate + zz_ef_equity_premium), at least the
#     account; a share's fair price -- it over the shares x the stock's multiplier (var:zz_ef_xmul_s -- the book's
#     unfilled orders, zz_ef_xch_mul_move); the gain of holding -- its yield at that price less the holder's rate;
#   trading: the shares of a company trade on the exchange its country trades on (var:zz_ef_xch of its owner) -- the
#     holders of that route (zz_ef_xch_post_sells_s, zz_ef_xch_post_buys_s, from zz_ef_xch_step; the book, the
#     clearing and the deals -- ld_exchange_trade.txt, kind s); no fund and no currency points for shares;
#   the dividends (Д.R6.8): a listed company's dividend goes to the holders of its shares by their part, its unsold
#     shares' part stays on its account; a company with no shares pays all of it to the pops of its country
#     (ld_company_accounts.txt). A holder abroad is paid through the clearing (its flow abroad and its pool).
# The numbers -- script_values/ld_company_shares_values.txt. Log: EFJ|...|comp (shares, the price), EFI (n_s, bad_s).

# ---- listing and IPO ------------------------------------------------------------------------------------------------

# COMPANY scope, from the listing's dispatcher (zz_ef_listing_mark, monthly): listed; the first time -- the IPO.
zz_ef_co_go_public = {
	if = {
		limit = { NOT = { has_variable = zz_ef_co_listed } }
		set_variable = { name = zz_ef_co_listed value = 1 }
	}
	if = {
		limit = { NOT = { has_variable = zz_ef_co_shares } }
		zz_ef_co_ipo = yes
	}
}
zz_ef_co_go_private = {
	if = {
		limit = { has_variable = zz_ef_co_listed }
		remove_variable = zz_ef_co_listed
	}
}

# COMPANY scope: the IPO (Д.R6.10) -- the shares issued at the capitalization at the fair price (one share -- one unit
# of its country's money at it), the offered part to the company's own holding (its country's zz_ef_rq_co_s), the rest
# to the pops of its country.
zz_ef_co_ipo = {
	save_temporary_scope_as = zz_ef_co_c
	set_variable = { name = zz_ef_co_shares value = zz_ef_co_cap_fair_v }
	set_variable = { name = zz_ef_co_ipo_date value = 1 }
	owner = {
		set_variable = { name = zz_ef_t_ipo value = { value = scope:zz_ef_co_c.var:zz_ef_co_shares multiply = zz_ef_co_ipo_share } }
		zz_ef_rq_add = { H = co K = s D = scope:zz_ef_co_c V = var:zz_ef_t_ipo }
		set_variable = { name = zz_ef_t_ipo value = { value = scope:zz_ef_co_c.var:zz_ef_co_shares subtract = var:zz_ef_t_ipo } }
		zz_ef_rq_add = { H = pp K = s D = scope:zz_ef_co_c V = var:zz_ef_t_ipo }
		remove_variable = zz_ef_t_ipo
	}
	if = {
		limit = { zz_ef_logs_on = yes }
		debug_log = "EFJ|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCompany.GetCountry.GetNameNoFormatting]|ipo|[THIS.GetCompany.GetNameNoFormatting]|shares [THIS.GetCompany.MakeScope.Var('zz_ef_co_shares').GetValue|0]|np_y [THIS.GetCompany.MakeScope.ScriptValue('zz_ef_co_np_y_v')|0]|acc [THIS.GetCompany.MakeScope.ScriptValue('zz_ef_co_acc_v')|0]|px [THIS.GetCompany.MakeScope.ScriptValue('zz_ef_co_px_v')|3]"
	}
}

# ---- the month ------------------------------------------------------------------------------------------------------

# COMPANY scope, monthly (zz_ef_co_month_step, after the last month's income and outflow): the year's net profit as a
# moving average of the months (zz_ef_co_np_months).
zz_ef_co_np_update = {
	if = { limit = { NOT = { has_variable = zz_ef_co_np_y } } set_variable = { name = zz_ef_co_np_y value = 0 } }
	change_variable = {
		name = zz_ef_co_np_y
		add = {
			value = zz_ef_co_in_last_v
			subtract = zz_ef_co_out_last_v
			multiply = 12
			subtract = var:zz_ef_co_np_y
			divide = zz_ef_co_np_months
		}
	}
}

# COUNTRY scope, monthly (zz_ef_co_month_step): the holders' totals of shares by the maps' passes -- a gone company's
# shares (its key drops out of the passes) leave them.
zz_ef_co_holders_month = {
	zz_ef_co_tot_by_pass = { H = pp }
	zz_ef_co_tot_by_pass = { H = bk }
	zz_ef_co_tot_by_pass = { H = co }
}
zz_ef_co_tot_by_pass = {
	if = {
		limit = { has_variable_map = zz_ef_rq_$H$_s }
		set_variable = { name = zz_ef_t_tbp value = 0 }
		save_temporary_scope_as = zz_ef_co_hd
		every_key_in_variable_map = {
			variable = zz_ef_rq_$H$_s
			save_temporary_scope_as = zz_ef_rq_key
			scope:zz_ef_co_hd = {
				zz_ef_rq_$H$_s_read = yes
				change_variable = { name = zz_ef_t_tbp add = var:zz_ef_rq_h }
			}
		}
		set_variable = { name = zz_ef_rq_tot_$H$_s value = var:zz_ef_t_tbp }
		remove_variable = zz_ef_t_tbp
	}
}

# COMPANY scope, from the dividends (zz_ef_co_month_step), var:zz_ef_co_div_last -- the month's dividend: paid to the
# holders of its shares by their part; the unsold shares' part stays on the account. Sets var:zz_ef_t_cdp -- what left
# the account (the country's companies' deposits move by it there).
zz_ef_co_div_holders = {
	save_temporary_scope_as = zz_ef_co_c
	set_variable = { name = zz_ef_t_cdp value = 0 }
	set_variable = { name = zz_ef_t_dps value = { value = var:zz_ef_co_div_last divide = { value = var:zz_ef_co_shares min = 1 } } }
	zz_ef_co_div_pay_list = { H = pp }
	zz_ef_co_div_pay_list = { H = bk }
	remove_variable = zz_ef_t_dps
}
zz_ef_co_div_pay_list = {
	if = {
		limit = { has_variable_list = zz_ef_co_h_$H$ }
		every_in_list = {
			variable = zz_ef_co_h_$H$
			save_temporary_scope_as = zz_ef_co_x
			scope:zz_ef_co_c = { save_temporary_scope_as = zz_ef_rq_key }
			zz_ef_rq_$H$_s_read = yes
			if = {
				limit = { var:zz_ef_rq_h > 0 }
				set_variable = { name = zz_ef_t_dv value = { value = var:zz_ef_rq_h multiply = scope:zz_ef_co_c.var:zz_ef_t_dps } }
				scope:zz_ef_co_c = { change_variable = { name = zz_ef_t_cdp add = scope:zz_ef_co_x.var:zz_ef_t_dv } }
				zz_ef_co_div_to_$H$ = yes
				remove_variable = zz_ef_t_dv
			}
		}
	}
}
# HOLDER country scope, scope:zz_ef_co_c, var:zz_ef_t_dv -- the dividend in the company's country's money: into the
# pops' deposits / the banks' capital; from abroad -- through the clearing (both countries' flows abroad and pools).
zz_ef_co_div_to_pp = {
	zz_ef_co_div_cross = yes
	if = { limit = { NOT = { has_variable = zz_ef_f_co_div } } set_variable = { name = zz_ef_f_co_div value = 0 } }
	change_variable = { name = zz_ef_pop_deposits add = var:zz_ef_t_dv }
	change_variable = { name = zz_ef_pop_savings add = var:zz_ef_t_dv }
	change_variable = { name = zz_ef_f_co_div add = var:zz_ef_t_dv }
}
zz_ef_co_div_to_bk = {
	zz_ef_co_div_cross = yes
	if = { limit = { NOT = { has_variable = zz_ef_bank_capital } } set_variable = { name = zz_ef_bank_capital value = 0 } }
	change_variable = { name = zz_ef_bank_capital add = var:zz_ef_t_dv }
}
# HOLDER country scope: a holder abroad -- var:zz_ef_t_dv into its money; the company's country pays it out of its pool
# and abroad, the holder takes it into its pool from abroad (the flows of the exchange's deals: zz_ef_xch_ext_acc,
# zz_ef_xch_pool_acc, read at the next weekly step -- zz_ef_xch_step)
zz_ef_co_div_cross = {
	if = {
		limit = { NOT = { scope:zz_ef_co_c.owner = this } }
		save_temporary_scope_as = zz_ef_co_x
		scope:zz_ef_co_c.owner = {
			set_variable = { name = zz_ef_xd_xe value = { value = scope:zz_ef_co_x.var:zz_ef_t_dv multiply = -1 } }
			zz_ef_co_flow_abroad = yes
		}
		set_variable = {
			name = zz_ef_t_dv
			value = {
				value = var:zz_ef_t_dv
				multiply = scope:zz_ef_co_c.owner.zz_ef_xch_gpm_v
				divide = zz_ef_xch_gpm_v
			}
		}
		set_variable = { name = zz_ef_xd_xe value = var:zz_ef_t_dv }
		zz_ef_co_flow_abroad = yes
	}
}
# COUNTRY scope, var:zz_ef_xd_xe: money in (+) / out (-) of the pool from / to abroad
zz_ef_co_flow_abroad = {
	add_investment_pool = var:zz_ef_xd_xe
	if = { limit = { NOT = { has_variable = zz_ef_xch_pool_acc } } set_variable = { name = zz_ef_xch_pool_acc value = 0 } }
	change_variable = { name = zz_ef_xch_pool_acc add = var:zz_ef_xd_xe }
	zz_ef_xch_ext_in = yes
	remove_variable = zz_ef_xd_xe
}

# Any scope, the world's month (zz_ef_world_month_ef): every exchange's list of the listed companies that trade on it
# (var:zz_ef_xch of their country) -- the buyers' candidates (zz_ef_xch_post_buys_s).
zz_ef_co_world_month = {
	every_country = {
		limit = { has_variable_list = zz_ef_xls }
		clear_variable_list = zz_ef_xls
	}
	every_country = {
		limit = {
			has_variable = zz_ef_xch
			exists = var:zz_ef_xch
			any_company = { has_variable = zz_ef_co_listed has_variable = zz_ef_co_shares }
		}
		save_temporary_scope_as = zz_ef_co_cty
		every_company = {
			limit = {
				has_variable = zz_ef_co_listed
				has_variable = zz_ef_co_shares
			}
			save_temporary_scope_as = zz_ef_co_c
			scope:zz_ef_co_cty.var:zz_ef_xch = { add_to_variable_list = { name = zz_ef_xls target = scope:zz_ef_co_c } }
		}
	}
}

# ---- the orders -----------------------------------------------------------------------------------------------------

# Holder scope, scope:zz_ef_xch_d -- a company: the gain of holding its shares, as zz_ef_xch_eval for the other kinds --
# the yield at the market price (zz_ef_co_yield_v) less the risk premium of its country and the cost of leaving;
# the fees and the cost of leaving are reckoned against its country (scope:zz_ef_xch_d is its owner meanwhile).
zz_ef_xch_eval_s = {
	scope:zz_ef_xch_d = { save_temporary_scope_as = zz_ef_xch_dco }
	scope:zz_ef_xch_dco.owner = { save_temporary_scope_as = zz_ef_xch_d }
	set_variable = { name = zz_ef_xe_base value = scope:zz_ef_xch_dco.zz_ef_co_yield_v }
	change_variable = { name = zz_ef_xe_base subtract = scope:zz_ef_xch_d.zz_ef_gov_rate_premium }
	change_variable = { name = zz_ef_xe_base subtract = zz_ef_xch_exit_v }
	set_variable = { name = zz_ef_xe_fee value = zz_ef_xch_fee_buy_v }
	set_variable = { name = zz_ef_xe_r value = zz_ef_deposit_rate }
	set_variable = { name = zz_ef_xe_rtr value = zz_ef_money_rate }
	set_variable = { name = zz_ef_xe_rcb value = zz_ef_money_rate }
	scope:zz_ef_xch_dco = { save_temporary_scope_as = zz_ef_xch_d }
}

# Holder scope, scope:zz_ef_xch_e: the sell orders of the holder T (pp | bk) for its shares that trade on our exchange --
# a small position at the lowest step, else at the fair price x (1 + the gain), kept if it would buy more.
zz_ef_xch_post_sells_s = {
	if = {
		limit = { has_variable_map = zz_ef_rq_$T$_s }
		set_variable = { name = zz_ef_xo_pf value = zz_ef_xch_portfolio_$T$ }
		save_temporary_scope_as = zz_ef_xo_h
		every_key_in_variable_map = {
			variable = zz_ef_rq_$T$_s
			save_temporary_scope_as = zz_ef_xch_d
			scope:zz_ef_xo_h = { zz_ef_xch_sell_one_s = { T = $T$ } }
		}
	}
}
zz_ef_xch_sell_one_s = {
	if = {
		limit = {
			scope:zz_ef_xch_d = { zz_ef_co_trades_on = { E = scope:zz_ef_xch_e } }
			NOT = { zz_ef_xch_banned_s = yes }
			NOT = { is_key_in_variable_map = { name = zz_ef_xa_$T$_s target = scope:zz_ef_xch_d } }
		}
		scope:zz_ef_xch_d = { save_temporary_scope_as = zz_ef_rq_key }
		zz_ef_rq_$T$_s_read = yes
		set_variable = { name = zz_ef_xo_a value = var:zz_ef_rq_h }
		if = {
			limit = { var:zz_ef_xo_a > 0 }
			set_variable = { name = zz_ef_xo_s value = 0 }
			set_variable = { name = zz_ef_xo_g value = { value = var:zz_ef_xo_a multiply = scope:zz_ef_xch_d.zz_ef_xq_gold_s } }
			if = {
				limit = { var:zz_ef_xo_g < zz_ef_xo_small_lim_v }
				set_variable = { name = zz_ef_xo_s value = 1 }
			}
			else = {
				zz_ef_xch_eval_s = yes
				zz_ef_xch_g_$T$ = yes
				if = {
					limit = { var:zz_ef_xe_g > var:zz_ef_xe_fee }
					# it would pay over the fair price for more: it keeps what it has
				}
				else = {
					set_variable = { name = zz_ef_xo_s value = zz_ef_xch_ask_step_v }
				}
			}
			if = {
				limit = {
					var:zz_ef_xo_s >= 1
					var:zz_ef_xo_s <= zz_ef_xq_top
				}
				zz_ef_xch_order_put = { T = $T$ K = s SIDE = s }
			}
		}
	}
}
# Holder scope (a company's country), scope:zz_ef_xch_e: the shares its companies have not sold yet (the IPO) offered at
# the fair price
zz_ef_xch_post_sells_co = {
	if = {
		limit = { has_variable_map = zz_ef_rq_co_s }
		save_temporary_scope_as = zz_ef_xo_h
		every_key_in_variable_map = {
			variable = zz_ef_rq_co_s
			save_temporary_scope_as = zz_ef_xch_d
			scope:zz_ef_xo_h = {
				if = {
					limit = {
						scope:zz_ef_xch_d = { zz_ef_co_trades_on = { E = scope:zz_ef_xch_e } }
						NOT = { is_key_in_variable_map = { name = zz_ef_xa_co_s target = scope:zz_ef_xch_d } }
					}
					scope:zz_ef_xch_d = { save_temporary_scope_as = zz_ef_rq_key }
					zz_ef_rq_co_s_read = yes
					set_variable = { name = zz_ef_xo_a value = var:zz_ef_rq_h }
					set_variable = { name = zz_ef_xo_s value = zz_ef_xq_mid }
					if = {
						limit = { var:zz_ef_xo_a > 0 }
						zz_ef_xch_order_put = { T = co K = s SIDE = s }
					}
				}
			}
		}
	}
}
# Holder scope, scope:zz_ef_xch_e: the buy orders for shares -- the best of the listed companies that trade on our
# exchange (its list zz_ef_xls) by the gain over the costs of buying, for the pops and for the banks; the week's budget
# at the bid (the same budget as for money).
zz_ef_xch_post_buys_s = {
	set_variable = { name = zz_ef_xo_bv_bk_s value = 0 }
	set_variable = { name = zz_ef_xo_bv_pp_s value = 0 }
	if = {
		limit = {
			scope:zz_ef_xch_e = { has_variable_list = zz_ef_xls }
			NOT = { has_law_or_variant = law_type:law_isolationism }
		}
		save_temporary_scope_as = zz_ef_xo_h
		scope:zz_ef_xch_e = {
			every_in_list = {
				variable = zz_ef_xls
				limit = { zz_ef_co_trades_on = { E = scope:zz_ef_xch_e } }
				save_temporary_scope_as = zz_ef_xch_d
				scope:zz_ef_xo_h = {
					if = {
						limit = { NOT = { zz_ef_xch_banned_s = yes } }
						zz_ef_xch_eval_s = yes
						zz_ef_xch_best = { T = bk K = s }
						zz_ef_xch_best = { T = pp K = s }
					}
				}
			}
		}
	}
	zz_ef_xch_buy_put = { T = bk K = s }
	zz_ef_xch_buy_put = { T = pp K = s }
}

# ---- the deals ------------------------------------------------------------------------------------------------------

# Holder scope (zz_ef_xch_deal), var:zz_ef_xd_x -- what we pay (+) / get (-) in our money: the pops -- out of / into their
# deposits, the banks -- their asset (zz_ef_bank_inv, re-valued at the shares' price: zz_ef_bank_inv_revalue), the
# company that sells its shares (the IPO) -- into its account; the money moves with the pool.
zz_ef_xch_pay_pp_s = { zz_ef_xch_pay_pp_m = yes }
zz_ef_xch_pay_bk_s = {
	zz_ef_bank_inv_add = { V = var:zz_ef_xd_x }
	zz_ef_xch_pool_pay = yes
}
zz_ef_xch_pay_co_s = {
	save_temporary_scope_as = zz_ef_co_ps
	scope:zz_ef_xch_d = {
		if = { limit = { NOT = { has_variable = zz_ef_co_acc } } set_variable = { name = zz_ef_co_acc value = 0 } }
		change_variable = { name = zz_ef_co_acc subtract = scope:zz_ef_co_ps.var:zz_ef_xd_x }
		if = { limit = { NOT = { has_variable = zz_ef_co_ipo_got } } set_variable = { name = zz_ef_co_ipo_got value = 0 } }
		change_variable = { name = zz_ef_co_ipo_got subtract = scope:zz_ef_co_ps.var:zz_ef_xd_x }
	}
	if = { limit = { NOT = { has_variable = zz_ef_comp_dep } } set_variable = { name = zz_ef_comp_dep value = 0 } }
	change_variable = { name = zz_ef_comp_dep subtract = var:zz_ef_xd_x }
	zz_ef_xch_pool_pay = yes
}
# the deal's fee for shares: the costs of buying reckoned against the company's country (zz_ef_xch_deal_fee_<K>,
# ld_exchange_trade.txt)
zz_ef_xch_deal_fee_s = {
	scope:zz_ef_xch_d = { save_temporary_scope_as = zz_ef_xch_dco }
	scope:zz_ef_xch_dco.owner = { save_temporary_scope_as = zz_ef_xch_d }
	set_variable = { name = zz_ef_xd_fee value = { value = zz_ef_xch_fee_buy_v multiply = var:zz_ef_xd_g } }
	scope:zz_ef_xch_dco = { save_temporary_scope_as = zz_ef_xch_d }
}
