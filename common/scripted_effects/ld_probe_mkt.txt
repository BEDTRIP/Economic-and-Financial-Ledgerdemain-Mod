# Probe (В.R4б.28): does a buyer pay for the part of its order the market did not fill? Every 4th week, France and
# Great Britain: the market's orders and balance (EFP|mkt) and every building with inputs -- what it paid for them
# (GetLastMaterialExpenses), the value of its consumption (GetTotalConsumptionValue), its input availability
# (CalcInputAvailabilityOutputMult), its revenue (EFP|bld). Paid for the whole order -- paid / value grows as the
# availability falls. Read by tools/ld_probe_mkt.py. Country scope, the logs only; removed after the probe.
zz_ef_probe_mkt = {
	if = {
		limit = {
			zz_ef_sched_week_mod4 = 0
			OR = {
				AND = { exists = c:FRA THIS = c:FRA }
				AND = { exists = c:GBR THIS = c:GBR }
			}
		}
		debug_log = "EFP|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|mkt|buy [THIS.GetCountry.GetMarket.GetTotalBuyOrders|0]|sell [THIS.GetCountry.GetMarket.GetTotalSellOrders|0]|bal [THIS.GetCountry.GetMarket.GetBalance|0]|gap [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_mkt_gap_week')|0]"
		every_scope_building = {
			limit = { level > 0 }
			debug_log = "EFP|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetBuilding.GetOwner.GetNameNoFormatting]|bld|t [THIS.GetBuilding.GetNameNoFormatting]|mat [THIS.GetBuilding.GetLastMaterialExpenses|2]|cv [THIS.GetBuilding.GetTotalConsumptionValue|2]|av [THIS.GetBuilding.CalcInputAvailabilityOutputMult|3]|rev [THIS.GetBuilding.GetLastRevenue|2]|sh [THIS.GetBuilding.HasAnyInputShortage]"
		}
	}
}
