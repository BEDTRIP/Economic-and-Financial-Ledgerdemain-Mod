# The engine's market gap of the week: the market's goods sold less bought by the orders, at
# the market's price -- money the engine adds (+) or takes away (-) between buyers and sellers. Country scope, the
# market's owner (the EFM line).
zz_ef_mkt_gap_week = {
	value = 0
	market = {
		every_market_goods = {
			add = {
				value = market_goods_sell_orders
				subtract = market_goods_buy_orders
				multiply = zz_ef_mg_price_v
			}
		}
	}
}
# MARKET_GOODS scope: the price of the goods in this market
zz_ef_mg_price_v = {
	value = goods.base_price
	multiply = {
		value = 1
		add = {
			value = market_goods_pricier
			min = 0
		}
		subtract = {
			value = market_goods_cheaper
			min = 0
		}
	}
}

# The companies' accounts (Д.R6.2; the log only, ld_acc_log.txt): every finished building level of the country, the
# buildings' cash a level, a company's own levels
zz_ef_all_levels_v = {
	value = 0
	every_scope_building = {
		limit = {
			level > 0
			NOT = { is_under_construction = yes }
		}
		add = level
	}
}
zz_ef_cash_per_level_v = {
	value = 0
	if = {
		limit = { zz_ef_all_levels_v > 0 }
		value = zz_ef_building_cash
		divide = zz_ef_all_levels_v
	}
}
