# The exchange (scripted_effects/ld_exchange.txt).
# the base fee of a deal, a share of its value (Д.R8.5: 0.5 %). The exchange's building (Д.R9в.3): its active trading
# method's factor -- the outcry (stock_exchange, joint_stock_companies), the banking houses (investment_banks,
# mutual_funds), the ticker (international_exchange_standards, modern_financial_instruments): the middle of the two
# technologies' steps of 15 % each (принято ночью); the economic system's law -- laissez-faire a bonus, the command
# economy a malus (принято ночью). A country without the building (not an exchange) -- each financial technology takes
# zz_ef_xch_fee_tech_cut of the base off, down to zz_ef_xch_fee_min of it.
zz_ef_xch_fee_base = 0.005
zz_ef_xch_fee_tech_cut = 0.15
zz_ef_xch_fee_min = 0.1
zz_ef_xch_fee_pm_outcry = 0.85
zz_ef_xch_fee_pm_houses = 0.55
zz_ef_xch_fee_pm_ticker = 0.25
zz_ef_xch_fee_law_laissez_faire = 0.9
zz_ef_xch_fee_law_command = 1.25
zz_ef_xch_fee_v = {
	value = 1
	if = {
		limit = { zz_ef_xb_has = yes }
		if = {
			limit = { var:zz_ef_fc_state = { any_scope_building = { is_building_type = building_ld_exchange has_active_production_method = pm_ld_exchange_ticker } } }
			value = zz_ef_xch_fee_pm_ticker
		}
		else_if = {
			limit = { var:zz_ef_fc_state = { any_scope_building = { is_building_type = building_ld_exchange has_active_production_method = pm_ld_exchange_banking_houses } } }
			value = zz_ef_xch_fee_pm_houses
		}
		else = {
			value = zz_ef_xch_fee_pm_outcry
		}
		if = { limit = { has_law_or_variant = law_type:law_laissez_faire } multiply = zz_ef_xch_fee_law_laissez_faire }
		else_if = { limit = { has_law_or_variant = law_type:law_command_economy } multiply = zz_ef_xch_fee_law_command }
	}
	else = {
		if = { limit = { has_technology_researched = stock_exchange } subtract = zz_ef_xch_fee_tech_cut }
		if = { limit = { has_technology_researched = joint_stock_companies } subtract = zz_ef_xch_fee_tech_cut }
		if = { limit = { has_technology_researched = investment_banks } subtract = zz_ef_xch_fee_tech_cut }
		if = { limit = { has_technology_researched = mutual_funds } subtract = zz_ef_xch_fee_tech_cut }
		if = { limit = { has_technology_researched = international_exchange_standards } subtract = zz_ef_xch_fee_tech_cut }
		if = { limit = { has_technology_researched = modern_financial_instruments } subtract = zz_ef_xch_fee_tech_cut }
		min = zz_ef_xch_fee_min
	}
	multiply = zz_ef_xch_fee_base
}
# The surcharge on a claim on another country (Д.R8б.18): the scope -- the buyer, scope:zz_ef_xch_d -- the debtor. None
# for one's own, between an overlord and its subject, in one market and with the treaty of foreign investment rights;
# otherwise by the buyer's trade policy: free trade 0.5 %, mercantilism 1 %, protectionism 2 %, the canton system 3 %
# (isolationism -- no deal at all, zz_ef_xch_banned).
zz_ef_xch_sur_free_trade = 0.005
zz_ef_xch_sur_mercantilism = 0.01
zz_ef_xch_sur_protectionism = 0.02
zz_ef_xch_sur_canton = 0.03
zz_ef_xch_surcharge_v = {
	value = 0
	if = {
		limit = { NOT = { zz_ef_xch_free_with = { D = scope:zz_ef_xch_d } } }
		if = { limit = { has_law_or_variant = law_type:law_free_trade } value = zz_ef_xch_sur_free_trade }
		else_if = { limit = { has_law_or_variant = law_type:law_mercantilism } value = zz_ef_xch_sur_mercantilism }
		else_if = { limit = { has_law_or_variant = law_type:law_protectionism } value = zz_ef_xch_sur_protectionism }
		else_if = { limit = { has_law_or_variant = law_type:law_canton_system } value = zz_ef_xch_sur_canton }
	}
}

# The exchange's building (Д.R9в.2): its levels by the exchange's capitalization in gold (var:zz_ef_xlm_cap) -- one,
# and one more for each doubling above zz_ef_xb_cap_base (the largest exchange of 1841, ~64 mln, -- 10 levels; принято
# ночью), up to 15.
zz_ef_xb_cap_base = 100000

	if = { limit = { zz_ef_xb_ratio_v >= 16 } add = 1 }
	if = { limit = { zz_ef_xb_ratio_v >= 32 } add = 1 }
	if = { limit = { zz_ef_xb_ratio_v >= 64 } add = 1 }
	if = { limit = { zz_ef_xb_ratio_v >= 128 } add = 1 }
	if = { limit = { zz_ef_xb_ratio_v >= 256 } add = 1 }
	if = { limit = { zz_ef_xb_ratio_v >= 512 } add = 1 }
