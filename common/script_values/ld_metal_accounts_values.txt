#############################################
# The metal accounts.
# Effects: scripted_effects/ld_metal_accounts.txt. Every unit of metal on an account was bought on the market
# (or is the start's stock, logged as such): the CB (the monetary authority's var:zz_ef_am_gold / zz_ef_am_silver,
# Д.R4а.2), the banks (var:zz_ef_bankm_gold / _silver, one per country -- by company in Ф3) and the pops
# (var:zz_ef_popm_gold / _silver).
#
# Units -- E&F's reserve units. Gold: a good x 100 (gold costs 100: 1 unit = 1 money at the base price). Silver: E&F
# counts it against gold by WEIGHT, 1:15.5 (silver_to_gold_rate, ours: the market prices / 15.5), so one silver
# good (base price 50 = half a gold good) is 50 gold units = 50 x 15.5 = 775 silver units -- the value in the
# reserves equals the market value at any prices. Was x 100 for silver too (Д2.7а): a silver standard's CB got
# 7.75 times less cover than it paid for.
#############################################

zz_ef_gold_units_per_good = 100
zz_ef_silver_units_per_good = 775

# the engine's money is one unit for every country -- 1 unit of money = 1 gold unit of the reserves
# at the base price (gold costs 100 = one good). The metal is counted in it at a fixed rate, the parity is not in it
# (the parity -- var:money_value_target_1, E&F's historical one -- sets only the rate between currencies and the labels
# in national units). Native metal units per unit of money: gold and bimetallism (counted in gold) 1, a silver
# standard -- silver at the market rate.
zz_ef_metal_per_money = {
	value = 1
	if = {
		limit = { zz_ef_metal_silver = yes }
		value = gold_to_silver_rate
	}
}
# Gold per unit of money: 1 (Д.R1б.1).
zz_ef_gold_per_money_par = 1

# The country keeps its metal in silver (silver standard; no CB and no gold / bimetallic standard -- Asia's silver
# economies) or in gold. Bimetallism: both, half and half.
zz_ef_metal_is_silver = {
	value = 0
	if = {
		limit = {
			OR = {
				zz_ef_metal_silver = yes
				AND = {
					NOT = { has_modifier = has_central_bank }
					NOT = { zz_ef_metal_gold = yes }
					NOT = { zz_ef_metal_bimet = yes }
				}
			}
		}
		value = 1
	}
}
zz_ef_metal_silver_share = {
	value = zz_ef_metal_is_silver
	if = {
		limit = { zz_ef_metal_bimet = yes }
		value = 0.5
	}
}

# ---- step 6: the banks' reserve norm (Д2.26 а) ------------------------------
# 15% at the key rate 6.5%, 10% at 2%, 20% at 12%, linear (two pieces), 5..25%.
zz_ef_reserve_norm = {
	value = zz_ef_money_rate
	subtract = 0.065
	if = {
		limit = { zz_ef_money_rate < 0.065 }
		multiply = 1.1111
	}
	else = {
		multiply = 0.9091
	}
	add = 0.15
	min = 0.05
	max = 0.25
}

# The ceiling of the banks' credit (Д2.26 а, Д2.27 а): business credit + consumer credit <= the banks' metal at
# parity / the reserve norm, times the cover's credit multiplier (1 at 40% cover, 0 at 20%). The room under it is
# what the banks may lend more: the CB's credit to them and the new consumer credit stop at it.
zz_ef_credit_now = {
	value = zz_ef_bc_debt
	add = zz_ef_cc_debt
}
zz_ef_credit_limit = {
	value = zz_ef_bankm_money
	divide = zz_ef_reserve_norm
	multiply = zz_ef_cb_cover_credit_mult
	min = 0
}
zz_ef_credit_room = {
	value = zz_ef_credit_limit
	subtract = zz_ef_credit_now
	min = 0
}

zz_ef_sav_norm_gdp = {

	min = 0
	multiply = zz_ef_sav_norm_gdp
}

# ---- step 7: the CB as an account (Д2.28 а) ----------------------------------------
# Its building's purchase, percent on the method's input: (40% - cover) x 5000 -- +500% at 30% cover, +1000% (the cap) at
# 20%, the method's own at 40%, none from 42% (the method buys ~44 gold a week for Britain); a weaker push (x 1500, at
# most +300%) let Britain's cover fall 0.39 -> 0.27 in two years at the cap (~0.9M a year against the outflow abroad and
# M2 +10% a year). Metal standards with a CB.
zz_ef_cb_buy_mult = {
	value = zz_ef_cover_normal
	subtract = zz_ef_cb_cover
	multiply = 5000
	min = -100
	max = 1000
	# until the CB's start metal (Д.R1б.2, var:zz_ef_cb_start_due): the method's own -- the start's cover reads 0.33 for
	# its first weeks, and buying by it sends gold in London to 161
	if = {
		limit = { has_variable = zz_ef_cb_start_due }
		value = 0
	}
}
# The sale of a surplus: 2% a week of the metal over 60% cover, in goods (bimetallism sells gold).
zz_ef_cb_sell_cover = 0.6
zz_ef_cb_sell_units = {
	value = 0
	if = {
		limit = {
			zz_ef_cb_cover > zz_ef_cb_sell_cover
			# not before the CB's start metal (Д.R1б.2: the first step reads 0.75 cover before the engine fills the
			# business cash -- Britain would sell 440 gold)
			NOT = { has_variable = zz_ef_cb_start_due }
		}
		value = zz_ef_cb_cover
		subtract = zz_ef_cb_sell_cover
		multiply = money_supply
		multiply = zz_ef_cb_valuation
		multiply = 0.02
		# over 100% cover another 5% of that excess a week (2% alone leaves the surplus countries at a cover of 2-3.6 --
		# the US 2.9, Belgium 3.6)
		if = {
			limit = { zz_ef_cb_cover > zz_ef_cb_sell_cover_high }
			add = {
				value = zz_ef_cb_cover
				subtract = zz_ef_cb_sell_cover_high
				multiply = money_supply
				multiply = zz_ef_cb_valuation
				multiply = 0.05
			}
		}
		# the cover counts foreign currency too: never more than 2% of the standard's own metal a week (5% over
		# 100% cover)
		max = zz_ef_cb_sell_max_units
	}
	min = 0
}
zz_ef_cb_sell_cover_high = 1
zz_ef_cb_sell_max_units = {
	value = zz_ef_cb_metal
	multiply = 0.02
	if = {
		limit = { zz_ef_cb_cover > zz_ef_cb_sell_cover_high }
		value = zz_ef_cb_metal
		multiply = 0.05
	}
}

# ---- the pops' hoard (Д.R4б.17; «Доверие к деньгам») -------------------------------
# The pops keep a target share of all they hold (savings + metal) in metal: the base + zz_ef_popm_share_distrust_k x the
# distrust (1 - the trust in money, Д.R7.5: ld_trust.txt -- the radicals raise it, the loyalists lower it; at the 1840
# world's shares, radicals 6 %, loyalists 1 %, 13 % as with the former coefficients). Over the target they bring a share of the excess a week to the
# CB (a metal standard with a CB: new notes at parity) or to the banks (a deposit; the metal is the banks'). Under the
# target nothing is forced: they buy through their need on the market.
zz_ef_popm_share_base = 0.1
zz_ef_popm_share_distrust_k = 0.5
zz_ef_popm_share_min = 0.02
zz_ef_popm_share_max = 0.6
zz_ef_popm_dishoard_speed = 0.01
zz_ef_popm_share_v = {
	value = 1
	subtract = zz_ef_trust_v
	multiply = zz_ef_popm_share_distrust_k
	add = zz_ef_popm_share_base
	min = zz_ef_popm_share_min
	max = zz_ef_popm_share_max
}
zz_ef_popm_wealth_v = {
	value = zz_ef_pop_savings
	add = zz_ef_popm_money
	min = 0
}
zz_ef_popm_target_v = {
	value = zz_ef_popm_wealth_v
	multiply = zz_ef_popm_share_v
}
# the share of the hoard brought out this week: speed x the excess over the target, of the hoard
zz_ef_popm_out_frac_v = {
	value = 0
	if = {
		limit = { zz_ef_popm_money > zz_ef_popm_target_v }
		value = zz_ef_popm_money
		subtract = zz_ef_popm_target_v
		multiply = zz_ef_popm_dishoard_speed
		multiply = zz_ef_step_weeks_v
		divide = zz_ef_popm_money
		max = 1
	}
}
# the hoard's share of all the pops hold now (the logs)
zz_ef_popm_share_now_v = {
	value = 0
	if = {
		limit = { zz_ef_popm_wealth_v > 0 }
		value = zz_ef_popm_money
		divide = zz_ef_popm_wealth_v
	}
}
