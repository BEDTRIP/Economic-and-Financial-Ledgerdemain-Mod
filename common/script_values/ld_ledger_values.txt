# the ledger's values (scripted_effects/ld_ledger.txt).
# The banks' book -- our accounts of the pool: liabilities (deposits of pops, of companies, of the investment fund and of other CBs,
# the CB's credit, the treasury's money, capital) less the loans (consumer, business). Before the first reconciliation
# the capital is 0.
zz_ef_bank_book_v = {
	value = 0
	add = zz_ef_ledger_deposits_v
	add = zz_ef_fd_dep_v
	add = zz_ef_nr_dep_v
	# (Д.R4б.31) the register's change the pool gets only at our step
	subtract = zz_ef_nr_dep_pending_v
	if = { limit = { has_variable = zz_ef_bank_cb_debt } add = var:zz_ef_bank_cb_debt }
	if = { limit = { has_variable = zz_ef_bank_gov_dep } add = var:zz_ef_bank_gov_dep }
	# (Д.R4б.33) the companies' deposits -- the engine's burnt money returned (ld_leak_back.txt)
	if = { limit = { has_variable = zz_ef_comp_dep } add = var:zz_ef_comp_dep }
	if = { limit = { has_variable = zz_ef_bank_capital } add = var:zz_ef_bank_capital }
	if = { limit = { has_variable = zz_ef_cc_debt } subtract = var:zz_ef_cc_debt }
	if = { limit = { has_variable = zz_ef_bc_debt } subtract = var:zz_ef_bc_debt }
	# (Д.R2.7) the banks' parts of foreign bonds (E&F's ai_privat_bank_bond_value_N) -- their asset
	subtract = zz_ef_bank_bonds
	# (Д.R5.2) the engine's debt at home -- the banks' loan to the government (ld_gov_loan.txt)
	subtract = zz_ef_bank_gov_loan_v
	# (Д.R4а.12) the fund's shares and the foreign money the banks bought -- their asset at its value
	if = { limit = { has_variable = zz_ef_bank_inv } subtract = var:zz_ef_bank_inv }
	# (Д.R4б.23) the state's building levels the banks bought (privatization) -- their asset at the price paid
	if = { limit = { has_variable = zz_ef_bank_priv } subtract = var:zz_ef_bank_priv }
}

# steps 10 and 12: the engine's price of a privatized building level is 250 x its construction points
# (PRIVATIZATION_PER_LEVEL_COST), the points a multiple of 100 -- the unit of the reconciliation's line «выкуп уровней»
zz_ef_lvl_unit = 25000
zz_ef_lvl_unit_neg = -25000
# a step's change within this of a whole multiple of the unit: the multiple is levels bought, the rest -- other money of
# the same step (it stays «прочее»). A small multiple -- zz_ef_lvl_tolerance_small: a noisy week of a large country
# often lands near one by chance; from zz_ef_lvl_big_buy (a railway level, 200 000) the pool's other moves of the same
# week (~5 000 in a large country) are allowed for, such a jump is rarely noise
zz_ef_lvl_tolerance_small = 1000
# the private owners' levels grown in a step that make a pool's multiple a purchase (the ownership fractions' noise below it)
zz_ef_lvl_privlv_min = 0.5
zz_ef_lvl_tolerance_big = 6000
zz_ef_lvl_big_buy = 100000
zz_ef_lvl_tolerance = {
	value = zz_ef_lvl_tolerance_small
	if = {
		limit = {
			OR = {
				zz_ef_lvl_mult_v >= zz_ef_lvl_big_buy
				zz_ef_lvl_mult_v <= zz_ef_lvl_big_buy_neg
			}
		}
		value = zz_ef_lvl_tolerance_big
	}
}
zz_ef_lvl_big_buy_neg = {
	value = zz_ef_lvl_big_buy
	multiply = -1
}
# 1 if var:zz_ef_t_lvl (a step's change, the pool's sign) is within zz_ef_lvl_tolerance of a whole negative multiple of
# the unit, else 0
zz_ef_lvl_is_buy_v = {
	value = 0
	if = {
		limit = {
			zz_ef_lvl_mult_v <= -24999
			zz_ef_lvl_rest_v <= zz_ef_lvl_tolerance
			zz_ef_lvl_rest_v >= zz_ef_lvl_tolerance_neg
		}
		value = 1
	}
}
# 1 if var:zz_ef_t_lvl is within zz_ef_lvl_tolerance of a whole positive multiple of the unit (the treasury's sign
# turned: the state paid -- the engine's nationalization of building levels), else 0
zz_ef_lvl_is_nat_v = {
	value = 0
	if = {
		limit = {
			zz_ef_lvl_mult_v >= 24999
			zz_ef_lvl_rest_v <= zz_ef_lvl_tolerance
			zz_ef_lvl_rest_v >= zz_ef_lvl_tolerance_neg
		}
		value = 1
	}
}
# 1 if our pool paid the state's levels sold this step: its multiple is the treasury's, the other sign (Д.R4б.23)

		}
		value = 1
	}
}

# Д.R4б.25: a step's fall of the pool's part (var:zz_ef_t_lvl < 0) taken as the levels grown's price, at most their
# count x zz_ef_lvl_price_max (a level costs 45-65K in Britain, multiples up to 200K)
zz_ef_lvl_price_max = 250000
# (min = a lower bound: the fall, but not below the levels' price; the levels grown -- the larger count, at least 1)

	value = var:zz_ef_t_lvl
	min = {
		value = var:zz_ef_d_complv
		min = var:zz_ef_d_privlv
		min = 1
		multiply = zz_ef_lvl_price_max
		multiply = -1
	}
}
