# The pops' cash at hand against their deposits (scripted_effects/ld_pop_cash.txt; Д.R7.3).
# the target of banknotes, a share of GDP at the reference trust and deposits earning at least inflation (Д.R7.3: M0 ~ 15 %
# of GDP; принято ночью)
zz_ef_pop_cash_base = 0.15
zz_ef_pop_cash_trust_ref = 0.8
# the target's rise per point of trust under the reference (принято ночью)
zz_ef_pop_cash_trust_k = 1
# the target's rise when the deposit rate is under inflation (принято ночью)
zz_ef_pop_cash_neg_real_add = 0.5
# the share of the gap moved a week (Д.R7.3)
zz_ef_pop_cash_move_share = 0.1

# COUNTRY scope: the marginals' share of the population (Д.R7.7, ld_rich_literacy.txt; 0 before the first month) -- no
# deposits, so no interest, exchange or credit: their savings (less the metal) are banknotes
zz_ef_marg_v = {
	value = 0
	if = { limit = { has_variable = zz_ef_marg } value = var:zz_ef_marg }
}
# the savings less the metal (at the start, before var:zz_ef_pop_savings -- the start's norm)
zz_ef_pop_cash_sav_v = {
	value = zz_ef_pop_start_savings
	subtract = zz_ef_popm_start_money
	if = { limit = { has_variable = zz_ef_pop_savings } value = var:zz_ef_pop_savings }
	min = 0
}
# COUNTRY scope: the banknotes the pops want -- the others' share of their target, the marginals' share of the savings;
# the target -- at most the savings
zz_ef_pop_cash_target_v = {
	value = zz_ef_pop_cash_want_v
	max = { value = zz_ef_pop_savings min = 0 }
}
zz_ef_pop_cash_want_v = {
	value = gdp
	min = 0
	multiply = zz_ef_pop_cash_base
	multiply = {
		value = zz_ef_pop_cash_trust_ref
		subtract = zz_ef_trust_v
		multiply = zz_ef_pop_cash_trust_k
		add = 1
		min = 0
	}
	if = {
		limit = { zz_ef_deposit_rate < zz_ef_inflation }
		multiply = { value = 1 add = zz_ef_pop_cash_neg_real_add }
	}
	multiply = { value = 1 subtract = zz_ef_marg_v }
	add = { value = zz_ef_pop_cash_sav_v multiply = zz_ef_marg_v }
}
# the step's move into banknotes (+) or into deposits (-): the share of the gap a week x the weeks of the step; in a panic
# (Д.R7.6, ld_trust.txt) the deposits go out zz_ef_panic_mult times faster
zz_ef_pop_cash_move_v = {
	value = zz_ef_pop_cash_target_v
	subtract = zz_ef_pop_cash
	multiply = {
		value = zz_ef_pop_cash_move_share
		multiply = zz_ef_step_weeks_v
		if = {
			limit = {
				has_variable = zz_ef_panic
				zz_ef_pop_cash_target_v > zz_ef_pop_cash
			}
			multiply = zz_ef_panic_mult
		}
		max = 1
	}
}
zz_ef_v_w_cash_dep = {
	value = 0
	if = { limit = { has_variable = zz_ef_w_cash_dep } value = var:zz_ef_w_cash_dep }
}
