# The companies' accounts (Д.R6.7, scripted_effects/ld_company_accounts.txt).
# COUNTRY scope: the accounts of the companies that exist, and the collector -- the companies' deposits over them.
zz_ef_co_acc_sum_v = {
	value = 0
	every_company = {
		limit = { has_variable = zz_ef_co_acc }
		add = var:zz_ef_co_acc
	}
}
zz_ef_co_rest_v = {
	value = zz_ef_comp_dep_v
	subtract = zz_ef_co_acc_sum_v
}
zz_ef_co_share_v = {
	value = 0
	if = { limit = { has_variable = zz_ef_co_share } value = var:zz_ef_co_share }
}
# the week's contributions to the pool that went into the companies' accounts (zz_ef_pop_deposits_step)
zz_ef_v_f_co_contrib = {
	value = 0
	if = { limit = { has_variable = zz_ef_f_co_contrib } value = var:zz_ef_f_co_contrib }
}
# COMPANY scope: the account and the last month's income (the log)
zz_ef_co_acc_v = {
	value = 0
	if = { limit = { has_variable = zz_ef_co_acc } value = var:zz_ef_co_acc }
}
zz_ef_co_in_last_v = {
	value = 0
	if = { limit = { has_variable = zz_ef_co_in_last } value = var:zz_ef_co_in_last }
}
# the week's business-credit service the companies paid out of their accounts (zz_ef_pop_deposits_step)
zz_ef_v_f_co_paid = {
	value = 0
	if = { limit = { has_variable = zz_ef_f_co_paid } value = var:zz_ef_f_co_paid }
}
# COMPANY scope (Д.R6.9): its part of the country's business credit -- the companies' share of zz_ef_bc_debt by its
# levels; its month's interest on it (the weekly rate zz_ef_cc_rate / 52 x 52 / 12)
zz_ef_co_lv_part_v = {
	value = 0
	if = {
		limit = {
			has_variable = zz_ef_co_lv
			owner ?= { has_variable = zz_ef_co_lv_sum var:zz_ef_co_lv_sum > 0 has_variable = zz_ef_co_share }
		}
		value = var:zz_ef_co_lv
		divide = owner.var:zz_ef_co_lv_sum
		multiply = owner.var:zz_ef_co_share
	}
}
zz_ef_co_debt_v = {
	value = 0
	if = {
		limit = { owner ?= { has_variable = zz_ef_bc_debt } }
		value = owner.var:zz_ef_bc_debt
		multiply = zz_ef_co_lv_part_v
	}
}
zz_ef_co_int_month_v = {
	value = zz_ef_co_debt_v
	multiply = owner.zz_ef_cc_rate
	divide = 12
}
zz_ef_co_out_last_v = {
	value = 0
	if = { limit = { has_variable = zz_ef_co_out_last } value = var:zz_ef_co_out_last }
}
# COUNTRY scope: the companies' part of the business credit; the collector's -- the rest
zz_ef_co_debt_sum_v = {
	value = 0
	every_company = { add = zz_ef_co_debt_v }
}
zz_ef_bc_debt_v = {
	value = 0
	if = { limit = { has_variable = zz_ef_bc_debt } value = var:zz_ef_bc_debt }
}
# Д.R6.8: the months of outflow a company keeps on its account; over it -- the dividend (принято ночью: 3)
zz_ef_co_reserve_months = 3
# COMPANY scope: the dividend due -- the account over its norm
zz_ef_co_div_due_v = {
	value = zz_ef_co_acc_v
	subtract = {
		value = zz_ef_co_out_last_v
		multiply = zz_ef_co_reserve_months
	}
	min = 0
}
zz_ef_co_div_last_v = {
	value = 0
	if = { limit = { has_variable = zz_ef_co_div_last } value = var:zz_ef_co_div_last }
}
# COUNTRY scope: the dividends paid to the pops' deposits, the window the deposits' card covers
zz_ef_v_w_co_div = {
	value = 0
	if = { limit = { has_variable = zz_ef_w_co_div } value = var:zz_ef_w_co_div }
}
