# Subjects-banks of a country (Д.R4б.1, scripted_effects/ld_bank_subjects.txt).

# COUNTRY scope: the country's built "Bank" levels, and the part the government owns («госбанки»; the ownership trigger
# fails on a building under construction).
zz_ef_bk_levels_v = {
	value = 0
	every_scope_building = {
		limit = {
			is_building_type = building_zz_ef_bank
			level > 0
			NOT = { is_under_construction = yes }
		}
		add = level
	}
}
zz_ef_bk_gov_levels_v = {
	value = 0
	every_scope_building = {
		limit = {
			is_building_type = building_zz_ef_bank
			level > 0
			NOT = { is_under_construction = yes }
			country_ownership_fraction > 0
		}
		add = {
			value = level
			multiply = country_ownership_fraction
		}
	}
}
# COUNTRY scope: «частные банкиры» -- the private levels without the companies' (scaled to fit)
zz_ef_bk_pri_levels_v = {
	value = var:zz_ef_bk_cl
	multiply = var:zz_ef_bk_ck
	multiply = -1
	add = var:zz_ef_bk_free
	min = 0
}
# COMPANY scope: its local books abroad (Д.R9а.2)
zz_ef_bk_books_n_v = {

		limit = { has_variable_list = zz_ef_bk_books }
		every_in_list = {
			variable = zz_ef_bk_books
			add = 1
		}
	}
}
# SUBJECT scope: the capital it keeps in the month's payout -- its norm; a local book -- all of it
zz_ef_bk_cap_hold_v = {
	value = zz_ef_bk_cap_keep_v
	if = {
		limit = {
			has_variable = zz_ef_bk_local
			has_variable = zz_ef_bk_cap
		}
		value = var:zz_ef_bk_cap
	}
}
zz_ef_bk_subj_n_v = {
	value = 0
	every_in_list = {
		variable = zz_ef_bk_subj
		add = 1
	}
}

# SUBJECT scope: its share of the country's levels; a country without banks -- all to «частные банкиры».
zz_ef_bk_share_v = {
	value = 0
	if = {
		limit = {
			exists = var:zz_ef_bk_cty
			var:zz_ef_bk_cty.var:zz_ef_bk_t > 0
		}
		value = var:zz_ef_bk_lv
		if = {
			limit = { var:zz_ef_bk_kind ?= flag:company }
			multiply = var:zz_ef_bk_cty.var:zz_ef_bk_ck
		}
		divide = var:zz_ef_bk_cty.var:zz_ef_bk_t
	}
	else_if = {
		limit = { var:zz_ef_bk_kind ?= flag:pri }
		value = 1
	}
}

# ---- the bank book by subject (Д.R4б.2, zz_ef_bk_book_step) ----

# the rate (принято ночью): a private subject's premium per unit of its capital ratio above the country's, and its
# bound; the appeal of the rate per unit of premium (1 + zz_ef_bk_appeal_k x premium: +-1 pp -> x1.2 / x0.8)
zz_ef_bk_prem_k = 0.25
zz_ef_bk_prem_max = 0.01
zz_ef_bk_appeal_k = 20
# a bankrupt company is out of the subjects this long (days; принято ночью)
zz_ef_bk_fail_days = 365
# a local book abroad not updated by its host this long is closed (the host is gone; zz_ef_bk_home_levels)
zz_ef_bk_book_seen_days = 14
# the capital a subject keeps, a share of its earning assets by kind (Д.R9б.4): loans, bonds and the privatised levels
# -- zz_ef_bk_cap_norm, the investments (fund shares, others' money and claims, revalued weekly) --
# zz_ef_bk_cap_norm_inv, so a paper gain of the revaluation stays as the reserve against the next one; above the norm --
# to the owners monthly (Д.R9.3)
zz_ef_bk_cap_norm = 0.1
zz_ef_bk_cap_norm_inv = 0.25
# the leverage (Д.R9.2, принято ночью): a subject's loans (consumer + business) at most its capital x
# zz_ef_bk_lev_base x (zz_ef_bk_lev_ref / the key rate), within [zz_ef_bk_lev_min, zz_ef_bk_lev_max] -- 10 at the
# default key rate (zz_ef_money_rate without a rate); the key rate's floor in the division
zz_ef_bk_lev_base = 10
zz_ef_bk_lev_ref = 0.065
zz_ef_bk_lev_min = 5
zz_ef_bk_lev_max = 20
zz_ef_bk_lev_rate_floor = 0.0025
# the capital a subject keeps above what its loans need at the leverage: the norm of Д.R9.3 is the larger of the two
# (принято ночью) -- with 10 % of its earning assets only, a bank of loans alone would be at its limit whenever the
# leverage is under 10
zz_ef_bk_cap_buffer = 1.25

# COUNTRY scope: the country's accounts the subjects' follow (those held as variables)
zz_ef_bk_c_gvd_v = { value = 0 if = { limit = { has_variable = zz_ef_bank_gov_dep } value = var:zz_ef_bank_gov_dep } }
zz_ef_bk_c_cbd_v = { value = 0 if = { limit = { has_variable = zz_ef_bank_cb_debt } value = var:zz_ef_bank_cb_debt } }
zz_ef_bk_c_cap_v = { value = 0 if = { limit = { has_variable = zz_ef_bank_capital } value = var:zz_ef_bank_capital } }
zz_ef_bk_c_cc_v = { value = 0 if = { limit = { has_variable = zz_ef_cc_debt } value = var:zz_ef_cc_debt } }
zz_ef_bk_c_bc_v = { value = 0 if = { limit = { has_variable = zz_ef_bc_debt } value = var:zz_ef_bc_debt } }
zz_ef_bk_c_inv_v = { value = 0 if = { limit = { has_variable = zz_ef_bank_inv } value = var:zz_ef_bank_inv } }
zz_ef_bk_c_prv_v = { value = 0 if = { limit = { has_variable = zz_ef_bank_priv } value = var:zz_ef_bank_priv } }
# COUNTRY scope: the banks' capital ratio -- capital / earning assets (loans, bonds, fund shares and foreign money)
zz_ef_bk_cty_capr_v = {
	value = zz_ef_bk_c_cap_v
	divide = {
		value = zz_ef_bk_c_cc_v
		add = zz_ef_bk_c_bc_v
		add = zz_ef_bank_bonds
		add = zz_ef_bk_c_inv_v
		add = zz_ef_bk_c_prv_v
		min = 1
	}
}

# SUBJECT scope: earning assets, capital ratio, the rate's premium (none for «госбанки»)
zz_ef_bk_assets_v = {
	value = 0
	if = { limit = { has_variable = zz_ef_bk_cc } add = var:zz_ef_bk_cc }
	if = { limit = { has_variable = zz_ef_bk_bc } add = var:zz_ef_bk_bc }
	if = { limit = { has_variable = zz_ef_bk_bnd } add = var:zz_ef_bk_bnd }
	if = { limit = { has_variable = zz_ef_bk_gvl } add = var:zz_ef_bk_gvl }
	if = { limit = { has_variable = zz_ef_bk_inv } add = var:zz_ef_bk_inv }
	if = { limit = { has_variable = zz_ef_bk_prv } add = var:zz_ef_bk_prv }
	min = 0
}
zz_ef_bk_capr_v = {
	value = 0
	if = { limit = { has_variable = zz_ef_bk_cap } value = var:zz_ef_bk_cap }
	divide = {
		value = zz_ef_bk_assets_v
		min = 1
	}
}
zz_ef_bk_prem_v = {
	value = 0
	if = {
		limit = { NOT = { var:zz_ef_bk_kind ?= flag:gov } }
		value = zz_ef_bk_capr_v
		subtract = var:zz_ef_bk_cty.var:zz_ef_bk_capr
		multiply = zz_ef_bk_prem_k
		min = {
			value = zz_ef_bk_prem_max
			multiply = -1
		}
		max = zz_ef_bk_prem_max
	}
}
# weights of the residual: the share of levels; x the appeal of the rate (deposits, loans); the liquidity gap (debt to
# the CB: earning assets above deposits and capital); earning assets (capital) -- each + the share, so a country whose
# subjects have none still shares by levels; the gap is net of the CB debt the subject already has
zz_ef_bk_w_lv_v = {
	value = var:zz_ef_bk_sh
}
zz_ef_bk_w_dep_v = {
	value = var:zz_ef_bk_prem
	multiply = zz_ef_bk_appeal_k
	add = 1
	min = 0
	multiply = var:zz_ef_bk_sh
}
# business credit (Д.R9.1, Д.R9.2): the free limit x the appeal of the loan rate (a cheaper bank lends more) + the share
# of levels (a country whose banks have no free limit shares the credit over it by levels)
zz_ef_bk_w_bc_v = {
	value = var:zz_ef_bk_prem
	multiply = zz_ef_bk_appeal_k
	multiply = -1
	add = 1
	min = 0
	multiply = zz_ef_bk_free_v
	add = var:zz_ef_bk_sh
}
zz_ef_bk_w_cbd_v = {
	value = zz_ef_bk_assets_v
	if = { limit = { has_variable = zz_ef_bk_dep } subtract = var:zz_ef_bk_dep }
	if = { limit = { has_variable = zz_ef_bk_fdd } subtract = var:zz_ef_bk_fdd }
	if = { limit = { has_variable = zz_ef_bk_cpd } subtract = var:zz_ef_bk_cpd }
	if = { limit = { has_variable = zz_ef_bk_nrd } subtract = var:zz_ef_bk_nrd }
	if = { limit = { has_variable = zz_ef_bk_gvd } subtract = var:zz_ef_bk_gvd }
	if = { limit = { has_variable = zz_ef_bk_cap } subtract = var:zz_ef_bk_cap }
	if = { limit = { has_variable = zz_ef_bk_cbd } subtract = var:zz_ef_bk_cbd }
	min = 0
	add = var:zz_ef_bk_sh
}
zz_ef_bk_w_cap_v = {
	value = zz_ef_bk_assets_v
	add = var:zz_ef_bk_sh
}
# COUNTRY scope: the key rate (without a CB -- the government's, zz_ef_cc_rate) and the leverage at it
zz_ef_bk_lev_v = {
	value = zz_ef_bk_lev_ref
	divide = {
		value = zz_ef_cc_rate
		subtract = zz_ef_cc_credit_premium
		min = zz_ef_bk_lev_rate_floor
	}
	multiply = zz_ef_bk_lev_base
	min = zz_ef_bk_lev_min
	max = zz_ef_bk_lev_max
}
# SUBJECT scope: its loan limit (capital x the country's leverage, var:zz_ef_bk_lev of the book step) and the free part
zz_ef_bk_limit_v = {
	value = 0
	if = {
		limit = {
			has_variable = zz_ef_bk_cap
			exists = var:zz_ef_bk_cty
			var:zz_ef_bk_cty = { has_variable = zz_ef_bk_lev }
		}
		value = var:zz_ef_bk_cap
		multiply = var:zz_ef_bk_cty.var:zz_ef_bk_lev
		min = 0
	}
}
zz_ef_bk_free_v = {
	value = zz_ef_bk_limit_v
	subtract = zz_ef_bk_loans_v
	min = 0
}
# SUBJECT scope: the capital its norm keeps -- the larger of the norms by kind of its earning assets and its loans x
# zz_ef_bk_cap_buffer / the leverage
zz_ef_bk_cap_keep_v = {
	value = zz_ef_bk_assets_v
	if = { limit = { has_variable = zz_ef_bk_inv } subtract = var:zz_ef_bk_inv }
	min = 0
	multiply = zz_ef_bk_cap_norm
	if = {
		limit = { has_variable = zz_ef_bk_inv }
		add = {
			value = var:zz_ef_bk_inv
			min = 0
			multiply = zz_ef_bk_cap_norm_inv
		}
	}
	if = {
		limit = {
			exists = var:zz_ef_bk_cty
			var:zz_ef_bk_cty = { has_variable = zz_ef_bk_lev }
		}
		min = {
			value = zz_ef_bk_loans_v
			multiply = zz_ef_bk_cap_buffer
			divide = var:zz_ef_bk_cty.var:zz_ef_bk_lev
		}
	}
}
# SUBJECT scope: its capital above the norm (zz_ef_bk_div_step); a local book abroad pays none -- its capital stays in
# the host (one capital of the company, Д.R9а.2: a payout abroad would be a cross-border flow; принято ночью)
zz_ef_bk_div_due_v = {
	value = 0
	if = {
		limit = {
			has_variable = zz_ef_bk_cap

		}
		value = var:zz_ef_bk_cap
		subtract = zz_ef_bk_cap_keep_v
		min = 0
	}
}
