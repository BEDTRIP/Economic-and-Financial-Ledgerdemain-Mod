# Deposits of non-residents (scripted_effects/ld_nr_deposits.txt).

# our money held abroad -- the deposits of the holders in our banks, money
zz_ef_nr_dep_v = {
	value = zz_ef_rq_out_m_v
}
# the register's change not yet in the pool: the clearing's settlements and the exchange's deals in our money since our
# last step, the holders' interest credited -- posted at our step (zz_ef_nr_dep_step), after the reconciliation; the
# banks' book leaves it out until then (Д.R4б.31: a week's swing -x / +x of «прочее», not a loss)
zz_ef_nr_dep_pending_v = {
	value = 0
	if = {
		limit = { has_variable = zz_ef_model_started }
		if = { limit = { has_variable = zz_ef_clr_a_cur_out } add = var:zz_ef_clr_a_cur_out }
		if = { limit = { has_variable = zz_ef_clr_a_own_back } subtract = var:zz_ef_clr_a_own_back }
		if = { limit = { has_variable = zz_ef_xch_nr_acc } add = var:zz_ef_xch_nr_acc }
		if = {
			limit = {
				zz_ef_nr_issuer = yes
				has_variable = zz_ef_rq_int_due
			}
			add = var:zz_ef_rq_int_due
		}
	}
}
# put into (+) / drawn out of (-) the pool this receiver / last window
zz_ef_v_f_nr_dep = {
	value = 0
	if = { limit = { has_variable = zz_ef_f_nr_dep } value = var:zz_ef_f_nr_dep }

# the export advantage's multiplier, +1 (= +1%) per 3% of GDP held abroad, at most 20
zz_ef_fx_holders_demand_per = 3
zz_ef_fx_holders_demand_max = 20
zz_ef_fx_holders_demand_m = {
	value = 0
	if = {
		limit = { zz_ef_raw_gdp > 0 }
		value = zz_ef_nr_dep_v
		divide = zz_ef_raw_gdp
		multiply = 100
		divide = zz_ef_fx_holders_demand_per
	}
	max = zz_ef_fx_holders_demand_max
	min = 0
}
