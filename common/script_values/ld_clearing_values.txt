# The clearing of payments with abroad (scripted_effects/ld_clearing.txt; Д.R8в.4, Д.R8в.11). docs/clearing-fx.md.

# Д.R8в.29: the share of a payment not in the payer's money (metal to a metal standard, else the receiver's money) is a
# balance of two wishes, weighted by the strength of the two currencies (zz_ef_clr_pair):
#   the receiver's demand -- by its trust in the payer's money, the payer's strength against the reference
#   (zz_ef_currency_strength): 100 % under 0.75, 30 % at 1.25 and over, linear between, the reference's money 15 %
#   (zz_ef_clr_metal_share, payer scope); a metal standard's CB under its cover target wants more of it
#   (zz_ef_clr_recv_boost);
#   the payer's offer -- by its cover (zz_ef_clr_cover_keep);
#   share = demand x w + offer x (1 - w), w = the receiver's GDP / the sum of both (the bigger economy has its way).
#   The currencies' strength is no weight: on metal it stays ~1 for all.
zz_ef_clr_metal_share_low = 0.75
zz_ef_clr_metal_share_high = 1.25
zz_ef_clr_metal_share_min = 0.3
zz_ef_clr_metal_share_ref = 0.15
zz_ef_clr_metal_share = {
	value = zz_ef_currency_strength
	subtract = zz_ef_clr_metal_share_low
	divide = {
		value = zz_ef_clr_metal_share_high
		subtract = zz_ef_clr_metal_share_low
	}
	multiply = {
		value = 1
		subtract = zz_ef_clr_metal_share_min
	}
	multiply = -1
	add = 1
	max = 1
	min = zz_ef_clr_metal_share_min
	if = {
		limit = { has_modifier = global_monetary_reference }
		value = zz_ef_clr_metal_share_ref
	}
}
# Payer scope: how much it is willing to pay other than its money, by its cover (Д.R8в.28, Д.R8в.29) -- only its
# metal over the target: nothing at the target (zz_ef_cover_normal), all at twice it, linear between. No CB (the
# banks' metal) -- all.
zz_ef_clr_cover_keep = {
	value = 1
	if = {
		limit = { zz_ef_clr_cb_ok = yes }
		value = zz_ef_cb_cover
		subtract = zz_ef_cover_normal
		divide = zz_ef_cover_normal
		min = 0
		max = 1
	}
}
# Receiver scope: its demand for metal x this -- a CB on a metal standard under its cover target wants more (x2 at a
# cover of 0); else 1.
zz_ef_clr_recv_boost = {
	value = 1
	if = {
		limit = {
			zz_ef_clr_cb_ok = yes
			zz_ef_clr_wants_metal = yes
			zz_ef_cb_cover < zz_ef_cover_normal
		}
		value = zz_ef_cover_normal
		subtract = zz_ef_cb_cover
		divide = zz_ef_cover_normal
		add = 1
	}
}
# Payer scope, the payer's own view for the log (EFX share): its trust curve x its offer
zz_ef_clr_own_share = {
	value = 1
	subtract = {
		value = zz_ef_clr_metal_share
		multiply = zz_ef_clr_cover_keep
	}
}
# Gold a unit of the scope country's money pays / claims abroad: its money against the parity (zz_ef_clr_gpm_own); no
# money of its own -- the world's average, weighted by GDP (zz_ef_world_acc).
zz_ef_clr_gold_per_money = {
	value = zz_ef_clr_gpm_own
	if = {
		limit = {
			NOT = { zz_ef_clr_gpm_own > 0 }
			has_global_variable = zz_ef_wld_gpm_avg
		}
		value = global_var:zz_ef_wld_gpm_avg
	}
	min = 0
}
# ... the country's own: its money against the parity (Д.R8а.4; a metal standard -- 1); no monetary system -- 0
zz_ef_clr_gpm_own = {
	value = 0
	if = {
		limit = { zz_ef_has_currency_value = yes }
		value = zz_ef_value_to_parity
	}
}
# The metal the payer's monetary authority can pay, in gold: the CB -- its standard's metal (zz_ef_cb_metal), the banks
# -- their gold and silver.
# Д.R8в.30: the floor of a CB's cover the clearing does not take its metal under -- half the target; the share of its
# reserves over it.
zz_ef_clr_cover_floor = {
	value = zz_ef_cover_normal
	multiply = 0.5
}
zz_ef_clr_over_floor = {
	value = 0
	if = {
		limit = { zz_ef_cb_cover > zz_ef_clr_cover_floor }
		value = zz_ef_cb_cover
		subtract = zz_ef_clr_cover_floor
		divide = zz_ef_cb_cover
	}
}
zz_ef_clr_metal_av_g = {
	value = 0
	if = {
		limit = { zz_ef_clr_cb_ok = yes }
		value = zz_ef_cb_metal
		if = {
			limit = { zz_ef_metal_silver = yes }
			multiply = silver_to_gold_rate
		}
		# only the metal over the floor of the cover (Д.R8в.30): below it the payer pays in claims, as a CB suspending
		# redemption would
		multiply = zz_ef_clr_over_floor
	}
	else = {
		value = zz_ef_bankm_gold
		add = {
			value = zz_ef_bankm_silver
			multiply = silver_to_gold_rate
		}
	}
	min = 0
}
