# a currency's strength against the reference currency (see
# scripted_triggers/ld_reference_currency_triggers.txt). COUNTRY scope.
# zz_ef_value_to_parity: a unit of the country's money against its parity (Д.R8а.4) -- see below.
# The reference's own value_to_parity is kept monthly in global_var:zz_ef_ref_vtp, its price index in
# global_var:zz_ef_ref_cpi (zz_ef_reference_strength_step, the world's monthly pass).
# !! MAINTENANCE !! Key-level REPLACE_OR_CREATE of E&F's difference_with_average_gold_exchange_
# rate_currencies and money_value_median_and_money_value_in_gold_ratio (script_values/
# 00_economic_scripted_value.txt:4970, 8542): our strength instead of the value / the median --
# they drive E&F's trade-route modifiers and the tooltip's "+15% against the average".
#############################################

# Д.R8а.4: 1 under a metal or exchange standard and with no monetary system (the market will move around the parity
# with the exchange); a fiat currency floats -- 1 at the transition, then x (the reference's prices / its own) since then
# (var:zz_ef_fiat_p0 / _r0, scripted_effects/ld_standard_switch.txt), x the reference's own value when the reference is
# fiat too (global_var:zz_ef_ref_vtp); the fiat reference itself (Д.R8б.28) -- the price of gold on its market at the
# transition over its price now (var:zz_ef_fiat_g0): gold dearer, every fiat money cheaper in gold. An external exchange
# standard -- its anchor's (var:zz_ef_anchor, scripted_effects/ld_anchor.txt; an anchor on fiat floats, and the subject
# with it).
zz_ef_value_to_parity_own = {
	value = 1
	if = {
		limit = {
			has_law = law_type:law_fiat_standard
			has_modifier = global_monetary_reference
			has_variable = zz_ef_fiat_g0
			var:zz_ef_fiat_g0 > 0
			zz_ef_gold_price_here_v > 0
		}
		value = var:zz_ef_fiat_g0
		divide = zz_ef_gold_price_here_v
		min = 0.01
		max = 100
	}
	else_if = {
		limit = {
			has_law = law_type:law_fiat_standard
			has_variable = zz_ef_fiat_p0
			has_variable = zz_ef_fiat_r0
			var:zz_ef_fiat_r0 > 0
			has_variable = zz_ef_prev_price
			var:zz_ef_prev_price > 0
			has_global_variable = zz_ef_ref_cpi
		}
		value = var:zz_ef_fiat_p0
		divide = var:zz_ef_prev_price
		multiply = global_var:zz_ef_ref_cpi
		divide = var:zz_ef_fiat_r0
		if = {
			limit = {
				has_global_variable = zz_ef_ref_vtp
				global_var:zz_ef_ref_vtp > 0
			}
			multiply = global_var:zz_ef_ref_vtp
		}
		min = 0.01
		max = 100
	}
}
# Country scope: the price of gold on its market against the base price (1 -- the base; 0 without a market)
zz_ef_gold_price_here_v = {
	value = 0
	if = {
		limit = { exists = market }
		value = market.mg:gold.market_goods_pricier
		add = 1
	}
}
zz_ef_value_to_parity = {
	value = zz_ef_value_to_parity_own
	if = {
		limit = { zz_ef_anchored = yes }
		value = var:zz_ef_anchor.zz_ef_value_to_parity_own
	}
}
zz_ef_currency_strength = {
	value = zz_ef_value_to_parity
	if = {
		limit = { has_global_variable = zz_ef_ref_vtp global_var:zz_ef_ref_vtp > 0 }
		divide = global_var:zz_ef_ref_vtp
	}
}
# (Д.R3.4) the trade modifier's multiplier from the REAL rate's skew against the reference --
# b = (1 / strength) x (the reference's price index / ours): a devaluation (strength down) makes our goods cheaper
# abroad, the export bonus; as our prices rise after it, b falls back and the bonus melts. (b - 1) x 40 (the slope of
# 1% of trade per 2.5% of skew), within -50..50. The price indices are the ones the
# weekly step keeps (var:zz_ef_prev_price; the reference's -- global_var:zz_ef_ref_cpi), 100 before the first step.
# Was (Е.10): (1 - strength) x 40 within -10..10, prices not counted.
zz_ef_currency_trade_m = {
	value = 1
	if = {
		limit = { zz_ef_currency_strength > 0.01 }
		divide = zz_ef_currency_strength
	}
	if = {
		limit = {
			has_global_variable = zz_ef_ref_cpi
			global_var:zz_ef_ref_cpi > 0
			has_variable = zz_ef_prev_price
			var:zz_ef_prev_price > 0
		}
		multiply = global_var:zz_ef_ref_cpi
		divide = var:zz_ef_prev_price
	}
	subtract = 1
	multiply = 40
	max = 50
	min = -50
}



# Ф.1: the banks' balance sheet -- assets (the pool as their reserves, business and
# consumer credit, bonds) minus liabilities (pops' deposits, debt to the CB); shown in the banks' card.
zz_ef_bank_capital = {
	value = zz_ef_pool
	add = zz_ef_bc_debt
	add = zz_ef_cc_debt
	add = zz_ef_bank_bonds
	subtract = zz_ef_pop_deposits
	subtract = zz_ef_bank_cb_debt
	# other CBs' deposits (our currency they hold)
	subtract = zz_ef_nr_dep_v
}
