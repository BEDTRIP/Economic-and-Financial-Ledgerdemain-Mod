# The construction regulator buys all four construction goods (Д.R8в.24): 10 units of any of them make a point
# (construction_goods_per_point); the split goes to what is offered on the market and is cheaper at the state's local
# prices. Effects -- scripted_effects/ld_construction_market.txt; docs/construction.md.

# Scope: state. Local price of a construction good: its base price x (1 + state_goods_pricier); the floor keeps a
# division safe at the bottom of the engine's price range.
zz_ef_reg_price_floor = 0.01
zz_ef_reg_p_wood = {
	value = 1
	sg:wood_construction = { add = state_goods_pricier }
	min = zz_ef_reg_price_floor
	multiply = wood_construction_good_base_price
}
zz_ef_reg_p_iron = {
	value = 1
	sg:iron_construction = { add = state_goods_pricier }
	min = zz_ef_reg_price_floor
	multiply = iron_construction_good_base_price
}
zz_ef_reg_p_steel = {
	value = 1
	sg:steel_construction = { add = state_goods_pricier }
	min = zz_ef_reg_price_floor
	multiply = steel_construction_good_base_price
}
zz_ef_reg_p_arc = {
	value = 1
	sg:arc_welded_construction = { add = state_goods_pricier }
	min = zz_ef_reg_price_floor
	multiply = arc_welded_construction_good_base_price
}

# Scope: state. What the market offers: sell orders of the state's market.
zz_ef_reg_supply_wood = {
	value = 0
	market = { mg:wood_construction = { add = market_goods_sell_orders } }
}
zz_ef_reg_supply_iron = {
	value = 0
	market = { mg:iron_construction = { add = market_goods_sell_orders } }
}
zz_ef_reg_supply_steel = {
	value = 0
	market = { mg:steel_construction = { add = market_goods_sell_orders } }
}
zz_ef_reg_supply_arc = {
	value = 0
	market = { mg:arc_welded_construction = { add = market_goods_sell_orders } }
}

# Scope: state. The cheapest price among the goods on offer.
zz_ef_reg_pmin = {
	value = 1000000
	if = { limit = { zz_ef_reg_supply_wood > 0 } max = zz_ef_reg_p_wood }
	if = { limit = { zz_ef_reg_supply_iron > 0 } max = zz_ef_reg_p_iron }
	if = { limit = { zz_ef_reg_supply_steel > 0 } max = zz_ef_reg_p_steel }
	if = { limit = { zz_ef_reg_supply_arc > 0 } max = zz_ef_reg_p_arc }
}

# Scope: state. Weight of a good in the purchase: its supply x (cheapest price / its price)^2.
zz_ef_reg_w_wood = {
	value = scope:zz_ef_reg_pmin
	divide = zz_ef_reg_p_wood
	save_temporary_value_as = zz_ef_reg_r
	multiply = scope:zz_ef_reg_r
	multiply = zz_ef_reg_supply_wood
}
zz_ef_reg_w_iron = {
	value = scope:zz_ef_reg_pmin
	divide = zz_ef_reg_p_iron
	save_temporary_value_as = zz_ef_reg_r
	multiply = scope:zz_ef_reg_r
	multiply = zz_ef_reg_supply_iron
}
zz_ef_reg_w_steel = {
	value = scope:zz_ef_reg_pmin
	divide = zz_ef_reg_p_steel
	save_temporary_value_as = zz_ef_reg_r
	multiply = scope:zz_ef_reg_r
	multiply = zz_ef_reg_supply_steel
}
zz_ef_reg_w_arc = {
	value = scope:zz_ef_reg_pmin
	divide = zz_ef_reg_p_arc
	save_temporary_value_as = zz_ef_reg_r
	multiply = scope:zz_ef_reg_r
	multiply = zz_ef_reg_supply_arc
}

# How far a week moves the shares toward the market's split (the rest is last week's): a step damps the swing of
# buying more of what got cheap
zz_ef_reg_step = 0.5

# Scope: state. A share of the purchase (var:zz_ef_reg_s_<good>), 0 without the variable; wood is the default.
zz_ef_reg_s_wood_v = {
	value = 1
	if = { limit = { has_variable = zz_ef_reg_s_wood } value = var:zz_ef_reg_s_wood }
}
zz_ef_reg_s_iron_v = {
	value = 0
	if = { limit = { has_variable = zz_ef_reg_s_iron } value = var:zz_ef_reg_s_iron }
}
zz_ef_reg_s_steel_v = {
	value = 0
	if = { limit = { has_variable = zz_ef_reg_s_steel } value = var:zz_ef_reg_s_steel }
}
zz_ef_reg_s_arc_v = {
	value = 0
	if = { limit = { has_variable = zz_ef_reg_s_arc } value = var:zz_ef_reg_s_arc }
}

# Scope: state. The base price of a unit of the mix (PSC's price model and the spending's split between the states).
zz_ef_reg_base_mix = {
	value = 0
	add = { value = zz_ef_reg_s_wood_v multiply = wood_construction_good_base_price }
	add = { value = zz_ef_reg_s_iron_v multiply = iron_construction_good_base_price }
	add = { value = zz_ef_reg_s_steel_v multiply = steel_construction_good_base_price }
	add = { value = zz_ef_reg_s_arc_v multiply = arc_welded_construction_good_base_price }
	min = 1
}

# The regulator's staffing counted at least this in its throughput (PSC_scripted_effects.txt): an empty regulator does
# not get a throughput of thousands for the week its workers come
zz_ef_reg_occupancy_floor = 0.1
