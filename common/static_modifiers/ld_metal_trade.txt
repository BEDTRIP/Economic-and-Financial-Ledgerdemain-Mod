# The metal accounts buy and sell through buildings (Ф1, Д2.14, Д2.25, Д2.28).
# Applied weekly by zz_ef_metal_week_step (scripted_effects/ld_metal_accounts.txt), add_modifier on the
# building with a multiplier, as E&F sizes its exchanges' inputs (bond_deemande):
#   * the bank (building_zz_ef_bank): percent on its methods' base input of metal (multiplier = the country's
#     purchase over the base, -100..+25000);
#   * the CB (building_bank): its standard's method buys 1 gold (10 silver) per workforce unit; the multiplier
#     (percent) steers it by the cover -- more under 40%, none from 50% (zz_ef_cb_buy_mult);
#   * the CB's sale of a surplus over 60% cover: goods a week of gold / silver out (multiplier = the goods).
# The engine does not buy a good through a building's own _add modifier when the method does not list it (below).
#############################################

# the bank: percent on its methods' base of 0.01 gold / 0.02 silver per workforce unit (a building's own
# goods_input_gold_add modifier buys nothing -- the input reads 0 with the modifier set)
zz_ef_bank_gold_buy = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_coins_positive.dds
	goods_input_gold_mult = 0.01
}

}

	goods_input_silver_mult = 0.01
